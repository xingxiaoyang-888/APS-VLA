"""Root-only byte-exact Git-index archive delivery for this heartbeat."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import subprocess
import sys

root = Path(__file__).resolve().parents[1]
base = 'results/formal_main_v2/'
required_name = base + 'preflight/parallel_review_batch_20261017T2250_git_required.json'
proof_name = base + 'preflight/parallel_review_batch_20261017T2250_git_delivery.json'
assert not (root / proof_name).exists()
digest = lambda data: hashlib.sha256(data).hexdigest()

def git(*args):
    return subprocess.check_output(['git', *args], cwd=root)

pins = {
    'scripts/ops/nonweight_archive.py': 'f81284b3c56fbfd1cce339c4ea98b20943c8cd0866b4e8b94c38ac3cdbd1e3c2',
    'scripts/ops/check_nonweight_archive_stage.py': 'ebe5d25ff5850fe37e73cec551332ccd2b446aa3c4bbb0e38b6b83ce7a6775f6',
}
for name, checksum in pins.items():
    assert digest((root / name).read_bytes()) == checksum
before = git('ls-files', '--stage', '-z')
changes = git('diff', '--cached', '--name-status', '-z')
head = git('rev-parse', 'HEAD').decode().strip()
assert head == '1a2f29afe8c85ae1068738323a7664989bfd7ce1'
existing_name = 'configs/manifests/formal_main_v2/libero_object/initial_states/development.json'
existing_sha256 = '6b7e6dd9fb5b6351ee47c7bb30eab911e7c144d700cd88f7e5162641738b2d6a'
existing_record = base + 'preflight/object_existing_development_panel_index_20261018.json'
existing = json.loads((root / existing_record).read_bytes())
assert existing['status'] == 'verified' and existing['head'] == head
assert existing['file']['path'] == existing_name and existing['file']['sha256'] == existing_sha256
assert not git('status', '--porcelain', '--', existing_name).strip()
for value in ((root / existing_name).read_bytes(), git('show', 'HEAD:' + existing_name), git('show', ':' + existing_name)):
    assert digest(value) == existing_sha256
required = json.loads((root / required_name).read_bytes())
command = [sys.executable, '-B', 'scripts/ops/check_nonweight_archive_stage.py',
    '--project-root', str(root),
    '--snapshot', base + 'nonweight_archives/v1/snapshots/long_dev600_20261018.json',
    '--snapshot-sha256', 'ac7e4dce68abeddb147dd412f83afffbc5041715868547d6f60811e5fca137f0',
    '--raw-path', base + 'libero_10/downstream/development/8c843c1d4ee8383c/attempt01/trials.jsonl',
    '--raw-path', base + 'preflight/long_sixth_development_raw_v1/17_trials.jsonl',
    '--required-files', required_name]
result = subprocess.run(command, cwd=root, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
assert result.returncode == 0, result.stderr.decode('utf-8', errors='replace')
report = json.loads(result.stdout)
assert report['status'] == 'verified_index_archive_delivery'
assert report['index_sha256'] == digest(before)
assert report['staged_changes_sha256'] == digest(changes)
assert report['head'] == head
assert report['frozen_files_verified_in_worktree_and_index'] == 173
assert report['required_count'] == len(required['files'])
assert len(report['raw_paths_restored_exactly']) == 2
assert report['gpu_seconds'] == 0 and not report['real_originals_changed']
assert before == git('ls-files', '--stage', '-z')
assert changes == git('diff', '--cached', '--name-status', '-z')
assert not git('status', '--porcelain', '--', existing_name).strip()
for value in ((root / existing_name).read_bytes(), git('show', 'HEAD:' + existing_name), git('show', ':' + existing_name)):
    assert digest(value) == existing_sha256
proof = dict(schema='formal_main_v2', kind='root_git_index_nonweight_delivery',
    reviewed_utc=datetime.now(timezone.utc).isoformat(), status='passed',
    root_guard_command=command, guard_result=report,
    separately_verified_existing_development_dependency=dict(path=existing_name,
        sha256=existing_sha256, HEAD_index_worktree_equal_before_and_after=True,
        identity_receipt=existing_record, identity_receipt_sha256=digest((root / existing_record).read_bytes()),
        never_modified_or_staged=True),
    unchanged_index_until_guard_return=True, scientific_settings_changed=False,
    no_weights_delivered=True, final_state_values_opened=False)
data = (json.dumps(proof, ensure_ascii=False, sort_keys=True, indent=2) + '\n').encode('utf-8')
with (root / proof_name).open('xb') as stream:
    stream.write(data)
subprocess.run(['git', 'add', '--', proof_name], cwd=root, check=True)
old_rows = set(before.split(b'\0')) - {b''}
new_rows = set(git('ls-files', '--stage', '-z').split(b'\0')) - {b''}
assert not old_rows - new_rows and len(new_rows - old_rows) == 1
assert next(iter(new_rows - old_rows)).split(b'\t', 1)[1].decode('utf-8') == proof_name
assert git('show', ':' + proof_name) == data
subprocess.run(['git', 'diff', '--cached', '--check'], cwd=root, check=True)
assert not git('diff', '--name-only').strip(), 'Unstaged work remains after guarded delivery'
print(json.dumps(dict(status='passed', proof=proof_name, sha256=digest(data),
    required_count=report['required_count'], index_sha256=report['index_sha256'],
    frozen_files=173, restored_raw_paths=2, commit_not_performed=True)))
