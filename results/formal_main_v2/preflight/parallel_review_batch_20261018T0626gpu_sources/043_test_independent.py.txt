def run_independent_cell(rt, directory, method, *, done=None, fail=None, exhausted=False):
    original = rt.module.run_episode
    policy, components, identity = author.setup_policy(rt, method)
    base = policy
    class BytePolicy:
        selector = base.selector
        def reset(self):
            return base.reset()
        def __call__(self, observation, instruction):
            chunk = base(observation, instruction)
            values = chunk.environment_actions.copy()
            values[0, 0] = -0.0
            return rt.module.PolicyChunk(chunk.normalized_actions, values, chunk.metadata)
    policy = BytePolicy()
    state = dict(initial_state=[1., 2.], initial_state_sha256=m.state_hash([1., 2.]))
    journal = m.Journal(directory, "synthetic-independent", 3600)
    budget = m.GlobalBudget()
    if exhausted:
        budget.counts["policy"] = 16
    env = author.Environment(done=done, fail=fail)
    result, error = None, None
    try:
        with adapter.install_action_execution_compat(project_root=ROOT, source_root=ROOT,
                runtime_module=rt.module, formal_trials_module=rt.trials,
                expected_adapter_sha256=m.ADAPTER_SHA, emit=journal.emit):
            result = m.run_cell(rt.module, policy, env, "synthetic-independent", state, identity,
                journal, budget, lambda: m.policy_counters(policy, components, budget))
    except ValueError as exc:
        error = exc
    finally:
        journal.close()
    assert rt.module.run_episode is rt.trials.run_episode is original
    assert not components.vla.vision_backbone.hooks and not components.vla.language_model.model.hooks
    return SimpleNamespace(result=result, error=error, env=env, journal=journal, budget=budget, components=components)


@pytest.mark.parametrize("method", ["dense", "uniform", "A", "B"])
def test_four_methods_preserve_original_f64_bytes_signed_zero_and_finite_overrange(runtime, tmp_path, method):
    x = run_independent_cell(runtime, tmp_path, method)
    assert x.error is None
    raw = x.result["raw_result"]
    expected = np.asarray(raw["chunks"][0]["environment_actions"], dtype="<f8")
    received = np.asarray(x.env.actions[10:], dtype="<f8")
    assert received.tobytes() == expected.tobytes()
    assert np.signbit(received[0, 0]) and received[7, 2] == 1.010742197890625
    assert raw["control_steps"] == 8 and len(x.env.actions) == 18
    assert raw["policy_calls"] == 1 and len(raw["chunks"]) == 1
    assert x.budget.counts == dict(policy=1, vla_backbone=1, vision=1)
    assert x.result["observed_counts"] == dict(vla_backbone=1, vision=1, action_head=1,
        packing=1, topk=int(method in ("A", "B")), scorer=int(method in ("A", "B")))
    assert x.journal.events[-1]["original_bindings_restored"] is True

def test_actual_three_scoped_writers_and_new_journal_are_compatible(runtime, tmp_path, monkeypatch):
    """Actual emit producers; only their source/model-file I/O is substituted."""
    meta = m.load_module(ROOT / "scripts/ops/formal_asset_metadata_adapter.py", "independent_actual_metadata_writer")
    goal = m.load_module(ROOT / "scripts/ops/formal_goal_component_adapter.py", "independent_actual_goal_writer")
    loader = m.load_module(ROOT / "src/cti_vla/evaluation/checkpoint_manifest.py", "independent_strict_json_loader")
    views = {s: dict(suite=s, original_path=str(ROOT / meta.ORIGINALS[s][0]),
        derived_path=str(ROOT / "configs/assets/formal_main_v2_strict" / (s + ".json"))) for s in meta.ORIGINALS}
    identity = dict(source_root=str(ROOT), views=views, watched_files={},
        original_loader_code_sha256=meta.code_sha(loader.load_checkpoint_manifest),
        directory_verifier_code_sha256=meta.code_sha(loader.verify_checkpoint_directory))
    monkeypatch.setattr(meta, "validate_asset_metadata_compat", lambda **kwargs: copy.deepcopy(identity))
    monkeypatch.setattr(meta, "_canonical_loader", lambda source: loader)

    def original_state_loader(path):
        return {"synthetic_tensor_key": 1}
    def original_remove_prefix(value):
        return value
    def original_load_components(*args, **kwargs):
        raise AssertionError("GPU component loading forbidden in CPU writer test")
    def original_torch_load(*args, **kwargs):
        raise AssertionError("torch.load forbidden in CPU writer test")
    component = SimpleNamespace(_load_component_state=original_state_loader,
        remove_uniform_ddp_prefix=original_remove_prefix, load_openvla_oft_components=original_load_components,
        torch=SimpleNamespace(load=original_torch_load))
    mappings = [dict(requested_path=str(tmp_path / (name + "150000.synthetic")),
        target_path=str(tmp_path / (name + "50000.synthetic"))) for name in ("action", "proprio")]
    goal_identity = dict(mappings=mappings, watched_files={},
        original_function_code_sha256={k: goal.code_sha(getattr(component, k)) for k in goal.SYMBOLS})
    monkeypatch.setattr(goal, "_verify_target", lambda row: Path(row["target_path"]))
    journal = m.Journal(tmp_path, "synthetic-independent", 3600)
    try:
        with adapter.install_action_execution_compat(project_root=ROOT, source_root=ROOT,
                runtime_module=runtime.module, formal_trials_module=runtime.trials,
                expected_adapter_sha256=m.ADAPTER_SHA, emit=journal.emit):
            for suite in ("libero_object", "libero_goal"):
                journal.context = {"suite": suite}
                with ExitStack() as stack:
                    stack.enter_context(meta.install_asset_metadata_compat(project_root=ROOT, source_root=ROOT,
                        registration_path=ROOT / meta.REGISTRATION, expected_adapter_sha256=m.sha(meta.__file__), emit=journal.emit))
                    if suite == "libero_goal":
                        stack.enter_context(goal._scoped_component_loader(component, goal_identity, journal.emit))
                    result = loader.load_checkpoint_manifest(ROOT / meta.ORIGINALS[suite][0])
                    assert result.files and result.total_bytes > 0
                    if suite == "libero_goal":
                        for row in mappings:
                            assert component._load_component_state(Path(row["requested_path"])) == {"synthetic_tensor_key": 1}
        journal.context = {}
        m.validate_scope_events(journal.events)
    finally:
        journal.close()
    assert [v["seq"] for v in journal.events] == list(range(len(journal.events)))
    assert len([v for v in journal.events if v["event"] == "asset_metadata_compat_load_complete"]) == 2
    assert len([v for v in journal.events if v["event"] == "goal_component_load_complete"]) == 2
    assert component._load_component_state is original_state_loader

@pytest.mark.parametrize("case", ["probe", "manifest", "adapter", "passed_string", "kind", "evidence", "root_independent", "output", "limits", "retry"])
def test_self_consistent_raw_sha_cannot_substitute_bad_registration_semantics(root_contract, case):
    x = root_contract
    if case in ("probe", "manifest", "adapter"):
        key = {"probe": "probe_sha256", "manifest": "manifest_sha256", "adapter": "action_adapter_sha256"}[case]
        x.docs["candidate_independent"]["candidate_identity"][key] = "f" * 64
    elif case == "passed_string":
        x.docs["candidate_independent"]["status"] = "passed_synthetic_wrong_review"
        x.reg["cpu_reviews"]["candidate_independent"]["accepted_status"] = "passed_synthetic_wrong_review"
    elif case == "kind":
        x.docs["root_acceptance"]["kind"] = "unrelated_CPU_review"
    elif case == "evidence":
        x.docs["root_acceptance"]["evidence_sha256"].pop("audited/manifest.json")
    elif case == "root_independent":
        x.docs["root_acceptance"]["candidate_independent_review"]["sha256"] = "f" * 64
    elif case == "output":
        x.reg["output"] = x.reg["output_root"] + "/" + x.args.job_id
    elif case == "limits":
        x.reg["limits"]["cells"] = 17
    elif case == "retry":
        x.reg["automatic_retry"] = True
    x.save(bind_independent=case != "root_independent")
    with pytest.raises(ValueError):
        m.validate_gpu_registration(x.args, x.manifest_sha)


@pytest.mark.parametrize("path", ["reviews/final_states.json", "reviews/test_state.json", "reviews/weights/anything.json", "../escape.json", "results/final/panel.json"])
def test_cpu_receipt_ref_lexically_rejected_before_resolution(root_contract, monkeypatch, path):
    x = root_contract
    x.docs["root_acceptance"]["candidate_files"]["probe"]["path"] = path
    x.save()
    original_resolve = Path.resolve
    def safe_resolve(p, *args, **kwargs):
        if p.as_posix().endswith(path) or "../escape.json" in p.as_posix():
            pytest.fail("forbidden metadata reached filesystem resolution")
        return original_resolve(p, *args, **kwargs)
    monkeypatch.setattr(Path, "resolve", safe_resolve)
    with pytest.raises(ValueError):
        m.validate_gpu_registration(x.args, x.manifest_sha)
