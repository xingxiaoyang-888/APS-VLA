"""Single root invocation of the registered capacity-gated engineering submitter."""
from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,subprocess
ROOT=Path(__file__).resolve().parents[1]
CHECK='results/formal_main_v2/preflight/action_execution_gpu_registration_check_0626.json'
OUT='results/formal_main_v2/preflight/action_execution_gpu_capacity_check_0626.json'
assert not (ROOT/OUT).exists()
check=json.loads((ROOT/CHECK).read_bytes())
assert check['status']=='passed_local_contract_no_submission'
command=check['command'].copy(); assert command.pop()=='--check-only'; command.append('--submit')
started=datetime.now(timezone.utc).isoformat()
run=subprocess.run(command,cwd=ROOT,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
result=json.loads(run.stdout) if run.returncode==0 else None
value=dict(schema='formal_main_v2',kind='root_registered_engineering_capacity_attempt',
    status='completed_capacity_check' if run.returncode==0 else 'failed_preserved_no_automatic_retry',
    started_utc=started,finished_utc=datetime.now(timezone.utc).isoformat(),command=command,
    exit_code=run.returncode,stdout=run.stdout.decode(),stderr=run.stderr.decode(),result=result,
    evidence_sha256={CHECK:hashlib.sha256((ROOT/CHECK).read_bytes()).hexdigest(),
        '.local/try_action_gpu_capacity_0626.py':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()},
    scientific_release=False,final_states_opened=False)
with (ROOT/OUT).open('x',encoding='utf-8',newline='\n') as stream: json.dump(value,stream,ensure_ascii=False,indent=2);stream.write('\n')
print(json.dumps(value,ensure_ascii=False))
assert run.returncode==0,run.stderr.decode()
