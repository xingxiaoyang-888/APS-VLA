def sha(data):
    return hashlib.sha256(data).hexdigest()


def git(*args):
    return subprocess.check_output(["git", *args], cwd=ROOT)


def safe_name(name, active_panels=()):
    """Reject sealed/weight aliases lexically before any payload IO."""
    assert isinstance(name, str) and name and "\\" not in name and ":" not in name, name
    pure = PurePosixPath(name)
    assert not pure.is_absolute() and pure.as_posix() == name, name
    assert all(part not in ("", ".", "..") for part in pure.parts), name
    low = tuple(part.lower() for part in pure.parts)
    assert "test" not in low and not any("sealed" in part for part in low), name
    assert "weights" not in low and "checkpoints" not in low, name
    assert pure.suffix.lower() not in DENIED_SUFFIXES, name
    assert not (pure.name.lower() == "panel.json" and "final" in low), name
    scientific_py = pure.suffix == ".py" and pure.parts[0] in ("src", "scripts", "tests")
    state_name = any(stem in part for stem in DENIED_STATE_STEMS for part in low)
    assert not state_name or scientific_py or name in active_panels, name
    if name in active_panels:
        assert name in PANEL_PINS, name
    return pure


def ref(name, active_panels=()):
    safe_name(name, active_panels)
    path = ROOT / name
    assert path.is_file() and not path.is_symlink(), name
    for parent in path.parents:
        if parent == ROOT:
            break
        assert not parent.is_symlink(), name
    data = path.read_bytes()
    return dict(path=name, sha256=sha(data), bytes=len(data))


def read_json(name, expected_sha=None, active_panels=()):
    row = ref(name, active_panels)
    if expected_sha is not None:
        assert row["sha256"] == expected_sha, name
    data = (ROOT / name).read_bytes()
    assert sha(data) == row["sha256"], name
    return json.loads(data)


def write_new(name, value):
    payload = (json.dumps(value, indent=2, sort_keys=True) + "\n").encode("utf-8")
    with (ROOT / name).open("xb") as stream:
        stream.write(payload)
    return payload


def validate_boundary_rows(rows):
    assert isinstance(rows, list) and len(rows) == len(EXPECTED_BOUNDARIES)
    assert {row["job_id"] for row in rows} == set(EXPECTED_BOUNDARIES)
    assert len({(row["suite"], row["budget"], row["seed"]) for row in rows}) == 3
    for row in rows:
        actual = (row["suite"], row["budget"], row["seed"], row["boundary"])
        assert actual == EXPECTED_BOUNDARIES[row["job_id"]], row["job_id"]
        assert type(row["boundary"]) is int
        assert row["allowed_next"] == (None if row["boundary"] == 5000 else row["boundary"] + 500)
    assert sum(row["boundary"] == 5000 for row in rows) == 0


def classify_current_pulls(batch, training_pulls):
    """No job inference or recursive expansion of old pull references."""
    listed = batch["current_terminal_pulls"]
    assert isinstance(listed, list) and listed and len(listed) == len(set(listed))
    current = set(listed)
    engineering = batch.get("new_engineering_evidence", [])
    assert isinstance(engineering, list) and len(engineering) == len(set(engineering))
    for name in engineering:
        safe_name(name)
        assert name in batch["evidence_sha256"], name
    assert training_pulls <= current and len(training_pulls) == 3
    for name in current:
        assert name.startswith(BASE + "pulls/") and name.endswith("/receipt.json"), name
        assert name in batch["evidence_sha256"], name
    extra = current - training_pulls
    assert extra <= set(engineering), sorted(extra - set(engineering))
    return current, extra, engineering


def resolve_evidence_name(name, expected_sha, expected_bytes=None):
    """Only one exact historical tuple may resolve away from its original path."""
    safe_name(name, PANEL_PINS)
    if name != CANONICAL_SOURCE["path"]:
        return name
    if expected_sha == HISTORICAL_SOURCE["sha256"]:
        assert expected_bytes is None or expected_bytes == HISTORICAL_SOURCE["bytes"]
        return HISTORICAL_ARCHIVE["path"]
    assert expected_sha == CANONICAL_SOURCE["sha256"], name
    assert expected_bytes is None or expected_bytes == CANONICAL_SOURCE["bytes"]
    return name


def check_source_version_resolution():
    """Revalidate the accepted exception without changing either old receipt."""
    resolution = read_json(SOURCE_VERSION_NAME, SOURCE_VERSION_SHA)
    prior = read_json(PRIOR_DELIVERY_NAME, PRIOR_DELIVERY_SHA)
    assert resolution["status"] == "exact_version_archived"
    assert resolution["kind"] == "single_historical_input_version_resolution"
    assert resolution["original_path"] == HISTORICAL_SOURCE["path"]
    assert resolution["original_sha256"] == HISTORICAL_SOURCE["sha256"]
    assert resolution["original_bytes"] == HISTORICAL_SOURCE["bytes"]
    assert resolution["canonical_bytes"] == CANONICAL_SOURCE["bytes"]
    assert resolution["existing_HEAD_and_index_sha256"] == CANONICAL_SOURCE["sha256"]
    assert resolution["exact_original_archive"] == HISTORICAL_ARCHIVE
    assert resolution["arbitrary_SHA_mismatches_allowed"] is False and resolution["guard_changed"] is False
    assert resolution["conversion"] == dict(JSON_value_equal=True, byte_equal_to_existing_HEAD=True,
                                            not_a_frozen_runtime_file=True, operation="replace CRLF by LF", replacements=19)
    assert prior["status"] == "passed" and prior["source_canonicalization_only"] is True
    assert prior["old_batch_and_diagnosis_unchanged"] is True
    assert prior["unrelated_evidence_mismatch_exemptions"] == 0
    assert prior["canonical_current_source"] == CANONICAL_SOURCE
    assert prior["exact_original_archive"] == HISTORICAL_ARCHIVE
    assert prior["historical_source_resolution"] == ref(SOURCE_VERSION_NAME)
    assert prior["guard_result"]["codec_sha256"] == PINS["scripts/ops/nonweight_archive.py"]
    assert prior["guard_result"]["checker_sha256"] == PINS["scripts/ops/check_nonweight_archive_stage.py"]
    assert prior["guard_result"]["frozen_files_verified_in_worktree_and_index"] == 173
    rows = [ref(SOURCE_VERSION_NAME), ref(PRIOR_DELIVERY_NAME)]
    for row in [CANONICAL_SOURCE, HISTORICAL_ARCHIVE,
                resolution["batch"], resolution["diagnosis"], resolution["rejected_required"]]:
        assert ref(row["path"]) == row, row["path"]
        rows.append(row)
    original = (ROOT / HISTORICAL_ARCHIVE["path"]).read_bytes()
    canonical = (ROOT / CANONICAL_SOURCE["path"]).read_bytes()
    assert sha(original) == HISTORICAL_SOURCE["sha256"] and sha(canonical) == CANONICAL_SOURCE["sha256"]
    assert original.count(b"\r\n") == 19 and original.replace(b"\r\n", b"\n") == canonical
    assert json.loads(original) == json.loads(canonical)
    assert sha(git("show", "HEAD:" + CANONICAL_SOURCE["path"])) == CANONICAL_SOURCE["sha256"]
    assert sha(git("show", ":" + CANONICAL_SOURCE["path"])) == CANONICAL_SOURCE["sha256"]
    assert not git("status", "--porcelain", "--", CANONICAL_SOURCE["path"]).strip()
    return rows

def main():

        assert name in evidence, name

                                          sha256=ref(SELF_NAME)["sha256"], bytes=ref(SELF_NAME)["bytes"])
    assert evidence[stage_archive_name] == ref(SELF_NAME)["sha256"]
    training_pulls = set()
    boundary_records = []
    for boundary in boundaries:
        job = boundary["job_id"]
        end = boundary["boundary"]
        assert isinstance(end, int) and 500 <= end <= 5000 and end % 500 == 0
        expected_next = None if end == 5000 else end + 500
        assert boundary["allowed_next"] == expected_next
        name = batch["audit_source_archive_map"][".local/root_pull_review_2250_" + job + ".json"]
        op = bound_json(name)
        assert op["status"] == "passed"
        for key in ("job_id", "suite", "budget", "seed", "boundary", "allowed_next",
                    "pair_gpu_seconds", "latest_gpu_seconds"):
            assert op[key] == boundary[key], (job, key)
        assert op["pull"] in current_pulls
        assert op["pull_sha256"] == evidence[op["pull"]]
        training_pulls.add(op["pull"])
        root_name = op["root_review"]
        assert evidence[root_name] == op["root_sha256"]
        review = bound_json(root_name)
        assert review["status"] == "passed" and review["job_id"] == job
        assert (review["suite"], review["visual_tokens"], review["seed"]) == (
            boundary["suite"], boundary["budget"], boundary["seed"])
        assert review["source_sha256"] == SOURCE_SHA
        assert review["completed_by_arm"] == {"A": end, "B": end}
        assert review["allowed_next_segment_end"] == expected_next
        assert review["pair_allocated_gpu_seconds"] == boundary["pair_gpu_seconds"]
        assert review["sealed_states_opened"] is False and review["weights_copied"] is False
        ind_name = root_name[:-5] + "_independent.json"
        independent = bound_json(ind_name)
        assert independent["status"] == "passed" and independent["job_id"] == job
        assert independent["allowed_next_segment_end"] == expected_next
        assert independent["read_evidence_sha256"][root_name] == evidence[root_name]
        if end == 5000:
            assert review["continuation"] == "formal_training_complete_no_extension"
            assert review["resume_required"] is False

        boundary_records.append(dict(job_id=job, boundary=end, allowed_next=expected_next,

    # Additional engineering evidence is explicit, never inferred from job IDs
    # or a historical receipt. There is no new development run in this batch.

        assert ref(name)["sha256"] == evidence[name], name
    pull_records = {}
    for name in sorted(current_pulls):

        assert pull["weights_copied"] is False and pull["sealed_initial_states_copied"] is False

        selected_ids = {row["job_id"] for row in selected}
        requested_ids = set(pull["requested_job_ids"])
        assert selected_ids == requested_ids, name
        accounts = pull["accounting_after"]["jobs"]
        for row in selected:
            assert not row["active_overlapping_job_ids"], name
            assert accounts[row["job_id"]]["status"] in TERMINAL, (name, row["job_id"])
            if name in training_pulls:
                assert row["source_sha256"] == SOURCE_SHA and row["mode"] == "train"
            else:
                assert name in engineering_names
                assert row["mode"] not in ("train", "development", "final"), name
                assert isinstance(row["source_sha256"], str) and len(row["source_sha256"]) == 64

        assert len(file_names) == len(set(file_names)), name
        pull_records[name] = pull

    def check_existing_panels():
        rows = {}
        for name, pin in active_panels.items():
            rows[name] = ref(name, active_panels)
            assert rows[name]["sha256"] == pin, name
            assert not git("status", "--porcelain", "--", name).strip(), name
            assert sha(git("show", "HEAD:" + name)) == sha(git("show", ":" + name)) == pin, name

    panel_rows = check_existing_panels()
    source_version_rows = check_source_version_resolution()
    required = {row["path"]: row for row in source_version_rows}
    source_aliases_used = []

    def add_bound_file(name, want, size=None):
        safe_name(name, active_panels)
        resolved = resolve_evidence_name(name, want, size)
        row = ref(resolved, active_panels)
        assert row["sha256"] == want and (size is None or row["bytes"] == size), name
        if resolved != name:
            assert row == HISTORICAL_ARCHIVE
            source_aliases_used.append(dict(requested=HISTORICAL_SOURCE, delivered=row,
                                            resolution=ref(SOURCE_VERSION_NAME)))
        if resolved not in active_panels:
            assert resolved not in required or required[resolved] == row, resolved
            required[resolved] = row

    for name, want in evidence.items():

        if name in current_pulls:
            for entry in pull_records[name]["files"]:

    for name in ordinary:

        if not name.endswith(".md"):
            # An unreviewed engineering output must not enter via bulk add.
            assert name in required, ("unregistered_changed_nonweight_file", name)
            assert required[name] == ref(name, active_panels), name

    # All input checks above precede the first file/index mutation. These proof
    # files document preexisting bytes; they do not deliver a new state payload.

        record = dict(schema="formal_main_v2", kind="root_unchanged_preexisting_development_panel_index_check",

                      HEAD_index_worktree_equal=True, path_staged_or_modified=False, guard_implementation_changed=False,

                      final_states_opened=False)
        write_new(PANEL_PROOFS[name], record)
        required[PANEL_PROOFS[name]] = ref(PANEL_PROOFS[name])
    write_new(REQUIRED_NAME, dict(schema="cti_nonweight_archive_stage_v1", kind="required_files",
                                  files=[required[name] for name in sorted(required)]))

    def add(names, label, force=False):
        spec = ROOT / ".local" / ("stage_0626tail_" + label + ".nul")
        with spec.open("xb") as stream:
            stream.write(b"".join((":(literal)" + name).encode("utf-8") + b"\0" for name in sorted(names)))

                       ["--pathspec-from-file=" + str(spec), "--pathspec-file-nul"], cwd=ROOT, check=True)

    add(ordinary + [REQUIRED_NAME], "ordinary")
    add(set(required) - raw_names, "required", True)
    before, changes = git("ls-files", "--stage", "-z"), git("diff", "--cached", "--name-status", "-z")
    assert all(ref(name)["sha256"] == want for name, want in PINS.items())
    command = [sys.executable, "-B", "scripts/ops/check_nonweight_archive_stage.py", "--project-root", str(ROOT),

    for name in sorted(raw_names):
        command.extend(["--raw-path", name])
    print(json.dumps(dict(status="staged_guard_running", required_count=len(required))), flush=True)
    process = subprocess.run(command, cwd=ROOT, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    assert process.returncode == 0, process.stderr.decode("utf-8", errors="replace")
    report = json.loads(process.stdout)

    assert before == git("ls-files", "--stage", "-z") and changes == git("diff", "--cached", "--name-status", "-z")
    assert check_existing_panels() == panel_rows
    assert check_source_version_resolution() == source_version_rows
    # Preserve the independent raw unified diff byte for byte. Its four blank
    # context lines contain the required leading patch space, not prose edits.
    raw_diff_source = '.local/action_execution_compat_v2_independent_0626/v1_to_v2.diff'
    raw_diff_sha = '60fe689611e43953d201d9c3c27d37a613fc0cde40c4ffbb9e582564f520bbfa'
    raw_diff_archives = [
        BASE + 'preflight/action_execution_compat_v2_CPU_evidence_0626/035_v1_to_v2.diff.txt',
        batch['audit_source_archive_map'][raw_diff_source],
    ]
    assert len(set(raw_diff_archives)) == 2 and ref(raw_diff_source)['sha256'] == raw_diff_sha
    for name in raw_diff_archives:
        assert evidence[name] == ref(name)['sha256'] == raw_diff_sha
        assert git('show', ':' + name) == (ROOT / raw_diff_source).read_bytes()
    text_check = ['git', 'diff', '--cached', '--check', '--', '.']
    text_check.extend(':(exclude,literal)' + name for name in raw_diff_archives)
    subprocess.run(text_check, cwd=ROOT, check=True)
    proof = dict(schema="formal_main_v2", kind="root_git_index_nonweight_delivery", status="passed",

                 separately_verified_existing_development_dependencies=panel_proof_names,
                 existing_panels_HEAD_index_worktree_equal_before_and_after=True, final_state_values_opened=False,

                 new_engineering_evidence=engineering_names,
                 historical_source_resolution=ref(SOURCE_VERSION_NAME), historical_source_alias_uses=source_aliases_used,
                 canonical_current_source=CANONICAL_SOURCE, exact_original_archive=HISTORICAL_ARCHIVE,
                 historical_receipts_recursively_expanded=False, old_receipts_modified=False,
