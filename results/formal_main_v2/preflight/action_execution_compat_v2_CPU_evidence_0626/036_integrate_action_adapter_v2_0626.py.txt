SOURCE='.local/action_execution_compat_v2_candidate/adapter.py'

def include(name,digest=None):
    assert not Path(name).is_absolute() and ':' not in name and '..' not in Path(name).parts
    assert not any('sealed' in p.lower() or p.lower() in ('test','weights','checkpoints') for p in Path(name).parts)
    path=ROOT/name
    assert path.is_file() and not path.is_symlink()
    actual=f.sha(path); assert digest is None or actual==digest,name
    assert name not in evidence or evidence[name]==actual
    evidence[name]=actual

for name,digest in ((SOURCE,SOURCE_SHA),(AUTHOR,AUTHOR_SHA),(INDEPENDENT,INDEPENDENT_SHA)):
    include(name,digest)
author,independent=f.read(ROOT/AUTHOR),f.read(ROOT/INDEPENDENT)
