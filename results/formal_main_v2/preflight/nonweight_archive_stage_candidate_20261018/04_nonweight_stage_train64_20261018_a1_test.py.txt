"""Synthetic Git repositories only, always below the project's .local directory.

Most cases replace only the frozen inventory pin/count with a complete tiny
fixture. A fresh-process case uses all real 173 source bytes and production pins.
The codec, Git-index extraction, restore and byte checks are never substituted.
"""
from copy import deepcopy
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys

import pytest


ROOT = Path(__file__).resolve().parents[1]
SCRIPT = ROOT / "scripts/ops/check_nonweight_archive_stage.py"
spec = importlib.util.spec_from_file_location("stage_under_test", SCRIPT)
s = importlib.util.module_from_spec(spec); spec.loader.exec_module(s)
a = s.load_codec()
RAW = "results/synthetic/trials.jsonl"
RAW_NEW = "results/synthetic/history.jsonl"
SNAPSHOT = a.PREFIX + "/snapshots/synthetic_stage.json"
REQUEST = "results/synthetic/archive_request.json"
REQUIRED = "results/synthetic/required_files.json"
REPORT = "results/synthetic/root_review.json"
OPAQUE = "configs/manifests/synthetic/initial_states/sealed_final.json"


def git(root, *args):
    assert ROOT / ".local" in root.absolute().parents
    env = {k: v for k, v in os.environ.items() if not k.startswith("GIT_")}
    result = subprocess.run(["git", "-c", "user.name=Synthetic Test", "-c", "user.email=synthetic@example.invalid",
        "-c", "commit.gpgsign=false", "-c", "core.hooksPath=" + str(root / "empty-hooks"), "-C", str(root), *args],
        env=env, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    assert result.returncode == 0, result.stderr.decode("utf-8", "replace")
    return result.stdout


def put(root, name, data):
    path = root / name; path.parent.mkdir(parents=True, exist_ok=True); path.write_bytes(data)
    return {"path": name, "sha256": a.digest(data), "bytes": len(data)}


def add(root, *paths):
    git(root, "add", "--", *paths)


def make_project(root, monkeypatch=None, *, actual_frozen=False):
    assert ROOT / ".local" in root.absolute().parents
    root.mkdir(parents=True)
    git(root, "init", "--quiet")
    put(root, ".gitattributes", b"* -text\n")
    for path in (s.CODEC, s.SELF): put(root, path, (ROOT / path).read_bytes())
    if actual_frozen:
        data = (ROOT / s.INVENTORY).read_bytes()
        frozen = json.loads(data)["files"]
        for path in frozen: put(root, path, (ROOT / path).read_bytes())
    else:
        frozen = {}
        for path, data in [("src/frozen.py", b"# synthetic frozen code\r\nx = 1\r\n"), ("AGENTS.md", b"Synthetic rules\n")]:
            frozen[path] = put(root, path, data)["sha256"]
        data = a.encoded({"files": frozen, "synthetic": True})
        monkeypatch.setattr(s, "INVENTORY_SHA256", a.digest(data))
        monkeypatch.setattr(s, "FROZEN_COUNT", len(frozen))
    put(root, s.INVENTORY, data)
    put(root, RAW, b'{"synthetic":"old trial"}\r\nno-final-LF')
    # This is fake opaque data. The production checker may list its name/size,
    # but must never retrieve its content from Git or the working tree.
    put(root, OPAQUE, b"synthetic-opaque-do-not-read")
    add(root, ".gitattributes", s.CODEC, s.SELF, s.INVENTORY, RAW, OPAQUE, *frozen)
    git(root, "commit", "--quiet", "-m", "synthetic baseline")
    raw_refs = [put(root, RAW, (root / RAW).read_bytes()), put(root, RAW_NEW, b"\x00\xffsynthetic history\n")]
    review = put(root, REPORT, b'{"synthetic":true,"status":"not_a_scientific_proof"}\n')
    request = {"schema": a.SCHEMA, "kind": "archive_request", "files": raw_refs, "provenance": [review]}
    put(root, REQUEST, a.encoded(request))
    packed = a.pack(root, REQUEST, SNAPSHOT)
    required = {"schema": s.SCHEMA, "kind": "required_files", "files": raw_refs + [review]}
    put(root, REQUIRED, a.encoded(required))
    git(root, "rm", "--cached", "--quiet", "--", RAW)
    archive = [p.relative_to(root).as_posix() for p in (root / a.PREFIX).rglob("*")
               if p.is_file() and p.name != ".writer.lock"]
    add(root, REPORT, REQUEST, REQUIRED, *archive)
    return {"root": root, "snapshot_sha": packed["snapshot_sha256"], "raw": [RAW, RAW_NEW],
            "frozen": frozen, "required": required}


@pytest.fixture
def project(tmp_path, monkeypatch):
    return make_project(tmp_path / "repo", monkeypatch)


def run(case, **kwargs):
    args = dict(project=case["root"], snapshot=SNAPSHOT, snapshot_sha256=case["snapshot_sha"],
                raw_paths=case["raw"], required_files=REQUIRED)
    args.update(kwargs)
    return s.check(**args)


def chunk_path(root):
    value = json.loads((root / SNAPSHOT).read_bytes())
    obj = json.loads((root / value["files"][0]["manifest"]["path"]).read_bytes())
    return obj["chunks"][0]["path"]


def change_required(case, edit):
    value = deepcopy(case["required"]); edit(value)
    put(case["root"], REQUIRED, a.encoded(value)); add(case["root"], REQUIRED)


def test_complete_index_restore_matches_retained_raw_and_index_never_changes(project, monkeypatch):
    root = project["root"]
    before_index = (root / ".git/index").read_bytes()
    before_raw = {p: (root / p).read_bytes() for p in project["raw"]}
    real_blob = s.Index.blob
    def guard(index, path):
        assert path != OPAQUE, "State-value content must not be loaded"
        return real_blob(index, path)
    monkeypatch.setattr(s.Index, "blob", guard)
    result = run(project)
    assert result["status"] == "verified_index_archive_delivery"
    assert result["allowed_raw_deletions"] == [RAW]
    assert result["frozen_files_verified_in_worktree_and_index"] == 2
    clone = root / result["scratch_clone"]
    for path, raw in before_raw.items():
        assert (root / path).read_bytes() == (clone / path).read_bytes() == raw
        assert git(root, "ls-files", "--", path) == b""
    assert (root / ".git/index").read_bytes() == before_index
    assert result["git_write_commands"] is result["scientific_completion_inferred"] is False


def test_real_173_frozen_bytes_fresh_process_cli(tmp_path):
    case = make_project(tmp_path / "real_source_copy", actual_frozen=True)
    root = case["root"]; before = (root / ".git/index").read_bytes()
    cmd = [sys.executable, "-B", str(SCRIPT), "--project-root", str(root), "--snapshot", SNAPSHOT,
           "--snapshot-sha256", case["snapshot_sha"], "--required-files", REQUIRED]
    for path in case["raw"]: cmd += ["--raw-path", path]
    result = subprocess.run(cmd, capture_output=True, text=True)
    assert result.returncode == 0, result.stderr
    value = json.loads(result.stdout)
    assert value["frozen_files_verified_in_worktree_and_index"] == 173
    assert value["frozen_inventory_sha256"] == s.INVENTORY_SHA256
    assert value["codec_sha256"] == s.CODEC_SHA256
    assert (root / ".git/index").read_bytes() == before

@pytest.mark.parametrize("target", ["snapshot", "request", "provenance", "manifest", "chunk"])
def test_worktree_cannot_supply_missing_index_archive_member(project, target):
    root = project["root"]; snap = json.loads((root / SNAPSHOT).read_bytes())
    names = {"snapshot": SNAPSHOT, "request": REQUEST, "provenance": REPORT,
             "manifest": snap["files"][0]["manifest"]["path"], "chunk": chunk_path(root)}
    path = names[target]; expected = (root / path).read_bytes()
    git(root, "rm", "--cached", "--quiet", "--", path)
    with pytest.raises(ValueError, match="Missing required index blob"): run(project)
    assert (root / path).read_bytes() == expected


@pytest.mark.parametrize("matching_worktree", [True, False])
def test_corrupt_staged_chunk_and_worktree_drift_rejected(project, matching_worktree):
    root = project["root"]; path = chunk_path(root); original = (root / path).read_bytes()
    put(root, path, b"synthetic corrupt blob"); add(root, path)
    if not matching_worktree: put(root, path, original)
    with pytest.raises(ValueError, match="differs from index|SHA/bytes differ"): run(project)


@pytest.mark.parametrize("mode", ["still_indexed", "extra_whitelist", "missing_whitelist", "duplicate", "markdown"])
def test_exact_raw_whitelist_and_no_duplicate_index_blob(project, mode):
    raw = list(project["raw"])
    if mode == "still_indexed": add(project["root"], RAW)
    elif mode == "extra_whitelist": raw.append("results/synthetic/not_in_snapshot.jsonl")
    elif mode == "missing_whitelist": raw.pop()

    elif mode == "markdown": raw.append("results/synthetic/report.md")

@pytest.mark.parametrize("required", [False, True])
def test_markdown_legacy_exception_does_not_override_explicit_delivery_sha(project, required):
    root = project["root"]; name = "results/synthetic/report.md"
    ref = put(root, name, b"same text\n"); add(root, name)
    put(root, name, b"same text\r\n")
    if required:
        change_required(project, lambda value: value["files"].append(ref))
        with pytest.raises(ValueError, match="differs from index"): run(project)
    else:
        assert run(project)["markdown_legacy_byte_exceptions"] == [name]


@pytest.mark.parametrize("where", ["worktree", "index_and_worktree", "inventory", "codec"])
def test_frozen_and_codec_pins_cannot_be_rebound(project, where):
    root = project["root"]
    if where in ("worktree", "index_and_worktree"):
        name = next(iter(project["frozen"]))
        put(root, name, b"changed frozen bytes")

def test_over_limit_index_size_rejected_before_payload_read(project, monkeypatch):
    # Bound this test's storage: fake only the Git size reply for one real object;
    # command protocol and all other actual index/blob checks remain unchanged.
    wanted = git(project["root"], "rev-parse", ":" + REPORT).strip().decode()
    original = s.Index.git
    def reply(index, *args, **kwargs):
        data = original(index, *args, **kwargs)
        if args[:1] == ("cat-file",) and "--batch-check=" in args[1]:
            rows = data.decode().splitlines()
            data = ("\n".join(wanted + " blob " + str(s.LIMIT + 1) if row.startswith(wanted + " ") else row for row in rows) + "\n").encode()
        return data
    monkeypatch.setattr(s.Index, "git", reply)
    monkeypatch.setattr(s.Index, "blob", lambda *args: pytest.fail("Oversize should fail before payload read"))
    with pytest.raises(ValueError, match="over-100MiB"): run(project)


@pytest.mark.parametrize("mode,stage", [("120000", "0"), ("160000", "0"), ("100644", "1")])
def test_unsafe_index_mode_and_unmerged_stage(project, monkeypatch, mode, stage):
    original = s.Index.git
    def reply(index, *args, **kwargs):
        data = original(index, *args, **kwargs)
        if args[:1] == ("ls-files",):
            rows = data.split(b"\0")
            head, name = rows[0].split(b"\t", 1); _, oid, _ = head.split()
            rows[0] = mode.encode() + b" " + oid + b" " + stage.encode() + b"\t" + name
            data = b"\0".join(rows)
        return data
    monkeypatch.setattr(s.Index, "git", reply)
    with pytest.raises(ValueError, match="Unmerged/symlink/submodule"): run(project)


def test_index_changed_during_restore_is_rejected(project, monkeypatch):
    real_load = s.load_codec
    def load():
        codec = real_load(); original = codec.restore
        def restore(*args, **kwargs):
            result = original(*args, **kwargs)
            put(project["root"], "results/synthetic/racing.json", b"synthetic index race")
            add(project["root"], "results/synthetic/racing.json")
            return result
        codec.restore = restore
        return codec
    monkeypatch.setattr(s, "load_codec", load)
    with pytest.raises(ValueError, match="index or HEAD changed"): run(project)


@pytest.mark.parametrize("where", ["raw", "required"])
def test_forbidden_delivery_rejected_before_state_payload_read(project, monkeypatch, where):
    forbidden = "results/synthetic/init_states_test.json" + ("l" if where == "raw" else "")
    if where == "raw":
        monkeypatch.setattr(s, "Index", lambda *args: pytest.fail("Invalid raw whitelist precedes any Git read"))
        with pytest.raises(ValueError): run(project, raw_paths=[forbidden])
    else:
        change_required(project, lambda value: value["files"].append({"path": forbidden, "sha256": "0" * 64, "bytes": 4}))
        real = s.Index.blob
        def guard(index, name):
            assert name != forbidden
            return real(index, name)
        monkeypatch.setattr(s.Index, "blob", guard)
        with pytest.raises(ValueError, match="State-value"): run(project)
