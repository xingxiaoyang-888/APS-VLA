def digest(data):
    return hashlib.sha256(data).hexdigest()


def read(relative, expected=None):
    path = ROOT / relative
    assert path.resolve().is_relative_to(ROOT)
    assert not any(s in relative.lower() for s in ('sealed_final', 'final_registry', '.pruned_init'))
    assert path.suffix.lower() not in {'.pt', '.pth', '.bin', '.ckpt', '.safetensors', '.init'}
    data = path.read_bytes()
    checksum = digest(data)
    if expected is not None:
        assert checksum == expected, relative
    if relative in READ:
        assert READ[relative] == checksum, relative
    READ[relative] = checksum
    return data


def jread(relative, expected=None):
    return json.loads(read(relative, expected))
