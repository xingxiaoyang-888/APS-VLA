def load(path, name):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module

def observation(i):
    return {"agentview_image": np.full((2, 2, 3), i, dtype=np.uint8),
            "robot0_eye_in_hand_image": np.full((2, 2, 3), i + 1, dtype=np.uint8),

def prefix():

def checkpoints():
    return [{"control_step": i, "observation_sha256": obs_sha(observation(i))} for i in range(0, 56, 8)]


class Env:
    def __init__(self, done_at=None):
        self.actions, self.states, self.done_at = [], [], done_at
        self.reset_count = 0

    def reset(self):
        self.reset_count += 1

    def set_init_state(self, value):
        self.states.append(value.copy())
        return observation(0)

    def step(self, action):
        self.actions.append(list(action))
        return observation(max(0, len(self.actions) - 10)), 0., len(self.actions) == self.done_at, {}

class Tensor:
    def __init__(self, values): self.values = list(values)
    def clone(self): return Tensor(self.values)
    def tolist(self): return list(self.values)


class RNG:
    def __init__(self):
        self.cpu, self.gpu = Tensor([11, 12]), [Tensor([21, 22])]
        self.cuda = SimpleNamespace(get_rng_state_all=lambda: [t.clone() for t in self.gpu],
                                    set_rng_state_all=lambda rows: setattr(self, "gpu", [t.clone() for t in rows]))
    def get_rng_state(self): return self.cpu.clone()
    def set_rng_state(self, value): self.cpu = value.clone()


@pytest.mark.parametrize("failure", [None, "extra_forward", "mutate_input", "exception", "nonfinite"])
def test_two_captures_same_input_rng_and_exception_cleanup(journal, failure):
    rng, seen = RNG(), []
    before = m.rng_state(rng)
    raw = observation(56)
    def capture(name, current, instruction):
        seen.append((name, obs_sha(current), instruction, random.random(), float(np.random.random()),
                     rng.cpu.tolist(), rng.gpu[0].tolist()))
        rng.cpu.values[0] += 1
        rng.gpu[0].values[0] += 1
        if failure == "exception": raise RuntimeError("synthetic capture exception")
        if failure == "mutate_input": current["robot0_eef_pos"][0] += 1
        array = np.zeros((8, 7), dtype=np.float32)
        array[0, 0] = np.inf if failure == "nonfinite" else 1.1
        return dict(normalized=array, physical=array.copy(), postprocessed=array.copy(),
                    calls={"vla_backbone": 2 if failure == "extra_forward" else 1, "vision": 1},
                    metadata={"synthetic": True})
    args = dict(snapshot_rng=lambda: m.rng_state(rng), reset_rng=lambda s: m.restore_rng(rng, s),
                hash_rng=m.rng_digest, observation_hash=obs_sha, journal=journal)
    if failure:
        with pytest.raises((RuntimeError, ValueError)): m.capture_pair(raw, "fixed instruction", capture, **args)
        assert not any(r["event"] == "capture_complete" for r in journal.events)
    else:
        result = m.capture_pair(raw, "fixed instruction", capture, **args)
        assert [r[0] for r in seen] == ["official_native_dense", "frozen_token_dense"]
        assert seen[0][1:] == seen[1][1:]
        assert result["total_VLA_forwards"] == 2 and result["env_steps_after_t56"] == 0
        assert result["success_collected"] is False and result["original_failed_values_available"] is False
        assert all(c["frozen_guard_rejects"] for c in result["captures"].values())
    assert m.rng_digest(m.rng_state(rng)) == m.rng_digest(before)
    assert obs_sha(raw) == obs_sha(observation(56))


class HookModule:
    def __init__(self): self.hooks, self.executed = [], 0
    def register_forward_pre_hook(self, fn):
        self.hooks.append(fn)
        return SimpleNamespace(remove=lambda: self.hooks.remove(fn))
    def forward(self):
        for hook in self.hooks: hook(self, ())
        self.executed += 1


def test_actual_pre_hooks_block_second_kernel_and_remove_handles():
    llm, vision, head = HookModule(), HookModule(), HookModule()
    component = SimpleNamespace(vla=SimpleNamespace(language_model=SimpleNamespace(model=llm), vision_backbone=vision),
                                action_head=SimpleNamespace(model=head))
    with pytest.raises(RuntimeError, match="second forward"):
        with m.forward_counters(component) as calls:
            llm.forward(); vision.forward(); head.forward(); head.forward()
            assert calls == dict(vla_backbone=1, vision=1, action_head=2)
            llm.forward()
    assert llm.executed == 1 and head.executed == 2
    assert not llm.hooks and not vision.hooks and not head.hooks


class Controller:
    def __init__(self, control_delta=True):
        self.use_delta = control_delta
        self.input_min, self.input_max = np.full(6, -1.), np.full(6, 1.)
        self.output_min, self.output_max = np.full(6, -.1), np.full(6, .1)
    def scale_action(self, action): return np.clip(action, self.input_min, self.input_max)
    def set_goal(self, action):
        if self.use_delta: return self.scale_action(action)
        return action

@pytest.mark.parametrize("mode", ["success", "missing_wrapper", "wrong_watch", "existing_output", "timeout", "child_failure"])
def test_launcher_exact_contract_and_one_child_only(tmp_path, monkeypatch, mode):
    module, inv, path, out = launcher_case(tmp_path, monkeypatch)
    if mode == "missing_wrapper": inv["watched_files"].pop(str(out / "source/probe.slurm"))
    if mode == "wrong_watch": inv["watched_files"][str(out / "source/probe.slurm")] = "0" * 64; inv["wrapper_sha256"] = "0" * 64

    monkeypatch.setattr(sys, "argv", [str(LAUNCHER), "--invocation", str(path), "--invocation-sha256", module.sha(path)])
    calls = []

        assert 0 < kwargs["timeout"] <= 890
        assert kwargs["cwd"] == module.S
        assert command.count("--execute-gpu") == 1 and command[1] == "-B"
        if mode == "timeout": raise module.subprocess.TimeoutExpired(command, kwargs["timeout"])
        return SimpleNamespace(returncode=1 if mode == "child_failure" else 0)
    monkeypatch.setattr(module.subprocess, "run", child)
    if mode in ("missing_wrapper", "existing_output"):
        with pytest.raises(ValueError): module.main()
        assert not calls
        assert not (module.R / "job_receipts/99992.json").exists()
    else:
        code = module.main()
        receipt = json.loads((module.R / "job_receipts/99992.json").read_bytes())
        assert receipt["new_trials"] == 0
        assert len(calls) == (0 if mode == "wrong_watch" else 1)

def test_gpu_body_no_env_step_or_training_and_guard_scope_no_panel_reopen():

    assert all(line > hf_update for line, module in imports if module == "transformers")

    monkeypatch.setattr(sys, "argv", ["synthetic_submit", "--root-review", "synthetic_root_review.json"])
    return module, state, actions


@pytest.mark.parametrize("mode", ["capacity", "pending", "duplicate", "prior_output", "remote_mismatch", "unknown_submission", "seven"])
def test_submit_capacity_duplicate_and_unknown_outcome(tmp_path, monkeypatch, mode):
    module, state, actions = submit_case(tmp_path, monkeypatch, mode)
    if mode in {"pending", "duplicate", "prior_output", "remote_mismatch"}:
        with pytest.raises(AssertionError): module.main()
        assert not any(isinstance(x, tuple) for x in actions)
    elif mode == "unknown_submission":
        with pytest.raises(RuntimeError, match="pending preserved"): module.main()
        assert state["pending_submission"] is not None
        assert len(state["registered_jobs"]) == 7
        assert len([x for x in actions if isinstance(x, tuple) and x[0] == "synthetic_transport" and x[1][0] == "SYNTHETIC_SSH_NEVER_EXECUTE"]) == 1
    else:
        module.main()
        if mode == "capacity":
            assert actions == ["refresh"] and len(state["registered_jobs"]) == 8
        else:
            assert len(state["registered_jobs"]) == 8 and state["pending_submission"] is None
            assert state["registered_jobs"][-1]["job_id"] == "99993"
            assert state["formal_main_progress"]["parallel_execution"]["registered_requested_active_GPU_jobs"] == 8


def test_submit_all_unknown_and_transition_states_reserve_slot(tmp_path, monkeypatch):
    module, _, _ = submit_case(tmp_path, monkeypatch, "capacity")
    for status in ["", "UNKNOWN", "STAGE_OUT", "RUNNING+", "PENDING", "SUBMITTED", "SUSPENDED", "COMPLETING"]:
        assert module.reserves_slot({"status": status})
    for status in ["COMPLETED", "CANCELLED by 15016", "FAILED+", "TIMEOUT", "OUT_OF_MEMORY"]:
        assert not module.reserves_slot({"status": status})
