"""Exercise existing CLI entrypoints only in the isolated synthetic CPU mirror."""
from pathlib import Path
import hashlib
import json
import subprocess
import sys
import time

HERE = Path(__file__).resolve().parent
ROOT = HERE.parent.parent
MIRROR = HERE / 'pytest_a1/synthetic_v3_CPU_only0'
STAGE = 'formal_main_v2_action_execution_compat_gpu_v3'
SOURCE = 'results/formal_main_v2/engineering/' + STAGE + '/source/'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    assert MIRROR.is_dir() and ROOT / '.local' in MIRROR.parents
    regpath = 'configs/experiments/' + STAGE + '.json'
    reg = json.loads((MIRROR / regpath).read_bytes())
    assert reg['synthetic_fixture_only'] is True and reg['real_submission_authorized_by_fixture'] is False
    manifest = reg['candidate_manifest']
    root = reg['cpu_reviews']['root_acceptance']
    commands = {
        'inspect_cpu': [sys.executable, '-B', str(HERE / 'probe.py'), '--project-root', str(MIRROR),
            '--source-root', str(MIRROR), '--manifest', manifest['path'], '--manifest-sha256', manifest['sha256'], '--inspect-cpu'],
        'check_only': [sys.executable, '-B', str(HERE / 'submit.py'), '--project-root', str(MIRROR),
            '--root-review', root['path'], '--root-review-sha256', root['sha256'],
            '--manifest', manifest['path'], '--manifest-sha256', manifest['sha256'],
            '--gpu-registration', regpath, '--gpu-registration-sha256', sha(MIRROR / regpath), '--check-only'],
    }
    rows = []
    for name, command in commands.items():
        start = time.monotonic()
        run = subprocess.run(command, cwd=ROOT, capture_output=True)
        seconds = time.monotonic() - start
        for suffix, data in (('stdout.json', run.stdout), ('stderr.log', run.stderr)):
            with (HERE / (name + '.' + suffix)).open('xb') as stream:
                stream.write(data)
        rows.append(dict(name=name,command=command,exit_code=run.returncode,wall_seconds=seconds,
            stdout_sha256=hashlib.sha256(run.stdout).hexdigest(),stderr_sha256=hashlib.sha256(run.stderr).hexdigest()))
    receipt = dict(schema='formal_main_v2',kind='isolated_synthetic_v3_actual_CLI_check',
        synthetic_typed_approval_objects=True, project_inputs_are_actual_bytes=True,
        real_project_registration_created=False, transport_executed=False, GPU_seconds=0, runs=rows)
    with (HERE / 'local_cli_execution.json').open('x',encoding='utf-8',newline='\n') as stream:
        json.dump(receipt,stream,indent=2);stream.write('\n')
    print(json.dumps(receipt))
    assert all(row['exit_code']==0 for row in rows)
    assert json.loads((HERE/'check_only.stdout.json').read_bytes())['GPU_submitted'] is False


if __name__ == '__main__':
    main()
