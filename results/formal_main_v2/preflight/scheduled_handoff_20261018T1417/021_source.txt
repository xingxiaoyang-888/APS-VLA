def sha(data):
    return hashlib.sha256(data).hexdigest()


def raw(path):
    return (ROOT / path).read_bytes()


def read(path):
    return json.loads(raw(path))


def write(name, data):
    path = HERE / name
    assert path.parent.resolve() == HERE and not path.exists(), name
    with path.open('xb') as stream:
        stream.write(data)


def output_json(name, value):
    write(name, (json.dumps(value, indent=2, ensure_ascii=False, allow_nan=False) + '\n').encode())


def reference(path, data=None):
    data = raw(path) if data is None else data
    return dict(path=path, sha256=sha(data), bytes=len(data))


def replace(text, old, new, count=None):
    if count is not None:
        assert text.count(old) == count, (old, text.count(old), count)
    else:
        assert old in text, old
    return text.replace(old, new)
