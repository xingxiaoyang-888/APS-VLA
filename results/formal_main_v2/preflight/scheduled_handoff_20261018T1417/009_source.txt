"""Freeze only this unsubmitted candidate's author CPU evidence."""
from datetime import datetime, timezone
from pathlib import Path
import hashlib
import json
import xml.etree.ElementTree as ET

HERE=Path(__file__).resolve().parent
ROOT=HERE.parent.parent


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def read(path):
    return json.loads(path.read_bytes())


def main():
    before=read(HERE/'baseline_sha256.json')
    for rel,digest in before.items(): assert sha(ROOT/rel)==digest,rel
    run=read(HERE/'pytest_a1_execution.json')
    assert run['exit_code']==0 and run['source_sha256_before']==run['source_sha256_after']
    suite=ET.parse(HERE/'pytest_a1.xml').getroot().find('testsuite')
    assert int(suite.attrib['tests'])==22
    assert all(int(suite.attrib[k])==0 for k in ('failures','errors','skipped'))
    cli=read(HERE/'local_cli_execution.json')
    assert all(r['exit_code']==0 for r in cli['runs'])
    evidence=dict(before)
    for name in ('AGENTS.md','.local/integrate_action_gpu_v2_0840.py','.local/submit_action_gpu_v2_excluded_0840.py'):
        evidence[name]=sha(ROOT/name)
    for file in HERE.iterdir():
        if file.is_file(): evidence[file.relative_to(ROOT).as_posix()]=sha(file)
    mirror=HERE/'pytest_a1/synthetic_v3_CPU_only0'
    for rel in ('configs/experiments/formal_main_v2_action_execution_compat_gpu_v3.json',
                'results/formal_main_v2/preflight/action_execution_gpu_v3_candidate_root_0840.json',
                'results/formal_main_v2/preflight/action_execution_gpu_v3_candidate_independent_0840.json',
                'configs/experiments/synthetic_node_policy.json','.local/synthetic_node_event.json',
                'results/experiment_supervisor/state.json'):
        file=mirror/rel;evidence[file.relative_to(ROOT).as_posix()]=sha(file)
    rows=read(HERE/'publication_plan.json')['files']
    for row in rows:
        file=ROOT/row['source']
        assert sha(file)==row['sha256'] and file.stat().st_size==row['bytes']
    value=dict(schema='formal_main_v2',version='action-execution-gpu-v3-preparation-author-0840-v1',
        kind='unsubmitted_GPU_v3_constructor_identity_repair_candidate_CPU_review',
        status='passed_author_CPU_candidate_pending_independent_root_registration_and_GPU',
        reviewed_utc=datetime.now(timezone.utc).isoformat(),preparer='train64',
        stage_key='formal_main_v2_action_execution_compat_gpu_v3',candidate_files=rows,
        tests=dict(command=run['command'],passed=22,failed=0,skipped=0,
            junit_seconds=float(suite.attrib['time']),wall_seconds=run['wall_seconds'],
            XML='.local/action_execution_gpu_v3_0840/pytest_a1.xml',
            log='.local/action_execution_gpu_v3_0840/pytest_a1.log'),
        CLI_runs=cli['runs'],CLI_approval_scope='Only explicitly synthetic typed receipts inside isolated .local mirror',
        implementation_scope=dict(
            actual_source173_and_four_development_panels=True,
            actual_per_suite_training_selected_checkpoint_metadata=True,
            weights_read=False,final_states_read=False,model_loaded=False,environment_created=False,
            GPU_runtime_executed=False,SSH_executed=False,Slurm_submitted=False,
            real_state_written=False,Git_written=False,
            synthetic_objects=['candidate/root typed approval receipts','node registration and pending state'],
            transport_test='check_output stand-in only captures the long stdin payload/short argv; no command execution',
            real_GPU_methods_not_retested=True),
        equivalence=dict(
            fixed16_scientific_manifest_subtrees_identical=True,
            probe_GPU_body_and_cell_loop_AST_identical=True,
            submit_py_byte_identical_including_stdin_transport=True,
            launcher_only_typed_CPU_metadata_gate_and_namespace_pins_changed=True,
            node_exclusion_only_exact_versioned_capability_changed=True,
            adapter_exact_independently_reviewed_b8a163_bytes=True,
            baseline_173_old_adapter_v2_and_failure_bytes_unchanged=True),
        historical_failed_GPU_job=dict(job_id='3030159',status='FAILED',GPU_seconds=85,
            retained=True,rerun=False,records_rewritten=False),
        new_GPU_seconds=0,production_published=False,root_registration_created=False,
        candidate_independent_review_pending=True,actual_GPU_acceptance_pending=True,
        source173_unchanged=True,read_evidence_sha256=evidence,evidence_sha256=evidence,
        next_steps_document='.local/action_execution_gpu_v3_0840/README.md',
        limitations=['Synthetic check-only success is not real-project root authorization.',
            'Node v3 policy must receive its own root approval before the scoped submission call.',
            'The existing 85-second v2 failure stays counted; no automatic retry or science release.'])
    for rel,digest in evidence.items(): assert sha(ROOT/rel)==digest,rel
    path=HERE/'candidate_review.json'
    with path.open('x',encoding='utf-8',newline='\n') as out:
        json.dump(value,out,ensure_ascii=False,indent=2);out.write('\n')
    print(json.dumps(dict(path=path.relative_to(ROOT).as_posix(),sha256=sha(path),evidence=len(evidence),
                         tests=22,actual_CLI_checks=2,GPU_seconds=0)))


if __name__=='__main__': main()
