"""Independent, local CPU-only audit of the fixed GPU v3 preparation."""
from __future__ import annotations
import ast
import difflib
import hashlib
import importlib.util
import json

import subprocess
import sys

import xml.etree.ElementTree as ET
from datetime import datetime, timezone

ROOT=Path(__file__).resolve().parents[2]
OUT=Path(__file__).resolve().parent
HERE=ROOT/'.local/action_execution_gpu_v3_0840'
OLD=ROOT/'results/formal_main_v2/engineering/formal_main_v2_action_execution_compat_gpu_v2/source'
STAGE='formal_main_v2_action_execution_compat_gpu_v3'
AUTHOR_SHA='6d1014cdf837e9f391a08a5ee8a6bc43ed0a6b4c8ca8477f8a98c768c67cbf8a'
sys.dont_write_bytecode=True

def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def read(path):return json.loads(Path(path).read_bytes())
def write(path,value):
    with Path(path).open('x',encoding='utf8',newline='\n') as f:
        json.dump(value,f,ensure_ascii=False,indent=2,sort_keys=True);f.write('\n')
def raw(path,data):
    with Path(path).open('xb') as f:f.write(data)
def rel(path):return Path(path).relative_to(ROOT).as_posix()
def nodes(path):
    return {n.name:ast.dump(n,include_attributes=False) for n in ast.parse(Path(path).read_bytes()).body
        if isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef,ast.ClassDef))}
def safe(ref):
    p=Path(ref);low=ref.lower().replace('\\','/')
    assert not p.is_absolute() and '..' not in p.parts
    assert 'sealed' not in low and '/weights/' not in low and '/checkpoints/' not in low
    assert p.suffix.lower() not in ('.pt','.pth','.bin','.safetensors')
    if '/initial_states/' in low:assert low in {
        'configs/manifests/formal_main_v2/'+s+'/initial_states/development.json'
        for s in ('libero_10','libero_spatial','libero_object','libero_goal')}
    return ROOT/p
def run(label,command):
    start=time.perf_counter()
    proc=subprocess.run(command,cwd=ROOT,capture_output=True,timeout=180)
    raw(OUT/(label+'.stdout.log'),proc.stdout);raw(OUT/(label+'.stderr.log'),proc.stderr)
    item=dict(command=command,exit_code=proc.returncode,wall_seconds=time.perf_counter()-start,
        stdout=rel(OUT/(label+'.stdout.log')),stdout_sha256=sha(OUT/(label+'.stdout.log')),
        stderr=rel(OUT/(label+'.stderr.log')),stderr_sha256=sha(OUT/(label+'.stderr.log')))
    write(OUT/(label+'.execution.json'),item)
    assert proc.returncode==0,item
    return item,proc.stdout

def main():
    start=time.perf_counter()
    assert sha(HERE/'candidate_review.json')==AUTHOR_SHA
    author=read(HERE/'candidate_review.json')
    evidence=dict(author['evidence_sha256'])
    evidence[rel(HERE/'candidate_review.json')]=AUTHOR_SHA
    publication=read(HERE/'publication_plan.json')
    assert len(publication['files'])==8 and publication['no_publication_performed'] is True
    for row in publication['files']:
        p=safe(row['source']);assert sha(p)==row['sha256'] and p.stat().st_size==row['bytes']
        evidence[row['source']]=row['sha256']
    evidence[rel(__file__)]=sha(__file__)
    for ref,h in evidence.items():assert sha(safe(ref))==h,('before',ref)
    before=dict(evidence);write(OUT/'inputs_before.json',before)
    inventory=read(ROOT/'results/formal_main_v2/source/formal_main_v2_run_564be0995e.json')
    assert len(inventory['files'])==173
    for ref,h in inventory['files'].items():assert before[ref]==h
    differences={}
    for name,expected in [('probe.py',{'prepare','validate_cpu_reviews'}),('launcher.py',{'check_cpu_review'})]:
        a,b=nodes(OLD/name),nodes(HERE/name)
        assert a.keys()==b.keys() and {k for k in a if a[k]!=b[k]}==expected
        differences[name]=sorted(expected)
    assert (OLD/'submit.py').read_bytes()==(HERE/'submit.py').read_bytes()
    assert (HERE/'probe.slurm').read_bytes()==(OLD/'probe.slurm').read_bytes().replace(
        b'formal_main_v2_action_execution_compat_gpu_v2',STAGE.encode())
    old_manifest,new_manifest=read(OLD/'candidate_manifest.json'),read(HERE/'candidate_manifest.json')
    assert {k for k in old_manifest if old_manifest[k]!=new_manifest[k]}=={
        'version','action_adapter','action_adapter_independent_review','read_evidence_sha256'}
    assert set(new_manifest)-set(old_manifest)=={'constructor_server_cpu','constructor_identity_repair'}
    assert new_manifest['suites']==old_manifest['suites'] and new_manifest['limits']==old_manifest['limits']
    assert new_manifest['limits']['cells']==new_manifest['limits']['max_vla_calls']==16
    oldmap,newmap=old_manifest['read_evidence_sha256'],new_manifest['read_evidence_sha256']
    assert all(newmap[k]==v for k,v in oldmap.items()) and len(set(newmap)-set(oldmap))==3
    old_node=(ROOT/'scripts/ops/formal_scheduler_node_exclusion.py').read_bytes()
    expected_node=old_node.replace(b'root-ssh-node-exclusion-v1',b'root-ssh-node-exclusion-v2-action-gpu-v3')
    for a,b in ((b'formal_main_v2_action_execution_compat_gpu_v2',STAGE.encode()),
        (b'action_execution_gpu_v2',b'action_execution_gpu_v3'),(b'gpu_v2',b'gpu_v3'),
        (sha(OLD/'probe.slurm').encode(),sha(HERE/'probe.slurm').encode()),
        (b'engineering v2 wrapper',b'engineering v3 wrapper'),(b'v2 invocation identity differs',b'v3 invocation identity differs')):
        expected_node=expected_node.replace(a,b)
    assert (HERE/'node_exclusion.py').read_bytes()==expected_node
    assert (HERE/'action_adapter.py').read_bytes()==(ROOT/'.local/action_constructor_server311_0840/adapter.py').read_bytes()
    assert (HERE/'constructor_independent_cpu.json').read_bytes()==(ROOT/'.local/action_constructor_server311_independent_0840/independent_review.json').read_bytes()
    prior=read(HERE/'prior_failure.json');assert prior['status']=='FAILED' and prior['GPU_seconds']==85
    for ref,h in prior['files_sha256'].items():assert sha(safe(ref))==h

    command=[sys.executable,'-B','-m','pytest','-p','no:cacheprovider',rel(HERE/'test_candidate.py'),
        '--basetemp='+rel(OUT/'pytest_a1'),'--junitxml='+rel(OUT/'tests.xml'),'-q']
    test_execution,_=run('tests',command)
    xml=ET.parse(OUT/'tests.xml').getroot();suites=list(xml.iter('testsuite'))
    counts={key:sum(int(x.attrib.get(key,0)) for x in suites) for key in ('tests','failures','errors','skipped')}
    assert counts==dict(tests=22,failures=0,errors=0,skipped=0),counts
    mirror=OUT/'pytest_a1/synthetic_v3_CPU_only0'
    reg_rel='configs/experiments/'+STAGE+'.json'
    reg=read(mirror/reg_rel);assert reg['synthetic_fixture_only'] is True and reg['real_submission_authorized_by_fixture'] is False
    manifest_ref=reg['candidate_manifest'];root_ref=reg['cpu_reviews']['root_acceptance']
    inspect_execution,inspect_stdout=run('inspect_cpu',[sys.executable,'-B',str(HERE/'probe.py'),
        '--project-root',str(mirror),'--source-root',str(mirror),'--manifest',manifest_ref['path'],
        '--manifest-sha256',manifest_ref['sha256'],'--inspect-cpu'])
    check_execution,check_stdout=run('check_only',[sys.executable,'-B',str(HERE/'submit.py'),
        '--project-root',str(mirror),'--root-review',root_ref['path'],'--root-review-sha256',root_ref['sha256'],
        '--manifest',manifest_ref['path'],'--manifest-sha256',manifest_ref['sha256'],
        '--gpu-registration',reg_rel,'--gpu-registration-sha256',sha(mirror/reg_rel),'--check-only'])
    checked=json.loads(check_stdout)
    assert checked['watched_files']==287 and checked['source_files']==173 and checked['engineering_cells']==16
    assert checked['scientific_trials']==0 and checked['remote_checked'] is False and checked['GPU_submitted'] is False
    spec=importlib.util.spec_from_file_location('independent_v3_launcher',HERE/'launcher.py')
    c=importlib.util.module_from_spec(spec);spec.loader.exec_module(c)
    closure=c.check_contract(mirror,root_ref={k:root_ref[k] for k in ('path','sha256')},manifest_ref=manifest_ref,
        registration_ref={'path':reg_rel,'sha256':sha(mirror/reg_rel)})
    rows=closure['deployment_files'];assert len(rows)==len({r['server'] for r in rows})==287
    assert sum(r['server'].startswith(c.S+'/') for r in rows)==173
    assert sum(r['server'].startswith(c.P+'/') and not r['server'].startswith(c.S+'/') for r in rows)==114
    write(OUT/'synthetic_checked_deployment_closure.json',dict(synthetic_typed_approvals=True,files=rows))
    for ref,h in before.items():assert sha(safe(ref))==h,('after',ref)
    write(OUT/'inputs_after.json',before)
    for p in OUT.iterdir():
        if p.is_file() and p.name not in ('audit.log','independent_review.json'):
            evidence[rel(p)]=sha(p)
    receipt=dict(schema='formal_main_v2',kind='action_execution_GPU_v3_candidate_independent_CPU_review',
        version='action-execution-gpu-v3-independent-0840-v1',
        status='passed_CPU_independent_candidate_pending_publication_root_registration_and_GPU',
        observed_utc=datetime.now(timezone.utc).isoformat(),author_review_sha256=AUTHOR_SHA,
        candidate_identity=read(HERE/'independent_review.template.json')['candidate_identity'],
        reviewed_candidate_files=publication['files'],published_files_verified=False,
        semantic_diff=dict(changed_functions=differences,submit_bytes_unchanged=True,wrapper_only_namespace=True,
            node_adapter_only_exact_version_stage_wrapper_replacement=True,action_adapter_exact_previously_reviewed_bytes=True,
            all16_scientific_manifest_subtrees_unchanged=True,all173_unchanged=True),
        independent_CPU_tests=dict(**counts,execution=test_execution),
        isolated_actual_CLI_runs=[inspect_execution,check_execution],
        synthetic_closure=dict(total=287,frozen_snapshot=173,project=114,
            includes_root_and_registration=2,real_publication_claimed=False),
        prior_failed_job=dict(job_id='3030159',status='FAILED',GPU_seconds=85,original_bytes_retained=True),
        typed_projection_advice=dict(template=rel(HERE/'independent_review.template.json'),
            after_root_exact_publication_status='passed_CPU_independent_review_pending_registered_GPU',
            requirements=['Archive this raw independent receipt unchanged and bind its SHA.',
                'Verify the eight publication path/SHA/byte rows before creating a typed projection.',
                'Populate candidate_files and evidence_sha256 with the exact three published probe/manifest/adapter refs.',
                'Require new root receipt, exact P/S closure, registration intent and current capacity/node scope before submission.']),
        limitations=['The two typed approvals and node pending/state objects in the CPU mirror are synthetic.',
            'Actual frozen sources, four development panels, selected5000 metadata and gate/closure code were used.',
            'No GPU model, simulator, controller or rollout execution ran; scientific hold remains.',
            'Runtime constructor tests were not repeated; the exact previously independently reviewed adapter bytes were verified.'],
        blockers=[],GPU_seconds=0,SSH_calls=0,Slurm_submissions=0,real_state_or_Git_changed=False,
        sealed_states_opened=False,weights_opened=False,wall_seconds=time.perf_counter()-start,
        read_evidence_sha256=dict(sorted(evidence.items())),evidence_sha256=dict(sorted(evidence.items())))
    write(OUT/'independent_review.json',receipt)
    print(json.dumps(dict(status=receipt['status'],receipt=rel(OUT/'independent_review.json'),sha256=sha(OUT/'independent_review.json'),
        tests=22,CLI_runs=2,closure=287,evidence=len(evidence),wall_seconds=receipt['wall_seconds'])))

if __name__=='__main__':main()
