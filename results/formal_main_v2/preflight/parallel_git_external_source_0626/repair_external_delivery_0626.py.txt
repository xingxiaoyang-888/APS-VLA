"""Deliver one untracked external source dependency as exact archived evidence.

Keep the rejected manifest and all prior evidence unchanged. The pinned byte
delivery checker remains unmodified; its new required list binds the archive.
"""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import subprocess
import sys

ROOT=Path(__file__).resolve().parents[1]
BASE='results/formal_main_v2/'
PRE=BASE+'preflight/'
PARENT='1bbaff1e2cb29d0e7d23d2857c11be326e0c1193'
OLD=PRE+'parallel_review_batch_20261018T0626_git_required.json'
NEW=PRE+'parallel_review_batch_20261018T0626_git_required_v2.json'
RESOLUTION=PRE+'parallel_git_external_source_resolution_0626.json'
PROOF=PRE+'parallel_review_batch_20261018T0626_git_delivery_v2.json'
ORIGINAL=dict(path='external/openvla-oft/prismatic/extern/hf/modeling_prismatic.py',bytes=49991,
    sha256='f6aba7898aed7a57405c1d68343086243c34506359f35da7625c9018ef943938')
ARCHIVE=PRE+'parallel_git_external_source_0626/modeling_prismatic.py.txt'
SNAPSHOT=BASE+'nonweight_archives/v1/snapshots/long_dev600_20261018.json'
SNAPSHOT_SHA='ac7e4dce68abeddb147dd412f83afffbc5041715868547d6f60811e5fca137f0'
PINS={'scripts/ops/nonweight_archive.py':'f81284b3c56fbfd1cce339c4ea98b20943c8cd0866b4e8b94c38ac3cdbd1e3c2',
    'scripts/ops/check_nonweight_archive_stage.py':'ebe5d25ff5850fe37e73cec551332ccd2b446aa3c4bbb0e38b6b83ce7a6775f6'}

def digest(data): return hashlib.sha256(data).hexdigest()
def git(*args): return subprocess.check_output(['git',*args],cwd=ROOT)
def ref(name):
    data=(ROOT/name).read_bytes()
    return dict(path=name,bytes=len(data),sha256=digest(data))
def write(name,value):
    data=(json.dumps(value,indent=2,sort_keys=True)+'\n').encode()
    p=ROOT/name;p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('xb') as stream:stream.write(data)
    return data

def main():
    assert git('rev-parse','HEAD').decode().strip()==PARENT
    assert not (ROOT/NEW).exists() and not (ROOT/PROOF).exists()
    old_bytes=(ROOT/OLD).read_bytes(); old=json.loads(old_bytes)
    assert len(old['files'])==1030
    matches=[r for r in old['files'] if r['path'].startswith('external/')]
    assert matches==[ORIGINAL] and ref(ORIGINAL['path'])==ORIGINAL
    assert not git('ls-files','--stage','--',ORIGINAL['path']).strip()
    assert not git('diff','--name-only').strip()
    for name,pin in PINS.items(): assert ref(name)['sha256']==pin
    batch=PRE+'parallel_review_batch_20261018T0626.json'
    assert ref(batch)['sha256']=='ab2101e565eca43574025a68c603b48a6fed03ae71a339eac0b8f6043d2b4876'
    p=ROOT/ARCHIVE;p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('xb') as stream:stream.write((ROOT/ORIGINAL['path']).read_bytes())
    archived=ref(ARCHIVE)
    assert archived['bytes']==ORIGINAL['bytes'] and archived['sha256']==ORIGINAL['sha256']
    helper=PRE+'parallel_git_external_source_0626/repair_external_delivery_0626.py.txt'
    with (ROOT/helper).open('xb') as stream:stream.write(Path(__file__).read_bytes())
    external_head=subprocess.check_output(['git','-C',str(ROOT/'external/openvla-oft'),'rev-parse','HEAD']).decode().strip()
    write(RESOLUTION,dict(schema='formal_main_v2',kind='exact_external_source_evidence_archive',status='exact_bytes_archived',
        recorded_utc=datetime.now(timezone.utc).isoformat(),original=ORIGINAL,archived=archived,
        external_repository_commit=external_head, external_repository_modified=False,
        original_is_not_parent_Git_index_blob=True, original_receipts_and_batch_unchanged=True,
        rejected_required=ref(OLD),batch=ref(batch),helper_archive=ref(helper),
        failure_observation=dict(helper='.local/stage_verify_parallel_0626_candidate.py',
            helper_sha256='0c6aed9feb94754ecc1e2a32b3227bd0d526c5027bf0384a909210a4bdbbab79',
            status='rejected',error_type='StageError',error='Missing required index blob: '+ORIGINAL['path']),
        resolution='Only this exact path/SHA/size tuple is delivered at an exact-byte source archive; no checksum exemption or broader alias.',
        frozen_source_files_changed=0,codec_and_guard_unchanged=True,GPU_seconds=0,final_states_opened=False))
    rows={r['path']:r for r in old['files'] if r!=ORIGINAL}
    for name in (ARCHIVE,helper,RESOLUTION,OLD):
        row=ref(name); assert name not in rows or rows[name]==row; rows[name]=row
    write(NEW,dict(schema='cti_nonweight_archive_stage_v1',kind='required_files',files=[rows[n] for n in sorted(rows)]))
    subprocess.run(['git','add','--',ARCHIVE,helper,RESOLUTION,NEW],cwd=ROOT,check=True)
    before,changes=git('ls-files','--stage','-z'),git('diff','--cached','--name-status','-z')
    snapshot=json.loads((ROOT/SNAPSHOT).read_bytes()); assert ref(SNAPSHOT)['sha256']==SNAPSHOT_SHA
    command=[sys.executable,'-B','scripts/ops/check_nonweight_archive_stage.py','--project-root',str(ROOT),
        '--snapshot',SNAPSHOT,'--snapshot-sha256',SNAPSHOT_SHA,'--required-files',NEW]
    for row in snapshot['files']: command.extend(['--raw-path',row['path']])
    print(json.dumps(dict(status='exact_archive_guard_running',required_count=len(rows))),flush=True)
    process=subprocess.run(command,cwd=ROOT,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    assert process.returncode==0,process.stderr.decode('utf-8',errors='replace')
    report=json.loads(process.stdout)
    assert report['status']=='verified_index_archive_delivery' and report['head']==PARENT
    assert report['index_sha256']==digest(before) and report['staged_changes_sha256']==digest(changes)
    assert report['required_count']==len(rows) and report['frozen_files_verified_in_worktree_and_index']==173
    assert len(report['raw_paths_restored_exactly'])==2 and report['gpu_seconds']==0
    assert before==git('ls-files','--stage','-z') and changes==git('diff','--cached','--name-status','-z')
    assert (ROOT/OLD).read_bytes()==old_bytes and ref(ORIGINAL['path'])==ORIGINAL
    assert git('show',':'+ARCHIVE)==(ROOT/ORIGINAL['path']).read_bytes()
    data=write(PROOF,dict(schema='formal_main_v2',kind='root_git_index_nonweight_delivery',status='passed',
        reviewed_utc=datetime.now(timezone.utc).isoformat(),guard_result=report,guard_command=command,
        unchanged_index_until_guard_return=True,exact_external_evidence_resolution=ref(RESOLUTION),
        rejected_manifest_preserved=ref(OLD),old_batch_unchanged=ref(batch),
        only_required_original_replaced=ORIGINAL,delivered_exact_archive=archived,
        no_checksum_exemptions=True,codec_and_guard_unchanged=True,final_states_opened=False,GPU_seconds=0))
    subprocess.run(['git','add','--',PROOF],cwd=ROOT,check=True)
    after=git('ls-files','--stage','-z'); added=set(after.split(b'\0'))-set(before.split(b'\0'))
    assert len(added)==1 and not set(before.split(b'\0'))-set(after.split(b'\0'))
    assert next(iter(added)).split(b'\t',1)[1].decode()==PROOF and git('show',':'+PROOF)==data
    subprocess.run(['git','diff','--cached','--check'],cwd=ROOT,check=True)
    assert not git('diff','--name-only').strip()
    print(json.dumps(dict(status='passed',proof=PROOF,sha256=digest(data),required_count=len(rows))),flush=True)

if __name__=='__main__':main()
