def obs_hash(obs):return m.signature(obs)
def actions():return np.linspace(-.9,.9,56*7).reshape(56,7).tolist()
def checks():return [{"control_step":i,"observation_sha256":obs_hash({"position":i})} for i in range(0,56,8)]

def test_replay_exact56_and_hard_stop(journal):
    env=FakeEnv();a=actions()
    gate,obs=m.replay_to56(env,[1.,2.],a,checks(),obs_hash,[0]*6+[-1],journal)
    assert env.actions==[[0]*6+[-1]]*10+a and gate.steps==66 and gate.locked
    assert obs=={"position":56} and env.resets==1
    with pytest.raises(RuntimeError,match="forbidden"):
        gate.step([0]*7,kind="recorded_control",index=56)
    assert len(env.actions)==66
    assert [x["control_step"] for x in journal.events if x["event"]=="observation_checkpoint"]==list(range(0,56,8))

class HookModule:
    def __init__(self):self.hooks=[]
    def register_forward_pre_hook(self,fn):
        self.hooks.append(fn)
        return SimpleNamespace(remove=lambda:self.hooks.remove(fn))
    def call(self):
        for h in self.hooks:h(self,())


def test_forward_hooks_count_and_cleanup_on_failure():
    modules=[HookModule() for _ in range(3)]
    c=SimpleNamespace(vla=SimpleNamespace(language_model=SimpleNamespace(model=modules[0]),vision_backbone=modules[1]),
        action_head=SimpleNamespace(model=modules[2]))
    with pytest.raises(RuntimeError):
        with m.forward_counters(c) as calls:
            for x in modules:x.call()
            assert calls=={"vla_backbone":1,"vision":1,"action_head":1}
            raise RuntimeError("synthetic")
    assert all(not x.hooks for x in modules)


class SyntheticController:
    def __init__(self,control_delta=True):
        self.use_delta=control_delta
        self.input_min=np.array([-1.,-1.]);self.input_max=np.array([1.,1.])
        self.output_min=np.array([-.1,-.1]);self.output_max=np.array([.1,.1])
    def scale_action(self,action):
        return np.clip(action,self.input_min,self.input_max)
    def set_goal(self,action):
        if self.use_delta:return self.scale_action(action)
        return action
