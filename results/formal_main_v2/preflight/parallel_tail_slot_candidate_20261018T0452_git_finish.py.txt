"""Preserve the one raw candidate log matched by the repository log ignore rule."""
from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,subprocess

R=Path(__file__).resolve().parents[1]
B='results/formal_main_v2/preflight/'
P=B+'parallel_tail_slot_candidate_20261018T0452.json'
O=B+'parallel_tail_slot_candidate_20261018T0452_git_finish.json'
S=B+'parallel_tail_slot_candidate_20261018T0452_git_finish.py.txt'
L=B+'action_execution_compat_candidate_0452/tests_initial.log'
PARENT='75da054987d670b710de3facd98533e0c314e7f9'
def g(*args):return subprocess.check_output(['git',*args],cwd=R)
def sha(p):return hashlib.sha256((R/p).read_bytes()).hexdigest()

assert g('rev-parse','HEAD').decode().strip()==PARENT
assert not g('diff','--cached','--name-only').strip()
assert not (R/O).exists() and not (R/S).exists()
r=json.loads((R/P).read_text())
names=set(r['evidence_sha256'])-{r['parent_full_delivery_proof']}
names.add(P)
for p,h in r['evidence_sha256'].items():assert sha(p)==h,p
assert L in names and sha(L)=='0a3583aa6218ea092a9ce14545990d62dc4e12c73e3662c30a291054b9134871'
ignored=g('check-ignore','--',L).decode().strip();assert ignored==L
changed=set(filter(None,g('ls-files','--modified','--others','--exclude-standard','-z').decode().split('\0')))
assert changed==names-{L},(changed-names,names-changed)
for p in names:
    assert p.startswith(('results/','docs/')) and (R/p).stat().st_size<1024*1024,p
payload=Path(__file__).read_bytes();(R/S).write_bytes(payload)
v=dict(schema='formal_main_v2',kind='root_small_tail_git_delivery',status='passed',reviewed_utc=datetime.now(timezone.utc).isoformat(),
    parent_commit=PARENT,tail_receipt=dict(path=P,sha256=sha(P)),
    original_writer_exit_code=1,original_writer_failure='The recorded files were complete; pre-stage changed-file equality omitted one existing repository-ignored *.log candidate artifact. No staging had begun and no data was missing.',
    original_writer_source=r['evidence_sha256'][B+'action_execution_compat_candidate_0452/root_record_tail_slot_candidate_0452.py.txt'],
    exact_ignored_log_retained=dict(path=L,sha256=sha(L)),global_ignore_rules_changed=False,
    source_archive=dict(path=S,sha256=hashlib.sha256(payload).hexdigest()),
    existing_raw_and_GPU_evidence_unchanged_from_parent=True,new_payload_max_bytes=1024*1024,
    candidate_only=True,candidate_independent_review_or_GPU_acceptance_claimed=False,
    new_weight_or_final_state_files=False,evidence_sha256={p:sha(p) for p in sorted(names)})
(R/O).write_bytes((json.dumps(v,ensure_ascii=False,indent=2,sort_keys=True)+'\n').encode())
names.update([S,O])
subprocess.run(['git','add','--',*sorted(names-{L})],cwd=R,check=True)
subprocess.run(['git','add','-f','--',L],cwd=R,check=True)
assert set(filter(None,g('diff','--cached','--name-only','-z').decode().split('\0')))==names
for p in names:assert g('show',':'+p)==(R/p).read_bytes(),p
subprocess.run(['git','diff','--cached','--check'],cwd=R,check=True)
assert not g('diff','--name-only').strip()
print(json.dumps(dict(status='passed',receipt=O,sha256=sha(O),files=len(names),cost=r['all_formal_cost_snapshot'])))
