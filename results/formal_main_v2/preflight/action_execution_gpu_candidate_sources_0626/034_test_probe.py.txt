class Clock:
    def __init__(self): self.value = 0.
    def perf_counter(self):
        self.value += .001
        return self.value


class FakeModule:
    def __init__(self): self.hooks = []; self.executed = 0
    def register_forward_pre_hook(self, fn):
        self.hooks.append(fn)
        return SimpleNamespace(remove=lambda: self.hooks.remove(fn))
    def __call__(self):
        for fn in list(self.hooks): fn(self, ())
        self.executed += 1

def observation(step):
    return dict(agentview_image=np.full((2,2,3),step,dtype=np.uint8),
        robot0_eye_in_hand_image=np.zeros((2,2,3),dtype=np.uint8),robot0_eef_pos=np.array([step,0.,0.]),
        robot0_eef_quat=np.array([0.,0.,0.,1.]),robot0_gripper_qpos=np.zeros(2))


class Environment:
    def __init__(self,done=None,fail=None): self.done=done; self.fail=fail; self.actions=[]; self.resets=0; self.sets=0
    def reset(self): self.resets+=1
    def set_init_state(self,value): self.sets+=1; return observation(0)
    def step(self,action):
        self.actions.append(copy.deepcopy(action))
        if len(self.actions)==self.fail: raise RuntimeError("synthetic env.step exception")
        return observation(len(self.actions)),0.,len(self.actions)==self.done,{}


def setup_policy(rt, method, *, failure=None, mutation=None):
    learned=method in ("A", "B"); tokens=512 if method=="dense" else 128
    identity=dict(suite="libero_10",method=method,method_id="dense" if method=="dense" else "uniform128" if method=="uniform" else method+"128_seed17",visual_tokens=tokens)
    if learned: identity.update(checkpoint_sha256="a"*64,training_seed=17)
    components=SimpleNamespace(vla=SimpleNamespace(language_model=SimpleNamespace(model=FakeModule()),vision_backbone=FakeModule()),
        action_head=SimpleNamespace(model=FakeModule()))
    estimator=FakeModule() if learned else None
    class Policy:
        selector=SimpleNamespace(estimator=estimator) if learned else None
        def reset(self): pass
        def __call__(self, obs, instruction):
            if failure=="exception": raise RuntimeError("synthetic policy exception")
            components.vla.vision_backbone()
            if learned: estimator(); rt.training.formal_hard_select()
            rt.packing.pack_equal_length(); components.vla.language_model.model(); components.action_head.model()
            if failure=="twice": components.vla.language_model.model()
            if failure=="watchdog": rt.clock.value += 61
            if failure=="mutation": obs["agentview_image"][0,0,0]=255
            if failure=="wrong_type": return object()
            indices=list(range(512)) if tokens==512 else list(range(64))+list(range(256,320))
            meta=dict(schema="formal_main_v2",suite=identity["suite"],method=method,visual_tokens=tokens,
                future_head_calls=0,policy_calls=1,vision_calls=1,packing_calls=1,scorer_calls=int(learned),language_cache="disabled",retained_indices=indices)
            if learned: meta.update(selector_checkpoint_sha256=identity["checkpoint_sha256"],training_seed=17)
            if mutation: mutation(meta)
            actions=np.linspace(-.9,.9,56).reshape(8,7); actions[7,2]=1.010742197890625
            if failure=="nan": actions[0,0]=np.nan
            return rt.module.PolicyChunk(np.zeros((8,7)),actions,meta)
    return Policy(),components,identity


def cell(rt,tmp_path,method="dense",done=None,fail=None,failure=None,mutation=None):
    journal=m.Journal(tmp_path,"000CPU",3600); budget=m.GlobalBudget()
    policy,components,identity=setup_policy(rt,method,failure=failure,mutation=mutation)
    env=Environment(done=done,fail=fail); state=dict(initial_state=[1.,2.],initial_state_sha256=m.state_hash([1.,2.]))
    try:
        with adapter.install_action_execution_compat(project_root=ROOT,source_root=ROOT,runtime_module=rt.module,
            formal_trials_module=rt.trials,expected_adapter_sha256=m.ADAPTER_SHA,emit=journal.emit):
            result=m.run_cell(rt.module,policy,env,"synthetic",state,identity,journal,budget,
                lambda:m.policy_counters(policy,components,budget))
        return result,env,journal,budget,components
    finally: journal.close()


@pytest.mark.parametrize("method",["dense","uniform","A","B"])
@pytest.mark.parametrize("done",[None,13])
def test_original_loop_real_scope_unchanged_queue_and_early_tail(runtime,tmp_path,method,done):
    original=runtime.module.run_episode
    result,env,journal,budget,c=cell(runtime,tmp_path,method,done)
    steps=8 if done is None else 3
    assert result["actual_control_steps"]==steps and len(env.actions)==10+steps
    assert len(result["unexecuted_chunk_tail"])==8-steps
    assert env.actions[10:]==result["raw_result"]["chunks"][0]["environment_actions"][:steps]
    assert result["raw_result"]["chunks"][0]["environment_actions"][7][2]==1.010742197890625
    assert env.resets==env.sets==1 and budget.counts==dict(policy=1,vla_backbone=1,vision=1)
    assert runtime.module.run_episode is original and runtime.trials.run_episode is original
    assert not c.vla.vision_backbone.hooks and not c.vla.language_model.model.hooks
    assert journal.events[-1]["event"]=="action_execution_candidate_exit"
