def write_json(root, rel, value):
    p = root / rel
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(value, sort_keys=True, indent=2) + '\n', encoding='utf-8', newline='\n')
    return dict(path=rel, sha256=c.sha(p))

def test_exact_closure_real_173_and_cpu_check_only(contract):
    checked = contract.check()
    assert len(checked['inventory']['files']) == 173
    shared = 'configs/experiments/formal_main_v2.json'
    assert checked['watched_files'][c.S + '/' + shared] == checked['watched_files'][c.P + '/' + shared]
    assert checked['watched_files'][c.P + '/' + contract.root_path] == contract.refs['root_review']['sha256']
    assert checked['watched_files'][c.P + '/' + contract.reg_path] == contract.refs['gpu_registration']['sha256']
    before = {p.relative_to(contract.project).as_posix(): c.sha(p) for p in contract.project.rglob('*') if p.is_file()}
    args = ['python', '-B', str(HERE / 'submit.py'), '--project-root', str(contract.project), '--check-only']
    for name, ref in contract.refs.items():
        args.extend(['--' + name.replace('_', '-'), ref['path'], '--' + name.replace('_', '-') + '-sha256', ref['sha256']])
    result = subprocess.run(args, capture_output=True, text=True, check=True)
    assert json.loads(result.stdout)['GPU_submitted'] is False
    assert before == {p.relative_to(contract.project).as_posix(): c.sha(p) for p in contract.project.rglob('*') if p.is_file()}


@pytest.mark.parametrize('change,match', [
    ('missing_snapshot', 'closure mismatch'), ('duplicate_server', 'duplicate/aliased'),
    ('server_alias', 'duplicate/aliased'), ('byte_count', 'bytes differ'),
    ('typed_status', 'typed candidate'), ('hash_cycle', 'hash cycle'),
    ('wrong_intent', 'registration intent'), ('unwatched_dispatch', 'unbound dispatch')])
def test_contract_rejects_identity_or_closure_damage(contract, change, match):
    if change == 'missing_snapshot':
        contract.root['deployment_files'].pop(0)
    elif change == 'duplicate_server':
        contract.root['deployment_files'].append(copy.deepcopy(contract.root['deployment_files'][0]))
    elif change == 'server_alias':
        contract.root['deployment_files'][0]['server'] += '.alias'
    elif change == 'byte_count':
        contract.root['deployment_files'][0]['bytes'] += 1
    elif change == 'typed_status':
        contract.root['status'] = contract.reg['cpu_reviews']['root_acceptance']['accepted_status'] = 'passed'
    elif change == 'hash_cycle':
        contract.root['evidence_sha256'][contract.reg_path] = '0' * 64
    elif change == 'unwatched_dispatch':
        del contract.root['evidence_sha256'][c.LAUNCHER]
    contract.sign()
    if change == 'wrong_intent':
        contract.root['gpu_registration_intent_sha256'] = '0' * 64
        ref = write_json(contract.project, contract.root_path, contract.root)
        contract.refs['root_review'] = ref
        contract.reg['cpu_reviews']['root_acceptance'].update(ref)
        contract.refs['gpu_registration'] = write_json(contract.project, contract.reg_path, contract.reg)
    with pytest.raises(ValueError, match=match):
        contract.check()

def jobs(statuses):
    return [dict(job_id=str(100 + n), experiment='formal_main_v2', stage_key='synthetic_stage_' + str(n),
                 status=status, gpus=0 if n == 0 else 1) for n, status in enumerate(statuses)]


def test_capacity_counts_cpu_pending_submitted_and_unknown():
    rows = jobs(['RUNNING', 'PENDING', 'SUBMITTED', 'COMPLETING', 'UNKNOWN', '', None, 'RUNNING'])
    active, available = d.submission_gate({'pending_submission': None}, rows)
    assert len(active) == 8 and not available
    rows[7]['status'] = 'COMPLETED+'
    assert d.submission_gate({'pending_submission': None}, rows)[1] is True


@pytest.mark.parametrize('damage', ['pending', 'prior_failed_stage', 'duplicate_job', 'duplicate_active_stage'])
def test_no_duplicate_or_uncertain_retry(damage):
    state = {'pending_submission': None}
    rows = jobs(['RUNNING', 'PENDING'])
    if damage == 'pending': state['pending_submission'] = {'known_job_id': None}
    if damage == 'prior_failed_stage': rows.append(dict(job_id='999', experiment='formal_main_v2', stage_key=c.STAGE, status='FAILED'))
    if damage == 'duplicate_job': rows[1]['job_id'] = rows[0]['job_id']
    if damage == 'duplicate_active_stage': rows[1]['stage_key'] = rows[0]['stage_key']
    with pytest.raises(ValueError): d.submission_gate(state, rows)


def fake_supervisor(project, rows):
    state = dict(pending_submission=None, registered_jobs=copy.deepcopy(rows), current_round_cost={},
                 formal_main_progress={'parallel_execution': {}})
    writes = []
    def write(path, value):
        writes.append((str(path), copy.deepcopy(value)))
    return SimpleNamespace(ROOT=project, P=c.P, H=c.H, STATE=project / 'synthetic_state.json',
        SSH=['NEVER_REAL_SSH'], supervisor_lock=lambda: nullcontext(), read=lambda path: state,
        write=write, refresh=lambda value: value['registered_jobs'], state=state, writes=writes)


def test_capacity_exit_precedes_any_deployment_or_pending_write(contract, monkeypatch):
    def forbidden(*args, **kwargs): raise AssertionError('deployment touched when full')
    f = fake_supervisor(contract.project, jobs(['RUNNING'] * 8))
    monkeypatch.setattr(d, 'remote_attempt_absent', forbidden)
    monkeypatch.setattr(d, 'remote_hashes', forbidden)
    result = d.execute_submit(contract.project, contract.check(), contract.refs, f)
    assert result['submitted'] is False and f.writes == []
    assert not (contract.project / c.INVOCATION).exists()


def test_ambiguous_sbatch_keeps_durable_pending_and_never_retries(contract, monkeypatch):
    checked = contract.check()
    f = fake_supervisor(contract.project, jobs(['RUNNING'] * 7))
    events = []
    monkeypatch.setattr(d, 'remote_attempt_absent', lambda f: events.append('checked_no_attempt'))
    def remote_hashes(f, names):
        if list(names) == [c.P + '/' + c.INVOCATION]:
            return {c.P + '/' + c.INVOCATION: c.sha(contract.project / c.INVOCATION)}
        return dict(checked['watched_files'])
    monkeypatch.setattr(d, 'remote_hashes', remote_hashes)
    def publish(f, local, server, expected):
        assert f.writes[-1][1]['pending_submission']['invocation_sha256'] == expected
        assert local == contract.project / c.INVOCATION
        events.append('invocation_publication_after_pending')
    monkeypatch.setattr(d, 'publish_missing', publish)
    def ambiguous(command, **kwargs):
        assert command[0] == 'NEVER_REAL_SSH'
        assert f.state['pending_submission'] is not None
        events.append('sbatch_once')
        return SimpleNamespace(returncode=255, stdout='', stderr='synthetic lost reply')
    monkeypatch.setattr(d.subprocess, 'run', ambiguous)
    with pytest.raises(RuntimeError, match='Unknown sbatch outcome'):
        d.execute_submit(contract.project, checked, contract.refs, f)
    assert events.count('sbatch_once') == 1 and len(f.state['registered_jobs']) == 7
    assert f.state['pending_submission']['status'] == 'prepared_before_invocation_publication'
    assert (contract.project / c.INVOCATION).is_file()
    with pytest.raises(ValueError, match='unresolved pending'):
        d.execute_submit(contract.project, checked, contract.refs, f)
    assert events.count('sbatch_once') == 1


def test_static_wrapper_and_launcher_execution_boundary():
    wrapper = (HERE / 'probe.slurm').read_text(encoding='utf-8')
    assert '\r' not in wrapper
    for token in ['#SBATCH --gres=gpu:l40:1', '#SBATCH --time=01:00:00', '#SBATCH --no-requeue',
                  'WRAPPER_SHA256', '"$H/venv/bin/python"', 'INVOCATION_SHA256']:
        assert token in wrapper
    bootstrap = wrapper.split("<<'PY'\n", 1)[1].rsplit('\nPY', 1)[0]
    ast.parse(bootstrap)
    assert bootstrap.index('hashlib.sha256(lp.read_bytes())') < bootstrap.index('os.execv(')
    launcher = (HERE / 'launcher.py').read_text(encoding='utf-8')
    ast.parse(launcher)
    ast.parse((HERE / 'submit.py').read_text(encoding='utf-8'))
    for flag in ['--manifest-sha256', '--gpu-registration-sha256', '--job-id', '--script-sha256', '--execute-gpu']:
        assert flag in launcher
    assert 'remaining = 3540' in launcher and "marker.open('x'" in launcher
    assert 'scientific_completion_inferred=False' in launcher
