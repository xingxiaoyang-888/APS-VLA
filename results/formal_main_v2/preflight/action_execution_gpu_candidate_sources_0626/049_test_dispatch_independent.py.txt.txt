    return [dict(job_id=str(i+100),experiment='formal_main_v2',stage_key='synthetic_'+str(i),status=s,gpus=0 if i==0 else 1)

class SimulatedRootSupervisor:
    """Run exact metadata remote-Python snippets against isolated files, not SSH."""
    def __init__(self,project,server,jobs,monkeypatch,reply='9901;synthetic\n'):
        self.ROOT=project;self.P=c.P;self.H=c.H;self.STATE=project/'synthetic_supervisor_state.json'
        self.state=dict(pending_submission=None,registered_jobs=copy.deepcopy(jobs),current_round_cost={},formal_main_progress={'parallel_execution':{}})
        self.SSH=['NEVER_ACTUAL_SSH'];self.server=server;self.server.mkdir();self.writes=[];self.transport=[]
        self.locked=False;self.refreshed=False;self.monkeypatch=monkeypatch;self.reply=reply;self.sbatch=0
        monkeypatch.setattr(d.subprocess,'run',self.run)
    def mapped_path(self,value):
        raw=str(value)
        if raw==c.P:return self.server
        if raw.startswith(c.P+'/'):return self.server/raw[len(c.P)+1:]
        p=REAL_PATH(value)
        if p==self.server or self.server in p.parents:return p
        raise AssertionError('remote metadata path escaped isolated project: '+raw)
    @contextmanager
    def supervisor_lock(self):
        assert not self.locked;self.locked=True
        try:yield
        finally:self.locked=False
    def read(self,path):assert path==self.STATE;return self.state
    def write(self,path,value):
        assert self.locked;self.writes.append((str(path),copy.deepcopy(value)))
    def refresh(self,state):
        assert self.locked and state is self.state;self.refreshed=True
        return [j for j in state['registered_jobs'] if j['experiment']=='formal_main_v2']
    def remote(self,command):
        assert self.locked and self.refreshed
        args=shlex.split(command);assert args[:2]==[c.H+'/venv/bin/python','-c']
        self.transport.append(('remote_python',command))
        module=ModuleType('pathlib');module.Path=self.mapped_path
        output=io.StringIO()
        with self.monkeypatch.context() as patch:
            patch.setitem(sys.modules,'pathlib',module)
            with redirect_stdout(output):exec(compile(args[2],'<exact-dispatch-remote-python>','exec'),{})
        return output.getvalue()
    def run(self,args,**kwargs):
        assert self.locked and self.refreshed
        if args[0]=='scp':
            assert args[:2]==['scp','-q'] and args[3].startswith('tongji-cvpr:'+c.P+'/')
            dst=self.mapped_path(args[3].split(':',1)[1]);assert not dst.exists()
            shutil.copyfile(args[2],dst);self.transport.append(('scp_upload_only',str(dst)))
            return SimpleNamespace(returncode=0)
        assert args[0]=='NEVER_ACTUAL_SSH' and 'sbatch --parsable' in args[1]
        assert self.state['pending_submission']['status']=='prepared_before_invocation_publication'
        self.sbatch+=1;self.transport.append(('synthetic_sbatch',args[1]))
        return SimpleNamespace(returncode=0,stdout=self.reply,stderr='')

    with pytest.raises(ValueError,match='already exists'):
        d.execute_submit(case.project,checked,case.refs,f)

def test_wrapper_fixed_resources_no_requeue_and_embedded_launch_guard():
    raw=(AUTHOR/'probe.slurm').read_bytes();assert b'\r' not in raw
    text=raw.decode();directives=[x for x in text.splitlines() if x.startswith('#SBATCH')]
    for value in ('--nodes=1','--ntasks=1','--cpus-per-task=7','--gres=gpu:l40:1','--mem=64G','--time=01:00:00','--no-requeue'):
        assert '#SBATCH '+value in directives
    assert 'CUDA_VISIBLE_DEVICES=' not in text and 'srun "$H/venv/bin/python" -B - ' in text
    python=text.split("<<'PY'\n",1)[1].rsplit('\nPY',1)[0];tree=ast.parse(python)
    exec_calls=[x for x in ast.walk(tree) if isinstance(x,ast.Call) and isinstance(x.func,ast.Attribute) and x.func.attr=='execv']
    assert len(exec_calls)==1
    assert "if hashlib.sha256(lp.read_bytes()).hexdigest()!=inv['launcher_sha256']" in python
    assert 'invocation SHA differs' in python and 'wrapper/launcher not bound' in python
    assert exec_calls[0].lineno>max(x.lineno for x in ast.walk(tree) if isinstance(x,ast.Raise))
