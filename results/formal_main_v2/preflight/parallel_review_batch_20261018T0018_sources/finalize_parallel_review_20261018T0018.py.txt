"""Close the root heartbeat with byte-exact evidence and all allocation costs."""
from pathlib import Path
from datetime import datetime, timezone, timedelta
import json, subprocess, sys
root = Path(__file__).resolve().parents[1]
sys.path.insert(0,str(root/'scripts'))
import formal_supervisor as f
base='results/formal_main_v2/'
batch=base+'preflight/parallel_review_batch_20261018T0018.json'
assert not (root/batch).exists()
ids=['3021362','3021349','3021350','3021351','3021353','3021370','3021418','3021463']
refs, sources, pulls, boundaries=set(),set(),set(),[]
for jid in ids:
    oprel='.local/root_pull_review_2250_'+jid+'.json'
    op=f.read(root/oprel)
    rr=op['root_review']; ir=rr.removesuffix('.json')+'_independent.json'
    r,i=f.read(root/rr),f.read(root/ir)
    assert op['status']==r['status']==i['status']=='passed'
    assert op['job_id']==r['job_id']==i['job_id']==jid
    assert f.sha(root/rr)==op['root_sha256']
    assert f.sha(root/op['pull'])==op['pull_sha256']
    assert r['completed_by_arm']==dict.fromkeys(('A','B'),op['boundary'])
    assert r['allowed_next_segment_end']==i['allowed_next_segment_end']==op['allowed_next']
    for path,checksum in i['read_evidence_sha256'].items():
        assert f.sha(root/path)==checksum,path
        (sources if path.startswith('.local/') else refs).add(path)
    assert i['read_evidence_sha256'][rr]==op['root_sha256']
    refs.update([rr,ir,op['pull']]);sources.add(oprel);pulls.add(op['pull'])
    if op['boundary']==5000:
        selected=rr.removesuffix('.json')+'_selected_server.json'
        assert i['read_evidence_sha256'][selected]==f.sha(root/selected)
        assert f.read(root/selected)['status']=='passed'
        refs.add(selected)
    for entry in f.read(root/op['pull'])['files']:
        assert f.sha(root/entry['relative'])==entry['sha256'],entry['relative']
        refs.add(entry['relative'])
    boundaries.append({k:op[k] for k in ('job_id','suite','budget','seed','boundary','allowed_next','pair_gpu_seconds','latest_gpu_seconds')})
object_root=base+'preflight/object_first_pair_development_root_review_v2.json'
object_sub=base+'preflight/object_first_pair_development_submission_root_v2.json'
object_review=f.read(root/object_root);sub=f.read(root/object_sub)
assert object_review['status']=='passed_cpu_only' and object_review['plan_rebuilt_equal']
assert object_review['completed_trials_before']==12 and object_review['additional_stop_after']==120
assert sub['status']=='submitted_pending_actual_GPU_review'
for path,checksum in object_review['read_evidence_sha256'].items():
    assert f.sha(root/path)==checksum,path
    (sources if path.startswith('.local/') else refs).add(path)
for path,checksum in sub['evidence_sha256'].items():
    assert f.sha(root/path)==checksum,path
    refs.add(path)
refs.update([object_root,object_sub])
sources.update(['.local/root_pull_review_one_20261017T2250.py','.local/review_selected_pair_server.py',
    '.local/fill_reviewed_pair_slot.py','.local/record_parallel_boundary.py',
    '.local/root_accept_boundary_0018.py','.local/reconcile_parallel_progress_20261018.py',
    '.local/finalize_parallel_review_20261018T0018.py','.local/stage_verify_parallel_0018.py'])
parent=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root).decode().strip()
assert parent=='9513cbe3c8059c7dc4fec2a7e3bcaef062897cc3'
prior=json.loads(subprocess.check_output(['git','show','HEAD:results/experiment_supervisor/state.json'],cwd=root))
old_ids={j['job_id'] for j in prior['registered_jobs']}
with f.supervisor_lock():
    state=f.read(f.STATE);jobs=f.refresh(state);p=state['formal_main_progress']
    active=[j for j in jobs if j['status'] in f.ACTIVE|{'SUBMITTED'}]
    assert len(active)==8 and len({j['stage_key'] for j in active})==8
    assert all(j['gpus']==1 for j in active)
    assert sum(j['mode']=='train' for j in active)==7
    assert sum(j['mode']=='development' and j['suite']=='libero_object' for j in active)==1
    assert p['trained_arms']==6 and p['root_and_independent_reviewed_arm_updates']==108000
    assert p['completed_development_trials']==612 and p['completed_final_trials']==0
    assert not p['test_opened'] and not p['global_release_created']
    assert p['first_object_development']['active_job_id']==sub['job_id']
    new_jobs=[j for j in jobs if j['job_id'] not in old_ids]
    assert len(new_jobs)==8 and {j['job_id'] for j in new_jobs}=={j['job_id'] for j in active}
    for j in new_jobs:
        assert f.sha(root/j['invocation'])==j['invocation_sha256']
        refs.update([j['invocation'],base+'submissions/'+j['job_id']+'.json'])
        if j['mode']=='train':
            key=f"{j['suite']}_k{j['visual_tokens']}_seed{j['training_seed']}"
            pair=p['parallel_execution']['pair_tracking'][key]
            assert pair['active_job_id']==j['job_id']
            assert f.sha(root/pair['latest_resume_precheck'])==pair['latest_resume_precheck_sha256']
            refs.add(pair['latest_resume_precheck'])
    folder=root/(base+'preflight/parallel_review_batch_20261018T0018_sources')
    assert not folder.exists();folder.mkdir()
    source_map={}
    for path in sorted(sources):
        src=root/path;assert src.is_file() and not src.is_symlink()
        dest=folder/(src.name+'.txt');assert not dest.exists()
        dest.write_bytes(src.read_bytes());name=dest.relative_to(root).as_posix()
        source_map[path]=name;refs.add(name)
    now=datetime.now(timezone.utc);local=now.astimezone(timezone(timedelta(hours=8)))
    value=dict(schema='formal_main_v2',kind='root_incremental_parallel_training_review',status='passed',
        reviewed_utc=now.isoformat(),parent_delivered_commit=parent,
        completed_training_arms=6,expected_training_arms=72,cumulative_reviewed_arm_updates=108000,
        newly_reviewed_arm_updates=8000,reviewed_boundaries=boundaries,
        completed_development_trials=612,new_development_trials=0,completed_final_trials=0,
        final_states_sealed=True,new_object_development_submission=sub['job_id'],
        object_additional_stop_after=120,object_expected_cumulative_trials=132,
        current_batch_closed_GPU_seconds=sum(x['latest_gpu_seconds'] for x in boundaries),
        closed_cost_is_included_in_snapshot_not_additional=True,all_formal_cost_snapshot=state['current_round_cost'],
        active_jobs=[{k:j.get(k) for k in ('job_id','status','mode','suite','visual_tokens','training_seed','segment_end')} for j in active],
        new_submission_job_ids=[j['job_id'] for j in new_jobs],current_terminal_pulls=sorted(pulls),
        pull_file_entries=sum(len(f.read(root/name)['files']) for name in pulls),
        audit_source_archive_map=source_map,evidence_sha256={path:f.sha(root/path) for path in sorted(refs)},
        scientific_settings_changed=False,weights_copied=False)
    f.write(root/batch,value)
    p['latest_incremental_review']=dict(path=batch,sha256=f.sha(root/batch))
    state['next_actions'].append(p['first_object_development']['next'])
    state['next_actions'].append('Spatial128 seed17 A/B5000 reviewed; prepare registered development with chosen A4000/B5000 identities after a GPU slot frees. L1 alone is not a closed-loop benefit.')
    state['next_actions'].append('Deliver exact nonweight evidence through the pinned Git-index archive guard; do not force-add the two large Long raw ledger paths.')
    state['updated_utc']=now.isoformat();state['updated_date']=local.strftime('%Y-%m-%d');f.write(f.STATE,state)
    text=(f'{local:%Y-%m-%d %H:%M}进度：完整训练6/72组，累计108000/360000次臂更新通过双重复核（30.0%）。'
        '本批8个500步分段全部验收；Spatial128 seed17完成5000，选A4000/B5000。7个单卡训练与1个Object开发作业并行。'
        f"正式轮累计临时{state['current_round_cost']['allocated_gpu_seconds']} GPU秒（{state['current_round_cost']['allocated_gpu_hours']:.6f}小时），包含运行中及历史失败分配。\n\n"
        'Object在原600条清单上继续下一120条，完成目标132条；现已验收仍为Long600＋Object12＝612条，本段结果尚待回存复核。'
        '其余已验收pair按空闲槽续跑，仅下一500步。最终0/35200且继续封存，未改变科学配置。'
        '\n\n[本批原始产物、审计与费用](../'+batch+')；[Spatial完整训练记录](正式Spatial首对5000步记录.md)。\n\n')
    for name in ('正式主实验规划.md','完整实验计划.md','实验进度.md'):
        doc=root/'docs'/name;heading,sep,body=doc.read_bytes().partition(b'\n');assert sep
        doc.write_bytes(heading+sep+b'\n'+text.encode('utf-8')+body)
    doc=root/'docs/正式Object首批开发登记.md'
    with doc.open('a',encoding='utf-8',newline='\n') as stream:
        stream.write(f'\n{local:%Y-%m-%d %H:%M}：第二段已登记并提交`{sub["job_id"]}`，同原账本显式resume，新增120、累计目标132/600，首12条不重跑。方法、任务、初态、环境与权重身份不变；下一分段须待实际终态双审。\n\n'
            '[v2登记](../'+base+'preflight/object_first_pair_development_registration_v2.json)；[root重建及提交](../'+object_sub+')。\n')
print(json.dumps(dict(receipt=batch,sha256=f.sha(root/batch),cost=value['all_formal_cost_snapshot'],
    completed_arms=6,reviewed_updates=108000,active_jobs=8,pull_files=value['pull_file_entries'])),flush=True)
