def sha(path):
    h = hashlib.sha256()
    with (ROOT / path).open("rb") as stream:
        for block in iter(lambda: stream.read(8 << 20), b""):
            h.update(block)
    return h.hexdigest()


def read(path):
    return json.loads((ROOT / path).read_text(encoding="utf-8"))


def main():
    for output in (MANIFEST, MANIFEST + ".evidence.json", PLAN, RECORD, LOG):
        if (ROOT / output).exists():
            raise ValueError("Preparation refuses existing output: " + output)
    for path, digest in EXPECTED.items():
        if sha(path) != digest:
            raise ValueError("Pinned preparation evidence differs: " + path)
    registration = read(REGISTRATION)

    inventory = read(INVENTORY)

    assert independent["allowed_next_segment_end"] is None and not independent["resume_required"]
    assert independent["root_review_sha256"] == EXPECTED[ROOT_REVIEW]
    assert independent["selected_server_review_sha256"] == EXPECTED[SELECTED_REVIEW]
    assert selected["remote_returncode"] == 0 and not selected["weights_copied"]

        assert sha(path) == digest, path
    assert {row["arm"] for row in selected["selected"]} == set(BEST)
    for item in selected["selected"]:
        arm = item["arm"]
        assert (item["selected_update"], item["expected_sha256"]) == BEST[arm]
        actual = [row for row in selected["server_selected_checks"] if row["arm"] == arm]
        assert len(actual) == 1 and actual[0]["exists"] and actual[0]["stable_during_hash"]
        assert actual[0]["path"] == item["path"] and actual[0]["sha256"] == item["expected_sha256"]
    state_path = "results/experiment_supervisor/state.json"
    state_before = sha(state_path)
    commands = [
        [sys.executable, "-B", "scripts/formal_execution_manifests.py", "development", "--suite", "libero_object",
         "--methods", *METHODS, "--source-inventory", INVENTORY, "--output", MANIFEST],
        [sys.executable, "-B", "scripts/ops/formal_downstream_supervisor.py", "plan",
         "--stage", "development", "--execution-manifest", MANIFEST, "--stop-after", "12"],
    ]
    runs = []
    with (ROOT / LOG).open("x", encoding="utf-8", newline="\n") as log:
        for command in commands:
            result = subprocess.run(command, cwd=ROOT, capture_output=True, text=True, encoding="utf-8",
                env={**os.environ, "PYTHONDONTWRITEBYTECODE": "1"})
            run = dict(command=command, returncode=result.returncode, stdout=result.stdout, stderr=result.stderr)
            log.write(json.dumps(run, ensure_ascii=False) + "\n")
            log.flush()
            runs.append(run)
            if result.returncode:
                raise RuntimeError("Original CPU entrypoint rejected preparation; retained in " + LOG)

    manifest, manifest_evidence = read(MANIFEST), read(MANIFEST + ".evidence.json")
    value = json.loads(runs[1]["stdout"])

    assert [row["identity"]["method_id"] for row in manifest["methods"]] == METHODS
    assert all(row["identity"]["future_head_deployed"] is False for row in manifest["methods"])
    for row in manifest["methods"][-2:]:
        arm = row["identity"]["method"]
        assert row["identity"]["checkpoint_sha256"] == BEST[arm][1]
        assert row["identity"]["model_revision"] == spec["model_revision"]
        assert row["checkpoint"] == next(item["path"] for item in selected["selected"] if item["arm"] == arm)

    compatibility = invocation["downstream_asset_compatibility"]

    assert value["resources"] == dict(job_name="cti-formal-downstream", account="p_p15016",
        qos="cpu-500_core-l40-8_card-a800-8_card", partition="L40", nodes=1, ntasks=1,
        cpus_per_task=7, gres="gpu:l40:1", mem="64G", time="04:00:00")
    assert not (ROOT / value["output_relative"]).exists()
    assert not (ROOT / value["invocation_relative"]).exists()

    assert len({row["trial_id"] for row in panel["states"]}) == 100

        stopping_rule="Original fixed state-major/method-minor order: first 2 development states times 6 methods. After root audit, resume the same 600-trial ledger; completed trials are never rerun.",
        slice_reason="First actual Object downstream interface and throughput check within the registered 4-hour allocation. Later slice sizes may depend on measured duration, never observed success outcomes.",
        scope="Only 6 of the registered 22 Object methods, each on the same 100 development states. This is not complete suite development, the global release, a final result, or proof of success/latency benefit.",
        reference_reuse="None: generate Object Dense/uniform references from its own registered base and environment. No Spatial, Long, pilot, or other-suite outcomes are reused.",
        resources=value["resources"], actual_object_downstream_gpu_verified=False,

            "Root holds the original supervisor lock, refreshes registered jobs, recomputes the identical plan SHA, and verifies no pending submission, duplicate pair/run, or more than 8 nonterminal requests.",
            "After the first 12 trials, root verifies the actual Object metadata/adapter restoration and raw closed-loop ledger before explicit continuation."],

        read_evidence_sha256=dict(sorted(evidence.items())))
    with (ROOT / RECORD).open("x", encoding="utf-8", newline="\n") as stream:

        stream.write("\n")
    print(json.dumps(dict(status=record["status"], registration=RECORD, registration_sha256=sha(RECORD),

        environment=ENVIRONMENT, target_trials=600, initial_engineering_slice=12, submitted=False), ensure_ascii=False))


if __name__ == "__main__":
    main()
