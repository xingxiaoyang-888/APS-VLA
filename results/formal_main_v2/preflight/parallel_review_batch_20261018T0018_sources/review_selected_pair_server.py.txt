"""Root-only selected-weight SHA check after strict completed-5000 review."""
from pathlib import Path, PurePosixPath
from datetime import datetime, timezone
import argparse
import hashlib
import json
import shlex
import subprocess
import sys

root = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(root / 'scripts'))
import formal_supervisor as f

parser = argparse.ArgumentParser()
parser.add_argument('--job-id', required=True)
parser.add_argument('--pull-receipt', type=Path, required=True)
args = parser.parse_args()
assert args.job_id.isdigit()

with f.supervisor_lock():
    state = f.read(f.STATE)
    jobs = f.refresh(state)
    job = next(j for j in jobs if j['job_id'] == args.job_id)
    assert job['mode'] == 'train' and job['status'] == 'COMPLETED' and job['exit_code'] == '0:0'
    assert job['segment_end'] == 5000 and state['pending_submission'] is None
    own = [j for j in jobs if j.get('stage_key') == job['stage_key']]
    assert max(own, key=lambda j: j['submitted_utc'])['job_id'] == args.job_id
    assert not any(j['status'] in f.ACTIVE | {'SUBMITTED'} for j in own)
    suite, budget, seed = job['suite'], job['visual_tokens'], job['training_seed']
    registration_rel = 'configs/experiments/formal_main_v2.json'
    registration = f.read(root / registration_rel)
    pair = next(p for p in registration['pairs'] if
                (p['suite'], p['visual_tokens'], p['seed']) == (suite, budget, seed))
    base = f'results/formal_main_v2/{suite}/k{budget}/seed{seed}/training'
    review_rel = base + '/segment_reviews/' + args.job_id + '.json'
    review = f.read(root / review_rel)
    assert review['status'] == 'passed' and review['job_id'] == args.job_id
    assert review['completed_by_arm'] == {'A': 5000, 'B': 5000}
    assert review['allowed_next_segment_end'] is None and review['resume_required'] is False
    assert review['continuation'] == 'formal_training_complete_no_extension'
    assert len(review['server_resume_checks']) == 2
    pull_path = args.pull_receipt if args.pull_receipt.is_absolute() else root / args.pull_receipt
    pull_rel = pull_path.resolve().relative_to(root.resolve()).as_posix()
    pull_sha = f.sha(pull_path)
    assert review['evidence_sha256'][pull_rel] == pull_sha
    inputs = {review_rel: f.sha(root / review_rel), pull_rel: pull_sha,
              registration_rel: f.sha(root / registration_rel), job['invocation']: f.sha(root / job['invocation'])}
    assert inputs[job['invocation']] == job['invocation_sha256'] == review['invocation_sha256']
    specs = []
    for arm in ('A', 'B'):
        receipt_rel = base + '/' + arm + '/deployment_receipt.json'
        report_rel = base + '/' + arm + '/training_report.json'
        for rel in (receipt_rel, report_rel):
            inputs[rel] = f.sha(root / rel)
            assert inputs[rel] == review['files'][rel]
        receipt, report = f.read(root / receipt_rel), f.read(root / report_rel)
        expected = str(PurePosixPath(pair['checkpoint_root_server']) / arm / 'best.pt')
        assert report['best_checkpoint'] == expected
        assert receipt['training_complete'] is True and receipt['completed_updates'] == 5000
        assert receipt['checkpoint_sha256'] == report['best_checkpoint_sha256']
        best = min(report['evaluations'], key=lambda x: (x['validation_metric'], x['update']))
        assert [x['update'] for x in report['evaluations']] == list(range(0, 5001, 500))
        assert best['update'] == report['best_update'] == receipt['metadata']['update']
        specs.append(dict(arm=arm, path=expected, expected_sha256=receipt['checkpoint_sha256'],
                          selected_update=best['update'], validation_metric=best['validation_metric']))
    code = '''import hashlib,json,os,stat
from pathlib import Path
specs=INPUT
out=[]
for row in specs:
 p=Path(row['path'])
 for parent in [*reversed(p.parents),p]:
  if parent.is_symlink(): raise ValueError('Symlink checkpoint path')
 fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW)
 with os.fdopen(fd,'rb') as handle:
  before=os.fstat(handle.fileno())
  if not stat.S_ISREG(before.st_mode): raise ValueError('Not a regular selected checkpoint')
  h=hashlib.sha256()
  for block in iter(lambda:handle.read(8<<20),b''): h.update(block)
  after=os.fstat(handle.fileno())
  current=os.stat(p,follow_symlinks=False)
  identity=lambda s:(s.st_dev,s.st_ino,s.st_size,s.st_mtime_ns)
  if not identity(before)==identity(after)==identity(current): raise ValueError('Selected checkpoint changed during hash')
 out.append(dict(arm=row['arm'],path=str(p),exists=True,bytes=after.st_size,sha256=h.hexdigest(),
  device=after.st_dev,inode=after.st_ino,mtime_ns=after.st_mtime_ns,stable_during_hash=True))
print(json.dumps(out))
'''.replace('INPUT', repr(specs), 1)
    answer = subprocess.run(f.SSH + [shlex.quote(f.H + '/venv/bin/python') + ' -B -'],
                            input=code, capture_output=True, text=True, encoding='utf-8', check=True, timeout=180)
    checks = json.loads(answer.stdout)
    assert len(checks) == 2 and {x['arm'] for x in checks} == {'A', 'B'}
    for spec in specs:
        actual = next(x for x in checks if x['arm'] == spec['arm'])
        assert actual['path'] == spec['path'] and actual['sha256'] == spec['expected_sha256']
        assert actual['stable_during_hash'] and actual['exists'] and actual['bytes'] > 0
    assert all(f.sha(root / rel) == digest for rel, digest in inputs.items())
    out_rel = base + '/segment_reviews/' + args.job_id + '_selected_server.json'
    assert not (root / out_rel).exists(), 'Keep existing immutable selected checkpoint proof'
    value = dict(schema='formal_main_v2', kind='root_selected_pair_server_SHA_review', status='passed',
        reviewed_utc=datetime.now(timezone.utc).isoformat(), job_id=args.job_id, suite=suite,
        visual_tokens=budget, seed=seed, source_sha256=review['source_sha256'],
        invocation_sha256=job['invocation_sha256'], registered_terminal_job=job,
        root_segment_review=review_rel, input_sha256=inputs, server_selected_checks=checks,
        selected=specs, probe_program_sha256=hashlib.sha256(code.encode('utf-8')).hexdigest(),
        probe_program=code, remote_returncode=answer.returncode, weights_copied=False,
        weights_deserialized=False, additional_GPU_seconds=0, final_states_opened=False,
        global_release=False, scope='Only two exact registered completed-pair best.pt files; not global72 checkpoint inventory.')
    f.write(root / out_rel, value)
    print(json.dumps({'output': out_rel, 'sha256': f.sha(root / out_rel), 'selected': specs}))
