def stable(path):
 require(not any(x.is_symlink() for x in (path,*path.parents)),'symlink forbidden')
 a=path.stat();require(stat.S_ISREG(a.st_mode),'not regular')
 data=path.read_bytes();b=path.stat()
 require((a.st_dev,a.st_ino,a.st_size,a.st_mtime_ns,a.st_ctime_ns)==(b.st_dev,b.st_ino,b.st_size,b.st_mtime_ns,b.st_ctime_ns),'concurrent byte change')
 return data
class Evidence:
