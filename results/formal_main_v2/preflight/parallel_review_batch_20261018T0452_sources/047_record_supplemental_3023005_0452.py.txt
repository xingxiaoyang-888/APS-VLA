"""Root binds the first Spatial256 seed17 boundary and its actual continuation."""
from pathlib import Path
from datetime import datetime, timezone
import json
import sys

ROOT=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(ROOT/'scripts'))
import formal_supervisor as f

NAME='results/formal_main_v2/preflight/parallel_review_supplement_3023005_0452.json'
assert not (ROOT/NAME).exists()
op_name='.local/root_pull_review_2250_3023005.json'
op=f.read(ROOT/op_name)
rr=op['root_review'];ir=rr.removesuffix('.json')+'_independent.json'
ind=f.read(ROOT/ir)
assert op['status']==ind['status']=='passed' and op['allowed_next']==ind['allowed_next_segment_end']==1000
assert op['latest_gpu_seconds']==op['pair_gpu_seconds']==4083
evidence=dict(ind['read_evidence_sha256'])
for n in (op_name,rr,ir,op['pull'],'.local/record_supplemental_3023005_0452.py'):
    evidence[n]=f.sha(ROOT/n)
for row in f.read(ROOT/op['pull'])['files']:
    assert row['relative'] not in evidence or evidence[row['relative']]==row['sha256']
    evidence[row['relative']]=row['sha256']
for n,h in evidence.items():assert f.sha(ROOT/n)==h,n
state=f.read(f.STATE);pair=state['formal_main_progress']['parallel_execution']['pair_tracking']['libero_spatial_k256_seed17']
assert pair['last_completed_job_id']=='3023005' and pair['last_root_reviewed_common_update']==500
assert pair['active_job_id'] and pair['active_job_id']!='3023005' and pair['allowed_next_segment_end']==1000
active=next(j for j in state['registered_jobs'] if j['job_id']==pair['active_job_id'])
assert active['suite']=='libero_spatial' and active['training_seed']==17 and active['visual_tokens']==256 and active['segment_end']==1000
assert active['status'] in f.ACTIVE|{'SUBMITTED'}
value=dict(schema='formal_main_v2',kind='root_additional_training_boundary_review',status='passed',
    reviewed_utc=datetime.now(timezone.utc).isoformat(),job_id='3023005',suite='libero_spatial',visual_tokens=256,seed=17,
    completed_updates_by_arm=dict(A=500,B=500),allowed_next_segment_end=1000,
    resumed_job_id=pair['active_job_id'],newly_reviewed_arm_updates=1000,newly_completed_arms=0,
    pair_allocated_GPU_seconds=4083,current_allocated_GPU_seconds=4083,
    cost_included_in_global_snapshot=True,root_review=rr,independent_review=ir,
    terminal_pull=op['pull'],evidence_sha256=evidence,scientific_settings_changed=False,
    final_states_opened=False,weights_copied=False)
f.write(ROOT/NAME,value)
print(json.dumps(dict(receipt=NAME,sha256=f.sha(ROOT/NAME),files=len(evidence))))
