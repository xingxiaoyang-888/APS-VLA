def check(value, label):
    if not value:
        raise ValueError(label)

def digest(data):
    return hashlib.sha256(data).hexdigest()

def sig(value):
    return digest(json.dumps(value, sort_keys=True, ensure_ascii=False, separators=(',', ':'), allow_nan=False).encode())

def raw(rel):
    path = Path(rel)
    if path.is_absolute():
        path = path.relative_to(ROOT)
    rel = path.as_posix()
    check('..' not in path.parts and 'state.json' != path.name and 'sealed' not in rel.lower()
          and path.suffix.lower() not in {'.pt', '.pth', '.bin', '.ckpt', '.safetensors', '.npz', '.init', '.pruned_init'}
          and not any(p.lower() in {'weights','checkpoints'} for p in path.parts)
          and '/downstream/final/' not in rel.lower(), 'forbidden read')
    data = (ROOT / rel).read_bytes()
    sha = digest(data)
    check(rel not in READ or READ[rel] == sha, 'evidence changed during reads: ' + rel)
    READ[rel] = sha
    return data

def read(rel):
    return json.loads(raw(rel))

def verify(rel, sha):
    check(digest(raw(rel)) == sha, 'SHA mismatch: ' + str(rel))

def lines(rel):
    data = raw(rel)
    check(data and data.endswith(b'\n'), 'unfinished journal: ' + str(rel))
    rows = [json.loads(line) for line in data.splitlines()]
    check(all(isinstance(row, dict) for row in rows), 'non-object journal row')
    return rows

def near(a, b, label='numeric evidence mismatch'):
    check(isinstance(a, (int, float)) and math.isfinite(a) and math.isclose(a, b, rel_tol=1e-7, abs_tol=1e-9), label)
