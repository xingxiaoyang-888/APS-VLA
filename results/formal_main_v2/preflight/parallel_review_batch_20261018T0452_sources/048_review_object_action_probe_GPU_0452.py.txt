"""Root recomputes capture-only engineering evidence; no scientific trial."""
from pathlib import Path
from datetime import datetime, timezone
import ast
import hashlib
import json
import sys
import textwrap
import numpy as np

ROOT=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(ROOT/'scripts'))
import formal_supervisor as f

BASE='results/formal_main_v2/'
RUN=BASE+'engineering/object_action_range_probe_v1/3023495/'
NAME=BASE+'preflight/object_action_range_probe_GPU_root_0452.json'
PULL=BASE+'pulls/20261018T052537Z_211f20566eec/receipt.json'
REG='configs/experiments/formal_main_v2_object_action_range_probe_v1.json'
INV=BASE+'invocations/libero_object_action_range_probe_v1_attempt01.json'
CPU=BASE+'preflight/object_action_range_probe_v1_CPU_root_0452.json'


def main():
    assert not (ROOT/NAME).exists()
    evidence={}
    def read(name):
        data=(ROOT/name).read_bytes();evidence[name]=hashlib.sha256(data).hexdigest()
        return json.loads(data)
    reg,inv,cpu,pull=(read(n) for n in (REG,INV,CPU,PULL))
    assert evidence[REG]==inv['registration_sha256']=='c87c76f8a3175bd760551c530f65990d08a88ab6a41d2771b44120e9cea25a3d'
    assert evidence[INV]=='726db906988c62cdbffd140a455355082459e6b70d7cbf8ffea7f7e9a45a4b23'
    assert evidence[CPU]==inv['root_review_sha256']=='79bcd5ddf80ef72619a0a1d7d95c8a0838de5876df011d532958ffe2dacd2a02'
    assert pull['status']=='verified_nonweight_snapshot' and pull['requested_job_ids']==['3023495']
    assert not pull['weights_copied'] and not pull['sealed_initial_states_copied']
    for row in pull['files']:
        assert f.sha(ROOT/row['relative'])==row['sha256'],row['relative']
        assert (ROOT/row['relative']).stat().st_size==row['bytes']
        evidence[row['relative']]=row['sha256']
    account=pull['accounting_after']['jobs']['3023495']
    assert account['status']=='COMPLETED' and account['exit_code']=='0:0' and account['elapsed_seconds']==263
    assert 'gres/gpu=1' in account['allocation']
    completion,cap,controller,inputs,job=(read(n) for n in (RUN+'completion.json',RUN+'captures.json',RUN+'controller.json',RUN+'inputs.json',BASE+'job_receipts/3023495.json'))
    assert completion['status']=='captured_pending_root_review' and completion['exit_code']==0 and completion['error'] is None
    assert completion['source_inputs_unchanged'] and not completion['scientific_complete']
    assert not completion['original_results_replaced'] and not completion['final_states_read'] and not completion['weights_copied']
    assert completion['candidate_sha256']==inv['entrypoint_sha256']==cpu['candidate_sha256']
    assert job['exit_code']==0 and job['invocation_sha256']==evidence[INV]
    assert job['watched_inputs_verified_before'] and job['watched_inputs_verified_after']
    assert completion['wall_seconds']<900 and job['wall_seconds']<900
    assert completion['result']['runtime']==reg['fixed_identity']['runtime']
    assert completion['result']['base_checkpoint_unchanged'] and completion['result']['model_parameters_unchanged']
    assert completion['result']['dense_identity']==reg['fixed_identity']['dense_deployment_identity']
    assert completion['result']['diagnostic']==cap
    assert completion['result']['controller_evidence_sha256']==evidence[RUN+'controller.json']
    for local,row in cpu['deployment_files'].items():
        assert f.sha(ROOT/local)==row['sha256'],local
        assert inv['watched_files'][row['server']]==row['sha256']
        evidence[local]=row['sha256']
    for path,h in inputs['watched_files'].items():
        assert completion['read_evidence_sha256'][path]==h,path
    source=read(reg['fixed_identity']['source_inventory'])
    assert len(source['files'])==173
    for rel,h in source['files'].items():
        assert f.sha(ROOT/rel)==h
        assert completion['read_evidence_sha256'][inv['source_root']+'/'+rel]==h
    raw_path=reg['original_evidence']['raw_ledger'];raw=(ROOT/raw_path).read_bytes()
    evidence[raw_path]=hashlib.sha256(raw).hexdigest()
    assert evidence[raw_path]==reg['read_evidence_sha256'][raw_path]
    rows=[json.loads(line) for line in raw.splitlines()]
    original=[r for r in rows if r.get('event')=='result' and r.get('trial_id')==reg['original_evidence']['trial_id']]
    assert len(original)==1;original=original[0]
    assert original['status']=='policy_error' and len(original['executed_actions'])==56 and len(original['chunks'])==7
    events_raw=(ROOT/(RUN+'events.jsonl')).read_bytes()
    assert hashlib.sha256(events_raw).hexdigest()==completion['events_sha256']
    assert len(events_raw)==completion['events_bytes'] and events_raw.endswith(b'\n')
    events=[json.loads(x) for x in events_raw.splitlines()]
    assert [x['seq'] for x in events]==list(range(len(events))) and len(events)==155
    assert all(x['job_id']=='3023495' for x in events)
    def of(name):return [r for r in events if r['event']==name]
    assert not of('probe_failure')
    starts,ends=of('environment_step_start'),of('environment_step_end')
    assert len(starts)==len(ends)==66
    expected=[[0.0]*6+[-1.0]]*10+original['executed_actions']
    for i,(s,e,a) in enumerate(zip(starts,ends,expected)):
        kind,index=('wait',i) if i<10 else ('recorded_control',i-10)
        assert s['step_kind']==e['step_kind']==kind and s['index']==e['index']==index
        assert s['action']==a and e['steps_confirmed']==i+1 and e['seq']==s['seq']+1
    checks=of('observation_checkpoint');assert len(checks)==7
    for actual,expected in zip(checks,reg['reconstruction']['mandatory_exact_checkpoints']):
        assert actual['control_step']==expected['control_step'] and actual['equal'] is True
        assert actual['actual_sha256']==actual['expected_sha256']==expected['observation_sha256']
    reconstruction=of('reconstruction_complete');assert len(reconstruction)==1
    assert reconstruction[0]['confirmed_environment_steps']==66 and reconstruction[0]['recorded_control_steps']==56
    assert reconstruction[0]['original_t56_observation_available'] is False
    assert cap['observation_sha256']==reconstruction[0]['new_t56_observation_sha256']
    start=of('capture_start');finish=of('capture_complete')
    assert [s['path'] for s in start]==[e['path'] for e in finish]==['official_native_dense','frozen_token_dense']
    assert starts[-1]['seq']<reconstruction[0]['seq']<start[0]['seq']
    assert start[0]['input_rng_sha256']==start[1]['input_rng_sha256']==cap['input_rng_sha256']
    assert start[0]['observation_sha256']==start[1]['observation_sha256']==cap['observation_sha256']
    assert start[0]['instruction_sha256']==start[1]['instruction_sha256']
    arrays={}
    for name in ('official_native_dense','frozen_token_dense'):
        c=cap['captures'][name]
        assert c==next(e['capture'] for e in finish if e['path']==name)
        assert c['calls']==dict(vision=1,vla_backbone=1,action_head=2 if name=='official_native_dense' else 1)
        arrays[name]={}
        for layer,item in c['arrays'].items():
            a=np.asarray(item['values'],dtype=item['dtype'])
            assert list(a.shape)==item['shape']==[8,7] and np.isfinite(a).all()
            assert hashlib.sha256(np.ascontiguousarray(a).tobytes()).hexdigest()==item['bytes_sha256']
            arrays[name][layer]=a
        post=arrays[name]['postprocessed']
        violating=[dict(row=int(i),column=int(j),value=float(post[i,j])) for i,j in np.argwhere(np.abs(post)>1.00001)]
        assert violating==c['violating_components'] and bool(violating)==c['frozen_guard_rejects']
    for layer in ('normalized','physical','postprocessed'):
        a,b=(arrays[n][layer] for n in ('official_native_dense','frozen_token_dense'))
        assert cap['comparisons'][layer]==dict(exact_equal=bool(np.array_equal(a,b)),maximum_absolute_difference=float(np.max(np.abs(a-b))))
        assert np.array_equal(a,b)
    measured=cap['captures']['frozen_token_dense']['violating_components']
    assert measured==[dict(row=7,column=2,value=1.010742197890625)]
    assert cap['total_VLA_forwards']==2 and cap['future_head_calls']==cap['DIE_calls']==cap['env_steps_after_t56']==0
    assert cap['success_collected'] is False and cap['original_failed_values_available'] is False
    assert controller['controller_type']=='robosuite.controllers.osc.OperationalSpaceController'
    assert controller['configured_control_delta'] is True and controller['actual_use_delta'] is True
    assert controller['set_goal_dispatches_to_bound_scale_action'] is True and controller['captured_action_applied'] is False
    assert controller['limits']['input_min']==[-1.]*6 and controller['limits']['input_max']==[1.]*6
    for name,relative in [('scale_action','controllers/base_controller.py'),('set_goal','controllers/osc.py')]:
        m=controller['methods'][name];expected=reg['actual_controller_evidence']['installed_source_reference_from_root_diagnosis'][relative]
        assert m['path']==expected['path'] and m['sha256']==expected['sha256']
        assert completion['read_evidence_sha256'][m['path']]==m['sha256']
    scale=ast.parse(textwrap.dedent(controller['methods']['scale_action']['source']))
    assert any(isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute) and n.func.attr=='clip' for n in ast.walk(scale))
    goals=ast.parse(textwrap.dedent(controller['methods']['set_goal']['source']))
    assert any(isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute) and n.func.attr=='scale_action' for n in ast.walk(goals))
    assert len(of('asset_metadata_compat_load_complete'))==1
    restored=of('asset_metadata_compat_exit');assert len(restored)==1 and not restored[0]['runner_raised']
    for key in ('original_loader_restored','directory_verifier_unchanged','binding_intact','original_loader_code_unchanged','directory_verifier_code_unchanged','frozen_source_and_inputs_unchanged'):
        assert restored[0][key] is True
    assert restored[0]['adapted_calls_attempted']==restored[0]['adapted_calls_completed']==1 and restored[0]['passthrough_calls']==0
    evidence['.local/review_object_action_probe_GPU_0452.py']=f.sha(ROOT/'.local/review_object_action_probe_GPU_0452.py')
    value=dict(schema='formal_main_v2',kind='root_bounded_action_range_GPU_review',status='passed_capture_only_diagnosis',
        reviewed_utc=datetime.now(timezone.utc).isoformat(),job_id='3023495',pull_receipt=PULL,
        invocation=INV,invocation_sha256=evidence[INV],completion=RUN+'completion.json',
        seven_exact_prefix_checkpoints=True,wait_steps=10,recorded_control_steps=56,environment_steps_total=66,
        VLA_forwards_total=2,extra_native_head_recaptures=1,environment_steps_after_t56=0,
        layers_exact_equal=['normalized','physical','postprocessed'],measured_violations=measured,
        actual_controller_use_delta=True,actual_bound_controller_clips_input=True,
        interpretation='The reconstructed observation gives exactly the same official/native and frozen/token actions. The CTI precheck rejects the finite1.010742 component that the bound official controller would clip. This is an execution-contract discrepancy, not a token-policy discrepancy in this captured case.',
        limitations=['The original failed eighth chunk and original t56 observation were not recorded; new values describe this exact-prefix reconstruction.',
            'No captured action was applied; no subsequent task success or replacement of the original failure is measured.',
            'This does not establish future-supervision benefits or release final evaluation.'],
        original_600_terminal_records_unchanged=True,new_trials=0,new_success_outcomes=0,
        allocated_GPU_seconds=263,cost_included_in_global_snapshot=True,all_173_frozen_source_unchanged=True,
        requires_independent_GPU_review=True,uniform_engineering_correction_not_yet_registered=True,
        new_rollouts_held=True,final_states_opened=False,weights_copied=False,evidence_sha256=evidence)
    f.write(ROOT/NAME,value)
    print(json.dumps(dict(receipt=NAME,sha256=f.sha(ROOT/NAME),evidence_files=len(evidence),measured_violations=measured)))


if __name__=='__main__':main()
