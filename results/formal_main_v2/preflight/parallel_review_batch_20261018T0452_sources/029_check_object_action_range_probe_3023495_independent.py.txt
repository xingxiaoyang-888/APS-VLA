def need(value, message):
    if not value:
        raise ValueError(message)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def sig(value):
    return sha(json.dumps(value, sort_keys=True, separators=(",", ":"),
                          ensure_ascii=False, allow_nan=False).encode())


class Evidence:
    def __init__(self):
        self.files = {}

    def path(self, relative):
        need(isinstance(relative, str), "non-string evidence path")
        pure = PurePosixPath(relative)
        need(not pure.is_absolute() and "\\" not in relative and ":" not in relative
             and ".." not in pure.parts and str(pure) == relative, "unsafe evidence path")
        low = relative.lower()
        if relative != PANEL and pure.suffix.lower() not in {".py", ".pyc", ".pyo"}:
            need(not any(x in low for x in ("init_state", "initial_state", "final_state", "test_state"))
                 and not any(x in pure.parts for x in ("test", "sealed", "sealed_final")), "state values prohibited")
        need(pure.suffix.lower() not in {".pt", ".pth", ".bin", ".safetensors", ".ckpt"}, "weight prohibited")
        path = ROOT / relative
        need(path.is_file() and path.resolve() == path, "evidence missing/alias: " + relative)
        return path

    def raw(self, relative, expected=None):
        raw = self.path(relative).read_bytes()
        digest = sha(raw)
        need(expected is None or digest == expected, "SHA mismatch: " + relative)
        need(relative not in self.files or self.files[relative] == digest, "changed between reads: " + relative)
        self.files[relative] = digest
        return raw

    def js(self, relative, expected=None):
        return json.loads(self.raw(relative, expected))

    def unchanged(self):
        for path, expected in list(self.files.items()):
            self.raw(path, expected)


def unpack_array(record):
    need(record["shape"] == [8, 7] and record["dtype"] in ("float32", "float64"), "array shape/type")
    value = np.array(record["values"], dtype=record["dtype"])
    need(value.shape == (8, 7) and np.isfinite(value).all(), "array not finite 8x7")
    need(sha(value.tobytes(order="C")) == record["bytes_sha256"], "array bytes SHA mismatch")
    return value
