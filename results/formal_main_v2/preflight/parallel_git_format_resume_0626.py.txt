"""Resume only Git delivery after two exact immutable-source EOF findings.

Keep the failed helper, batch and 502-file manifest unchanged. Re-run the pinned
archive checker on an extended manifest and preserve the original test bytes.
No SSH, scientific job, state update, source normalization or commit occurs.
"""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import runpy
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[3]
BASE = "results/formal_main_v2/preflight/"
SELF = BASE + "parallel_git_format_resume_0626.py.txt"
OLD = ".local/stage_verify_parallel_0626gpu_candidate.py"
OLD_SHA = "64a068601263c3eac6171824fa699c634ddd371800b6c42315933621bad32019"
BATCH = BASE + "parallel_review_batch_20261018T0626gpu.json"
BATCH_SHA = "679910f98aa08697fa87fa3498191932949032383186327fb05a644989dc6db5"
REQUIRED = BASE + "parallel_review_batch_20261018T0626gpu_git_required.json"
REQUIRED_SHA = "1b5bf054fbcb3aed2a254f7fe02398005be5f3a7b427dad97d8107cf61b2cc4a"
EXTENDED = BASE + "parallel_review_batch_20261018T0626gpu_git_required_v2.json"
RESOLUTION = BASE + "parallel_git_format_resolution_0626.json"
PROOF = BASE + "parallel_review_batch_20261018T0626gpu_git_delivery.json"
RAW_SOURCE = ".local/action_execution_gpu_acceptance_independent_0626/test_independent.py"
RAW_SHA = "6237f14904ad4a7a7535ea194e9e074163a9d27011e33dcd3a4e73fb3c9a457e"
FORMAT_PATHS = (
    BASE + "action_execution_gpu_candidate_sources_0626/043_test_independent.py.txt",
    BASE + "parallel_review_batch_20261018T0626gpu_sources/043_test_independent.py.txt",
)


def digest(data):
    return hashlib.sha256(data).hexdigest()


def git(*args):
    return subprocess.check_output(["git", *args], cwd=ROOT)


def row(name):
    path = ROOT / name
    assert path.is_file() and not path.is_symlink(), name
    data = path.read_bytes()
    return dict(path=name, sha256=digest(data), bytes=len(data))


def write_new(name, value):
    data = (json.dumps(value, indent=2, sort_keys=True) + "\n").encode()
    with (ROOT / name).open("xb") as stream:
        stream.write(data)
    return data


def check_format():
    source = (ROOT / RAW_SOURCE).read_bytes()
    assert digest(source) == RAW_SHA and len(source) == 16715
    for name in FORMAT_PATHS:
        assert (ROOT / name).read_bytes() == source == git("show", ":" + name), name
    expected = "".join(name + ":288: new blank line at EOF.\n" for name in FORMAT_PATHS).encode()
    check = subprocess.run(["git", "diff", "--cached", "--check"], cwd=ROOT,
                           stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    assert check.returncode == 2 and check.stdout == expected and check.stderr == b"", check
    rest = subprocess.run(["git", "diff", "--cached", "--check", "--", "."] +
                          [":(exclude,literal)" + name for name in FORMAT_PATHS], cwd=ROOT,
                          stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    assert rest.returncode == 0 and rest.stdout == rest.stderr == b"", rest
    return dict(returncode=check.returncode, stdout=check.stdout.decode(), stderr="",
                exact_path_line_kind_and_full_bytes_required=True, other_findings_allowed=False)


def main():
    assert ROOT == Path("F:/CTI-VLA") and Path(__file__).resolve() == ROOT / SELF
    for name, pin in ((OLD, OLD_SHA), (BATCH, BATCH_SHA), (REQUIRED, REQUIRED_SHA)):
        assert row(name)["sha256"] == pin, name
    original = runpy.run_path(str(ROOT / OLD), run_name="format_resume_read_only_helpers")
    assert git("rev-parse", "HEAD").decode().strip() == original["PARENT"]
    assert not git("diff", "--name-only").strip()
    for name, pin in original["PINS"].items():
        assert row(name)["sha256"] == pin and digest(git("show", ":" + name)) == pin
    batch = json.loads((ROOT / BATCH).read_bytes())
    original["validate_boundary_rows"](batch["reviewed_boundaries"])
    for name, pin in batch["evidence_sha256"].items():
        resolved = original["resolve_evidence_name"](name, pin)
        assert row(resolved)["sha256"] == pin, name
    original["check_source_version_resolution"]()
    for name, pin in original["PANEL_PINS"].items():
        assert row(name)["sha256"] == pin
        assert digest(git("show", "HEAD:" + name)) == digest(git("show", ":" + name)) == pin
        assert not git("status", "--porcelain", "--", name).strip()
    findings = check_format()
    for name in (EXTENDED, RESOLUTION, PROOF):
        assert not (ROOT / name).exists(), name
    old_required = json.loads((ROOT / REQUIRED).read_bytes())
    assert len(old_required["files"]) == 502
    resolution = dict(schema="formal_main_v2", kind="exact_immutable_archive_format_resolution",
        status="preserved_bytes_with_two_exact_EOF_findings", created_utc=datetime.now(timezone.utc).isoformat(),
        original_helper=row(OLD), original_batch=row(BATCH), original_required=row(REQUIRED),
        initial_failure=dict(exit_code=1, stage="git diff --cached --check after pinned archive check",
            exception="subprocess.CalledProcessError: return code 2", findings=findings,
            original_guard_report_not_saved=True, fresh_archive_verification_required=True, GPU_seconds=0),
        source=row(RAW_SOURCE), preserved_copies=[row(name) for name in FORMAT_PATHS],
        resume_helper=row(SELF), scientific_settings_changed=False, original_receipts_modified=False,
        archive_codec_and_checker_changed=False, sealed_states_opened=False, GPU_submitted=False)
    write_new(RESOLUTION, resolution)
    extra = [row(REQUIRED), row(SELF), row(RESOLUTION)]
    required = old_required["files"] + extra
    assert len(required) == len({r["path"] for r in required}) == 505
    write_new(EXTENDED, dict(schema="cti_nonweight_archive_stage_v1", kind="required_files",
                             files=sorted(required, key=lambda r: r["path"])))
    subprocess.run(["git", "add", "--", SELF, RESOLUTION, EXTENDED], cwd=ROOT, check=True)
    before, changes = git("ls-files", "--stage", "-z"), git("diff", "--cached", "--name-status", "-z")
    snapshot = original["read_json"](original["SNAPSHOT_NAME"], original["SNAPSHOT_SHA"])
    command = [sys.executable, "-B", "scripts/ops/check_nonweight_archive_stage.py", "--project-root", str(ROOT),
        "--snapshot", original["SNAPSHOT_NAME"], "--snapshot-sha256", original["SNAPSHOT_SHA"],
        "--required-files", EXTENDED]
    for name in sorted(r["path"] for r in snapshot["files"]):
        command.extend(["--raw-path", name])
    print(json.dumps(dict(status="pinned_archive_guard_running", required_count=505)), flush=True)
    process = subprocess.run(command, cwd=ROOT, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    assert process.returncode == 0, process.stderr.decode(errors="replace")
    report = json.loads(process.stdout)
    assert report["status"] == "verified_index_archive_delivery" and report["head"] == original["PARENT"]
    assert report["index_sha256"] == digest(before) and report["staged_changes_sha256"] == digest(changes)
    assert report["required_count"] == 505 and report["frozen_files_verified_in_worktree_and_index"] == 173
    assert len(report["raw_paths_restored_exactly"]) == 2 and report["gpu_seconds"] == 0
    assert before == git("ls-files", "--stage", "-z") and changes == git("diff", "--cached", "--name-status", "-z")
    assert check_format() == findings
    original["check_source_version_resolution"]()
    for name, pin in original["PANEL_PINS"].items():
        assert row(name)["sha256"] == digest(git("show", "HEAD:" + name)) == digest(git("show", ":" + name)) == pin
        assert not git("status", "--porcelain", "--", name).strip()
    proof = dict(schema="formal_main_v2", kind="root_git_index_nonweight_delivery", status="passed",
        reviewed_utc=datetime.now(timezone.utc).isoformat(), guard_result=report, guard_command=command,
        original_failed_helper=row(OLD), exact_format_resolution=row(RESOLUTION), resume_helper=row(SELF),
        original_batch=row(BATCH), original_required=row(REQUIRED), original_502_required_unchanged=True,
        unchanged_index_until_guard_return=True, scientific_settings_changed=False, no_weights_delivered=True,
        existing_panels_HEAD_index_worktree_equal_before_and_after=True, final_state_values_opened=False,
        training_boundaries=batch["reviewed_boundaries"], new_engineering_evidence=batch["new_engineering_evidence"],
        old_receipts_modified=False, GPU_submitted=False)
    data = write_new(PROOF, proof)
    subprocess.run(["git", "add", "--", PROOF], cwd=ROOT, check=True)
    old = set(before.split(b"\0")) - {b""}
    new = set(git("ls-files", "--stage", "-z").split(b"\0")) - {b""}
    assert not old - new and len(new - old) == 1
    assert next(iter(new - old)).split(b"\t", 1)[1].decode() == PROOF
    assert git("show", ":" + PROOF) == data and not git("diff", "--name-only").strip()
    check_format()
    print(json.dumps(dict(status="passed", proof=PROOF, sha256=digest(data), required_count=505,
                         frozen_files=173, restored_raw_paths=2, format_findings_preserved=2)), flush=True)


if __name__ == "__main__":
    main()
