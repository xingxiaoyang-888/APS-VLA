from pathlib import Path
from datetime import datetime, timezone
import json
import shlex
import subprocess
import sys

ROOT=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(ROOT/'scripts'))
import formal_supervisor as f

with f.supervisor_lock():
    state=f.read(f.STATE)
    jobs=f.refresh(state)
    selected=[j for j in jobs if j['status'] in f.ACTIVE and j['mode']=='train']
    bound=[{k:j.get(k) for k in ('job_id','suite','visual_tokens','training_seed','segment_end','run_root','status','gpus','node','elapsed_seconds')} for j in selected]
    code='''from pathlib import Path
from datetime import datetime,timezone
import json
jobs=JOBS
result={'schema':'formal_main_v2','observed_utc':datetime.now(timezone.utc).isoformat(),'provisional_live_read':True,'not_a_completion_receipt':True,'jobs':[]}
for j in jobs:
 r=dict(j,arms={},failures=[])
 root=Path(j['run_root'])
 for arm in ('A','B'):
  rows=[]
  for p in sorted((root/arm/'attempts').glob('*/cost_events.jsonl')):
   lines=p.read_bytes().split(b'\\n')[:-1]
   rows.extend(json.loads(line) for line in lines if line)
  own=[x for x in rows if str(x.get('job_id'))==j['job_id']]
  starts=[{k:x.get(k) for k in ('attempt_id','utc','resumed','completed_update','next_sample_cursor')} for x in own if x['event']=='attempt_start']
  updates=[x for x in own if x['event']=='update_end']
  commits=[x for x in own if x['event']=='checkpoint_commit']
  r['arms'][arm]={'this_job_attempt_starts':starts,'this_job_updates_completed_in_memory':updates[-1]['update'] if updates else None,'this_job_last_checkpoint_update':commits[-1]['completed_update'] if commits else None,'this_job_last_update_seconds':updates[-1].get('seconds') if updates else None,'last_event':own[-1]['event'] if own else None,'last_event_utc':own[-1].get('utc') if own else None}
  r['failures'].extend(x for x in own if x['event']=='attempt_end' and x.get('status')=='failed')
 result['jobs'].append(r)
print(json.dumps(result))
'''.replace('JOBS',repr(bound))
    response=subprocess.run(f.SSH+[shlex.quote(f.H+'/venv/bin/python')+' -B -'],input=code,capture_output=True,text=True,encoding='utf-8',check=True)
    live=json.loads(response.stdout)
    stamp=datetime.fromisoformat(live['observed_utc']).strftime('%Y%m%dT%H%M%SZ')
    relative='results/formal_main_v2/live/parallel_'+stamp+'.json'
    f.write(ROOT/relative,live)
    parallel=state['formal_main_progress']['parallel_execution']
    parallel['active_trainings']=[dict(j,not_a_completion_receipt=True) for j in live['jobs']]
    parallel['active_acceptances']=[j['job_id'] for j in jobs if j['mode']=='acceptance' and j['status'] in f.ACTIVE]
    parallel['latest_live_observation']={'path':relative,'sha256':f.sha(ROOT/relative),'observed_utc':live['observed_utc']}
    parallel['registered_requested_active_GPU_jobs']=sum(j['status'] in f.ACTIVE|{'SUBMITTED'} for j in jobs)
    parallel['actual_running_allocated_GPUs']=sum(j.get('gpus',0) for j in jobs if j['status']=='RUNNING')
    state['updated_utc']=datetime.now(timezone.utc).isoformat()
    f.write(f.STATE,state)
    print(json.dumps({'live':relative,'cost':state['current_round_cost'],'jobs':live['jobs']},ensure_ascii=False))
