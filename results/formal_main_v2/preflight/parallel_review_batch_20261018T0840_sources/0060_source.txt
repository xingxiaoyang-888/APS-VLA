def digest(data):
    return hashlib.sha256(data).hexdigest()


def raw(name, expected=None, size=None):
    assert isinstance(name, str) and '\\' not in name and ':' not in name
    p = PurePosixPath(name)
    assert not p.is_absolute() and '..' not in p.parts and p.as_posix() == name
    low = name.lower()
    assert not any(x in {'sealed', 'test', 'weights', 'checkpoints', 'final'} or 'sealed' in x for x in low.split('/'))
    assert p.suffix.lower() in {'.py', '.json', '.jsonl', '.txt', '.md', '.sh', '.slurm', '.toml', '.xml', '.log', '.out', '.err'}
    state_name = any(x in low for x in ('init_state', 'initial_state', 'final_state', 'test_state'))
    assert not state_name or name in PANELS or p.suffix == '.py'
    assert p.name.lower() not in {'state.json', 'panel.json'}
    path = ROOT / name
    for part in (path, *path.parents):
        assert not part.is_symlink() and not (getattr(part.lstat(), 'st_file_attributes', 0) & 0x400)
        if part == ROOT:
            break
    assert path.resolve().is_relative_to(ROOT)
    data = path.read_bytes()
    h = digest(data)
    assert expected is None or h == expected, (name, 'sha', h, expected)
    assert size is None or len(data) == size, (name, 'size')
    assert name not in seen or seen[name] == h, (name, 'changed during review')
    seen[name] = h
    sizes[name] = len(data)
    return data


def read(name, expected=None):
    return json.loads(raw(name, expected))
