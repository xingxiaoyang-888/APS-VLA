"""Root preserves current nonweight evidence and verifies exact Git index bytes.

Explicit finish step only. No SSH, job submission, model loading or sealed reads.
Existing large nonweight archives stay unchanged and remain covered by their
previous successful restore proof. New files are checked individually here.
"""
from pathlib import Path, PurePosixPath
from datetime import datetime,timezone,timedelta
import argparse,hashlib,json,subprocess,sys

ROOT=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(ROOT/'scripts'))
import formal_supervisor as f
BASE='results/formal_main_v2/'
BATCH=BASE+'preflight/parallel_review_batch_20261018T0840.json'
PROOF=BASE+'preflight/parallel_review_batch_20261018T0840_git_delivery.json'
PARENT='249cee0c64ba398627075f372e71f220c8dba26c'
JOBS=['3025080','3025196','3025205','3025581','3025841','3025845','3025874','3025875','3026207','3026219','3026224']
TERMINAL={'COMPLETED','FAILED','CANCELLED','TIMEOUT','OUT_OF_MEMORY','NODE_FAIL','PREEMPTED','BOOT_FAIL','DEADLINE','REVOKED'}
PANELS={
 'configs/manifests/formal_main_v2/libero_spatial/initial_states/development.json':'07718b648bfb767472506c48b0ef1837a5e6af992c2460e7567c275064809c91',
 'configs/manifests/formal_main_v2/libero_object/initial_states/development.json':'6b7e6dd9fb5b6351ee47c7bb30eab911e7c144d700cd88f7e5162641738b2d6a',
 'configs/manifests/formal_main_v2/libero_10/initial_states/development.json':'46a06087483151e36198167e7d4e5313c50b63172e09128d65a3650112f0a268',
 'configs/manifests/formal_main_v2/libero_goal/initial_states/development.json':'ce8e3d31ab8f7616c87dd4d4031af41825713907ac184c6ea5b01d9040985aa9'}

def git(*args):return subprocess.check_output(['git',*args],cwd=ROOT)
def sha(data):return hashlib.sha256(data).hexdigest()
def payload(name):
    safe(name);data=(ROOT/name).read_bytes()
    if name in PANELS:assert sha(data)==PANELS[name]
    return data
def ref(name):
    data=payload(name);return dict(path=name,sha256=sha(data),bytes=len(data))
def read(name):return json.loads(payload(name))
def save(name,data):
    path=ROOT/name;path.parent.mkdir(parents=True,exist_ok=True)
    with path.open('xb') as s:s.write(data)
def write(name,value):save(name,(json.dumps(value,indent=2,ensure_ascii=False)+'\n').encode())
def safe(name):
    assert isinstance(name,str) and '\\' not in name
    p=PurePosixPath(name);assert not p.is_absolute() and p.as_posix()==name and '..' not in p.parts and ':' not in name
    assert p.suffix.lower() in {'.json','.jsonl','.txt','.md','.py','.xml','.log','.out','.err','.diff','.slurm','.toml','.sh','.yaml','.yml'},name
    low=[x.lower() for x in p.parts]
    assert not set(low)&{'test','sealed','weights','checkpoints'} and not any('sealed' in x for x in low),name
    state_name=any(s in x for s in ('init_state','initial_state','final_state','test_state') for x in low)
    assert not state_name or p.suffix=='.py' or name in PANELS,name
    assert not (p.name=='panel.json' and 'final' in p.parts),name
    path=ROOT/name;assert path.is_file() and path.resolve().is_relative_to(ROOT)
    for q in (path,*path.parents):
        assert not q.is_symlink() and not (getattr(q.lstat(),'st_file_attributes',0)&0x400),name
        if q==ROOT:break
    assert path.stat().st_size<95*1024*1024,name

def main():
    ap=argparse.ArgumentParser();ap.add_argument('--source',action='append',default=[])
    ap.add_argument('--engineering-receipt',action='append',default=[])
    ap.add_argument('--summary',required=True,help='Root-written current substantive Chinese summary, relative UTF8 text file.')
    a=ap.parse_args()
    assert git('rev-parse','HEAD').decode().strip()==PARENT
    assert not git('diff','--cached','--name-only').strip()
    assert not (ROOT/BATCH).exists() and not (ROOT/PROOF).exists()
    inventory_name=BASE+'source/formal_main_v2_run_564be0995e.json'
    assert sha(payload(inventory_name))=='5baef5e7adc942b9d9c58eb7afe3faf69417f4d7939791997876652238a12926'
    inventory=read(inventory_name)
    assert len(inventory['files'])==173
    for n,h in inventory['files'].items():assert sha(payload(n))==h,n
    # Do not open existing sealed panels, model checkpoints or large archived raw trials.
    prior=json.loads(git('show','HEAD:results/experiment_supervisor/state.json'))
    original_state_sha=sha(payload('results/experiment_supervisor/state.json'))
    state=read('results/experiment_supervisor/state.json');p=state['formal_main_progress'];par=p['parallel_execution']
    assert p['trained_arms']==28 and p['completed_final_trials']==0 and not p['test_opened'] and not p['global_release_created']
    assert p['completed_development_trials']==1212 and state['pending_submission'] is None
    formal=[j for j in state['registered_jobs'] if j.get('experiment')=='formal_main_v2']
    assert len(formal)==len(state['registered_jobs'])
    active=[j for j in formal if not str(j.get('status','')).split() or str(j['status']).split()[0].rstrip('+') not in TERMINAL]
    assert len(active)<=8 and len({j['stage_key'] for j in active})==len(active)
    references=set();sources=set();expected={}
    def include(name,digest=None):
        actual=sha(payload(name));assert digest is None or digest==actual,name
        assert name not in expected or expected[name]==actual,name
        expected[name]=actual
        (sources if name.startswith('.local/') else references).add(name)
    for name in set(a.source)|{Path(__file__).relative_to(ROOT).as_posix(),a.summary}:include(name)
    boundaries=[];pulls=set()
    for jid in JOBS:
        op_name='.local/root_pull_review_2250_'+jid+'.json';op=read(op_name)
        rr=op['root_review'];ir=rr[:-5]+'_independent.json';review=read(rr);ind=read(ir)
        assert op['status']==review['status']==ind['status']=='passed'
        assert jid==op['job_id']==review['job_id']==ind['job_id']
        assert review['completed_by_arm']==dict.fromkeys(('A','B'),op['boundary'])
        assert ind['allowed_next_segment_end']==review['allowed_next_segment_end']==op['allowed_next']
        assert ind['read_evidence_sha256'][rr]==sha(payload(rr))==op['root_sha256']
        for name,h in ind['read_evidence_sha256'].items():
            include(name,h)
        include(rr);include(ir);pulls.add(op['pull']);include(op_name)
        log='.local/audit_training_segment_'+jid+'.log'
        if (ROOT/log).exists():include(log)
        boundaries.append({k:op[k] for k in ('job_id','suite','budget','seed','boundary','allowed_next','pair_gpu_seconds','latest_gpu_seconds')})
    failure=BASE+'preflight/cuda_startup_root_observation_0840.json'
    assert sha(payload(failure))=='a6bb5a481e3576f9fb87892d29e37eb6425891a5d55872a589561042384d8ef2'
    engineering=sorted(set(a.engineering_receipt)|{failure})
    for name in engineering:
        value=read(name);include(name)
        for key in ('read_evidence_sha256','evidence_sha256'):
            for n,h in value.get(key,{}).items():
                include(n,h)
        for row in value.get('deployment_files',[]):
            include(row['path'],row['sha256']);assert ref(row['path'])['bytes']==row['bytes']
    engineering_pull=BASE+'pulls/20261018T134013Z_1f65a8e31587/receipt.json'
    assert sha(payload(engineering_pull))=='b5ea9f9522071caefee87446fef0fd183bf4d98b40398daae1b5d465708064a6'
    ep=read(engineering_pull);assert ep['requested_job_ids']==['3030159']
    ej=ep['accounting_after']['jobs']['3030159']
    assert ej['status']==ej['raw_status']=='FAILED' and ej['exit_code']=='1:0' and ej['elapsed_seconds']==85
    assert 'gres/gpu=1' in ej['allocation'].split(',')
    all_pulls=pulls|set(read(failure)['current_terminal_pulls'])|{engineering_pull}
    assert len(pulls)==6 and len(all_pulls)==9
    for name in sorted(all_pulls):
        pull=read(name);include(name)
        assert pull['status']=='verified_nonweight_snapshot' and pull['weights_copied'] is False and pull['sealed_initial_states_copied'] is False
        selected={j['job_id'] for j in pull['selected_jobs']};assert selected==set(pull['requested_job_ids'])
        for job in pull['selected_jobs']:
            assert not job['active_overlapping_job_ids']
            if name==engineering_pull:assert job['mode']=='engineering' and job['job_id']=='3030159'
            else:assert job['mode']=='train'
            assert pull['accounting_after']['jobs'][job['job_id']]['status'] in TERMINAL
        for row in pull['files']:
            actual=ref(row['relative']);assert actual['sha256']==row['sha256'] and actual['bytes']==row['bytes'],row['relative']
            include(row['relative'],row['sha256'])
            if row.get('previous_version'):
                old=Path(row['previous_version']).relative_to(ROOT).as_posix()
                assert old==(Path(name).parent/'previous_versions'/row['relative']).as_posix()
                include(old,row['previous_sha256'])
        for filename in ('inventory_before.json','inventory_after.json'):
            n=(Path(name).parent/filename).as_posix();read(n);include(n)
    archive={}
    for ix,n in enumerate(sorted(sources),1):
        assert n.startswith('.local/')
        safe(n);path=ROOT/n;assert path.stat().st_size<20*1024*1024
        dest=BASE+f'preflight/parallel_review_batch_20261018T0840_sources/{ix:03d}_{path.name}.txt'
        source=payload(n);assert sha(source)==expected[n],n
        save(dest,source);archive[n]=ref(dest);include(dest,expected[n])
    old_ids={j['job_id'] for j in prior['registered_jobs']}
    newjobs=[j for j in state['registered_jobs'] if j['job_id'] not in old_ids]
    updates=sum(2*x.get('last_root_reviewed_common_update',0) for x in par['pair_tracking'].values())
    assert updates==157000
    # Freeze all new/changed payload names explicitly. No `.local` directory bulk add.
    allowed=('docs','configs/experiments/formal_main_v2_action_execution_compat_gpu_v2.json',
        'configs/experiments/formal_main_v2_node_exclusion_v1.json','scripts/ops/formal_scheduler_node_exclusion.py',
        'results/formal_main_v2','results/experiment_supervisor/state.json','results/slurm')
    changed=sorted(set(n for n in git('ls-files','--modified','--others','--exclude-standard','-z','--',*allowed).decode().split('\0') if n))
    for n in changed:safe(n)
    assert not set(changed)&set(inventory['files'])
    assert not any('/nonweight_archives/' in n for n in changed),'Historical archive delta is outside this delivery.'
    all_unstaged=set(n for n in git('ls-files','--modified','--others','--exclude-standard','-z').decode().split('\0') if n)
    assert set(changed)==all_unstaged,('unbound_changes',all_unstaged-set(changed))
    now=datetime.now(timezone.utc);local=now.astimezone(timezone(timedelta(hours=8)))
    summary=payload(a.summary).decode('utf8').strip()
    for name in ('正式主实验规划.md','完整实验计划.md','实验进度.md'):
        file=ROOT/'docs'/name;first,sep,body=payload('docs/'+name).partition(b'\n');assert sep
        add=f'\n{local:%Y-%m-%d %H:%M}：{summary}\n\n[本批边界、失败与成本凭据](../{BATCH})。\n\n'
        file.write_bytes(first+sep+add.encode()+body)
    p['root_and_independent_reviewed_arm_updates']=updates
    state['current_work']=summary
    state['updated_utc']=now.isoformat();state['updated_date']=local.strftime('%Y-%m-%d')
    value=dict(schema='formal_main_v2',kind='root_parallel_review_and_failure_preservation',status='passed',
        reviewed_utc=now.isoformat(),parent_commit=PARENT,completed_training_arms=28,expected_training_arms=72,
        cumulative_reviewed_arm_updates=updates,reviewed_boundaries=boundaries,
        cost_snapshot=state['current_round_cost'],cost_is_total_Slurm_allocation_not_additive_to_other_receipts=True,
        failed_startup_job_ids=['3028460','3028468','3028522'],failed_startup_GPU_seconds=226,
        failed_action_engineering_job_id='3030159',failed_action_engineering_GPU_seconds=85,
        failures_included_in_cost_snapshot=True,new_jobs=newjobs,active_jobs=active,
        completed_development_trials=1212,completed_final_trials=0,final_states_sealed=True,
        weights_copied=False,scientific_configuration_changed=False,audit_source_archive_map=archive,
        current_training_pulls=sorted(pulls),current_terminal_pulls=sorted(all_pulls),engineering_receipts=engineering,
        engineering_statuses={n:read(n).get('status','status_not_asserted_by_record') for n in engineering},
        evidence_sha256={n:sha(payload(n)) for n in sorted(references)},
        historical_large_archives_unchanged=True)
    write(BATCH,value)
    p['latest_incremental_review']={'path':BATCH,'sha256':sha(payload(BATCH))}
    assert sha(payload('results/experiment_supervisor/state.json'))==original_state_sha,'Supervisor state changed during delivery; refuse stale write.'
    (ROOT/'results/experiment_supervisor/state.json').write_bytes((json.dumps(state,ensure_ascii=False,indent=2)+'\n').encode())
    changed=sorted(set(n for n in git('ls-files','--modified','--others','--exclude-standard','-z','--',*allowed).decode().split('\0') if n))
    for n in changed:safe(n)
    stage_names=sorted(set(changed)|references-set(PANELS))
    rows=[ref(n) for n in stage_names]
    spec=ROOT/'.local/stage_0840.nul';assert not spec.exists()
    spec.write_bytes(b''.join((':(literal)'+n).encode()+b'\0' for n in stage_names))
    subprocess.run(['git','add','-f','--pathspec-from-file='+str(spec),'--pathspec-file-nul'],cwd=ROOT,check=True)
    actual=set(git('diff','--cached','--name-only','-z').decode().rstrip('\0').split('\0'))
    assert set(changed)<=actual<=set(stage_names) and not actual&set(inventory['files']) and not actual&set(PANELS)
    for row in rows:
        indexed=git('show',':'+row['path']);assert sha(indexed)==row['sha256'] and len(indexed)==row['bytes'],row['path']
    for name in references:
        expected_row=ref(name)
        assert expected_row['sha256']==expected[name],name
        indexed=git('show',':'+name)
        assert sha(indexed)==expected_row['sha256'] and len(indexed)==expected_row['bytes'],('missing_or_changed_index_evidence',name)
    for name,row in archive.items():
        assert row['sha256']==expected[name]
        indexed=git('show',':'+row['path']);assert sha(indexed)==row['sha256'] and len(indexed)==row['bytes'],name
    for n,h in inventory['files'].items():assert sha(git('show',':'+n))==h,n
    fmt=subprocess.run(['git','diff','--cached','--check'],cwd=ROOT,capture_output=True)
    assert fmt.returncode in (0,2)
    # Exact raw archival output may contain original whitespace. Record without rewriting.
    findings=fmt.stdout.decode('utf8')
    flagged=[line.rsplit(':',2)[0] for line in findings.splitlines() if ': trailing whitespace.' in line or ': new blank line at EOF.' in line]
    assert all(n in actual and n.startswith(BASE) and ('/engineering/' in n or '/preflight/' in n) for n in flagged),findings
    assert not fmt.stderr
    assert not git('diff','--name-only').strip()
    proof=dict(schema='formal_main_v2',kind='root_current_nonweight_index_delivery',status='passed',
        verified_utc=datetime.now(timezone.utc).isoformat(),parent_commit=PARENT,batch=ref(BATCH),
        files=rows,frozen173_worktree_and_index_verified=True,historical_large_archives_unchanged=True,
        immutable_reference_index_closure_verified=len(references),source_archive_index_closure_verified=len(archive),
        no_weights_delivered=True,final_states_opened=False,GPU_seconds=0,
        exact_archived_whitespace_check={'returncode':fmt.returncode,'stdout':findings,'stderr':fmt.stderr.decode()})
    write(PROOF,proof);subprocess.run(['git','add','--',PROOF],cwd=ROOT,check=True)
    assert git('show',':'+PROOF)==payload(PROOF)
    assert not git('diff','--name-only').strip()
    print(json.dumps({'status':'passed','proof':PROOF,'sha256':ref(PROOF)['sha256'],'staged_files':len(actual)+1,'completed_arms':28,'reviewed_updates':updates,'active_jobs':len(active),'cost':state['current_round_cost']}))

if __name__=='__main__':
    with f.supervisor_lock():main()
