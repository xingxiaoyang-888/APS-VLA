"""Finish the independent audit; preserve the first strict-stderr harness failure."""
from __future__ import annotations
import ast
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import sys

ROOT=Path(__file__).resolve().parents[2]
OUT=Path(__file__).resolve().parent
sys.dont_write_bytecode=True

def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def read(path):return json.loads(Path(path).read_bytes())
def write(path,value):
    with Path(path).open('x',encoding='utf8',newline='\n') as f:
        json.dump(value,f,ensure_ascii=False,indent=2,sort_keys=True);f.write('\n')

evidence=read(OUT/'inputs_before.json')
for ref,expected in evidence.items():
    assert sha(ROOT/ref)==expected,ref
candidate=read(ROOT/'.local/action_constructor_server311_0840/candidate_review.json')
adapter=ROOT/'.local/action_constructor_server311_0840/adapter.py'
original=ROOT/'scripts/ops/formal_action_execution_compat.py'
old,new=ast.parse(original.read_bytes()),ast.parse(adapter.read_bytes())
assert len(old.body)==len(new.body)
changed=[]
for a,b in zip(old.body,new.body):
    if ast.dump(a,include_attributes=False)!=ast.dump(b,include_attributes=False):
        if isinstance(a,ast.Assign):
            assert isinstance(b,ast.Assign) and a.targets[0].id==b.targets[0].id=='VERSION'
            changed.append('VERSION')
        else:
            assert isinstance(a,ast.FunctionDef) and isinstance(b,ast.FunctionDef)
            assert a.name==b.name=='_check_chunk_construction';changed.append(a.name)
assert changed==['VERSION','_check_chunk_construction']
runs=[]
for tag,count in [('python310',14),('python312',16)]:
    execution=read(OUT/(tag+'.execution.json'))
    assert execution['exit_code']==0
    for key in ('stdout','stderr'):
        assert sha(ROOT/execution[key])==execution[key+'_sha256']
    report=read(ROOT/execution['stdout'])
    warning=(ROOT/execution['stderr']).read_bytes()
    if tag=='python310':assert warning==b''
    else:
        expected=(str(ROOT/'.local/action_constructor_server311_0840/cpu_check.py')+
            ':57: DeprecationWarning: co_lnotab is deprecated, use co_lines instead.\r\n'+
            '  differences=[x for x in fields if hasattr(actual,x) and getattr(actual,x)!=getattr(class_only,x)]\r\n')
        assert warning==expected.encode('utf8')
        assert report['previous_verifier']['status']=='reproduced_failure'
        assert report['isolated_class_different_fields']==['co_code']
    assert report['status']=='passed_CPU_only' and report['checks_passed']==count
    assert len(set(report['checks']))==len(report['checks'])==count
    assert report['candidate_sha256']==sha(adapter) and report['original_sha256']==sha(original)
    assert report['actual_code_sha256']==report['full_source_code_sha256']
    assert report['source173_unchanged'] and report['no_models_loaded'] and report['no_environment_created']
    assert report['no_GPU_calls'] and not report['torch_imported'] and not report['tensorflow_imported'] and not report['final_states_opened']
    runs.append(dict(execution=execution,report=report))
remote_ref='results/formal_main_v2/preflight/action_constructor_server311_candidate_cpu_0840.json'
remote=read(ROOT/remote_ref);report=read(ROOT/remote_ref)
remote_report=json.loads(remote['stdout'])
assert remote['exit_code']==remote['SSH_exit_code']==remote['GPU_seconds']==0
assert remote['stderr']==remote['SSH_stderr']==''
assert remote_report['status']=='passed_CPU_only' and remote_report['checks_passed']==16
assert remote_report['python'].startswith('3.11.7 ')
assert remote_report['actual_code_sha256']==remote_report['full_source_code_sha256']
assert remote_report['candidate_sha256']==sha(adapter)
assert remote_report['previous_verifier']['status']=='reproduced_failure'
assert remote_report['source173_unchanged'] and remote_report['no_GPU_calls']
assert not remote_report['torch_imported'] and not remote_report['final_states_opened']
for ref,expected in evidence.items():assert sha(ROOT/ref)==expected,ref
write(OUT/'inputs_after.json',evidence)
for p in sorted(OUT.iterdir()):
    if p.is_file() and p.name not in ('independent_review.json','finalize.log'):
        evidence[p.relative_to(ROOT).as_posix()]=sha(p)
receipt=dict(schema='formal_main_v2',kind='action_constructor_identity_independent_CPU_review',
    version='server311-full-module-code-reference-independent-v1',status='passed_CPU_review_GPU_acceptance_pending',
    observed_utc=datetime.now(timezone.utc).isoformat(),candidate_sha256=sha(adapter),
    cpu_check_sha256=candidate['cpu_check']['sha256'],candidate_review_sha256=sha(ROOT/'.local/action_constructor_server311_0840/candidate_review.json'),
    exact_AST_changed_nodes=changed,all_other_adapter_AST_nodes_unchanged=True,
    strict_code_marshal_version=2,code_identity_fields_normalized_or_omitted=False,
    local_CPU_runs=runs,independently_rerun_checks=30,
    independent_harness_first_attempt=dict(status='stopped_after_both_CPU_processes_passed',
        reason='Overstrict empty-stderr assertion rejected Python 3.12 co_lnotab deprecation warning.',
        original_script='audit.py',original_log='audit.log',
        correction='This finalizer accepts only the exact warning bytes; original candidate, check and all first-run files preserved.',
        checks_rerun_after_harness_correction=False),
    root_server311_evidence=dict(path=remote_ref,sha256=sha(ROOT/remote_ref),checks_passed=16,
        independently_connected_to_server=False,report=remote_report),
    source173_and_original_adapter_unchanged=True,new_GPU_seconds=0,previous_failed_job='3030159',previous_failed_GPU_seconds_retained=85,
    SSH_calls=0,Slurm_submissions=0,state_or_Git_changed=False,sealed_states_opened=False,weights_opened=False,
    blockers=[],limitations=[
        'Independent execution used local Python 3.10.11 (14 checks) and 3.12.14 (16 checks).',
        'Server Python 3.11.7 CPU result (16 checks) is root-provided evidence, not independent SSH execution.',
        'No model, simulator or GPU acceptance ran; unchanged adapter behavior is supported by exact AST comparison, not a new rollout.',
        'Original registered adapter and 173 frozen files are unchanged; candidate is not yet a deployed repair.'],
    local_CPU_wall_seconds=sum(x['execution']['wall_seconds'] for x in runs),
    read_evidence_sha256=dict(sorted(evidence.items())))
write(OUT/'independent_review.json',receipt)
print(json.dumps(dict(receipt=(OUT/'independent_review.json').relative_to(ROOT).as_posix(),sha256=sha(OUT/'independent_review.json'),
    status=receipt['status'],independent_checks=30,root_server311_checks=16,evidence_count=len(evidence))))
