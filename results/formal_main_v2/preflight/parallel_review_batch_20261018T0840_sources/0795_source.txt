from pathlib import Path
from datetime import datetime,timezone
import json,shlex,subprocess,sys
ROOT=Path(__file__).resolve().parents[1];sys.path.insert(0,str(ROOT/'scripts'))
import formal_supervisor as f
TERMINAL={'COMPLETED','FAILED','CANCELLED','TIMEOUT','OUT_OF_MEMORY','NODE_FAIL','PREEMPTED','BOOT_FAIL','DEADLINE','REVOKED'}
with f.supervisor_lock():
    state=f.read(f.STATE);jobs=f.refresh(state)
    active=[j for j in jobs if not str(j.get('status','')).split() or str(j['status']).split()[0].rstrip('+') not in TERMINAL]
    eng=next(j for j in jobs if j['job_id']=='3030159')
    script='''from pathlib import Path
import json
p=Path(RUN)
events=p/'events.jsonl'
rows=[]
if events.exists():
 for line in events.read_bytes().split(b'\\n')[:-1]:
  if line: rows.append(json.loads(line))
completion=p/'completion.json'
c=json.loads(completion.read_bytes()) if completion.exists() else None
print(json.dumps(dict(files=sorted(x.name for x in p.iterdir()) if p.exists() else [],last_events=rows[-4:],completed=bool(c),completion_status=c.get('status') if c else None,error=c.get('error') if c else None)))
'''.replace('RUN',repr(eng['run_root']))
    result=subprocess.run(f.SSH+[shlex.quote(f.H+'/venv/bin/python')+' -B -'],input=script,capture_output=True,text=True,encoding='utf8',check=True)
    observation=json.loads(result.stdout)
    now=datetime.now(timezone.utc)
    name='results/formal_main_v2/live/project_'+now.strftime('%Y%m%dT%H%M%SZ')+'.json'
    value=dict(schema='formal_main_v2',kind='registered_jobs_and_engineering_live_observation',observed_utc=now.isoformat(),
        completed_training_arms=state['formal_main_progress']['trained_arms'],reviewed_arm_updates=sum(2*x.get('last_root_reviewed_common_update',0) for x in state['formal_main_progress']['parallel_execution']['pair_tracking'].values()),
        cost=state['current_round_cost'],active_jobs=active,engineering_job=eng,engineering_live=observation,
        live_only_not_root_acceptance=True,final_states_opened=False,weights_copied=False)
    assert not (ROOT/name).exists();f.write(ROOT/name,value)
    par=state['formal_main_progress']['parallel_execution'];par['registered_requested_active_GPU_jobs']=len(active)
    par['actual_running_allocated_GPUs']=sum(j.get('gpus',0) for j in active if j['status']=='RUNNING')
    par['latest_all_jobs_observation']={'path':name,'sha256':f.sha(ROOT/name)}
    state['updated_utc']=now.isoformat();f.write(f.STATE,state)
    print(json.dumps(dict(path=name,sha256=f.sha(ROOT/name),active=[{k:j.get(k) for k in ('job_id','mode','status','node')} for j in active],cost=value['cost'],engineering_live=observation),ensure_ascii=False))
