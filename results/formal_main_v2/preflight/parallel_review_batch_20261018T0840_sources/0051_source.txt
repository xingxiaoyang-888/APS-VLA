def require(value, message):
    if not value:

def digest(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for part in iter(lambda: stream.read(8 << 20), b''):
            h.update(part)
    return h.hexdigest()

    require(not any('sealed' in x or x in {'weights','checkpoints','test'} for x in low), 'forbidden path')
    require(p.suffix.lower() not in {'.pt','.pth','.bin','.ckpt','.safetensors','.npz','.npy','.pkl','.pickle','.init','.pruned_init'}, 'weight/state payload')
    require('final_registry' not in relative and p.name not in {'panel.json','state.json'}, 'state payload')
    require(not state or p.suffix == '.py' or relative in panels or relative in {CAND+'synthetic_shape_only_project_a2/'+x for x in panels}, 'unregistered initial state')
    target = ROOT / relative
    require(target.absolute() == target.resolve() and ROOT in target.resolve().parents, 'resolved path alias')

def record(relative, expected=None):
    path = safe(relative); actual = digest(path)
    require(expected is None or actual == expected, 'SHA differs: ' + relative)
    require(relative not in seen or seen[relative] == actual, 'changed read: ' + relative)
    seen[relative] = actual
    return path

def read(relative, expected=None):
    return json.loads(record(relative, expected).read_bytes())

def load(relative, name):
    path = record(relative)
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec); sys.modules[name] = module
    exec(compile(path.read_bytes(), str(path), 'exec'), module.__dict__)
    return module
