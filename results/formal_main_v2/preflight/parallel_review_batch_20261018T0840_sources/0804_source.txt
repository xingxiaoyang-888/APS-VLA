"""Rehash the independent review's complete input map before root bookkeeping."""
from pathlib import Path
import hashlib, json, subprocess, sys
root = Path(__file__).resolve().parents[1]
job = sys.argv[1]
assert job.isdigit()
op = json.loads((root / ('.local/root_pull_review_2250_' + job + '.json')).read_bytes())
name = op['root_review'].removesuffix('.json') + '_independent.json'
review = json.loads((root / name).read_bytes())
assert op['status'] == review['status'] == 'passed' and review['job_id'] == job
evidence = review['read_evidence_sha256']
assert evidence[op['root_review']] == op['root_sha256']
for path, checksum in evidence.items():
    assert hashlib.sha256((root / path).read_bytes()).hexdigest() == checksum, path
assert review['allowed_next_segment_end'] == op['allowed_next']
print(json.dumps(dict(status='full_evidence_map_rehashed',job=job,files=len(evidence),
    independent_sha256=hashlib.sha256((root/name).read_bytes()).hexdigest())),flush=True)
subprocess.run([sys.executable,'-B','.local/record_parallel_boundary.py',job],cwd=root,check=True)
