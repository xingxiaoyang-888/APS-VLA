"""Preserve observed startup failures and actionable state, without retrying."""
from pathlib import Path
from datetime import datetime,timezone
import json,sys
ROOT=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(ROOT/'scripts'))
import formal_supervisor as f
NAME='results/formal_main_v2/preflight/cuda_startup_root_observation_0840.json'
assert not (ROOT/NAME).exists()
with f.supervisor_lock():
    s=f.read(f.STATE);jobs=f.refresh(s);p=s['formal_main_progress'];par=p['parallel_execution']
    failed=[next(j for j in jobs if j['job_id']==x) for x in ('3028460','3028468','3028522')]
    assert all(j['status']=='FAILED' and j['exit_code']=='1:0' and j['node']=='gpu4020' for j in failed)
    assert sum(j['elapsed_seconds']*j['gpus'] for j in failed)==226
    probe='results/formal_main_v2/preflight/cuda_startup_server_paths_0840.json'
    assert f.sha(ROOT/probe)=='5e5afef2f7bc9e52db23df3f7cd5c7e3e7923c0b5bcd839eaddd84964f6617a4'
    result=f.read(ROOT/probe)['result']
    for row in result['pairs'][:2]: assert not row['checkpoint_root_exists'] and not row['checkpoint_root_files']
    old='results/formal_main_v2/libero_spatial/k64/seed23/training/segment_reviews/3025205.json'
    oldreview=f.read(ROOT/old)
    expected={row['path']:row['sha256'] for row in oldreview['server_resume_checks']}
    assert {row['path']:row['sha256'] for row in result['resume_checks']}==expected
    assert all(row['stable_during_hash'] for row in result['resume_checks'])
    pulls=['results/formal_main_v2/pulls/20261018T124922Z_99c13ac6c07f/receipt.json',
           'results/formal_main_v2/pulls/20261018T125523Z_f18157f47643/receipt.json']
    evidence={probe:f.sha(ROOT/probe),old:f.sha(ROOT/old)}
    for name in pulls:
        pull=f.read(ROOT/name)
        assert pull['status']=='verified_nonweight_snapshot' and not pull['weights_copied'] and not pull['sealed_initial_states_copied']
        evidence[name]=f.sha(ROOT/name)
        for row in pull['files']:
            assert f.sha(ROOT/row['relative'])==row['sha256'],row['relative']
            evidence[row['relative']]=row['sha256']
    active=[j for j in jobs if j['status'] in f.ACTIVE|{'SUBMITTED'}]
    assert len(active)<=8 and len({j['stage_key'] for j in active})==len(active)
    updates=sum(2*x.get('last_root_reviewed_common_update',0) for x in par['pair_tracking'].values())
    value=dict(schema='formal_main_v2',kind='root_CUDA_startup_failure_observation',
        status='diagnosed_before_training_recovery_adapter_pending_review',observed_utc=datetime.now(timezone.utc).isoformat(),
        failures=failed,failed_GPU_seconds=226,cost_included_in_total_not_additional=True,
        cost_snapshot=s['current_round_cost'],current_terminal_pulls=pulls,evidence_sha256=evidence,
        confirmed='Three registered requests rejected CUDA before model loading or any new training attempt.',
        cause_not_established='Some other registered requests on gpu4020 remain active; logs do not identify a specific GPU, driver cause or node-wide outage.',
        zero_start_failed_jobs=['3028460','3028468'],existing_resume_failed_job='3028522',
        existing_resume_checkpoint_SHA_unchanged=True,retry_submitted=False,
        recovery_required='Explicit typed startup-failure proof; preserve all failed history and costs. Object target500 remains uninitialized; Spatial target1500 must restore1000/cursor8000.',
        scheduling_correction_pending='Root-only future sbatch exclusion of gpu4020 under a reviewed recorded transport adapter; leave current healthy allocations unchanged.',
        active_job_ids=[j['job_id'] for j in active],completed_training_arms=p['trained_arms'],reviewed_arm_updates=updates,
        scientific_settings_changed=False,weights_downloaded=False,final_states_opened=False)
    f.write(ROOT/NAME,value)
    p['CUDA_startup_issue']=dict(status=value['status'],observation=NAME,sha256=f.sha(ROOT/NAME),
        failed_job_ids=[j['job_id'] for j in failed],failed_GPU_seconds=226,
        unverified_retries_forbidden=True,healthy_training_continues=True,
        next=value['recovery_required'],scheduler_next=value['scheduling_correction_pending'])
    p['root_and_independent_reviewed_arm_updates']=updates
    par['registered_requested_active_GPU_jobs']=len(active)
    par['actual_running_allocated_GPUs']=sum(j['gpus'] for j in active if j['status']=='RUNNING')
    s['current_work']=f'Formal28/72 complete; {updates} reviewed arm updates. {len(active)} healthy registered GPU requests. Three CUDA startup failures retained; node exclusion and typed recovery review pending. Additional scientific rollouts held; final sealed.'
    s['next_actions']=[value['scheduling_correction_pending'],value['recovery_required'],
        p['downstream_action_range_issue']['next'],
        'Six additional terminal500 pairs have stable nonweight pull and root review; independent audits remain required before continuation. All72+16+four-suite development required before final release.']
    s['updated_utc']=value['observed_utc'];f.write(f.STATE,s)
print(json.dumps(dict(receipt=NAME,sha256=f.sha(ROOT/NAME),active=len(active),reviewed=updates,cost=s['current_round_cost'])))
