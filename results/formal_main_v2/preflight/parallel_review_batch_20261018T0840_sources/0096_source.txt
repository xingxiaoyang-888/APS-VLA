def load(name):

    mod = importlib.util.module_from_spec(spec); spec.loader.exec_module(mod); return mod

def job_snapshot(job_id):
    row = json.loads((ROOT / f'results/formal_main_v2/submissions/{job_id}.json').read_bytes())
    case = m.CASES[job_id]
    pull = json.loads((ROOT / case['pull']).read_bytes())
    row.update(pull['accounting_after']['jobs'][job_id], gpus=1)
    return row

class VirtualCanonical(recovery.Evidence):
    """Only the not-yet-published canonical proof/repair is virtual."""
    def __init__(self, root, proof, repair=None):
        super().__init__(root); self.proof = proof; self.repair = repair
        self.ref = m.proof_relative(job_snapshot(proof['job_id']))
    def read(self, name):
        if name == self.ref:
            self.watch(name); return deepcopy(self.proof)
        if self.repair and name == self.repair[0]:
            self.watch(name); return deepcopy(self.repair[1])
        return super().read(name)
    def watch(self, name):
        if name == self.ref:
            raw = json.dumps(self.proof, sort_keys=True).encode()
        elif self.repair and name == self.repair[0]:
            raw = json.dumps(self.repair[1], sort_keys=True).encode()
        else:
            return super().watch(name)
        self.hashes[str(name)] = m.sha(raw); return self.hashes[str(name)]


def make_repair(proof):
    c = m.CASES[proof['job_id']]; job = job_snapshot(proof['job_id'])
    e = VirtualCanonical(ROOT, proof)
    value = dict(status='reviewed', stage_key=job['stage_key'], target_source_sha256=m.SOURCE_SHA,
        segment_end=c['target'], failed_job_ids=[job['job_id']],
        startup_failure_reviews={m.proof_relative(job):e.watch(m.proof_relative(job))},
        automatic_retry=False, new_attempt_limit=1)
    if proof['runtime_failure_evidence']:
        value['runtime_compatibility'] = proof['runtime_failure_evidence']['runtime_compatibility']
    digest = m.sha(json.dumps(value, sort_keys=True).encode())
    return f'.local/cuda_startup_recovery_0840/synthetic_repairs/{digest}.json', value


@pytest.mark.parametrize('job_id', list(m.CASES))
def test_real_failures_reconstruct_bundles_and_remain_failed(bundles, job_id):
    proof = bundles[job_id]; e = VirtualCanonical(ROOT, proof)
    checked = m.require_saved(e, job=job_snapshot(job_id))
    assert checked['status'] == 'verified_startup_failure'
    assert checked['ordinary_segment_completed'] is False
    assert checked['current_allocation_new_updates'] == checked['current_allocation_arm_initializations'] == 0
    assert checked['failed_allocation_gpu_seconds'] == m.CASES[job_id]['seconds']
    assert checked['allowed_next_segment_end'] == m.CASES[job_id]['target']


@pytest.mark.parametrize('job_id', list(m.CASES))
def test_recovery_plan_uses_original_target_explicit_resume_and_live_bytes(bundles, monkeypatch, job_id):
    proof = bundles[job_id]; repair = make_repair(proof)
    monkeypatch.setattr(recovery, 'Evidence', lambda root: VirtualCanonical(root, proof, repair))
    job = job_snapshot(job_id)
    jobs = [job]
    if job_id == '3028522':
        for old in ('3024716', '3025205'):
            row = json.loads((ROOT / f'results/formal_main_v2/submissions/{old}.json').read_bytes())
            pull = json.loads((ROOT / m.SPATIAL_PULL).read_bytes())
            row.update(pull['accounting_after']['jobs'][old], gpus=1); jobs.append(row)
    plan = recovery.build_recovery_plan(ROOT, dict(pending_submission=None), jobs,
        failed_job_id=job_id, repair_relative=repair[0])
    assert plan['args']['resume'] is True
    assert plan['args']['segment'] == m.CASES[job_id]['target']
    assert all(j['status'] == ('FAILED' if j['job_id'] == job_id else 'COMPLETED') for j in jobs)
    assert plan['proof']['next_sample_cursor'] == dict.fromkeys(('A','B'), 8000 if job_id == '3028522' else 0)

@pytest.mark.parametrize('mutation', ['later_job', 'pending', 'eight_unknown', 'wrong_target', 'omitted_failure'])
def test_duplicate_pending_capacity_and_bad_repair_fail_closed(bundles, monkeypatch, mutation):
    proof = bundles['3028460']; repair = make_repair(proof); job = job_snapshot('3028460')
    state = dict(pending_submission=None); jobs = [job]
    if mutation == 'later_job': jobs += [{**job, 'job_id':'synthetic_new', 'submitted_utc':'9999', 'status':'SUBMITTED'}]
    if mutation == 'pending': state['pending_submission'] = {'stage_key':job['stage_key']}
    if mutation == 'eight_unknown':
        jobs += [dict(experiment=m.SCHEMA, job_id='synthetic'+str(i), status='UNKNOWN', stage_key='other'+str(i)) for i in range(8)]
    if mutation == 'wrong_target': repair[1]['segment_end'] = 1000
    if mutation == 'omitted_failure': repair[1]['failed_job_ids'] = []
    monkeypatch.setattr(recovery, 'Evidence', lambda root: VirtualCanonical(root, proof, repair))
    with pytest.raises(ValueError): recovery.build_recovery_plan(ROOT,state,jobs,failed_job_id=job['job_id'],repair_relative=repair[0])

@pytest.mark.parametrize('arm', ['A', 'B'])
def test_frozen_training_resume_true_is_fresh_only_without_checkpoint_evidence(example, tmp_path, arm):
    # Actual frozen trainer, synthetic tiny tensors/CPU only. No real checkpoint
    # is opened and the original 5000-step scientific configuration is untouched.
    partial = run_toy(example, tmp_path, arm=arm, resume=True, stop_after_updates=2)
    assert partial['training_updates_completed'] == 2
    event_paths = list((tmp_path/'results/attempts').glob('*/cost_events.jsonl'))
    starts = [json.loads(line) for p in event_paths for line in p.read_text().splitlines()
              if json.loads(line)['event'] == 'attempt_start']
    assert len(starts) == 1 and starts[0]['resumed'] is False
    assert starts[0]['completed_update'] == starts[0]['next_sample_cursor'] == 0
    prior_bytes = {p:p.read_bytes() for p in (tmp_path/'weights').iterdir()}
    (tmp_path/'weights/resume.pt').unlink()  # only this isolated synthetic file
    with pytest.raises(ValueError, match='silently reuse'):
        run_toy(example, tmp_path, arm=arm, resume=True, stop_after_updates=2)
    assert all(p.read_bytes() == raw for p,raw in prior_bytes.items() if p.name != 'resume.pt')
    assert len(list((tmp_path/'results/attempts').iterdir())) == 1
