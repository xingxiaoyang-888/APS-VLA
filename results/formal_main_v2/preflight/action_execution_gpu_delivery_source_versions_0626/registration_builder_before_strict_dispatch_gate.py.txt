"""Root preregisters one bounded engineering attempt; never submits it."""
from pathlib import Path
from datetime import datetime, timezone
import importlib.util
import hashlib
import json
import subprocess
import xml.etree.ElementTree as ET

ROOT=Path(__file__).resolve().parents[1]
BASE='results/formal_main_v2/'
PUB=BASE+'preflight/action_execution_gpu_candidate_publication_0626.json'
REVIEW=BASE+'preflight/action_execution_gpu_candidate_root_0626.json'
REG='configs/experiments/formal_main_v2_action_execution_compat_gpu_v1.json'
DISPATCH='.local/action_execution_gpu_root_dispatch_candidate_0626/'
def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def read(name): return json.loads((ROOT/name).read_bytes())
def write_new(name,value):
    p=ROOT/name; p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('xb') as out: out.write((json.dumps(value,ensure_ascii=False,indent=2)+'\n').encode())
def ref(name): return dict(path=name,sha256=sha(ROOT/name))
def main():
    publication=read(PUB)
    assert publication['status']=='prepared_pending_root_registration'
    for name,digest in publication['evidence_sha256'].items(): assert sha(ROOT/name)==digest,name
    spec=importlib.util.spec_from_file_location('_root_gpu_dispatch_registration',ROOT/(DISPATCH+'launcher.py'))
    c=importlib.util.module_from_spec(spec); spec.loader.exec_module(c)
    out=c.OUT+'/source/'
    for n in ('probe.py','candidate_manifest.json','launcher.py','probe.slurm','submit.py'):
        assert out+n in publication['evidence_sha256'],n
    indname=BASE+'preflight/action_execution_gpu_candidate_independent_0626.json'
    ind=read(indname)
    assert ind['status']=='passed_CPU_independent_review_pending_registered_GPU' and not ind['bounded_CPU_blockers_remaining']
    manifest_ref=ref(out+'candidate_manifest.json'); manifest=read(manifest_ref['path'])
    adapter_ind=manifest['action_adapter_independent_review']
    identity=ind['candidate_identity']; assert identity['probe_sha256']==sha(ROOT/(out+'probe.py'))
    root_run=read('.local/action_execution_gpu_root_cpu_0626/run.json')
    suites=ET.parse(ROOT/'.local/action_execution_gpu_root_cpu_0626/tests.xml').getroot()
    count=sum(int(s.get('tests','0')) for s in suites.iter('testsuite'))
    assert count==53 and all(int(s.get(k,'0'))==0 for s in suites.iter('testsuite') for k in ('failures','errors','skipped'))
    assert root_run['exit_code']==0 and root_run['source_and_frozen_files_unchanged']
    reg=dict(schema='formal_main_v2',version=c.STAGE,status='registered_single_attempt',gpu_submission_authorized=True,
        candidate_manifest=manifest_ref,entrypoint=dict(path=c.P+'/'+c.ENTRY,sha256=identity['probe_sha256']),
        output_root=c.P+'/'+c.OUT,gpus=1,attempt=1,automatic_retry=False,suite_order=list(c.SUITES),method_order=list(c.METHODS),limits=c.LIMITS,
        cpu_reviews=dict(action_adapter_independent=dict(path=adapter_ind['path'],sha256=adapter_ind['sha256'],accepted_status=c.ADAPTER_INDEPENDENT_STATUS),
            candidate_independent=dict(**ref(indname),accepted_status=ind['status']),
            root_acceptance=dict(path=REVIEW,sha256='0'*64,accepted_status='passed_CPU_root_review_pending_registered_GPU')),
        purpose='One fixed16cell real-controller engineering acceptance; no scientific rollout or success/latency claim',
        final_states_opened=False,original_results_replaced=False,failed_Object_trial_automatically_replayed=False,
        stage_capacity_condition='Only a free slot within eight registered active requests; no preemption or ninth pending job')
    evidence={}
    def include(name,digest=None):
        c.lexical_relative(name); value=sha(ROOT/name)
        assert digest is None or value==digest,name
        evidence[name]=value
    for name in (PUB,indname,BASE+'preflight/action_execution_gpu_candidate_author_0626.json',
            BASE+'preflight/action_execution_gpu_dispatch_author_0626.json',BASE+'preflight/action_execution_gpu_dispatch_independent_0626.json',
            '.local/action_execution_gpu_root_cpu_0626/run.json','.local/action_execution_gpu_root_cpu_0626/tests.xml',
            '.local/action_execution_gpu_root_cpu_0626/tests.log','.local/register_action_execution_gpu_0626.py',
            c.LAUNCHER,c.WRAPPER,c.SUBMITTER): include(name)
    for row in ind['candidate_files'].values(): include(row['path'],row['sha256'])
    review=dict(schema='formal_main_v2',version=c.VERSION,kind='bounded_action_execution_GPU_candidate_root_CPU_review',
        status='passed_CPU_root_review_pending_registered_GPU',reviewed_utc=datetime.now(timezone.utc).isoformat(),
        candidate_identity=identity,candidate_files=ind['candidate_files'],candidate_independent_review=ref(indname),
        dispatch_contract=c.DISPATCH,gpu_registration_path=REG,gpu_registration_intent_sha256=c.registration_intent(reg),
        evidence_sha256=evidence,root_complete_CPU_tests=53,independent_candidate_CPU_tests=26,
        GPU_submitted=False,real_controller_acceptance=False,scientific_trials=0,engineering_cells=16,
        final_states_opened=False,original_results_replaced=False,scientific_settings_changed=False,weights_copied=False,
        scope='CPU and preregistration only; actual GPU and controller must still pass terminal/root/independent review before deployment or historical reuse',
        historical_failures_retained=True)
    inventory=read(c.INVENTORY)
    rows={}
    def need(name,digest,server=None):
        path=c.safe_local(ROOT,name); assert sha(path)==digest,name
        server=server or c.P+'/'+name
        row=dict(path=name,server=server,sha256=digest,bytes=path.stat().st_size)
        assert server not in rows or rows[server]==row
        rows[server]=row
    for name,digest in inventory['files'].items(): need(name,digest,c.S+'/'+name)
    for name,digest in manifest['read_evidence_sha256'].items(): need(name,digest)
    for name,digest in evidence.items(): need(name,digest)
    for cpu in (ind,review):
        for item in cpu['candidate_files'].values(): need(item['path'],item['sha256'])
    for name,item in reg['cpu_reviews'].items():
        if name!='root_acceptance': need(item['path'],item['sha256'])
    for name,digest in ((c.INVENTORY,c.INVENTORY_SHA),(manifest_ref['path'],manifest_ref['sha256']),
            (c.ENTRY,identity['probe_sha256']),(c.ADAPTER,c.ADAPTER_SHA),(c.SUPERVISOR,c.SUPERVISOR_SHA)): need(name,digest)
    def refs(value):
        if isinstance(value,dict):
            if 'path' in value and 'sha256' in value and not str(value['path']).startswith('/'): need(value['path'],value['sha256'])
            for child in value.values(): refs(child)
        elif isinstance(value,list):
            for child in value: refs(child)
    refs(manifest)
    for row in manifest['suites'].values():
        for contract in row.get('compatibility',{}).values():
            for name,digest in contract['files'].items(): need(name,digest)
    review['deployment_files']=[rows[name] for name in sorted(rows)]
    # Validate CPU dispatcher source identities using the separate review record.
    dispatch_ind=read(BASE+'preflight/action_execution_gpu_dispatch_independent_0626.json')
    assert dispatch_ind['status'].startswith('passed'), 'Independent dispatch acceptance required'
    review['dispatch_independent_review']=ref(BASE+'preflight/action_execution_gpu_dispatch_independent_0626.json')
    write_new(REVIEW,review)
    reg['cpu_reviews']['root_acceptance']['sha256']=sha(ROOT/REVIEW)
    write_new(REG,reg)
    checked=c.check_contract(ROOT,root_ref=ref(REVIEW),manifest_ref=manifest_ref,registration_ref=ref(REG))
    assert len(checked['watched_files'])==len(rows)+2
    command=['python','-B',str(ROOT/c.SUBMITTER),'--project-root',str(ROOT),
        '--root-review',REVIEW,'--root-review-sha256',sha(ROOT/REVIEW),
        '--manifest',manifest_ref['path'],'--manifest-sha256',manifest_ref['sha256'],
        '--gpu-registration',REG,'--gpu-registration-sha256',sha(ROOT/REG),'--check-only']
    run=subprocess.run(command,cwd=ROOT,capture_output=True,text=True)
    receipt=dict(schema='formal_main_v2',kind='root_real_local_dispatch_contract_check',
        status='passed_local_contract_no_submission' if run.returncode==0 else 'failed',
        command=command,exit_code=run.returncode,stdout=run.stdout,stderr=run.stderr,
        root_review=ref(REVIEW),registration=ref(REG),manifest=manifest_ref,watched_files=len(rows)+2,
        GPU_submitted=False,final_states_opened=False)
    write_new(BASE+'preflight/action_execution_gpu_registration_check_0626.json',receipt)
    assert run.returncode==0,run.stderr
    print(json.dumps(receipt,ensure_ascii=False))

if __name__=='__main__': main()
