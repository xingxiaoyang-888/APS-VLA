def local(name):
    path = PurePosixPath(name)
    assert not path.is_absolute() and ".." not in path.parts and "\\" not in name
    assert not any("sealed" in part.lower() or part.lower() in ("checkpoints", "weights", "test", "final") for part in path.parts)
    assert path.suffix.lower() not in (".pt", ".pth", ".bin", ".safetensors", ".npz", ".npy", ".pkl", ".init")
    assert name != "results/experiment_supervisor/state.json" and "accounting" not in path.parts
    if "initial_states" in path.parts:
        assert name == "configs/manifests/formal_main_v2/libero_object/initial_states/development.json"
    p = ROOT / name
    assert p.is_file() and not p.is_symlink()
    return p


def sha(name, expected=None):
    h = hashlib.sha256()
    with local(name).open("rb") as stream:
        for block in iter(lambda: stream.read(8 << 20), b""):
            h.update(block)
    value = h.hexdigest()
    assert expected is None or value == expected, (name, value, expected)
    assert name not in seen or seen[name] == value, name
    seen[name] = value
    return value


def read(name):
    sha(name)
    return json.loads(local(name).read_bytes())


def signature(value):
    return hashlib.sha256(json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()).hexdigest()


def immutable_json(name, value):
    with (ROOT / name).open("x", encoding="utf-8", newline="\n") as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write("\n")
