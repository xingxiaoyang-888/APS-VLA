def path(name):
    p = PurePosixPath(name)
    assert not p.is_absolute() and ".." not in p.parts and "\\" not in name
    assert not any("sealed" in s.lower() or s.lower() in ("weights", "checkpoints", "accounting") for s in p.parts)
    assert p.name != "state.json" and "/downstream/final/" not in name
    assert p.suffix.lower() not in (".pt", ".pth", ".bin", ".safetensors", ".pkl", ".init", ".npy", ".npz")
    if "initial_states" in p.parts:
        assert name == "configs/manifests/formal_main_v2/libero_object/initial_states/development.json"
    f = ROOT / name
    assert f.is_file() and not f.is_symlink()
    return f


def watch(name, expected=None):
    h = hashlib.sha256()
    with path(name).open("rb") as stream:
        for block in iter(lambda: stream.read(8 << 20), b""):
            h.update(block)
    digest = h.hexdigest()
    assert expected is None or digest == expected, (name, digest, expected)
    assert name not in seen or seen[name] == digest, name
    seen[name] = digest
    return digest


def read(name):
    watch(name)
    return json.loads(path(name).read_bytes())


def lines(name):
    watch(name)
    data = path(name).read_bytes()
    assert data.endswith(b"\n") and all(data.splitlines())
    return [json.loads(row) for row in data.splitlines()]


def sig(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, ensure_ascii=False,
        separators=(",", ":"), allow_nan=False).encode()).hexdigest()


def remote(name):
    assert name.startswith(SNAP) or name.startswith(P), name
    return name[len(SNAP):] if name.startswith(SNAP) else name[len(P):]


def finite(value):
    return isinstance(value, (int, float)) and not isinstance(value, bool) and math.isfinite(value)


def verify_map(mapping, old_files):
    # Only two known append/rebuilt outputs have historical aliases. A digest
    # is never skipped; its old bytes must still be in the pinned first archive.
    for name, digest in mapping.items():
        if name in (RUN + "/trials.jsonl", RUN + "/closed_loop_report.json") and digest == old_files[name]["sha256"]:
            saved = old_files[name]["archive"]
            watch(saved, digest)
            historical_resolutions[name + "@" + digest] = saved
        else:
            watch(name, digest)


def validate_trial_pairs(rows, panel, identities):
    assert len(rows) == 266 and rows[0]["event"] == rows[25]["event"] == "invocation"
    assert [r["slurm_job_id"] for r in rows if r["event"] == "invocation"] == JOBS
    by_job = {job: [] for job in JOBS}
    current = None
    for row in rows:
        if row["event"] == "invocation":
            current = row["slurm_job_id"]
        else:
            assert current in by_job
            by_job[current].append(row)
    assert [len(by_job[job]) for job in JOBS] == [24, 240]
    for events in by_job.values():
        assert [r["event"] for r in events] == [v for _ in range(len(events)//2) for v in ("start", "result")]
    events = by_job[JOBS[0]] + by_job[JOBS[1]]
    expected = [(state, method) for state in panel["states"] for method in METHODS][:132]
    results, initial_observations = [], {}
    calls = {job: Counter(policy=0, vision=0, packing=0, scorer=0, future_head=0, executed_control_actions=0) for job in JOBS}
    for index, (state, method) in enumerate(expected):
        start, result = events[2*index:2*index+2]
        job = JOBS[int(index >= 12)]
        shared = dict(suite="libero_object", split="development", method=method,
            state_trial_id=state["trial_id"], trial_id=state["trial_id"]+":"+method,
            initial_state_sha256=state["initial_state_sha256"], source_state_index=state["source_state_index"],
            task_id=state["task_id"], task_name=state["task_name"], environment_seed=0,
            runtime_seed=17, configured_wait_steps=10, action_chunk=8)
        assert all(start.get(k) == v and result.get(k) == v for k, v in shared.items())
        assert result["status"] == "completed" and type(result["success"]) is bool and result["wait_steps"] == 10
        control = result["control_steps"]
        assert type(control) is int and 0 < control <= 280
        assert (result["success"] and result["termination"] == "environment_success") or (not result["success"] and result["termination"] == "task_horizon" and control == 280)
        chunks = result["chunks"]
        assert [c["control_step"] for c in chunks] == list(range(0, control, 8)) and result["policy_calls"] == len(chunks)
        assert len(result["executed_actions"]) == control
        assert [action for chunk in chunks for action in chunk["environment_actions"]][:control] == result["executed_actions"]
        mid = identities[method]
        expected_meta = dict(schema="formal_main_v2", suite="libero_object", method=mid["method"], visual_tokens=mid["visual_tokens"],
            per_camera_budgets=[mid["visual_tokens"]//2]*2, policy_calls=1, vision_calls=1, packing_calls=1,
            scorer_calls=int(mid["method"] in ("A", "B")), future_head_calls=0, language_cache="disabled",
            diagnostic_indices_recorded=True, inputs_version="libero-oft-eval-rotate180-jpeg-lanczos3-crop09-boundsq99-v1",
            action_version="libero-oft-boundsq99-gripper-sign-invert-chunk8-v1", packing_version="oft-original-positions-v1")
        if mid["method"] in ("A", "B"):
            expected_meta.update(selector_checkpoint_sha256=mid["checkpoint_sha256"], training_seed=17, pair_id=mid["pair_id"])
        for chunk in chunks:
            meta = chunk["metadata"]
            assert {k: v for k, v in meta.items() if k != "retained_indices"} == expected_meta
            kept = meta["retained_indices"]
            assert kept == sorted(set(kept)) and len(kept) == mid["visual_tokens"]
            assert all(type(k) is int and 0 <= k < 512 for k in kept) and sum(k < 256 for k in kept) == mid["visual_tokens"]//2
            if mid["method"] == "uniform":
                dy, dx = {256:(2,1), 128:(2,2), 64:(4,2)}[mid["visual_tokens"]]
                grid = [y*16+x for y in range(0,16,dy) for x in range(0,16,dx)]
                assert kept == grid+[k+256 for k in grid]
            elif mid["method"] == "dense":
                assert kept == list(range(512))
            for key in ("normalized_actions", "environment_actions"):
                assert len(chunk[key]) == 8 and all(len(a) == 7 and all(finite(v) for v in a) for a in chunk[key])
                if key == "environment_actions":
                    assert all(abs(v) <= 1.00001 for a in chunk[key] for v in a)
            assert len(chunk["observation_sha256"]) == 64 and all(v in "0123456789abcdef" for v in chunk["observation_sha256"])
            for key in ("policy", "vision", "packing", "scorer", "future_head"):
                calls[job][key] += meta[key+"_calls"]
        first = chunks[0]["observation_sha256"]
        assert initial_observations.setdefault(state["trial_id"], first) == first
        assert all(finite(result[k]) and result[k] >= 0 for k in ("wall_seconds", "attempt_wall_seconds"))
        calls[job]["executed_control_actions"] += control
        results.append(result)
    assert len({r["trial_id"] for r in results}) == 132 and len(initial_observations) == 22
    assert [r["trial_id"] for r in results] == [s["trial_id"]+":"+m for s,m in expected]
    assert Counter(r["method"] for r in results) == dict.fromkeys(METHODS,22)
    return results, initial_observations, calls
