def local(name):
    path = PurePosixPath(name)
    assert not path.is_absolute() and ".." not in path.parts and "\\" not in name
    assert not any("sealed" in part.lower() or part.lower() in ("checkpoints", "weights", "test", "final", "accounting") for part in path.parts)
    assert path.suffix.lower() not in (".pt", ".pth", ".bin", ".safetensors", ".npz", ".npy", ".pkl", ".init")
    assert "initial_states" not in path.parts and name != "results/experiment_supervisor/state.json"
    result = ROOT / name
    assert result.is_file() and not result.is_symlink(), name
    return result


def watch(name, expected=None):
    digest = hashlib.sha256(local(name).read_bytes()).hexdigest()
    assert expected is None or digest == expected, (name, digest, expected)
    assert name not in seen or seen[name] == digest, name
    seen[name] = digest
    return digest


def read(name, expected=None):
    watch(name, expected)
    return json.loads(local(name).read_bytes())


def signature(value):
    return hashlib.sha256(json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()).hexdigest()


def relative(server_path):
    assert server_path.startswith(P), server_path
    return server_path[len(P):]


def validate_hard_artifact(path, expected, *, arm, update, windows, job_ids):
    value = read(path, expected)
    assert value["arm"] == arm and value["update"] == update and value["job_id"] in job_ids
    assert value["metric"] == "task_macro_hard_validation_teacher_l1"
    rows = value["observations"]
    assert [row["window_id"] for row in rows] == windows["validation_window_ids"]
    assert len(rows) == len({row["window_id"] for row in rows}) == value["windows"] == 165
    by_task = defaultdict(list)
    window_rows = {row["window_id"]: row for row in windows["windows"]}
    for row in rows:
        original = window_rows[row["window_id"]]
        assert original["split"] == "validation"
        assert (row["episode_id"], row["task_id"]) == (original["episode_id"], original["task_id"])
        indices = row["retained_indices"]
        assert len(indices) == len(set(indices)) == 128 and indices == sorted(indices)
        assert all(type(i) is int and 0 <= i < 512 for i in indices)
        assert sum(i < 256 for i in indices) == sum(i >= 256 for i in indices) == 64
        actions = row["normalized_actions"]
        assert len(actions) == 8 and all(len(a) == 7 and all(math.isfinite(x) for x in a) for a in actions)
        assert all(math.isfinite(row[k]) and row[k] >= 0 for k in ("teacher_l1", "demonstration_l1"))
        by_task[str(row["task_id"])].append(row)
    assert set(by_task) == set(map(str, range(10))) and value["tasks"] == 10
    per_task = {task: {"windows": len(items), "teacher_l1": sum(r["teacher_l1"] for r in items) / len(items),
        "demonstration_l1": sum(r["demonstration_l1"] for r in items) / len(items)} for task, items in sorted(by_task.items())}
    assert per_task == value["per_task"]
    teacher = sum(r["teacher_l1"] for r in per_task.values()) / 10
    expert = sum(r["demonstration_l1"] for r in per_task.values()) / 10
    assert teacher == value["validation_metric"] and expert == value["demonstration_l1"]
    assert sum(r["teacher_l1"] for r in rows) / 165 == value["window_weighted_teacher_l1_diagnostic"]
    return dict(update=update, task_macro_teacher_l1=teacher, task_macro_demonstration_l1=expert,
        windows=165, tasks=10, artifact=path, artifact_sha256=seen[path], per_task=per_task)
