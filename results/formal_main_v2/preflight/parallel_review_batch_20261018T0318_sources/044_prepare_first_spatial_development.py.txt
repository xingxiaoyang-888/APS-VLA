def local(path):
    name = PurePosixPath(path)
    assert not name.is_absolute() and ".." not in name.parts and "\\" not in path
    assert not any("sealed" in part.lower() or part.lower() in ("test", "final", "checkpoints", "weights") for part in name.parts)
    assert name.suffix.lower() not in (".pt", ".pth", ".bin", ".safetensors", ".npz", ".npy", ".pkl", ".init")
    assert "accounting" not in name.parts
    if "initial_states" in name.parts:
        assert path == "configs/manifests/formal_main_v2/libero_spatial/initial_states/development.json"
    value = ROOT / path
    assert value.is_file() and not value.is_symlink()
    return value


def sha(path):
    h = hashlib.sha256()
    with local(path).open("rb") as stream:
        for block in iter(lambda: stream.read(8 << 20), b""):
            h.update(block)
    return h.hexdigest()


def read(path):
    return json.loads(local(path).read_bytes())


def signature(value):
    return hashlib.sha256(json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()).hexdigest()


def existing_development(state):
    jobs = []
    for job in state["registered_jobs"]:
        if job.get("experiment") != "formal_main_v2":
            continue
        if job.get("suite") == "libero_spatial" and (job.get("mode") == "development" or "/downstream/" in str(job.get("run_root", ""))):
            jobs.append({key: job.get(key) for key in ("job_id", "mode", "status", "run_root", "invocation")})
    return dict(registered_downstream_jobs=jobs,
        execution_files=[p.relative_to(ROOT).as_posix() for p in (ROOT / "results/formal_main_v2/execution").glob("*spatial*")],
        invocation_files=[p.relative_to(ROOT).as_posix() for p in (ROOT / "results/formal_main_v2/invocations").glob("*spatial*development*")],
        runtime_directory_exists=(ROOT / "results/formal_main_v2/libero_spatial/downstream").exists())
