from pathlib import Path
from datetime import datetime,timezone
import ast,hashlib,json,shlex,subprocess,sys
root=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(root/'scripts'))
import formal_supervisor as f
base='results/formal_main_v2/'
run=base+'libero_object/downstream/development/8c843c1d4ee8383c/attempt01'
out=base+'preflight/object_action_range_diagnosis_0318.json'
assert not (root/out).exists()
rr=base+'downstream_reviews/3021844.json';r=f.read(root/rr)
assert r['status']=='passed' and r['job_id']=='3021844' and r['completed_trials']==600
ledger=root/run/'trials.jsonl';assert f.sha(ledger)==r['files'][run+'/trials.jsonl']
rows=[json.loads(line) for line in ledger.read_bytes().splitlines()]
errors=[x for x in rows if x.get('event')=='result' and x['status']=='policy_error']
assert len(errors)==1
x=errors[0]
assert x['method']=='dense' and x['task_id']==8 and x['source_state_index']==49 and x['control_steps']==56
assert x['policy_calls']==8 and len(x['chunks'])==7
assert x['error']=='environment action outside normalized controller [-1, 1] range'
cfg=f.read(root/run/'config.json');revision=cfg['source_verification']['openvla_oft']['revision']
assert revision=='e4287e94541f459edc4feabc4e181f537cd569a8'
repo=root/'external/openvla-oft'
assert subprocess.check_output(['git','-C',str(repo),'rev-parse','HEAD']).decode().strip()==revision
official={}
for name,functions in [('prismatic/extern/hf/modeling_prismatic.py',['_unnormalize_actions']),
                       ('experiments/robot/libero/run_libero_eval.py',['process_action','run_episode'])]:
    local=(repo/name).read_bytes()
    blob=subprocess.check_output(['git','-C',str(repo),'show',revision+':'+name])
    assert local.replace(b'\r\n',b'\n')==blob
    lines=blob.decode().splitlines();tree=ast.parse(blob)
    selected=[]
    for node in ast.walk(tree):
        if isinstance(node,(ast.FunctionDef,ast.AsyncFunctionDef)) and node.name in functions:
            selected.append(dict(name=node.name,line=node.lineno,end_line=node.end_lineno,source='\n'.join(lines[node.lineno-1:node.end_lineno])))
    assert {v['name'] for v in selected}==set(functions)
    official[name]=dict(revision=revision,git_blob_sha256=hashlib.sha256(blob).hexdigest(),local_file_sha256=hashlib.sha256(local).hexdigest(),
        local_line_endings_normalized_match_git_blob=True,functions_extracted_from_exact_git_blob=True,functions=selected)
runtime='src/cti_vla/evaluation/libero_runtime.py'
assert f.sha(root/runtime)==cfg['source_files'][runtime]
code="""from pathlib import Path
import ast,hashlib,json,importlib.util,importlib.metadata
spec=importlib.util.find_spec('robosuite');assert spec and spec.origin
package=Path(spec.origin).parent
files={}
for relative,names in [('controllers/base_controller.py',['scale_action']),('controllers/osc.py',['set_goal'])]:
 p=package/relative;data=p.read_bytes();tree=ast.parse(data);lines=data.decode().splitlines();functions=[]
 for node in ast.walk(tree):
  if isinstance(node,ast.FunctionDef) and node.name in names:
   functions.append(dict(name=node.name,line=node.lineno,end_line=node.end_lineno,source='\\n'.join(lines[node.lineno-1:node.end_lineno])))
 assert {v['name'] for v in functions}==set(names)
 files[relative]=dict(path=str(p),sha256=hashlib.sha256(data).hexdigest(),functions=functions)
print(json.dumps(dict(robosuite_version=importlib.metadata.version('robosuite'),files=files,read_only=True,model_loaded=False,GPU_used=False)))
"""
answer=subprocess.run(f.SSH+[shlex.quote(f.H+'/venv/bin/python')+' -B -'],input=code,
    capture_output=True,text=True,encoding='utf-8',check=True,timeout=60)
server=json.loads(answer.stdout)

assert 'np.clip(action, self.input_min, self.input_max)' in scale

           'The failing eighth returned chunk has no recorded action values. All56 executed actions came from the preceding seven valid chunks.',
           'Pinned official OFT unnormalization is an affine transform without clipping; its rollout passes processed actions directly to env.step without this additional range rejection.',
           'The installed robosuite controller scale_action clips input to its configured bounds internally. This source-only probe does not replay the failed observation.'],
    limitations=['Exact failed magnitude/component and prediction values cannot be recovered from this ledger.',
                 'This establishes an additional frozen CTI precheck versus the official execution path; it does not show whether the aborted official trajectory would succeed.',

    official_source_functions=official,installed_controller=server,remote_program=code,remote_returncode=answer.returncode,
    next='Retain the error and cost. Before any downstream runtime change, register a bounded equivalence/diagnostic plan against pinned official action handling and review implications for all methods, existing development identities, and O2A. No outcome-based rerun or clipping patch; final release stays blocked.',
    source_changes=False,training_changes=False,new_trials=0,GPU_seconds=0,weights_copied=False,final_states_opened=False,

        'configs/assets/runtime_sources.json','.local/diagnose_object_action_range_0318.py',
        base+'preflight/object_action_range_diagnosis_0318_failures/attempt01_source.py.txt',
        base+'preflight/object_action_range_diagnosis_0318_failures/attempt01_failure.json')})
