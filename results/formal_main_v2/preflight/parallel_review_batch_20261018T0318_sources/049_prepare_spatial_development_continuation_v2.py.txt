def local(name):
    assert isinstance(name, str) and name and "\\" not in name and ":" not in name, name
    path = PurePosixPath(name)
    assert not path.is_absolute() and path.as_posix() == name and ".." not in path.parts, name
    lower = [part.lower() for part in path.parts]
    assert not any("sealed" in part or part in ("checkpoints", "weights", "test", "final") for part in lower), name
    assert path.suffix.lower() not in (".pt", ".pth", ".bin", ".safetensors", ".npz", ".npy", ".pkl", ".init", ".h5", ".hdf5"), name
    assert name not in ("results/experiment_supervisor/state.json", BASE + "slurm_accounting.json"), name
    scientific_py = path.suffix == ".py" and path.parts[0] in ("src", "scripts", "tests")
    state_payload = any(stem in part for stem in ("init_state", "initial_state", "final_state", "test_state") for part in lower)
    assert not state_payload or scientific_py or name == PANEL, name
    value = ROOT / name
    assert value.is_file() and not value.is_symlink(), name
    for parent in value.parents:
        if parent == ROOT:
            break
        assert not parent.is_symlink(), name
    return value


def sha(name, expected=None):
    h = hashlib.sha256()
    with local(name).open("rb") as stream:
        for block in iter(lambda: stream.read(8 << 20), b""):
            h.update(block)
    value = h.hexdigest()
    assert expected is None or value == expected, (name, value, expected)
    assert name not in seen or seen[name] == value, name
    seen[name] = value
    return value


def read(name):
    digest = sha(name)
    data = local(name).read_bytes()
    assert hashlib.sha256(data).hexdigest() == digest, name
    return json.loads(data)


def signature(value):
    return hashlib.sha256(json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()).hexdigest()


def immutable_json(name, value):
    with (ROOT / name).open("x", encoding="utf-8", newline="\n") as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write("\n")


def parse_args():
    parser = argparse.ArgumentParser(description=__doc__)
    defaults = {
        "prior-review": BASE + "downstream_reviews/" + PRIOR_JOB + ".json",
        "independent-review": PRE + "spatial_first_development_real_independent_20261018.json",
        "archive": PRE + "spatial_first_development_raw_v1/receipt.json",
        "integration": PRE + "spatial_first_development_root_integration_20261018.json",
    }
    for flag, default in defaults.items():
        parser.add_argument("--" + flag, default=default)
        parser.add_argument("--" + flag + "-sha256", required=True)
    parser.add_argument("--additional-stop-after", type=int, default=REMAINING,
                        help="Root-chosen next consecutive trial count; default588, multiple of6, no automatic shrinking.")
    args = parser.parse_args()
    for key in defaults:
        assert re.fullmatch(r"[a-f0-9]{64}", getattr(args, key.replace("-", "_") + "_sha256"))
    assert 6 <= args.additional_stop_after <= REMAINING and args.additional_stop_after % 6 == 0
    return args
