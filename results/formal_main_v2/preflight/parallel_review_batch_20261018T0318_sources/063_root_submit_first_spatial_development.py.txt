"""Root-only first Spatial development submission through unchanged controller."""
from pathlib import Path
from types import SimpleNamespace
from datetime import datetime, timezone
import hashlib, json, shlex, subprocess, sys, time

root=Path(__file__).resolve().parents[1]
sys.path[:0]=[str(root/'scripts'),str(root/'scripts/ops')]
import formal_supervisor as f
import formal_downstream_supervisor as c
base='results/formal_main_v2/'
pre=base+'preflight/spatial_first_pair_development_'
regrel,planrel=pre+'registration_v1.json',pre+'plan_v1.json'
reviewrel,subrel,freshrel=pre+'root_review_v1.json',pre+'submission_root_v1.json',pre+'selected_server_v1.json'
assert all(not (root/n).exists() for n in (reviewrel,subrel,freshrel))
assert f.sha(root/regrel)=='060dc1fd7ec4ab3ec35db1bfb8b3a54f92b5fd23f6ae3b9fd88e453eae4a4438'
assert f.sha(root/planrel)=='e8c4f5e782501237de21197fc24533ea1e1698f310d7d2e8034c906348bed600'
reg,expected=f.read(root/regrel),f.read(root/planrel)
for name,checksum in reg['read_evidence_sha256'].items():
    assert f.sha(root/name)==checksum,name
manifest=f.read(root/reg['execution_manifest'])
panel=f.read(root/manifest['development_panel'])
all_ids=[s['trial_id']+':'+method for s in panel['states'] for method in reg['methods']]
assert len(all_ids)==len(set(all_ids))==600 and all_ids[:12]==reg['exact_first12_trial_ids']
assert manifest['environment']==dict(max_steps=220,action_chunk=8,wait_steps=10,environment_seed=0,runtime_seed=17)
assert expected['invocation']['resume'] is False and expected['invocation']['prior_job_ids']==[]
assert expected['invocation']['stop_after']==12 and expected['output_must_exist'] is False
assert not c.resume_prefix_inputs(expected['invocation'])
assert reg['resources']['time']=='04:00:00' and reg['resources']['gres']=='gpu:l40:1'
assert not (root/expected['output_relative']).exists() and not (root/expected['invocation_relative']).exists()
args=SimpleNamespace(command='submit',stage='development',execution_manifest=reg['execution_manifest'],
    source_inventory=base+'source/formal_main_v2_run_564be0995e.json',output_relative=None,
    freeze_directory=None,profile_panel=None,prior_review=None,collection_review=None,repair_record=None,
    resume=False,stop_after=12,expected_plan_sha256=reg['plan_sha256'],root_submit=True)
started=time.perf_counter()
with f.supervisor_lock():
    state=f.read(f.STATE);assert state['pending_submission'] is None
    jobs=f.refresh(state);live=[j for j in c.formal_jobs(state) if j['status'] not in c.TERMINAL]
    assert len(live)<8
    assert not any(j['run_root']==expected['invocation']['run_root'] for j in jobs)
    p=state['formal_main_progress']
    assert 'first_spatial_development' not in p
    assert p['completed_development_trials']==732 and not p['test_opened'] and not p['global_release_created']
    completed=p['completed_training_pairs']['libero_spatial_k128_seed17']
    assert completed['job_id']=='3021362' and completed['allowed_next_segment_end'] is None
    print(json.dumps(dict(status='root_spatial_plan_reconstruction_running',active_requests=len(live))),flush=True)
    rebuilt=c.plan(args,state,project=root)
    assert rebuilt==expected
    assert rebuilt['plan_sha256']=='353d4a16c3426c8f7a1836e389f936fd2a1b672286fc1bd43a2191b01ab84693'
    for name,checksum in reg['read_evidence_sha256'].items():
        assert f.sha(root/name)==checksum,name
    oldrel=reg['selected_server_review'];old=f.read(root/oldrel)
    assert old['status']=='passed' and old['job_id']=='3021362' and old['suite']=='libero_spatial'
    specs=old['selected'];assert len(specs)==2 and {x['arm'] for x in specs}=={'A','B'}
    assert specs==reg['selected_checkpoints']
    for spec in specs:
        row=next(x for x in manifest['methods'] if x['identity']['method']==spec['arm'])
        assert row['checkpoint']==spec['path'] and row['identity']['checkpoint_sha256']==spec['expected_sha256']
    assert not any(j['status'] not in c.TERMINAL and j['stage_key']==old['registered_terminal_job']['stage_key'] for j in jobs)
    code=old['probe_program'];assert hashlib.sha256(code.encode()).hexdigest()==old['probe_program_sha256']
    response=subprocess.run(f.SSH+[shlex.quote(f.H+'/venv/bin/python')+' -B -'],input=code,
        capture_output=True,text=True,encoding='utf-8',check=True,timeout=180)
    checks=json.loads(response.stdout);assert len(checks)==2 and {x['arm'] for x in checks}=={'A','B'}
    for spec in specs:
        actual=next(x for x in checks if x['arm']==spec['arm'])
        assert actual['path']==spec['path'] and actual['sha256']==spec['expected_sha256']
        assert actual['stable_during_hash'] and actual['exists'] and actual['bytes']>0
    f.write(root/freshrel,dict(schema='formal_main_v2',kind='root_development_presubmission_selected_server_SHA_review',
        status='passed',reviewed_utc=datetime.now(timezone.utc).isoformat(),suite='libero_spatial',
        selected=specs,server_selected_checks=checks,prior_selected_review=oldrel,probe_program=code,
        probe_program_sha256=old['probe_program_sha256'],remote_returncode=response.returncode,
        inputs_sha256={n:f.sha(root/n) for n in (oldrel,regrel,planrel,reg['execution_manifest'],'.local/root_submit_first_spatial_development.py')},
        weights_copied=False,weights_deserialized=False,additional_GPU_seconds=0,final_states_opened=False))
    refs=dict(reg['read_evidence_sha256'])
    for name in (regrel,planrel,freshrel,'.local/root_submit_first_spatial_development.py'):
        refs[name]=f.sha(root/name)
    f.write(root/reviewrel,dict(schema='formal_main_v2',kind='root_first_spatial_development_reconstruction',
        status='passed_cpu_only',reviewed_utc=datetime.now(timezone.utc).isoformat(),registration=regrel,plan=planrel,
        plan_sha256=reg['plan_sha256'],plan_rebuilt_equal=True,current_state_fresh_under_supervisor_lock=True,
        active_requests_before=len(live),completed_trials_before=0,initial_stop_after=12,total_target=600,
        exact_first12_fixed=True,original_controller_unchanged=True,reconstruction_wall_seconds=time.perf_counter()-started,
        fresh_selected_server_review=freshrel,fresh_selected_server_review_sha256=f.sha(root/freshrel),
        actual_spatial_downstream_GPU_verified=False,metadata_or_component_adaptation=False,
        final_states_opened=False,weights_copied=False,scientific_settings_changed=False,GPU_seconds=0,
        read_evidence_sha256=dict(sorted(refs.items()))))
    # Transport checks all source/support/input identities; the separate root
    # probe above checks selected weights, and the runtime loader repeats it.
    job=c.submit_plan(rebuilt,state,project=root,supervisor=f,transport=c.Transport(root),expected_plan_sha256=reg['plan_sha256'])
    record=dict(schema='formal_main_v2',kind='root_first_spatial_development_submission',
        status='submitted_pending_actual_GPU_review',recorded_utc=datetime.now(timezone.utc).isoformat(),
        job_id=job['job_id'],invocation=job['invocation'],invocation_sha256=job['invocation_sha256'],
        plan_sha256=reg['plan_sha256'],completed_before=0,initial_stop_after=12,total_target=600,
        root_reconstruction=reviewrel,original_transport_prepare_passed=True,no_prior_trials_to_rerun=True,
        final_states_opened=False,weights_copied=False,scientific_settings_changed=False,
        evidence_sha256={n:f.sha(root/n) for n in (regrel,planrel,reviewrel,freshrel,job['invocation'],base+'submissions/'+job['job_id']+'.json')})
    f.write(root/subrel,record)
    p['first_spatial_development']=dict(status='first12_submitted_pending_actual_GPU_review',completed_trials=0,
        target_first_subset_trials=600,unique_verified_development_initial_states=0,active_job_id=job['job_id'],
        active_job_status=job['status'],latest_job_id=job['job_id'],next_stop_after=12,expected_cumulative_trials=12,
        registration=regrel,root_plan_review=reviewrel,latest_plan=planrel,latest_plan_sha256=reg['plan_sha256'],
        invocation=job['invocation'],invocation_sha256=job['invocation_sha256'],verified_segments=[],
        latest_submission_root_review=subrel,latest_submission_root_review_sha256=f.sha(root/subrel),
        next='Review raw first12 Spatial trials, source/adapter restoration and selected identities, one-forward/hard packing and zero future head, environment220/10/8, and every allocation cost before own explicit continuation. No final release.')
    p['parallel_execution']['registered_requested_active_GPU_jobs']=len(live)+1
    state['updated_utc']=record['recorded_utc'];f.write(f.STATE,state)
print(json.dumps(dict(status=record['status'],job_id=job['job_id'],review=reviewrel,root_sha256=f.sha(root/reviewrel),
    submission=subrel,submission_sha256=f.sha(root/subrel),fresh_selected=freshrel,fresh_selected_sha256=f.sha(root/freshrel))),flush=True)
