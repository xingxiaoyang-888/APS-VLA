def digest(data):
    return hashlib.sha256(data).hexdigest()


def read(path, expected=None):
    path = Path(path)
    rel = path.relative_to(ROOT).as_posix()
    assert not any(p in {'weights', 'checkpoints', 'sealed', 'final'} for p in path.parts)
    assert path.suffix.lower() not in {'.pt', '.pth', '.bin', '.npz', '.npy', '.init', '.safetensors'}
    assert path.name != 'state.json'
    data = path.read_bytes()
    actual = digest(data)
    assert expected is None or actual == expected
    assert rel not in READ or READ[rel] == actual
    READ[rel] = actual
    return data


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def function_tree(data, name):
    return next(n for n in ast.parse(data).body if isinstance(n, ast.FunctionDef) and n.name == name)


def test_exact_reversible_change_and_unchanged_transaction(modules):
    new, old = modules
    before = read(OLD / 'submit.py').decode()
    after = read(NEW / 'submit.py').decode()
    old_line = "    value = json.loads(f.remote(shlex.quote(c.H + '/venv/bin/python') + ' -c ' + shlex.quote(script)))\n"
    new_lines = "    # Send the potentially large source/path payload through stdin, not Windows argv.\n    command = shlex.quote(c.H + '/venv/bin/python') + ' -B -'\n    value = json.loads(subprocess.check_output(f.SSH + [command], input=script,\n        text=True, encoding='utf-8'))\n"
    assert after.replace(new_lines, old_line, 1) == before
    for name in ('submission_gate', 'reserves_slot', 'execute_submit', 'publish_missing', 'remote_attempt_absent', 'publish_local', 'main'):
        assert ast.dump(function_tree(before, name)) == ast.dump(function_tree(after, name))
    old_launcher = read(OLD / 'launcher.py')
    assert read(NEW / 'launcher.py') == old_launcher.replace(STAGE1.encode(), STAGE2.encode()).replace(
        b'root-action-execution-engineering-dispatch-v1', b'root-action-execution-engineering-dispatch-v2-stdin')
    assert read(NEW / 'probe.slurm') == read(OLD / 'probe.slurm').replace(STAGE1.encode(), STAGE2.encode())
    assert read(NEW / 'probe.py') == read(OLD / 'probe.py').replace(
        ('GPU_VERSION = "' + STAGE1 + '"').encode(), ('GPU_VERSION = "' + STAGE2 + '"').encode())
    # The paths/SHA source and response validator are identical, not replaced.
    old_node, new_node = function_tree(before, 'remote_hashes'), function_tree(after, 'remote_hashes')
    assert ast.dump(old_node.body[0]) == ast.dump(new_node.body[0])
    assert ast.dump(old_node.body[-2]) == ast.dump(new_node.body[-2])
    assert new.c.S == old.c.S and new.c.SOURCE_SHA == old.c.SOURCE_SHA

def make_shim(tmp_path):
    shim = tmp_path / 'stdin_server_shim.py'
    shim.write_text("import sys,shlex\ncommand=shlex.split(sys.argv[1])\n"
        "assert command[0].endswith('/venv/bin/python')\n"
        "assert command[1:]==['-B','-']\n"
        "source=sys.stdin.read()\nexec(compile(source,'<unmodified-remote-hash-program>','exec'))\n", encoding='utf-8')
    return [sys.executable, '-B', str(shim)]


def test_actual_windows_293_keys_old_206_new_stdin_byte_hashes(modules, tmp_path):
    assert os.name == 'nt', 'This audit requires actual Windows CreateProcess, no skipped substitute.'
    new, old = modules
    prefix = make_shim(tmp_path)
    paths, expected = [], {}
    for i in range(293):
        name = f'{i:03d}_' + 'long_metadata_' * 2 + ("空格 'quote'.json" if i == 0 else '.json')
        path = tmp_path / name
        data = b'\x00\xff\r\n' + ('synthetic only ' + str(i)).encode()
        if i < 291:
            path.write_bytes(data)
        paths.append(str(path.absolute()))
        expected[paths[-1]] = digest(data) if i < 291 else None
    old_size = []
    def local_argv(command):
        args = prefix + [command]
        old_size.append(len(subprocess.list2cmdline(args)))
        return subprocess.check_output(args, text=True, encoding='utf-8')
    with pytest.raises(OSError) as error:
        old.remote_hashes(SimpleNamespace(remote=local_argv), paths)
    assert error.value.winerror == 206 and old_size[0] > 32767
    received = new.remote_hashes(SimpleNamespace(SSH=prefix), paths)
    assert received == expected and len(received) == 293
    command = shlex.quote(new.c.H + '/venv/bin/python') + ' -B -'
    short_size = len(subprocess.list2cmdline(prefix + [command]))
    assert short_size < 1024
    # Re-read a changed synthetic byte payload; old SHA must not be reused.
    Path(paths[0]).write_bytes(b'changed\n')
    changed = new.remote_hashes(SimpleNamespace(SSH=prefix), paths)
    assert changed[paths[0]] == digest(b'changed\n') != received[paths[0]]
    assert {k:v for k,v in changed.items() if k != paths[0]} == {k:v for k,v in received.items() if k != paths[0]}
    (HERE / 'windows_long_argv_actual_metrics.json').write_text(json.dumps(dict(
        old_argv_characters=old_size[0], old_winerror=error.value.winerror,
        new_argv_characters=short_size, exact_keys=293, existing=291, missing=2,
        actual_Windows_process=True, unmodified_remote_program=True,
        binary_CRLF_quote_Unicode_bytes_checked=True, changed_source_SHA_not_cached=True,
        SSH_calls=0, GPU_calls=0), indent=2) + '\n', encoding='utf-8')


@pytest.mark.parametrize('reply', [
    '{}', '{"a":null,"b":"' + 'f'*64 + '","extra":null}',
    '{"a":false,"b":null}', '{"a":"' + 'F'*64 + '","b":null}', 'not JSON',
])
def test_invalid_reply_raises_with_actual_local_process(modules, reply):
    new, _ = modules
    fake_ssh = [sys.executable, '-B', '-c', 'import sys;sys.stdin.read();sys.stdout.write(' + repr(reply) + ')']
    with pytest.raises(ValueError):
        new.remote_hashes(SimpleNamespace(SSH=fake_ssh), ['a', 'b'])


def test_nonzero_transport_propagates_even_if_stdout_is_valid_json(modules):
    new, _ = modules
    fake_ssh = [sys.executable, '-B', '-c', 'import sys;sys.stdin.read();print("{\\"a\\":null}");sys.exit(29)']

        new.remote_hashes(SimpleNamespace(SSH=fake_ssh), ['a'])
    assert error.value.returncode == 29

def test_real_prior_failure_preserved_prepublication_and_no_automatic_retry(modules):
    new, _ = modules
    path = ROOT / 'results/formal_main_v2/preflight/action_execution_gpu_submission_0626.json'
    data = read(path, '9e18b0149bed9eb2b95ef194b4f25a323c2691a0e8dcc4f5e13c4d858003a2e2')
    assert read(NEW / 'observed_submission_failure_v1.json') == data
    failure = json.loads(data)
    assert failure['exit_code'] == 1 and failure['result'] is None
    assert 'WinError 206' in failure['stderr'] and 'observed = remote_hashes(f, watched)' in failure['stderr']
    assert failure['automatic_retry'] is False and failure['scientific_release'] is False

    assert 'Unknown sbatch outcome; pending preserved, no retry' in body
    assert new.c.DISPATCH['maximum_attempts'] == 1 and new.c.DISPATCH['automatic_retry'] is False
