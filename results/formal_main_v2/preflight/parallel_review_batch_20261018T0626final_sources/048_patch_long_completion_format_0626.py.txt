"""Apply the sole fixed raw-diff format guard after the existing generator.

This script only preserves and edits the not-yet-run delivery helper. It never
executes that helper, Git, the archive checker, SSH or any experiment.
"""
from pathlib import Path
import ast
import difflib
import hashlib
import json

ROOT = Path(__file__).resolve().parents[1]
TARGET = ROOT / ".local/stage_verify_parallel_0626final_candidate.py"
BEFORE = ROOT / ".local/stage_verify_parallel_0626final_before_format.py.txt"
CHANGES = ROOT / ".local/stage_verify_parallel_0626final_format_diff.json"
EXPECTED_BEFORE = "29f1c905eeed3b352c0c90b3a66086c05b41722e81fb58514eb7cf33bc0b32c0"
SOURCE = ROOT / ".local/action_execution_gpu_dispatch_transport_v2_0626/v1_to_v2.diff"
SOURCE_SHA = "6c5f4cbcfb14dd8c5a301b5ffefb8d0364309f569712463edde1474ff1dd37f7"


def sha(data):
    return hashlib.sha256(data).hexdigest()


def replace_once(text, before, after):
    assert text.count(before) == 1, (before, text.count(before))
    return text.replace(before, after, 1)


def main():
    original = TARGET.read_bytes()
    assert sha(original) == EXPECTED_BEFORE and b"\r" not in original
    raw_diff = SOURCE.read_bytes()
    assert sha(raw_diff) == SOURCE_SHA and len(raw_diff) == 2314
    assert raw_diff.splitlines()[43] == b" "
    assert not BEFORE.exists() and not CHANGES.exists()
    text = original.decode("utf-8")
    text = replace_once(text,
        'SELF_NAME = ".local/stage_verify_parallel_0626final_candidate.py"\n',
        'SELF_NAME = ".local/stage_verify_parallel_0626final_candidate.py"\n'
        'FORMAT_DIFF_SOURCE = ".local/action_execution_gpu_dispatch_transport_v2_0626/v1_to_v2.diff"\n'
        'FORMAT_DIFF_SHA = "6c5f4cbcfb14dd8c5a301b5ffefb8d0364309f569712463edde1474ff1dd37f7"\n'
        'FORMAT_DIFF_ARCHIVE_PREFIX = BASE + "preflight/parallel_review_batch_20261018T0626final_sources/"\n')
    text = replace_once(text,
        '    assert evidence[stage_archive_name] == ref(SELF_NAME)["sha256"]\n',
        '    assert evidence[stage_archive_name] == ref(SELF_NAME)["sha256"]\n'
        '''    format_archive_name = batch["audit_source_archive_map"][FORMAT_DIFF_SOURCE]
    safe_name(format_archive_name)
    assert format_archive_name.startswith(FORMAT_DIFF_ARCHIVE_PREFIX)
    assert PurePosixPath(format_archive_name).name.endswith("_v1_to_v2.diff.txt")
    assert {name for name, pin in evidence.items() if pin == FORMAT_DIFF_SHA} == {format_archive_name}
    format_source = (ROOT / FORMAT_DIFF_SOURCE).read_bytes()
    assert ref(FORMAT_DIFF_SOURCE)["sha256"] == sha(format_source) == FORMAT_DIFF_SHA
    assert len(format_source) == 2314 and format_source.splitlines()[43] == b" "
    assert ref(format_archive_name)["sha256"] == FORMAT_DIFF_SHA
    assert (ROOT / format_archive_name).read_bytes() == format_source

    def check_exact_format():
        assert (ROOT / FORMAT_DIFF_SOURCE).read_bytes() == format_source
        assert (ROOT / format_archive_name).read_bytes() == format_source == git("show", ":" + format_archive_name)
        expected = (format_archive_name + ":44: trailing whitespace.\\n+ \\n").encode("utf-8")
        check = subprocess.run(["git", "diff", "--cached", "--check"], cwd=ROOT,
                               stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        assert check.returncode == 2 and check.stdout == expected and check.stderr == b"", check
        rest = subprocess.run(["git", "diff", "--cached", "--check", "--", ".",
                               ":(exclude,literal)" + format_archive_name], cwd=ROOT,
                              stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        assert rest.returncode == 0 and rest.stdout == rest.stderr == b"", rest
        return dict(source=ref(FORMAT_DIFF_SOURCE), preserved_copy=ref(format_archive_name),
                    index_equals_original_bytes=True, returncode=check.returncode,
                    stdout=check.stdout.decode("utf-8"), stderr="", expected_line=44,
                    expected_line_bytes_hex="20", other_findings_allowed=False)

''')
    text = replace_once(text,
        '    subprocess.run(["git", "diff", "--cached", "--check"], cwd=ROOT, check=True)\n'
        '    proof = dict(',
        '    format_findings = check_exact_format()\n'
        '    proof = dict(')
    text = replace_once(text,
        '                 stage_helper_archive=ref(stage_archive_name),\n',
        '                 stage_helper_archive=ref(stage_archive_name),\n'
        '                 exact_immutable_raw_diff_format_findings=format_findings,\n')
    text = replace_once(text,
        '    subprocess.run(["git", "diff", "--cached", "--check"], cwd=ROOT, check=True)\n',
        '    assert check_exact_format() == format_findings\n')
    ast.parse(text)
    updated = text.encode("utf-8")
    changes = dict(schema="formal_main_v2", kind="exact_unexecuted_delivery_helper_patch",
                   before_sha256=sha(original), after_sha256=sha(updated),
                   fixed_source_sha256=SOURCE_SHA, stage_main_executed=False,
                   Git_executed=False, scientific_source_modified=False,
                   unified_diff_lines=list(difflib.unified_diff(
                       original.decode("utf-8").splitlines(keepends=True),
                       text.splitlines(keepends=True),
                       fromfile="stage_verify_parallel_0626final_before_format.py",
                       tofile="stage_verify_parallel_0626final_candidate.py")))
    with BEFORE.open("xb") as stream:
        stream.write(original)
    assert TARGET.read_bytes() == original
    TARGET.write_bytes(updated)
    with CHANGES.open("x", encoding="utf-8", newline="\n") as stream:
        json.dump(changes, stream, ensure_ascii=False, indent=2)
        stream.write("\n")
    print(json.dumps(dict(status="patched_not_executed", before_sha256=sha(original),
                          after_sha256=sha(updated), changes=str(CHANGES.relative_to(ROOT)))))


if __name__ == "__main__":
    main()
