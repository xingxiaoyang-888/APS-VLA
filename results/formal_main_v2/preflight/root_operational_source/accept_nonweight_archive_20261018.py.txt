from pathlib import Path
from datetime import datetime, timezone
import base64
import hashlib
import json
import sys

root = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(root / 'scripts'))
import formal_supervisor as f

base = 'results/formal_main_v2/preflight/'
ir = base + 'nonweight_archive_independent_20261018.json'
rt = base + 'nonweight_archive_real_roundtrip_20261018.json'
out = base + 'nonweight_archive_root_acceptance_20261018.json'
assert not (root / out).exists()
i, t = f.read(root / ir), f.read(root / rt)
assert f.sha(root / ir) == 'fdb2c16d26d6289b483658ef853944458ccb2c2e69d1eda1dab4d0d716395fc7'
assert i['status'] == 'passed_after_bounded_guard_repair_with_explicit_scope_limits'
assert i['ready_for_root_synthetic_roundtrip'] and not i['real_raw_migration_authorized_by_this_receipt']
assert t['status'] == 'passed' and t['source_input']['bytes'] == 86664845
assert t['real_source_size_mtime_sha_unchanged'] and t['existing_raw_files_migrated'] == 0
assert f.sha(root / t['codec_implementation']) == t['codec_implementation_sha256']
embedded = {}
for name, row in i['embedded_audit_files'].items():
    assert row['encoding'] == 'base64'
    data = base64.b64decode(row['data'], validate=True)
    assert len(data) == row['bytes'] and hashlib.sha256(data).hexdigest() == row['sha256']
    assert row['original_path'] not in embedded
    embedded[row['original_path']] = dict(receipt=ir, entry=name, sha256=row['sha256'], bytes=row['bytes'])
refs = {ir, rt, t['source_input']['path'], t['archive_receipt']}
resolved = {}
for rel, want in i['read_evidence_sha256'].items():
    assert f.sha(root / rel) == want, rel
    if rel.startswith('.local/'):
        if rel in embedded:
            assert embedded[rel]['sha256'] == want
            resolved[rel] = embedded[rel]
        else:
            assert rel == '.local/check_scientific_stage.py'
            dest = base + 'root_operational_source/check_scientific_stage_20261018.py.txt'
            assert not (root / dest).exists()
            (root / dest).write_bytes((root / rel).read_bytes())
            assert f.sha(root / dest) == want
            refs.add(dest)
            resolved[rel] = dict(archive=dest, sha256=want)
    else:
        refs.add(rel)
for name in ['verify_archive_real_bytes_20261018.py', 'accept_nonweight_archive_20261018.py']:
    src = root / '.local' / name
    dest = base + 'root_operational_source/' + name + '.txt'
    assert not (root / dest).exists()
    (root / dest).write_bytes(src.read_bytes())
    assert f.sha(root / dest) == f.sha(src)
    refs.add(dest)
tests = i['independent_tests']['tests']
assert len(tests['cases']) == 26 and all(x['status'] == 'passed' for x in tests['cases'])
author = i['candidate_versions']['v2']['author_tests']['summary']
assert int(author['tests']) == 69 and int(author['skipped']) == 3
assert int(author['failures']) == int(author['errors']) == 0
value = dict(schema='formal_main_v2', kind='root_nonweight_archive_engineering_acceptance',
    status='passed_engineering_preparation', reviewed_utc=datetime.now(timezone.utc).isoformat(),
    independent_review=ir, independent_review_sha256=f.sha(root / ir),
    code='scripts/ops/nonweight_archive.py', code_sha256=f.sha(root / 'scripts/ops/nonweight_archive.py'),
    unit_tests=dict(cases=69, passed=66, skipped=3, failures=0, reason_for_skips='Windows real symlink privilege absent; retained explicitly. Four actual junction cases passed in the independent suite, not a claim that real symlinks were tested.'),
    independent_tests=dict(cases=26, passed=26, skipped=0, failures=0),
    bounded_repair=dict(v1_retained=i['candidate_versions']['v1']['receipt'],
        v2_retained=i['candidate_versions']['v2']['receipt'],
        original_blocker='Seven lexical paths for sealed initial/final/test state files were accepted by the first candidate; only lexical probes used, no real state values accessed.',
        repair='Reject state-name stems and the test directory before reading the target. Preserve legal released final trial JSONL paths. Science sources and original results unchanged.'),
    real_bytes_roundtrip=dict(receipt=rt, receipt_sha256=f.sha(root / rt),
        source_bytes=86664845, compressed_bytes=t['total_compressed_chunk_bytes'],
        chunks=len(t['compressed_chunks']), same_bytes_after_restore=True,
        repeated_restore_idempotent=True, writes_only_in_isolated_project_copies=True),
    resolved_ignored_dependencies=resolved,
    scope_limits=['Existing raw files have not been migrated, removed, rewritten or ignored.',
        'The current Git staging checker is unchanged. Before a future raw-to-archive Git transition, add an exact-path staged-blob reconstruction check and include every chunk, mapping and provenance dependency.',
        'No multi-directory crash-atomicity claim; interrupted publication may leave verified files and must be completed idempotently.',
        'A passed codec is not an experiment release, closed-loop result, or source of evidence for future supervision benefit.'],
    frozen_scientific_files_unchanged=173, GPU_seconds=0, weights_copied=False,
    sealed_state_values_read=False, scientific_settings_changed=False,
    read_evidence_sha256={rel:f.sha(root / rel) for rel in sorted(refs)})
f.write(root / out, value)
print(json.dumps(dict(receipt=out, sha256=f.sha(root / out), permanent_evidence=len(refs),
    embedded_dependencies=len(embedded), source_bytes=86664845,
    compressed_bytes=t['total_compressed_chunk_bytes'], status=value['status'])))
