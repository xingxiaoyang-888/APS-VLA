"""Root derivative of immutable actual5000 reviews; no model or sealed data I/O."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json

ROOT = Path(__file__).resolve().parents[1]
BASE = Path('results/formal_main_v2/libero_10/k128/seed17/training')
OUT = ROOT / 'results/formal_main_v2/preflight/first_completed_pair_summary_20261017.json'
assert not OUT.exists()
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
read = lambda p: json.loads((ROOT / p).read_text(encoding='utf-8'))
refs = {}
def bound(rel):
    p = ROOT / rel
    refs[str(rel).replace('\\', '/')] = sha(p)
    return read(rel)

root = bound(BASE / 'segment_reviews/3016464.json')
ind = bound(BASE / 'segment_reviews/3016464_independent.json')
selected = bound(BASE / 'segment_reviews/3016464_selected_server.json')
assert root['status'] == ind['status'] == selected['status'] == 'passed'
assert root['completed_by_arm'] == {'A': 5000, 'B': 5000}
assert root['allowed_next_segment_end'] is None and not root['resume_required']
assert ind['formal_training_complete'] and not ind['global_release']
for rel, digest in ind['read_evidence_sha256'].items():
    assert sha(ROOT / rel) == digest, rel
jobs = root['registered_pair_jobs_at_review']
assert len(jobs) == 10 and all(j['status'] == 'COMPLETED' and j['exit_code'] == '0:0' and j['gpus'] == 1 for j in jobs)
assert sorted(j['segment_end'] for j in jobs) == list(range(500, 5001, 500))
bill = sum(j['gpus'] * j['elapsed_seconds'] for j in jobs)
assert bill == root['pair_allocated_gpu_seconds'] == ind['pair_training_allocated_gpu_seconds'] == 41261
summary = dict(schema='formal_main_v2', kind='first_complete_formal_pair_observed_summary',
    created_utc=datetime.now(timezone.utc).isoformat(), suite='libero_10', visual_tokens=128, seed=17,
    status='root_independent_and_actual_selected_server_SHA_passed', trained_arms=2,
    total_required_arms=72, source_sha256=root['source_sha256'], jobs=jobs, arms={},
    selected_server_review=refs[str(BASE / 'segment_reviews/3016464_selected_server.json').replace('\\', '/')],
    allocated_GPU_seconds=bill, allocated_GPU_hours=bill/3600,
    pair_committed_arm_updates=10000, pair_window_uses=80000,
    validation_window_forwards=2*11*165, cache_event_counts=root['cache_event_counts'],
    cache_generation_component_seconds=root['cache_generation_component_seconds'],
    next_training_boundary=None, formal_development_trials=0, final_trials=0,
    final_states_sealed=True, weights_copied=False,
    limitations=[
        'Per-arm40000 uses are40000 unique windows for this seed, drawn from the full68819-window pool; A/B share the same order and are not80000 independent windows.',
        'Teacher validation L1 is only the registered checkpoint-selection diagnostic; no closed-loop benefit or full inference speed claim.',
        'This pair crossed serial and concurrent execution phases. Its GPU bill is not an eight-GPU wallclock forecast or a causal A/B training-cost comparison.',
        'Update, validation and cache-generation phase times overlap and are not added to the Slurm bill. Shared cache generation is charged where it actually occurred; independent-run cache replay remains a separate later artifact.',
        'All10 pair jobs have no failed/replayed training; project-wide failed engineering allocations are retained in the full project bill, outside this pair subtotal.',
        'Other suites, budgets, remaining seeds, all development/final evaluation and full/component latency costs remain to be measured.'
    ])
for arm in ('A', 'B'):
    report = bound(BASE / arm / 'training_report.json')
    receipt = bound(BASE / arm / 'deployment_receipt.json')
    assert report['training_updates_completed'] == 5000
    assert report['committed_samples'] == report['unique_committed_windows'] == 40000
    assert report['train_pool_windows'] == 68819 and report['validation_pool_windows'] == 165
    assert receipt['checkpoint_sha256'] == report['best_checkpoint_sha256']
    assert selected['input_sha256'][str(BASE / arm / 'training_report.json').replace('\\', '/')] == refs[str(BASE / arm / 'training_report.json').replace('\\', '/')]
    keys = ('best_update', 'best_validation_metric', 'best_checkpoint', 'best_checkpoint_sha256',
            'committed_samples', 'unique_committed_windows', 'train_pool_windows', 'validation_pool_windows',
            'effective_epochs', 'completed_full_pool_epochs', 'initial_scorer_sha256', 'sample_order_sha256',
            'completed_update_work_seconds_including_replays', 'completed_validation_work_seconds_including_replays',
            'checkpoint_write_seconds', 'peak_cuda_memory_allocated_bytes', 'peak_cuda_memory_reserved_bytes')
    summary['arms'][arm] = {key: report[key] for key in keys}
    summary['arms'][arm]['allocated_memory_GiB_peak'] = report['peak_cuda_memory_allocated_bytes'] / 2**30
    summary['arms'][arm]['reserved_memory_GiB_peak'] = report['peak_cuda_memory_reserved_bytes'] / 2**30
    summary['arms'][arm]['mean_update_work_seconds'] = report['completed_update_work_seconds_including_replays'] / 5000
    summary['arms'][arm]['mean_full_panel_validation_seconds'] = report['completed_validation_work_seconds_including_replays'] / 11
summary['source_receipts_sha256'] = refs
summary['read_evidence_count'] = len(ind['read_evidence_sha256'])
with OUT.open('x', encoding='utf-8', newline='\n') as stream:
    json.dump(summary, stream, ensure_ascii=False, indent=2)
    stream.write('\n')
print(json.dumps({'output':OUT.relative_to(ROOT).as_posix(), 'sha256':sha(OUT), 'GPU_hours':bill/3600,
                  'arms':{a:summary['arms'][a] for a in ('A','B')}}, ensure_ascii=False))
