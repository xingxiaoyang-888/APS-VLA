from pathlib import Path
import subprocess,json,hashlib
root=Path.cwd();changed=subprocess.check_output(['git','diff','--cached','--name-only','-z']).decode('utf-8').rstrip('\0').split('\0')
entries={}
for row in subprocess.check_output(['git','ls-files','--stage','-z']).split(b'\0'):
 if row:
  h,p=row.split(b'\t',1);mode,oid,stage=h.split();entries[p.decode('utf-8')]=(mode,oid,stage)
bad=[];weight=[];large=[]
for p in changed:
 q=root/p
 if not q.is_file():raise ValueError('Unexpected removal '+p)
 if p.endswith(('.md',)):continue
 b=q.read_bytes();want=hashlib.sha1(b'blob '+str(len(b)).encode()+b'\0'+b).hexdigest().encode()
 if want!=entries[p][1]:bad.append(p)
 if q.suffix in ('.pt','.pth','.bin','.ckpt','.safetensors','.pkl','.pickle'):weight.append(p)
 if len(b)>100*1024*1024:large.append((p,len(b)))
inventory=json.loads((root/'results/formal_main_v2/source/formal_main_v2_run_564be0995e.json').read_text(encoding='utf-8'));frozenchanged=[]
for p,h in inventory['files'].items():
 assert hashlib.sha256((root/p).read_bytes()).hexdigest()==h,p
 if p in changed:frozenchanged.append(p)
print(json.dumps(dict(staged_files=len(changed),bytes_changed_by_git=bad,weights=weight,oversize=large,frozen_files_changed=frozenchanged),ensure_ascii=False));assert not bad and not weight and not large and not frozenchanged
