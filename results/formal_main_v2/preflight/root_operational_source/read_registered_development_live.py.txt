from pathlib import Path
from datetime import datetime, timezone
import json
import shlex
import subprocess
import sys

root = Path.cwd()
sys.path.insert(0, str(root / 'scripts'))
import formal_supervisor as f

with f.supervisor_lock():
    state = f.read(f.STATE)
    jobs = f.refresh(state)
    development = state['formal_main_progress']['first_long_development']
    job = next(j for j in jobs if j['job_id'] == development['active_job_id'])
    assert job['mode'] == 'development' and job['status'] in f.ACTIVE
    code = '''from pathlib import Path
from datetime import datetime,timezone
from collections import Counter
import json
p=Path(RUN_ROOT)/'trials.jsonl'
counts=Counter();statuses=Counter();ids=[];last=None
if p.exists():
 with p.open('rb') as stream:
  for raw in stream:
   if not raw.endswith(b'\\n'):continue
   row=json.loads(raw);counts[row['event']]+=1;last=row['created_utc']
   if row['event']=='invocation':ids.append(row['slurm_job_id'])
   if row['event']=='result':statuses[row['status']]+=1
print(json.dumps(dict(observed_utc=datetime.now(timezone.utc).isoformat(),not_a_terminal_receipt=True,events=sum(counts.values()),event_counts=dict(counts),invocation_job_ids=ids,live_terminal_trials=counts['result'],live_error_results=statuses['error'],result_status_counts=dict(statuses),last_event_utc=last)))
'''.replace('RUN_ROOT', repr(job['run_root']))
    result = subprocess.run(f.SSH + [shlex.quote(f.H + '/venv/bin/python') + ' -B -'], input=code, capture_output=True, text=True, encoding='utf-8', check=True)
    live = json.loads(result.stdout)
    live.update({k: job[k] for k in ('job_id', 'status', 'run_root', 'invocation_sha256')})
    live['source_program'] = code
    live['historical_probe_correction'] = {
        'path': 'results/formal_main_v2/live/long_development_20261017T140359Z.json',
        'reason': 'Prior root live-only counter used job_id/utc instead of slurm_job_id/created_utc, giving missing invocation IDs/timestamp. Counts were correct; this probe uses actual ledger keys. No experiment, job, scientific source, or GPU allocation changed.'
    }
    stamp = datetime.fromisoformat(live['observed_utc']).strftime('%Y%m%dT%H%M%SZ')
    rel = 'results/formal_main_v2/live/long_development_' + stamp + '.json'
    assert not (root / rel).exists()
    f.write(root / rel, live)
    development.update(latest_live=rel, active_job_status=job['status'])
    state['updated_utc'] = datetime.now(timezone.utc).isoformat()
    f.write(f.STATE, state)
    print(json.dumps({'live': rel, 'invocation_job_ids': live['invocation_job_ids'], 'live_terminal_trials': live['live_terminal_trials'], 'last_event_utc': live['last_event_utc'], 'cost': state['current_round_cost']}))
