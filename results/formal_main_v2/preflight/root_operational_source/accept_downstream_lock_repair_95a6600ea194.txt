"""Root bounded audit addendum and actual first development bookkeeping."""
from pathlib import Path
from datetime import datetime, timezone
import argparse
import json
import sys
import xml.etree.ElementTree as ET

root = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(root / 'scripts'))
import formal_supervisor as f

parser = argparse.ArgumentParser()
parser.add_argument('--independent', required=True)
parser.add_argument('--independent-sha256', required=True)
args = parser.parse_args()
evidence = {}

def check(name, expected=None):
    path = root / name
    path.resolve().relative_to(root)
    assert path.suffix not in ('.pt', '.pth', '.safetensors', '.bin', '.pkl', '.pickle')
    checksum = f.sha(path)
    assert expected is None or checksum == expected, name
    assert name not in evidence or evidence[name] == checksum, name
    evidence[name] = checksum
    return checksum

def read(name, expected=None):
    check(name, expected)
    return f.read(root / name)

old_path = 'results/formal_main_v2/preflight/asset_downstream_consumers_root_acceptance_20261017.json'
old = read(old_path, '0c8ee679a4058a837b9fe1458f53387c1af67be8262c4790316296477bb457a8')
repair_path = 'results/formal_main_v2/preflight/lock_receipt_repair_v1/repair_review.json'
repair = read(repair_path, 'a1c8d81222562a3814e9b954bd7cdf39b6c6db0f22cf0f48419b47f9ed2be655')
ind = read(args.independent, args.independent_sha256)
assert ind['status'].startswith('passed')
archive_path = 'results/formal_main_v2/preflight/long_first_development_raw_v1/receipt.json'
archive = read(archive_path, 'f179d79f2e88c59bb1a86b9b4a5ae4113d5ebd1c631fcf2a7575b080c3db1aa6')
aliases = {row['original']: row['archive'] for row in archive['files']}
source = 'scripts/review_formal_downstream.py'
old_source_archive = 'results/formal_main_v2/preflight/lock_receipt_repair_v1/review_formal_downstream.before.py.txt'
for name, checksum in old['read_evidence_sha256'].items():
    check(old_source_archive if name == source else name, checksum)
for receipt in (repair, ind):
    for name, checksum in receipt['read_evidence_sha256'].items():
        check(aliases.get(name, name), checksum)
consumers = dict(old['consumers_sha256'])
consumers[source] = repair['new_source_sha256']
for name, checksum in consumers.items():
    check(name, checksum)
cases = ET.parse(root / repair['tests']['junit']['path']).getroot().findall('.//testcase')
assert len(cases) == 50 and all(not any(c.find(k) is not None for k in ('error', 'failure', 'skipped')) for c in cases)
review_path = 'results/formal_main_v2/downstream_reviews/3016981.json'
review = read(review_path, '7bb724e15cf904058ea038e54d1d162c862d9f53512943b73935cdf406e9e4bb')
actual_ind_path = 'results/formal_main_v2/preflight/long_first_development_real_independent_20261017.json'
actual_ind = read(actual_ind_path, 'd1a9846f443471c58abfd15a5774a10c262ec87f41e5de680fb7c67d3132a194')
assert actual_ind['status'].startswith('passed')
for name, checksum in actual_ind['read_evidence_sha256'].items():
    check(aliases.get(name, name), checksum)
assert review['status'] == 'passed' and review['scientific_complete'] is False
assert review['completed_trials'] == 12 and review['expected_trials'] == 600
assert review['cost']['all_attempt_gpu_seconds'] == 437
assert actual_ind['completed_trials'] == 12 and actual_ind['final_trials_completed'] == 0
assert review['reviewer_source_sha256'] == repair['new_source_sha256']
for row in archive['files']:
    check(row['archive'], row['sha256'])
    assert (root / row['archive']).stat().st_size == row['bytes']

helper_bytes = Path(__file__).read_bytes()
helper_archive = 'results/formal_main_v2/preflight/root_operational_source/accept_downstream_lock_repair_' + f.sha(Path(__file__))[:12] + '.txt'
helper_target = root / helper_archive
if helper_target.exists():
    assert helper_target.read_bytes() == helper_bytes
else:
    helper_target.parent.mkdir(parents=True, exist_ok=True)
    helper_target.write_bytes(helper_bytes)
check(helper_archive)

out_rel = 'results/formal_main_v2/preflight/downstream_lock_root_acceptance_20261017.json'
assert not (root / out_rel).exists()
gate = dict(schema='formal_main_v2', kind='root_downstream_bounded_CPU_acceptance_addendum',
    status='passed_cpu_consumer_integration', reviewed_by='root', reviewed_utc=datetime.now(timezone.utc).isoformat(),
    previous_acceptance=old_path, previous_acceptance_sha256=check(old_path),
    previous_157_cases_retained_not_claimed_rerun=True, current_focused_cases_passed=50,
    bounded_fix=repair_path, bounded_fix_sha256=check(repair_path),
    independent_review=args.independent, independent_review_sha256=args.independent_sha256,
    consumers_sha256=consumers, frozen_files_verified=173, deployment_science_changed=False,
    historical_source_archive={source: old_source_archive}, historical_mutable_output_archives=aliases,
    original_rejection_retained=True, additional_GPU_seconds=0, actual_GPU_job_not_rerun=True,
    actual_Long_development_root_review=review_path, actual_Long_development_independent_review=actual_ind_path,
    actual_Long_development_trials_verified=12, actual_Object_Goal_downstream_GPU_verified=False,
    global_final_release_created=False, final_states_opened=False,
    read_evidence_sha256=dict(sorted(evidence.items())))
with f.supervisor_lock():
    state = f.read(f.STATE)
    jobs = f.refresh(state)
    job = next(j for j in jobs if j['job_id'] == '3016981')
    assert job['status'] == 'COMPLETED' and job['exit_code'] == '0:0' and job['elapsed_seconds'] == 437
    with (root / out_rel).open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(gate, stream, ensure_ascii=False, indent=2)
        stream.write('\n')
    p = state['formal_main_progress']
    integration = p['downstream_asset_consumer_integration']
    integration.update(root_review=out_rel, root_review_sha256=f.sha(root / out_rel),
        prior_acceptance=old_path, production_enabled=True, actual_GPU_downstream_verified=True,
        actual_GPU_verified_scope='Long first12 development trials only; Object/Goal downstream not yet run',
        focused_repair_tests_passed=50, previous157_tests_not_rerun_on_repaired_source=True)
    first = p['first_long_development']
    first.update(status='first12_root_and_independent_verified_same600_run_incomplete', job_status='COMPLETED',
        completed_trials=12, root_review=review_path, root_review_sha256=check(review_path),
        independent_review=actual_ind_path, independent_review_sha256=check(actual_ind_path),
        raw_archive=archive_path, allocated_GPU_seconds=437, actual_policy_calls=491)
    assert p['completed_development_trials'] in (0, 12)
    p['completed_development_trials'] = 12
    assert p['completed_final_trials'] == 0
    audit = p['downstream_entrypoint_audit']
    audit.update(status='first_Long_development_GPU_root_and_independent_verified', gpu_execution=True,
        latest_consumer_acceptance=out_rel, remaining_work='Complete all registered development panels and later globally released final evaluation/timing.')
    audit['root_artifact_review_producer'].update(status='actual_Long_first12_passed_root_and_independent',
        current_script_sha256=repair['new_source_sha256'], actual_root_review=review_path)
    audit['runtime_fingerprint_repair']['actual_Long_GPU_verified'] = True
    state['updated_utc'] = datetime.now(timezone.utc).isoformat()
    f.write(f.STATE, state)
print(json.dumps(dict(gate=out_rel, sha256=f.sha(root / out_rel), evidence_files=len(evidence),
    actual_development_trials=12, development_GPU_seconds=437, new_GPU_seconds=0)))
