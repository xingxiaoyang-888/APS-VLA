"""Publish two absent nonweight config copies from their immutable snapshot."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import shlex
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / 'scripts'))
import formal_supervisor as f

probe_rel = 'results/formal_main_v2/preflight/long_first_pair_development_server_inputs_v1.json'
probe = f.read(ROOT / probe_rel)
blocked = [r for r in probe['actual']['rows'] if r['status'] == 'blocked']
assert len(blocked) == 2 and all(not r['exists'] for r in blocked)
inventory = f.read(ROOT / 'results/formal_main_v2/source/formal_main_v2_run_564be0995e.json')
expected_names = {'configs/experiments/formal_main_v2.json','configs/experiments/formal_main_v2/libero_10_k128_seed17.json'}
rows = []
for row in blocked:
    relative = row['path'][len(f.P)+1:]
    assert relative in expected_names
    assert inventory['files'][relative] == f.sha(ROOT / relative) == row['expected_sha256']
    rows.append(dict(source=inventory['source_root_server']+'/'+relative, target=row['path'], sha256=row['expected_sha256']))
code = '''from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,os,uuid
rows=ROWS
out=[]
for row in rows:
 src,dst=Path(row['source']),Path(row['target'])
 if src.is_symlink() or src.resolve()!=src or dst.is_symlink() or dst.resolve()!=dst:raise ValueError('Noncanonical config path')
 data=src.read_bytes()
 if hashlib.sha256(data).hexdigest()!=row['sha256']:raise ValueError('Frozen config changed')
 existed=dst.exists()
 if existed:
  if not dst.is_file() or hashlib.sha256(dst.read_bytes()).hexdigest()!=row['sha256']:raise ValueError('Existing config differs; refuse overwrite')
 else:
  dst.parent.mkdir(parents=True,exist_ok=True)
  tmp=dst.parent/(dst.name+'.'+uuid.uuid4().hex+'.partial')
  with tmp.open('xb') as stream:stream.write(data)
  if hashlib.sha256(tmp.read_bytes()).hexdigest()!=row['sha256']:raise ValueError('Copy SHA changed')
  os.link(tmp,dst)
  tmp.unlink()
 if hashlib.sha256(dst.read_bytes()).hexdigest()!=row['sha256']:raise ValueError('Published config SHA differs')
 out.append(dict(row,bytes=len(data),previously_present=existed,identical_to_frozen_snapshot=True))
print(json.dumps(dict(utc=datetime.now(timezone.utc).isoformat(),files=out)))
'''.replace('ROWS', repr(rows), 1)
reply = subprocess.run(f.SSH + [shlex.quote(f.H+'/venv/bin/python')+' -B -'], input=code,
    capture_output=True,text=True,encoding='utf-8',check=True,timeout=180)
actual = json.loads(reply.stdout)
record = dict(schema='formal_main_v2',kind='root_missing_nonweight_config_publication',status='verified',
    created_utc=datetime.now(timezone.utc).isoformat(), preflight=probe_rel,preflight_sha256=f.sha(ROOT/probe_rel),
    diagnosis='Training read these exact configurations from the immutable snapshot; the downstream registration also requires canonical project-root copies, which were absent.',
    bounded_fix='Publish only the two identical snapshot files with SHA validation and no existing-file replacement; no changed source, settings, methods or checkpoint.',
    actual=actual,source_program=code,source_program_sha256=hashlib.sha256(code.encode()).hexdigest(),
    GPU_seconds=0,jobs_submitted=0,weights_read_or_copied=False,sealed_state_values_read=False,scientific_settings_changed=False)
out = ROOT/'results/formal_main_v2/preflight/long_first_pair_development_config_sync_v1.json'
assert not out.exists()
f.write(out,record)
print(json.dumps({'status':'verified','files':2,'record':out.relative_to(ROOT).as_posix(),'GPU_seconds':0}))
