from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import subprocess
import sys

root = Path(__file__).resolve().parents[1]
base = root / 'results/formal_main_v2/preflight'
target = base / 'long_first_pair_development_plan_v5.json'
record = base / 'long_first_pair_development_registration_v5.json'
assert not target.exists() and not record.exists()
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
prior = 'results/formal_main_v2/downstream_reviews/3018505.json'
review = json.loads((root / prior).read_text('utf-8'))
assert review['status'] == 'passed' and review['completed_trials'] == 372
assert review['cost']['all_attempt_gpu_seconds'] == 6580
manifest = 'results/formal_main_v2/execution/libero_10_development_first_pair_v1.json'
args = [sys.executable, '-B', 'scripts/ops/formal_downstream_supervisor.py', 'plan',
        '--stage', 'development', '--execution-manifest', manifest, '--resume',
        '--prior-review', prior, '--stop-after', '120']
r = subprocess.run(args, cwd=root, capture_output=True, text=True, encoding='utf-8')
assert r.returncode == 0, r.stderr
plan = json.loads(r.stdout)
inv = plan['invocation']
assert inv['resume'] is True and inv['stop_after'] == 120
assert inv['argv'].count('--stop-after') == 1
assert inv['argv'][inv['argv'].index('--stop-after') + 1] == '120'
assert inv['argv'].count('--resume') == 1
assert plan['output_relative'] == 'results/formal_main_v2/libero_10/downstream/development/8c843c1d4ee8383c/attempt01'
target.write_text(r.stdout, encoding='utf-8', newline='\n')
archive = 'results/formal_main_v2/preflight/long_fourth_development_raw_v1/receipt.json'
x = dict(schema='formal_main_v2', kind='root_first_long_development_continuation_preregistration',
    registered_utc=datetime.now(timezone.utc).isoformat(), suite='libero_10', methods=inv['method_ids'],
    plan=target.relative_to(root).as_posix(), plan_file_sha256=sha(target), plan_sha256=plan['plan_sha256'],
    execution_manifest=manifest, execution_manifest_sha256=sha(root / manifest),
    previous_root_review=prior, previous_root_review_sha256=sha(root / prior), previous_job_id='3018505',
    previous_completed_trials=372, previous_allocation_gpu_seconds=6580, expected_total_trials=600,
    additional_stop_after=120, expected_cumulative_trials_after_segment=492, resume=True,
    output_relative=plan['output_relative'], raw_archive=archive, raw_archive_sha256=sha(root / archive),
    engineering_reason='Retain the previous120-additional-trial slice, same600-trial ledger and original fixed order. No outcome-based changes or completed-trial repeats.',
    scientific_settings_changed=False,
    requirements_before_submit=['Actual3018505 independent review passed and bound',
        'Independent review of this exact plan', 'Fresh root accounting fewer than8 active requests and no overlapping run',
        'Original controller Transport exact server input hashes and explicit same-ledger resume'],
    production_submission_not_performed_by_this_record=True, final_states_opened=False, additional_GPU_seconds=0,
    command=args, historical_v3_preparation_correction='An earlier unsaved one-off assertion incorrectly expected --stop-after120 at argv tail. The original controller appends --resume. Check the named value and resume flag explicitly; controller/source/scientific settings unchanged. Earlier process exited before writing a plan/registration or submitting; no additional GPU allocation.')
record.write_text(json.dumps(x, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
print(json.dumps({'plan': target.relative_to(root).as_posix(), 'plan_sha256': plan['plan_sha256'],
    'record_sha256': sha(record), 'plan_file_sha256': sha(target)}))
