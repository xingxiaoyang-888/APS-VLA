"""Root-only immutable heartbeat evidence and brief progress publication."""
from pathlib import Path
from datetime import datetime, timezone, timedelta
import json
import sys

root = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(root / 'scripts'))
import formal_supervisor as f

base = 'results/formal_main_v2/'
batch = base + 'preflight/parallel_review_batch_20261017T1815.json'
assert not (root / batch).exists()
cases = [('k128/seed23', '3019840'), ('k128/seed41', '3019841'),
         ('k256/seed41', '3019843'), ('k64/seed41', '3019845')]
pulls = [base + 'pulls/' + p + '/receipt.json' for p in (
    '20261017T182054Z_d19f65961b0c', '20261017T182213Z_15f995a9ca6e',
    '20261017T182343Z_185b0d0ae8e6', '20261017T182515Z_d8021c194703')]
refs = set(pulls)
boundaries = []
audit_source_files = {
    '3019840': ['audit_training_segment_3019840.py', 'audit_training_segment_3019840.log'],
    '3019841': ['deploy64_audit_training_segment_3019841.py', 'deploy64_audit_training_segment_3019841.log'],
    '3019843': ['audit_training_segment_3019843_profile_long.py', 'audit_training_segment_3019843_profile_long_a1.log', 'audit_training_segment_3019843_profile_long_a1_invocation.json'],
    '3019845': ['audit_training_segment_3019845.py', 'audit_training_segment_3019845.log'],
}
audit_source_archive_map = {}
for pair, job in cases:
    prefix = base + 'libero_10/' + pair + '/training/segment_reviews/' + job
    rr, ir = prefix + '.json', prefix + '_independent.json'
    r, i = f.read(root / rr), f.read(root / ir)
    assert r['status'] == i['status'] == 'passed'
    assert r['job_id'] == i['job_id'] == job
    evidence = i['read_evidence_sha256']
    for name, checksum in evidence.items():
        assert f.sha(root / name) == checksum, name
    assert evidence[rr] == f.sha(root / rr)
    assert r['allowed_next_segment_end'] == i['allowed_next_segment_end']
    assert len(r['server_resume_checks']) == 2
    refs.update([rr, ir])
    boundaries.append(dict(job_id=job, pair='libero_10/' + pair,
        completed_by_arm=r['completed_by_arm'], next_boundary=r['allowed_next_segment_end'],
        pair_allocated_gpu_seconds=r['pair_allocated_gpu_seconds'],
        root_rehashed_independent_evidence_files=len(evidence)))
    # Preserve the independent audit's executable source and original output.
    for source_name in audit_source_files[job]:
        src = root / '.local' / source_name
        assert src.exists(), str(src)
        name = base + 'preflight/root_operational_source/' + source_name + '.txt'
        dest = root / name
        assert not dest.exists()
        dest.write_bytes(src.read_bytes())
        refs.add(name)
        audit_source_archive_map['.local/' + source_name] = name
for name in pulls:
    for row in f.read(root / name)['files']:
        assert f.sha(root / row['relative']) == row['sha256'], row['relative']
source_archive = base + 'preflight/root_operational_source/finalize_parallel_review_20261017T1815.py.txt'
assert not (root / source_archive).exists()
(root / source_archive).write_bytes(Path(__file__).read_bytes())
refs.add(source_archive)

with f.supervisor_lock():
    s = f.read(f.STATE)
    jobs = f.refresh(s)
    p = s['formal_main_progress']
    active = [j for j in jobs if j['status'] in f.ACTIVE | {'SUBMITTED'}]
    assert len(active) == 8 and all(j['mode'] == 'train' and j['gpus'] == 1 for j in active)
    assert len({j['stage_key'] for j in active}) == 8
    assert p['trained_arms'] == 2 and p['root_and_independent_reviewed_arm_updates'] == 75000
    assert p['completed_development_trials'] == 600 and p['completed_final_trials'] == 0
    assert not p['test_opened'] and not p['global_release_created']
    closed = [next(j for j in jobs if j['job_id'] == job) for _, job in cases]
    assert all(j['status'] == 'COMPLETED' and j['exit_code'] == '0:0' for j in closed)
    assert sum(j['elapsed_seconds'] * j['gpus'] for j in closed) == 21806
    new_jobs = [next(j for j in jobs if j['job_id'] == job) for job in ('3020306', '3020307', '3020309', '3020310')]
    for j in new_jobs:
        assert f.sha(root / j['invocation']) == j['invocation_sha256']
        refs.update([j['invocation'], base + 'submissions/' + j['job_id'] + '.json'])
        pair = p['parallel_execution']['pair_tracking'][f"{j['suite']}_k{j['visual_tokens']}_seed{j['training_seed']}"]
        assert pair['active_job_id'] == j['job_id']
        refs.add(pair['latest_resume_precheck'])
        assert f.sha(root / pair['latest_resume_precheck']) == pair['latest_resume_precheck_sha256']
    now = datetime.now(timezone.utc)
    local = now.astimezone(timezone(timedelta(hours=8)))
    value = dict(schema='formal_main_v2', kind='root_incremental_parallel_training_review',
        reviewed_utc=now.isoformat(), completed_training_arms=2, expected_training_arms=72,
        cumulative_reviewed_arm_updates=75000, newly_reviewed_arm_updates=4000,
        reviewed_boundaries=boundaries, completed_development_trials=600,
        new_development_trials=0, completed_final_trials=0, final_states_sealed=True,
        new_terminal_jobs=[dict(job_id=j['job_id'], GPU_seconds=j['elapsed_seconds'] * j['gpus']) for j in closed],
        new_closed_GPU_seconds=21806, all_formal_cost_snapshot=s['current_round_cost'],
        active_jobs=[dict(job_id=j['job_id'], status=j['status'], suite=j['suite'], budget=j['visual_tokens'], seed=j['training_seed'], segment_end=j['segment_end']) for j in active],
        new_submissions=[j['job_id'] for j in new_jobs],
        pull_file_entries=sum(len(f.read(root / name)['files']) for name in pulls),
        audit_source_archive_map=audit_source_archive_map,
        evidence_sha256={name: f.sha(root / name) for name in sorted(refs)},
        scientific_settings_changed=False, weights_copied=False)
    f.write(root / batch, value)
    p['latest_incremental_review'] = dict(path=batch, sha256=f.sha(root / batch))
    s['next_actions'].append('Preserve both exact raw development originals locally; deliver them through the registered nonweight archive. Do not force-add raw paths. Use the pinned index-only reconstruction guard for Git delivery.')
    s['updated_utc'] = now.isoformat()
    s['updated_date'] = local.strftime('%Y-%m-%d')
    f.write(f.STATE, s)
    line = (f"{local:%Y-%m-%d %H:%M}进度：完整训练2/72组，累计75000次臂更新通过root及独立复核，8个单卡训练作业并行。"
            f"正式轮累计临时{s['current_round_cost']['allocated_gpu_seconds']} GPU秒（{s['current_round_cost']['allocated_gpu_hours']:.6f}小时），含运行中与历史失败分配。"
            '本次Long128 seed23/41验收到3000步，Long256/64 seed41验收到2000步，均无训练重放；4个空闲槽已由其他验收通过的组接续。\n\n'
            '[本轮原始回存、双重审计及成本](../' + batch + ')。开发仍600条、最终0/35200；没有新增闭环结论，封存初态未开放。5000步、11次验证和全部科学设置不变。\n\n')
    for name in ('正式主实验规划.md', '完整实验计划.md', '实验进度.md'):
        doc = root / 'docs' / name
        old = doc.read_bytes()
        heading, sep, body = old.partition(b'\n')
        assert sep
        doc.write_bytes(heading + sep + b'\n' + line.encode('utf-8') + body)
print(json.dumps(dict(receipt=batch, sha256=f.sha(root / batch), cost=value['all_formal_cost_snapshot'],
    reviewed_updates=75000, active_jobs=8, pulls=value['pull_file_entries'])))
