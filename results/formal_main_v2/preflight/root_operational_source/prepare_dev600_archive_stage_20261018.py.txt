from pathlib import Path
import json,hashlib,subprocess
root=Path.cwd();base='results/formal_main_v2/';folder=base+'preflight/long_dev600_archive_v1/'
req={}
def add(p,want=None):
 b=(root/p).read_bytes();h=hashlib.sha256(b).hexdigest();assert want is None or h==want,p
 req[p]={'path':p,'sha256':h,'bytes':len(b)}
for pull in [base+'pulls/20261017T174014Z_6a431952b07c/receipt.json',base+'pulls/20261017T174821Z_ce8ffa1139ca/receipt.json']:
 add(pull)
 for r in json.loads((root/pull).read_bytes())['files']:add(r['relative'],r['sha256'])
for arc in [base+'preflight/long_sixth_development_raw_v1/receipt.json',base+'preflight/nonweight_archive_stage_candidate_20261018/receipt.json']:
 add(arc)
 for r in json.loads((root/arc).read_bytes())['files']:add(r['archive'],r['sha256'])
for p in ['.gitignore','scripts/ops/check_nonweight_archive_stage.py','tests/test_nonweight_archive_stage.py',base+'preflight/nonweight_archive_stage_independent_20261018.json',base+'preflight/nonweight_archive_stage_root_acceptance_20261018.json',base+'preflight/long_sixth_development_real_independent_20261018.json',base+'libero_spatial/k128/seed17/training/segment_reviews/3019842.json',base+'libero_spatial/k128/seed17/training/segment_reviews/3019842_independent.json',folder+'migration_registration.json',folder+'exact_ignore_registration.json',folder+'original.gitignore.txt',folder+'proposed.gitignore.txt']:
 add(p)
out=root/(folder+'required_files.json');out.write_text(json.dumps({'schema':'cti_nonweight_archive_stage_v1','kind':'required_files','files':[req[k] for k in sorted(req)]},indent=2)+'\n',encoding='utf-8',newline='\n')
s=json.loads((root/(base+'nonweight_archives/v1/snapshots/long_dev600_20261018.json')).read_bytes());raw=[r['path'] for r in s['files']]
changed=subprocess.check_output(['git','ls-files','--modified','--others','--exclude-standard','-z','--','results/formal_main_v2']).decode().split('\0')
ordinary=sorted({p for p in changed if p and p not in raw})
subprocess.run(['git','add','--','.gitignore','scripts/ops/check_nonweight_archive_stage.py','tests/test_nonweight_archive_stage.py',*ordinary],check=True)
subprocess.run(['git','add','-f','--',*[p for p in req if p not in raw]],check=True)
# Only after every chunk and listed provenance file is concretely staged, remove
# the one old raw index entry. Originals are never removed from the filesystem.
assert all((root/p).is_file() for p in raw)
subprocess.run(['git','rm','--cached','--',raw[0]],check=True)
assert all((root/p).is_file() for p in raw)
print(json.dumps({'required':len(req),'raw_index_removed':[raw[0]],'local_originals_retained':True}))
