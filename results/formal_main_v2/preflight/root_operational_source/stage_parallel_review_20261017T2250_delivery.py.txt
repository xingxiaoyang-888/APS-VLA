"""Exact delivery plus a separately checked, unchanged existing development panel."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import subprocess

root = Path(__file__).resolve().parents[1]
base = 'results/formal_main_v2/'
batch_name = base + 'preflight/parallel_review_batch_20261017T2250.json'
required_name = base + 'preflight/parallel_review_batch_20261017T2250_git_required.json'
assert not (root / required_name).exists()

def ref(name):
    path = root / name
    assert path.is_file() and not path.is_symlink()
    data = path.read_bytes()
    return dict(path=name, sha256=hashlib.sha256(data).hexdigest(), bytes=len(data))

batch = json.loads((root / batch_name).read_bytes())
current_pulls = set()
for boundary in batch['reviewed_boundaries']:
    operation_key = '.local/root_pull_review_2250_' + boundary['job_id'] + '.json'
    archived_operation = batch['audit_source_archive_map'][operation_key]
    assert ref(archived_operation)['sha256'] == batch['evidence_sha256'][archived_operation]
    operation = json.loads((root / archived_operation).read_bytes())
    assert operation['status'] == 'passed' and operation['job_id'] == boundary['job_id']
    assert operation['pull_sha256'] == batch['evidence_sha256'][operation['pull']]
    current_pulls.add(operation['pull'])
assert len(current_pulls) == len(batch['reviewed_boundaries']) == 8
snapshot_name = base + 'nonweight_archives/v1/snapshots/long_dev600_20261018.json'
snapshot = json.loads((root / snapshot_name).read_bytes())
raw_names = {r['path'] for r in snapshot['files']}
assert len(raw_names) == 2
required = {}
existing_name = 'configs/manifests/formal_main_v2/libero_object/initial_states/development.json'
existing_sha256 = '6b7e6dd9fb5b6351ee47c7bb30eab911e7c144d700cd88f7e5162641738b2d6a'
assert batch['evidence_sha256'][existing_name] == existing_sha256
assert ref(existing_name)['sha256'] == existing_sha256
assert not subprocess.check_output(['git', 'status', '--porcelain', '--', existing_name], cwd=root).strip()
for revision in ('HEAD:', ':'):
    assert hashlib.sha256(subprocess.check_output(['git', 'show', revision + existing_name], cwd=root)).hexdigest() == existing_sha256
existing_record = base + 'preflight/object_existing_development_panel_index_20261018.json'
assert not (root / existing_record).exists()
(root / existing_record).write_text(json.dumps(dict(schema='formal_main_v2',
    kind='root_unchanged_preexisting_development_panel_index_check', status='verified',
    checked_utc=datetime.now(timezone.utc).isoformat(), file=ref(existing_name),
    head=subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root).decode().strip(),
    HEAD_index_worktree_equal=True, path_staged_or_modified=False,
    scope='Only this exact previously committed Object development panel. The generic pinned archive guard deliberately rejects state-named required payloads. This existing dependency is verified separately against HEAD, index and worktree before/after that guard. No final state path is read; no new or modified payload is exempt.',
    guard_implementation_changed=False, final_states_opened=False), indent=2) + '\n', encoding='utf-8', newline='\n')
for name, want in batch['evidence_sha256'].items():
    row = ref(name)
    assert row['sha256'] == want
    if name != existing_name:
        required[name] = row
    # Historical receipts are immutable evidence, but their unrelated mutable
    # progress paths may have advanced. Expand only this batch's verified pulls;
    # all direct evidence and each current-pull file still require exact bytes.
    if name in current_pulls:
        receipt = json.loads((root / name).read_bytes())
        for entry in receipt['files']:
            row = ref(entry['relative'])
            assert row['sha256'] == entry['sha256']
            required[row['path']] = row
# Additional terminal work after the8-pull batch was sealed: explicitly cover
# every one of the12 actual Object files, not only their containing receipt.
object_review_name = base + 'downstream_reviews/3021394.json'
object_review = json.loads((root / object_review_name).read_bytes())
assert object_review['status'] == 'passed' and object_review['completed_trials'] == 12
assert len(object_review['files']) == 12
for name, checksum in object_review['files'].items():
    row = ref(name)
    assert row['sha256'] == checksum
    required[name] = row
required[object_review_name] = ref(object_review_name)
required[existing_record] = ref(existing_record)
for name in (batch_name, snapshot_name, 'results/experiment_supervisor/state.json', base + 'slurm_accounting.json'):
    required[name] = ref(name)
for row in snapshot['files']:
    actual = ref(row['path'])
    assert all(actual[k] == row[k] for k in ('path', 'sha256', 'bytes'))
    required[actual['path']] = actual

archive_name = base + 'preflight/root_operational_source/stage_parallel_review_20261017T2250_delivery.py.txt'
assert not (root / archive_name).exists()
(root / archive_name).write_bytes(Path(__file__).read_bytes())
required[archive_name] = ref(archive_name)
changed = subprocess.check_output(['git', 'ls-files', '--modified', '--others', '--exclude-standard', '-z', '--',
    'docs', 'results/formal_main_v2', 'results/experiment_supervisor/state.json', 'results/slurm'], cwd=root).decode('utf-8').split('\0')
ordinary = sorted({name for name in changed if name and name not in raw_names})
assert existing_name not in ordinary and existing_name not in required
for name in ordinary:
    if not name.endswith('.md'):
        required[name] = ref(name)
(root / required_name).write_text(json.dumps(dict(schema='cti_nonweight_archive_stage_v1', kind='required_files',
    files=[required[k] for k in sorted(required)]), indent=2) + '\n', encoding='utf-8', newline='\n')
def add_exact(names, label, force=False):
    pathspec = root / '.local' / ('stage_2250_' + label + '.nul')
    assert not pathspec.exists()
    pathspec.write_bytes(b''.join((':(literal)' + name).encode('utf-8') + b'\0' for name in sorted(names)))
    command = ['git', 'add'] + (['-f'] if force else []) + ['--pathspec-from-file=' + str(pathspec), '--pathspec-file-nul']
    subprocess.run(command, cwd=root, check=True)
add_exact(ordinary + [required_name], 'ordinary')
# Every ignored log here comes from the verified terminal pull; no broad force-add.
add_exact(set(required) - raw_names, 'required', force=True)
print(json.dumps(dict(required_files=required_name, required_count=len(required), retained_raw_paths=sorted(raw_names),
    separately_verified_existing_development_dependency=existing_record,
    ordinary_count=len(ordinary), required_sha256=ref(required_name)['sha256'])))
