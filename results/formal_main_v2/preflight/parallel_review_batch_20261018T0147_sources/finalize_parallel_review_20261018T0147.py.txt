from pathlib import Path
from datetime import datetime, timezone, timedelta
import json, subprocess, sys

root = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(root/'scripts'))
import formal_supervisor as f

base = 'results/formal_main_v2/'
batch = base+'preflight/parallel_review_batch_20261018T0147.json'
assert not (root/batch).exists()
ids = ['3021594','3021595','3021578','3021579','3021580','3021581','3021596']
refs, sources, pulls, boundaries = set(), set(), set(), []

def collect(mapping):
    for path, checksum in mapping.items():
        assert f.sha(root/path) == checksum, path
        (sources if path.startswith('.local/') else refs).add(path)

for jid in ids:
    oprel = '.local/root_pull_review_2250_'+jid+'.json'
    op = f.read(root/oprel)
    rr = op['root_review']; ir = rr.removesuffix('.json')+'_independent.json'
    r, i = f.read(root/rr), f.read(root/ir)
    assert op['status'] == r['status'] == i['status'] == 'passed'
    assert op['job_id'] == r['job_id'] == i['job_id'] == jid
    assert f.sha(root/rr) == op['root_sha256']
    assert f.sha(root/op['pull']) == op['pull_sha256']
    assert r['completed_by_arm'] == dict.fromkeys(('A','B'),op['boundary'])
    assert r['allowed_next_segment_end'] == i['allowed_next_segment_end'] == op['allowed_next']
    collect(i['read_evidence_sha256'])
    assert i['read_evidence_sha256'][rr] == op['root_sha256']
    refs.update([rr,ir,op['pull']]); sources.add(oprel); pulls.add(op['pull'])
    assert op['boundary'] < 5000
    for entry in f.read(root/op['pull'])['files']:
        assert f.sha(root/entry['relative']) == entry['sha256'], entry['relative']
        refs.add(entry['relative'])
    boundaries.append({k:op[k] for k in ('job_id','suite','budget','seed','boundary','allowed_next','pair_gpu_seconds','latest_gpu_seconds')})

    refs.add(entry['relative'])
