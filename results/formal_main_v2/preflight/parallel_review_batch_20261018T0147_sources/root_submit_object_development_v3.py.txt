"""Root reconstructs the reviewed Object continuation under the original lock."""
from pathlib import Path
from types import SimpleNamespace
from datetime import datetime, timezone
import hashlib, json, shlex, subprocess, sys, time

root = Path(__file__).resolve().parents[1]
sys.path[:0] = [str(root / 'scripts'), str(root / 'scripts/ops')]
import formal_supervisor as f
import formal_downstream_supervisor as c

base = 'results/formal_main_v2/'
pre = base + 'preflight/object_first_pair_development_'
regrel, planrel = pre + 'registration_v3.json', pre + 'plan_v3.json'
reviewrel, submissionrel = pre + 'root_review_v3.json', pre + 'submission_root_v3.json'
assert not (root / reviewrel).exists() and not (root / submissionrel).exists()
assert f.sha(root / regrel) == 'fb88e03892b9beddcc89e751c8be7b21f335da5b238bfaa8c3fb0a553bfdb3ef'
assert f.sha(root / planrel) == '3f7e56b5367ab139a1a2dfedd2115a14aeed91f7fce805840548266b8daf3754'
reg, expected = f.read(root / regrel), f.read(root / planrel)
for name, checksum in reg['read_evidence_sha256'].items():
    assert f.sha(root / name) == checksum, name
manifest = f.read(root / reg['execution_manifest'])
panel = f.read(root / manifest['development_panel'])
all_ids = [x['trial_id'] + ':' + method for x in panel['states'] for method in reg['methods']]
assert len(all_ids) == len(set(all_ids)) == 600
assert reg['prior_terminal_trial_ids'] == all_ids[:132]
assert reg['expected_next_trial_ids'] == all_ids[132:600]
assert manifest['environment'] == dict(max_steps=280, action_chunk=8, wait_steps=10, environment_seed=0, runtime_seed=17)
assert c.resume_prefix_inputs(expected['invocation']) == reg['resume_prefix_inputs']
assert reg['previous_allocation_gpu_seconds'] == 1787
assert reg['additional_stop_after'] == 468 and reg['expected_cumulative_trials_after_segment'] == 600
assert reg['engineering_estimate']['guarantee'] is False and reg['engineering_estimate']['allocation_limit_seconds'] == 14400
assert reg['resources']['time'] == '04:00:00' and reg['resources']['gres'] == 'gpu:l40:1'
args = SimpleNamespace(command='submit', stage='development', execution_manifest=reg['execution_manifest'],
    source_inventory=base+'source/formal_main_v2_run_564be0995e.json', output_relative=None,
    freeze_directory=None, profile_panel=None, prior_review=reg['previous_root_review'],
    collection_review=None, repair_record=None, resume=True, stop_after=468,
    expected_plan_sha256=reg['plan_sha256'], root_submit=True)
started = time.perf_counter()
with f.supervisor_lock():
    state = f.read(f.STATE)
    assert state['pending_submission'] is None
    jobs = f.refresh(state)
    live = [j for j in c.formal_jobs(state) if j['status'] not in c.TERMINAL]
    assert len(live) < 8
    assert not any(j['run_root'] == expected['invocation']['run_root'] for j in live)
    progress = state['formal_main_progress']
    dev = progress['first_object_development']
    assert dev['completed_trials'] == 132
    assert progress['completed_development_trials'] == 732
    assert not progress['test_opened'] and not progress['global_release_created']
    print(json.dumps(dict(status='root_reconstruction_running', active_requests=len(live), plan_sha256=reg['plan_sha256'])), flush=True)
    rebuilt = c.plan(args, state, project=root)
    assert rebuilt == expected
    assert rebuilt['plan_sha256'] == '1cf4ea60ea49be35bb1b75899957f12980196c62f1ac70889b4773428d8682ef'
    for name, checksum in reg['read_evidence_sha256'].items():
        assert f.sha(root / name) == checksum, name
    # Transport checks registered source/input and mutable ledger prefixes;
    # root separately hashes the selected weights immediately before submit.
    selectedrel = reg['actual_selected_server_review']
    old_selected = f.read(root/selectedrel)
    assert old_selected['status'] == 'passed' and old_selected['job_id'] == '3021103'
    assert old_selected['suite'] == 'libero_object' and old_selected['visual_tokens'] == 128
    specs = old_selected['selected']
    assert len(specs) == 2 and {x['arm'] for x in specs} == {'A','B'}
    for spec in specs:
        row = next(x for x in manifest['methods'] if x['identity']['method'] == spec['arm'])
        assert row['checkpoint'] == spec['path']
        assert row['identity']['checkpoint_sha256'] == spec['expected_sha256']
    assert not any(j['status'] not in c.TERMINAL and j['stage_key'] == old_selected['registered_terminal_job']['stage_key'] for j in jobs)
    code = old_selected['probe_program']
    assert hashlib.sha256(code.encode('utf-8')).hexdigest() == old_selected['probe_program_sha256']
    answer = subprocess.run(f.SSH+[shlex.quote(f.H+'/venv/bin/python')+' -B -'],
        input=code,capture_output=True,text=True,encoding='utf-8',check=True,timeout=180)
    checks = json.loads(answer.stdout)
    assert len(checks) == 2 and {x['arm'] for x in checks} == {'A','B'}
    for spec in specs:
        actual = next(x for x in checks if x['arm'] == spec['arm'])
        assert actual['path'] == spec['path'] and actual['sha256'] == spec['expected_sha256']
        assert actual['stable_during_hash'] and actual['exists'] and actual['bytes'] > 0
    selectedfresh = pre+'selected_server_v3.json'
    assert not (root/selectedfresh).exists()
    f.write(root/selectedfresh,dict(schema='formal_main_v2',kind='root_development_presubmission_selected_server_SHA_review',
        status='passed',reviewed_utc=datetime.now(timezone.utc).isoformat(),suite='libero_object',
        selected=specs,server_selected_checks=checks,prior_selected_review=selectedrel,
        probe_program=code,probe_program_sha256=old_selected['probe_program_sha256'],remote_returncode=answer.returncode,
        inputs_sha256={n:f.sha(root/n) for n in (selectedrel,regrel,planrel,reg['execution_manifest'],'.local/root_submit_object_development_v3.py')},
        weights_copied=False,weights_deserialized=False,additional_GPU_seconds=0,final_states_opened=False,
        clarification='Original Transport checks watched source/input and ledger prefixes. This separate root probe supplies fresh selected weight hashes; the runtime loader repeats checkpoint checks. Candidate wording attributing selected-weight hashing to Transport is corrected here.'))
    refs = dict(reg['read_evidence_sha256'])
    for name in (regrel, planrel, selectedfresh, '.local/root_submit_object_development_v3.py'):
        refs[name] = f.sha(root / name)
    review = dict(schema='formal_main_v2', kind='root_object_development_continuation_reconstruction',
        status='passed_cpu_only', reviewed_utc=datetime.now(timezone.utc).isoformat(),
        registration=regrel, plan=planrel, plan_sha256=reg['plan_sha256'], plan_rebuilt_equal=True,
        current_state_fresh_under_supervisor_lock=True, active_requests_before=len(live),
        completed_trials_before=132, additional_stop_after=468, cumulative_target=600, total_target=600,
        first132_exact_original_prefix=True, expected_remaining468_exact=True, same_run_explicit_resume=True,
        source_controller_unchanged=True, reconstruction_wall_seconds=time.perf_counter()-started,
        fresh_selected_server_review=selectedfresh,fresh_selected_server_review_sha256=f.sha(root/selectedfresh),
        selected_weight_SHA_checked_by_separate_root_probe=True,
        Transport_scope='Original watched source/input and ledger prefixes; selected weights verified by the separately bound root probe and again by runtime loader.',
        final_states_opened=False, weights_copied=False, scientific_settings_changed=False,
        GPU_seconds=0, read_evidence_sha256=dict(sorted(refs.items())))
    f.write(root / reviewrel, review)
    # The original transport verifies all watched server source/input and
    # mutable prefix identities, retains pending on ambiguous submission.
    job = c.submit_plan(rebuilt, state, project=root, supervisor=f, transport=c.Transport(root),
        expected_plan_sha256=reg['plan_sha256'])
    dev.update(status='remaining468_submitted_pending_actual_GPU_review', active_job_id=job['job_id'],
        active_job_status=job['status'], latest_job_id=job['job_id'], next_stop_after=468,
        expected_cumulative_trials=600, registration=regrel, root_plan_review=reviewrel,
        latest_plan=planrel, latest_plan_sha256=reg['plan_sha256'], invocation=job['invocation'],
        invocation_sha256=job['invocation_sha256'],
        next='Review the same-ledger cumulative600 after this468-trial continuation; preserve prior132 exactly and no reruns. If timeout, audit all partial records and allocation cost before separately registered explicit recovery. This finishes only the original six of22 Object deployment instances; no final release.')
    sr = dict(schema='formal_main_v2', kind='root_object_development_continuation_submission',
        status='submitted_pending_actual_GPU_review', recorded_utc=datetime.now(timezone.utc).isoformat(),
        job_id=job['job_id'], invocation=job['invocation'], invocation_sha256=job['invocation_sha256'],
        plan_sha256=reg['plan_sha256'], completed_before=132, additional_stop_after=468,
        cumulative_target=600, total_target=600, root_reconstruction=reviewrel,
        original_transport_prepare_passed=True, no_completed_trial_reruns=True,
        final_states_opened=False, weights_copied=False, scientific_settings_changed=False,
        evidence_sha256={name:f.sha(root/name) for name in
            (regrel,planrel,reviewrel,selectedfresh,job['invocation'],base+'submissions/'+job['job_id']+'.json')})
    f.write(root / submissionrel, sr)
    dev['latest_submission_root_review'] = submissionrel
    dev['latest_submission_root_review_sha256'] = f.sha(root / submissionrel)
    state['updated_utc'] = sr['recorded_utc']
    f.write(f.STATE, state)
print(json.dumps(dict(status=sr['status'],job_id=job['job_id'],review=reviewrel,
    review_sha256=f.sha(root/reviewrel),submission=submissionrel,sha256=f.sha(root/submissionrel))),flush=True)
