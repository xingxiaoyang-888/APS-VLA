"""Small CPU-only contract checks; no repository ledger or model is read."""
import copy
import unittest

import audit_legacy as audit


def terminal():
    md = dict(suite="libero_10", method="dense", visual_tokens=512,
              per_camera_budgets=[256, 256], policy_calls=1, vision_calls=1,
              packing_calls=1, future_head_calls=0, scorer_calls=0,
              language_cache="disabled")
    chunk = dict(control_step=0, observation_sha256="a" * 64,
                 environment_actions=[[0.0] * 7 for _ in range(8)],
                 normalized_actions=[[1.5] * 7 for _ in range(8)], metadata=md)
    return dict(chunks=[chunk], executed_actions=copy.deepcopy(chunk["environment_actions"]),
                control_steps=8, action_chunk=8, configured_wait_steps=10, wait_steps=10,
                method="dense", suite="libero_10", policy_calls=1, status="completed",
                success=True, termination="environment_success", trial_id="synthetic")


class ScopeTests(unittest.TestCase):
    def test_normalized_overrange_is_not_environment_rejection(self):
        result = audit.inspect_terminal(terminal(), 520)
        self.assertEqual(result["maximum_abs_normalized"], 1.5)
        self.assertFalse(result["reuse_authorized"])
        self.assertFalse(result["new_adapter_equivalence_proven"])

    def test_threshold_inclusive(self):
        x = terminal()
        x["chunks"][0]["environment_actions"][0][0] = audit.THRESHOLD
        x["executed_actions"][0][0] = audit.THRESHOLD
        self.assertTrue(audit.inspect_terminal(x, 520)["executed_exact_float64_queue_prefix"])

    def test_environment_overrange_rejected(self):
        x = terminal(); x["chunks"][0]["environment_actions"][0][0] = 1.00002
        with self.assertRaisesRegex(ValueError, "old threshold"):
            audit.inspect_terminal(x, 520)

    def test_nonfinite_and_bad_shape_rejected(self):
        for name in ("environment_actions", "normalized_actions"):
            for replacement in (float("nan"), float("inf"), True):
                with self.subTest(name=name, replacement=replacement):
                    x = terminal(); x["chunks"][0][name][0][0] = replacement
                    with self.assertRaisesRegex(ValueError, "nonfinite/non-numeric"):
                        audit.inspect_terminal(x, 520)
            x = terminal(); x["chunks"][0][name].pop()
            with self.assertRaisesRegex(ValueError, "row count"):
                audit.inspect_terminal(x, 520)

    def test_exact_prefix_rejects_changed_value_or_signed_zero(self):
        for value in (0.01, -0.0):
            with self.subTest(value=value):
                x = terminal(); x["executed_actions"][0][0] = value
                with self.assertRaisesRegex(ValueError, "exact float64 queue prefix"):
                    audit.inspect_terminal(x, 520)

    def test_early_success_allows_partial_final_chunk(self):
        x = terminal(); x["control_steps"] = 3; x["executed_actions"] = x["executed_actions"][:3]
        self.assertEqual(audit.inspect_terminal(x, 520)["executed_control_actions"], 3)

    def test_normal_missing_call_is_rejected(self):
        x = terminal(); x["policy_calls"] = 2
        with self.assertRaisesRegex(ValueError, "normal policy/chunk coverage"):
            audit.inspect_terminal(x, 520)

    def test_original_error_is_retained_with_unknown_eighth_chunk(self):
        x = terminal(); x.update(suite="libero_object", control_steps=56, policy_calls=8,
            status="policy_error", success=False, termination="ValueError", error_phase="policy",
            error="environment action outside normalized controller [-1, 1] range",
            trial_id="libero_object:pick_up_the_chocolate_pudding_and_place_it_in_the_basket:state-0049:env-0:dense")
        chunk = x["chunks"][0]; chunk["metadata"]["suite"] = "libero_object"
        x["chunks"] = [dict(copy.deepcopy(chunk), control_step=8*i) for i in range(7)]
        x["executed_actions"] = [copy.deepcopy(row) for c in x["chunks"] for row in c["environment_actions"]]
        result = audit.inspect_terminal(x, 280)
        self.assertEqual(result["missing_chunk_values"], "unknown_not_reconstructed")
        self.assertFalse(result["all_policy_calls_have_recorded_chunks"])
        self.assertFalse(result["reuse_authorized"])

    def test_forbidden_ledger_rejected_before_io(self):
        for path in ("results/formal_main_v2/libero_10/downstream/final/abc/attempt01/trials.jsonl",
                     "results/formal_main_v2/libero_10/downstream/development/abc/attempt02/trials.jsonl",
                     "../trials.jsonl", "F:/CTI-VLA/weights.pt"):
            with self.subTest(path=path), self.assertRaisesRegex(ValueError, "pinned development paths"):
                audit.guard_ledger(path)

    def test_duplicate_keys_nonfinite_json_and_bad_future_calls_rejected(self):
        for data in ('{"a":1,"a":2}', '{"a":NaN}'):
            with self.assertRaises(ValueError):
                audit.decode(data)
        x = terminal(); x["chunks"][0]["metadata"]["future_head_calls"] = 1
        with self.assertRaisesRegex(ValueError, "future/cache"):
            audit.inspect_terminal(x, 520)


if __name__ == "__main__":
    unittest.main(verbosity=2)
