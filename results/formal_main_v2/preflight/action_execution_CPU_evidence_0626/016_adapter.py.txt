"""Undeployed CPU candidate for the registered uniform execution compatibility.

There is no command-line or GPU entrypoint. This scope alone is not authorization
to run real rollouts: actual-controller acceptance, execution identities, ledger
equivalence, invocation accounting and root/independent release remain external.
"""
from __future__ import annotations

import __future__
import ast
import copy
from contextlib import contextmanager
import hashlib
import json
import marshal
from pathlib import Path
import sys
import threading
from types import CodeType, FunctionType, ModuleType

VERSION = "formal-main-v2-controller-exec-cpu-candidate-v1"
REGISTRATION = "configs/experiments/formal_main_v2_action_execution_compat_v1.json"
REGISTRATION_SHA256 = "0917bff60e632a5be3c2db50b16daa0b904e4754a3ba31334802060f75d3eefe"
INVENTORY = "results/formal_main_v2/source/formal_main_v2_run_564be0995e.json"
INVENTORY_SHA256 = "5baef5e7adc942b9d9c58eb7afe3faf69417f4d7939791997876652238a12926"
SOURCE_SHA256 = "dd3e5450e80b8e1f05a78acf6a05daf393f7362420fd8e87a8e6f5ec03feb61e"
RUNTIME = "src/cti_vla/evaluation/libero_runtime.py"
RUNTIME_SHA256 = "2fc1bacbd1268a36f2e31451c70a650be4f8d0c8f6803e7c8b2a3d45f701c781"
TRIALS = "src/cti_vla/evaluation/formal_trials.py"
TRIALS_SHA256 = "93a0971dbc8bbf4cb462490b0b00a337d92f015af83b798bb635163a9cb864f6"
RUNTIME_NAME = "cti_vla.evaluation.libero_runtime"
TRIALS_NAME = "cti_vla.evaluation.formal_trials"
CODE_MARSHAL_VERSION = 2
_LOCK = threading.Lock()

_BRANCH = ast.parse('''if bool((np.abs(chunk.environment_actions) > 1.00001).any()):
    raise ValueError("environment action outside normalized controller [-1, 1] range")
''').body[0]


def file_sha256(path):
    digest = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for part in iter(lambda: stream.read(8 << 20), b""):
            digest.update(part)
    return digest.hexdigest()


def signature(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, ensure_ascii=False,
        separators=(",", ":"), allow_nan=False).encode()).hexdigest()


def code_sha256(code):
    return hashlib.sha256(marshal.dumps(code, CODE_MARSHAL_VERSION)).hexdigest()


def ast_text(node):
    return ast.dump(node, include_attributes=False)


def _path(root, relative):
    raw = str(relative).replace("\\", "/")
    path = Path(raw)
    parts = raw.lower().split("/")
    if (path.is_absolute() or ":" in raw or ".." in path.parts or
            any("sealed" in part or part in ("weights", "checkpoints", "initial_states", "init_states", "test") for part in parts)
            or path.suffix.lower() in (".pt", ".pth", ".bin", ".safetensors", ".init", ".pruned_init")
            or "final_registry" in raw.lower() or path.name == "panel.json"):
        raise ValueError("forbidden non-source evidence path")
    target = Path(root) / path
    if target.resolve() != target.absolute() or Path(root).resolve() not in target.resolve().parents:
        raise ValueError("source/evidence path escapes its declared root")
    return target


def _read(path, expected, watched):
    data = Path(path).read_bytes()
    if hashlib.sha256(data).hexdigest() != expected:
        raise ValueError("pinned evidence SHA mismatch: " + str(path))
    watched[str(path)] = expected
    return data


def verify_inputs(project_root, source_root, expected_adapter_sha256):
    watched = {}
    if file_sha256(__file__) != expected_adapter_sha256:
        raise ValueError("candidate self SHA differs")
    watched[str(Path(__file__).absolute())] = expected_adapter_sha256
    registration = json.loads(_read(_path(project_root, REGISTRATION), REGISTRATION_SHA256, watched))
    inventory = json.loads(_read(_path(project_root, INVENTORY), INVENTORY_SHA256, watched))
    contract = registration["implementation_contract"]
    if (registration["schema"] != "formal_main_v2" or registration["version"] != "formal-main-v2-controller-exec-v1"
            or contract["target_module"] != RUNTIME or contract["target_module_sha256"] != RUNTIME_SHA256
            or inventory["source_sha256"] != SOURCE_SHA256 or len(inventory["files"]) != 173
            or inventory["files"].get(TRIALS) != TRIALS_SHA256
            or inventory["files"].get(RUNTIME) != RUNTIME_SHA256):
        raise ValueError("unknown source/registration contract")
    runtime_bytes = None
    trials_bytes = None
    for relative, digest in inventory["files"].items():
        data = _read(_path(source_root, relative), digest, watched)
        if relative == RUNTIME:
            runtime_bytes = data
        elif relative == TRIALS:
            trials_bytes = data
    # Pin the known from-import alias without executing the rest of formal_trials.
    tree = ast.parse(trials_bytes)
    aliases = [node for node in ast.walk(tree) if isinstance(node, ast.ImportFrom)
        and node.module == RUNTIME_NAME and any(item.name == "run_episode" for item in node.names)]
    if len(aliases) != 1 or [(x.name, x.asname) for x in aliases[0].names] != [("run_episode", None)]:
        raise ValueError("unknown formal_trials run_episode alias")
    return watched, runtime_bytes


def _code_in(code, names):
    for name in names:
        found = [item for item in code.co_consts if isinstance(item, CodeType) and item.co_name == name]
        if len(found) != 1:
            raise ValueError("ambiguous frozen code object: " + name)
        code = found[0]
    return code


def _check_function(function, expected_code, module, name):
    if (type(function) is not FunctionType or function.__globals__ is not module.__dict__
            or function.__closure__ is not None or function.__name__ != name
            or function.__module__ != module.__name__
            or code_sha256(function.__code__) != code_sha256(expected_code)):
        raise ValueError("loaded callable is not the pinned frozen function: " + name)


def derive_function(runtime, source_bytes):
    """Return the one permitted AST edit; never replace or reimplement the loop."""
    if hashlib.sha256(source_bytes).hexdigest() != RUNTIME_SHA256:
        raise ValueError("unknown frozen function source bytes")
    original = runtime.run_episode
    filename = original.__code__.co_filename
    compiled = compile(source_bytes, filename, "exec", dont_inherit=True)
    _check_function(original, _code_in(compiled, ["run_episode"]), runtime, "run_episode")
    _check_function(runtime._observation_hash, _code_in(compiled, ["_observation_hash"]), runtime, "_observation_hash")
    _check_function(runtime.PolicyChunk.__post_init__, _code_in(compiled, ["PolicyChunk", "__post_init__"]), runtime, "__post_init__")
    if (original.__defaults__ is not None or original.__kwdefaults__ != dict(wait_steps=10, max_steps=220, policy_watchdog_seconds=60.0)
            or tuple(runtime.DUMMY_ACTION) != (0., 0., 0., 0., 0., 0., -1.)):
        raise ValueError("frozen defaults or dummy action differ")
    functions = [node for node in ast.parse(source_bytes).body if isinstance(node, ast.FunctionDef) and node.name == "run_episode"]
    if len(functions) != 1 or functions[0].decorator_list:
        raise ValueError("unknown target function definition")
    before = functions[0]
    after = copy.deepcopy(before)
    matches = []
    def visit(node, path=()):
        for field, value in ast.iter_fields(node):
            if isinstance(value, list):
                for index, child in enumerate(value):
                    if isinstance(child, ast.AST):
                        if ast_text(child) == ast_text(_BRANCH):
                            matches.append((value, index, path + ((field, index),), child))
                        else:
                            visit(child, path + ((field, index),))
            elif isinstance(value, ast.AST):
                visit(value, path + ((field, None),))
    visit(after)
    if len(matches) != 1:
        raise ValueError("exact registered range branch not unique")
    sequence, index, path, removed = matches[0]
    del sequence[index]
    modified_ast = ast_text(after)
    # Restore and compare the whole function, including every other nested node.
    sequence.insert(index, removed)
    if ast_text(after) != ast_text(before):
        raise ValueError("AST changes exceed the single range branch")
    del sequence[index]
    namespace = {}
    exec(compile(ast.Module(body=[after], type_ignores=[]), filename, "exec",
        flags=__future__.annotations.compiler_flag, dont_inherit=True), runtime.__dict__, namespace)
    replacement = namespace["run_episode"]
    if (replacement.__globals__ is not original.__globals__ or replacement.__defaults__ != original.__defaults__
            or replacement.__kwdefaults__ != original.__kwdefaults__ or replacement.__annotations__ != original.__annotations__):
        raise ValueError("derived function calling contract differs")
    proof = dict(original_function_ast_sha256=signature(ast_text(before)),
        replacement_function_ast_sha256=signature(modified_ast), removed_branch_ast_sha256=signature(ast_text(removed)),
        exact_removed_node_path=[list(item) for item in path], removed_if_count=1,
        other_AST_nodes_unchanged=True, original_code_sha256=code_sha256(original.__code__),
        replacement_code_sha256=code_sha256(replacement.__code__), code_marshal_version=CODE_MARSHAL_VERSION)
    return original, replacement, proof


def _snapshot_function(function):
    return dict(code=function.__code__, defaults=function.__defaults__, kwdefaults=copy.deepcopy(function.__kwdefaults__),
        annotations=copy.deepcopy(function.__annotations__), attributes=copy.deepcopy(function.__dict__))


def _function_same(function, saved):
    return (function.__code__ is saved["code"] and function.__defaults__ == saved["defaults"]
        and function.__kwdefaults__ == saved["kwdefaults"] and function.__annotations__ == saved["annotations"]
        and function.__dict__ == saved["attributes"])


def _restore_function(function, saved):
    function.__code__ = saved["code"]
    function.__defaults__ = saved["defaults"]
    function.__kwdefaults__ = copy.deepcopy(saved["kwdefaults"])
    function.__annotations__ = copy.deepcopy(saved["annotations"])
    function.__dict__.clear()
    function.__dict__.update(copy.deepcopy(saved["attributes"]))


@contextmanager
def install_action_execution_compat(*, project_root, source_root, runtime_module, formal_trials_module,
        expected_adapter_sha256, emit=None):
    """Explicit CPU-candidate scope; controller/runtime release is not implied."""
    if not _LOCK.acquire(blocking=False):
        raise RuntimeError("duplicate or nested action-execution installation")
    emit = emit or (lambda event, **fields: None)
    installed = False
    runner_raised = False
    saved = None
    try:
        watched, source_bytes = verify_inputs(project_root, source_root, expected_adapter_sha256)
        for module, name, relative in ((runtime_module, RUNTIME_NAME, RUNTIME), (formal_trials_module, TRIALS_NAME, TRIALS)):
            path = _path(source_root, relative)
            if (type(module) is not ModuleType or module.__name__ != name or sys.modules.get(name) is not module
                    or Path(module.__file__).resolve() != path.resolve()):
                raise ValueError("unknown loaded module origin/binding: " + name)
        original, replacement, ast_proof = derive_function(runtime_module, source_bytes)
        if (formal_trials_module.run_episode is not original
                or Path(original.__code__.co_filename).resolve() != _path(source_root, RUNTIME).resolve()):
            raise ValueError("formal_trials alias or function origin differs")
        saved, patched_saved = _snapshot_function(original), _snapshot_function(replacement)
        dependencies = (runtime_module.PolicyChunk, runtime_module._observation_hash, runtime_module.DUMMY_ACTION)
        post_init_code, observation_code = dependencies[0].__post_init__.__code__, dependencies[1].__code__
        identity = dict(schema="formal_main_v2", kind="candidate_action_execution_identity", version=VERSION,
            candidate_only=True, deployment_authorized=False, adapter_sha256=expected_adapter_sha256,
            registration_sha256=REGISTRATION_SHA256, source_inventory_sha256=INVENTORY_SHA256,
            frozen_source_sha256=SOURCE_SHA256, frozen_source_files=173, source_root=str(Path(source_root)),
            target_module=RUNTIME, target_module_sha256=RUNTIME_SHA256, alias_module=TRIALS, alias_module_sha256=TRIALS_SHA256,
            runtime_python=sys.version, ast_proof=ast_proof, watched_files=watched,
            methods="uniform scope for dense/uniform/A/B; no method argument or branch",
            action_transformation="none; existing finite PolicyChunk passed unchanged",
            controller_verified=False, real_environment_acceptance_pending=True)
        identity["identity_sha256"] = signature(identity)
        emit("action_execution_candidate_enter", identity=copy.deepcopy(identity))
        runtime_module.run_episode = replacement
        formal_trials_module.run_episode = replacement
        installed = True
        try:
            yield copy.deepcopy(identity)
        except BaseException:
            runner_raised = True
            raise
        finally:
            intact = (runtime_module.run_episode is replacement and formal_trials_module.run_episode is replacement
                and _function_same(original, saved) and _function_same(replacement, patched_saved))
            helper_intact = (runtime_module.PolicyChunk is dependencies[0] and runtime_module._observation_hash is dependencies[1]
                and runtime_module.DUMMY_ACTION is dependencies[2]
                and dependencies[0].__post_init__.__code__ is post_init_code and dependencies[1].__code__ is observation_code)
            _restore_function(original, saved)
            runtime_module.run_episode = original
            formal_trials_module.run_episode = original
            restored = runtime_module.run_episode is original and formal_trials_module.run_episode is original and _function_same(original, saved)
            changed = [path for path, digest in watched.items() if not Path(path).is_file() or file_sha256(path) != digest]
            emit("action_execution_candidate_exit", identity_sha256=identity["identity_sha256"],
                original_bindings_restored=restored, scope_binding_and_code_intact=intact,
                finite_shape_and_observation_helpers_intact=helper_intact, source_inputs_unchanged=not changed,
                changed_inputs=changed, runner_raised=runner_raised, candidate_only=True, deployment_authorized=False)
            if not (intact and helper_intact and restored) or changed:
                raise RuntimeError("action-execution candidate scope/source tampering; bindings restored")
    finally:
        # Also protect an unexpected exception during installation itself.
        if installed and saved is not None:
            _restore_function(original, saved)
            runtime_module.run_episode = original
            formal_trials_module.run_episode = original
        _LOCK.release()
