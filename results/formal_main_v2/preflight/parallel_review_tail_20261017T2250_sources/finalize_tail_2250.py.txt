"""Root-only late heartbeat boundary and slot-refill receipt."""
from pathlib import Path
from datetime import datetime, timezone, timedelta
import json, subprocess, sys

root = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(root / 'scripts'))
import formal_supervisor as f
base = 'results/formal_main_v2/'
name = base + 'preflight/parallel_review_tail_20261017T2250.json'
assert not (root / name).exists()
op_name = '.local/root_pull_review_2250_3021352.json'
op = f.read(root / op_name)
rr = op['root_review']
ir = rr.removesuffix('.json') + '_independent.json'
r, i = f.read(root / rr), f.read(root / ir)
assert r['status'] == i['status'] == 'passed'
assert r['job_id'] == i['job_id'] == '3021352'
assert r['completed_by_arm'] == dict.fromkeys(('A', 'B'), 3000)
assert r['allowed_next_segment_end'] == i['allowed_next_segment_end'] == 3500
assert op['pair_gpu_seconds'] == 21283 and op['latest_gpu_seconds'] == 2682
refs = {rr, ir, op['pull']}
sources = {op_name, '.local/root_pull_review_one_20261017T2250.py',
    '.local/audit_training_segment_3021352.py', '.local/audit_training_segment_3021352.log',
    '.local/record_parallel_boundary.py', '.local/reconcile_parallel_progress_20261018.py',
    '.local/finalize_tail_2250.py', '.local/stage_verify_tail_2250.py', '.local/fill_reviewed_pair_slot.py'}
for rel, checksum in i['read_evidence_sha256'].items():
    assert f.sha(root / rel) == checksum, rel
    if rel.startswith('.local/'):
        sources.add(rel)
    else:
        refs.add(rel)
assert i['read_evidence_sha256'][rr] == f.sha(root / rr) == op['root_sha256']
assert f.sha(root / op['pull']) == op['pull_sha256']
for row in f.read(root / op['pull'])['files']:
    assert f.sha(root / row['relative']) == row['sha256']
    refs.add(row['relative'])
parent = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root).decode().strip()
assert parent == 'b1e3c864a07bfc2659e0a8dc1a5a4feb75ad2fda'
before = json.loads(subprocess.check_output(['git', 'show', 'HEAD:results/experiment_supervisor/state.json'], cwd=root))
old_ids = {j['job_id'] for j in before['registered_jobs']}
with f.supervisor_lock():
    state = f.read(f.STATE)
    jobs = f.refresh(state)
    active = [j for j in jobs if j['status'] in f.ACTIVE | {'SUBMITTED'}]
    assert len(active) == 8 and len({j['stage_key'] for j in active}) == 8, 'Refill an available reviewed slot before finalizing'
    assert all(j['mode'] == 'train' and j['gpus'] == 1 for j in active)
    p = state['formal_main_progress']
    assert p['trained_arms'] == 4 and p['root_and_independent_reviewed_arm_updates'] == 100000
    assert p['completed_development_trials'] == 612 and p['completed_final_trials'] == 0
    assert not p['test_opened'] and not p['global_release_created']
    new_jobs = [j for j in jobs if j['job_id'] not in old_ids]
    assert new_jobs and all(j in active for j in new_jobs)
    for job in new_jobs:
        assert f.sha(root / job['invocation']) == job['invocation_sha256']
        pair = p['parallel_execution']['pair_tracking'][f"{job['suite']}_k{job['visual_tokens']}_seed{job['training_seed']}"]
        assert pair['active_job_id'] == job['job_id']
        refs.update([job['invocation'], base + 'submissions/' + job['job_id'] + '.json', pair['latest_resume_precheck']])
        assert f.sha(root / pair['latest_resume_precheck']) == pair['latest_resume_precheck_sha256']
    folder = root / (base + 'preflight/parallel_review_tail_20261017T2250_sources')
    assert not folder.exists()
    folder.mkdir()
    source_map = {}
    for rel in sorted(sources):
        source = root / rel
        assert source.is_file() and not source.is_symlink()
        destination = folder / (source.name + '.txt')
        assert not destination.exists()
        destination.write_bytes(source.read_bytes())
        source_map[rel] = destination.relative_to(root).as_posix()
        refs.add(source_map[rel])
    now = datetime.now(timezone.utc)
    receipt = dict(schema='formal_main_v2', kind='root_late_parallel_boundary_and_refill', status='passed',
        reviewed_utc=now.isoformat(), parent_delivered_commit=parent,
        job_id='3021352', suite='libero_10', budget=64, seed=23, common_update=3000,
        allowed_next_segment_end=3500, root_review=rr, independent_review=ir, pull_receipt=op['pull'],
        current_job_gpu_seconds=2682, pair_gpu_seconds=21283, all_costs_included_in_snapshot_not_added_twice=True,
        completed_training_arms=4, reviewed_arm_updates=100000, completed_development_trials=612,
        completed_final_trials=0, all_formal_cost_snapshot=state['current_round_cost'],
        active_jobs=[{k:j.get(k) for k in ('job_id','status','suite','visual_tokens','training_seed','segment_end')} for j in active],
        new_submission_job_ids=[j['job_id'] for j in new_jobs],
        audit_source_archive_map=source_map, evidence_sha256={rel:f.sha(root / rel) for rel in sorted(refs)},
        final_states_opened=False, weights_copied=False, scientific_settings_changed=False)
    f.write(root / name, receipt)
    p['latest_incremental_review'] = dict(path=name, sha256=f.sha(root / name))
    state['updated_utc'] = now.isoformat()
    state['updated_date'] = now.astimezone(timezone(timedelta(hours=8))).strftime('%Y-%m-%d')
    f.write(f.STATE, state)
    local = now.astimezone(timezone(timedelta(hours=8)))
    line = (f'{local:%Y-%m-%d %H:%M}补充：Long64 seed23已到3000步并通过双重复核，累计100000/360000臂更新（27.8%）；完整训练仍4/72。'
        f"8个单卡训练请求保持并行，累计临时{state['current_round_cost']['allocated_gpu_seconds']} GPU秒（{state['current_round_cost']['allocated_gpu_hours']:.6f}小时），含运行中和失败分配。"
        '空闲卡由已复核的其他pair接续；本pair下一仅3500，不重启或跳过恢复验收。开发612条、最终0/35200，继续封存。'
        '\n\n[末次分段、补槽与全成本](../' + name + ')。\n\n')
    for docname in ('正式主实验规划.md','完整实验计划.md','实验进度.md'):
        path = root / 'docs' / docname
        heading, sep, body = path.read_bytes().partition(b'\n')
        assert sep
        path.write_bytes(heading + sep + b'\n' + line.encode('utf-8') + body)
print(json.dumps(dict(receipt=name, sha256=f.sha(root / name), active=8,
    reviewed_updates=100000, new_jobs=receipt['new_submission_job_ids'], cost=receipt['all_formal_cost_snapshot'])))
