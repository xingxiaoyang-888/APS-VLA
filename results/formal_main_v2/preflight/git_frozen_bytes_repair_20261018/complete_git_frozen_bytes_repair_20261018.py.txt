from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import subprocess

root = Path(__file__).resolve().parents[1]
base = 'results/formal_main_v2/preflight/'
folder = base + 'git_frozen_bytes_repair_20261018/'
def sha(data): return hashlib.sha256(data).hexdigest()
def read(rel): return json.loads((root / rel).read_bytes())
def git(*args): return subprocess.check_output(['git', *args], cwd=root)
registration = folder + 'registration.json'
independent = base + 'git_frozen_bytes_repair_independent_20261018.json'
assert sha((root / registration).read_bytes()) == 'eb3007aaf98331e03c1f71e6d108dd2f46c6373c30c943f8ea3a73b498f899b9'
assert sha((root / independent).read_bytes()) == 'c6c1acde3cd654fb980a46e0ab21692f21ee9e56898e7042144d9aab457f131f'
r, i = read(registration), read(independent)
assert i['status'] == 'passed' and i['phase'] == 'proposal_before_root_stage'
assert git('rev-parse', 'HEAD').decode().strip() == r['git_parent']
inventory = read(r['source_inventory'])
assert sha((root / r['source_inventory']).read_bytes()) == r['source_inventory_sha256']
assert len(inventory['files']) == 173
for rel, want in inventory['files'].items():
    assert sha((root / rel).read_bytes()) == want, rel
assert sha((root / '.gitattributes').read_bytes()) == r['proposed_attributes_sha256']
proposal = (root / r['proposed_attributes_archive']).read_bytes()
assert sha(proposal) == r['proposed_attributes_sha256']
for row in r['files']:
    raw = (root / row['path']).read_bytes()
    old = git('cat-file', 'blob', 'HEAD:' + row['path'])
    assert sha(raw) == row['frozen_sha256'] and sha(old) == row['old_git_sha256']
    assert raw.replace(b'\r\n', b'\n') == old
subprocess.run(['git', 'add', '--renormalize', '--', *[x['path'] for x in r['files']]], cwd=root, check=True)
checked = {}
for rel, want in inventory['files'].items():
    assert sha((root / rel).read_bytes()) == want, rel
    assert sha(git('cat-file', 'blob', ':' + rel)) == want, rel
    checked[rel] = want
assert git('cat-file', 'blob', ':.gitattributes') == proposal
old_bindings = []
for row in i['historical_codec_binding']:
    receipt = read(row['receipt'])
    assert sha((root / row['receipt']).read_bytes()) == row['sha256']
    assert receipt['read_evidence_sha256']['.gitattributes'] == r['original_attributes_sha256']
    assert sha((root / row['resolve_original_bytes_at']).read_bytes()) == r['original_attributes_sha256']
    old_bindings.append(row)
value = dict(schema='formal_main_v2', kind='root_git_frozen_byte_preservation_review',
    status='passed_stage_before_commit', reviewed_utc=datetime.now(timezone.utc).isoformat(),
    registration=registration, registration_sha256=sha((root / registration).read_bytes()),
    independent_review=independent, independent_review_sha256=sha((root / independent).read_bytes()),
    git_parent=r['git_parent'], changed_git_representation_paths=[x['path'] for x in r['files']],
    worktree_source_files_rewritten=0, local_and_index_frozen_files_verified=173,
    source_inventory=r['source_inventory'], source_inventory_sha256=r['source_inventory_sha256'],
    source_sha256=r['source_sha256'], checked_frozen_sha256=checked,
    attributes_sha256=r['proposed_attributes_sha256'], historical_attribute_bindings=old_bindings,
    preexisting_index_changes_unstaged_with_worktree_preserved=True,
    first_attempt_correction=folder+'attempt1_correction.json',
    runtime_semantics_changed=False, server_writes=0, new_GPU_seconds=0,
    sealed_state_values_read=False, archive_payload_migrated=False)
out = folder + 'root_review.json'
assert not (root / out).exists()
(root / out).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8', newline='\n')
source_archive = folder + 'complete_git_frozen_bytes_repair_20261018.py.txt'
(root / source_archive).write_bytes(Path(__file__).read_bytes())
print(json.dumps(dict(root_review=out, sha256=sha((root / out).read_bytes()), checked=173)))
