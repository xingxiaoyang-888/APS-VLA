"""Root closes six real boundaries and preserves this heartbeat's CPU evidence."""
from pathlib import Path
from datetime import datetime, timezone, timedelta
import argparse
import importlib.util
import json
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / 'scripts'))
import formal_supervisor as f
BASE = 'results/formal_main_v2/'
BATCH = BASE + 'preflight/parallel_review_batch_20261018T0626.json'
PARENT = '1bbaff1e2cb29d0e7d23d2857c11be326e0c1193'
JOBS = ['3023228', '3023516', '3023299', '3023274', '3023285', '3023249']
EXPECTED_NEW = {'3024981', '3024991', '3025000', '3025005', '3025017', '3025080'}

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--engineering-receipt', action='append', default=[])
    args = ap.parse_args()
    assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT).decode().strip() == PARENT
    assert not (ROOT / BATCH).exists()
    prior = json.loads(subprocess.check_output(['git', 'show', 'HEAD:results/experiment_supervisor/state.json'], cwd=ROOT))
    old_ids = {j['job_id'] for j in prior['registered_jobs']}
    assert prior['formal_main_progress']['trained_arms'] == 18
    refs, sources, pulls, boundaries = set(), set(), set(), []
    spec = importlib.util.spec_from_file_location('_delivery_paths_0626', ROOT / '.local/stage_verify_parallel_0626_candidate.py')
    guard = importlib.util.module_from_spec(spec); spec.loader.exec_module(guard)

    def include(name, checksum=None):
        guard.safe_name(name, guard.PANEL_PINS)
        assert 'final_registry' not in name.lower(), name
        if name == guard.HISTORICAL_SOURCE['path'] and checksum == guard.HISTORICAL_SOURCE['sha256']:
            for row in guard.check_source_version_resolution(): include(row['path'], row['sha256'])
            name = guard.HISTORICAL_ARCHIVE['path']
        path = ROOT / name
        assert path.is_file() and not path.is_symlink(), name
        for parent in path.parents:
            if parent == ROOT: break
            assert not parent.is_symlink(), name
        if name in guard.PANEL_PINS: assert f.sha(path) == guard.PANEL_PINS[name]
        if checksum is not None: assert f.sha(path) == checksum, name
        (sources if name.startswith('.local/') else refs).add(name)

    for jid in JOBS:
        operation = '.local/root_pull_review_2250_' + jid + '.json'
        op = f.read(ROOT / operation)
        rrname = op['root_review']; irname = rrname.removesuffix('.json') + '_independent.json'
        rr, ir = f.read(ROOT / rrname), f.read(ROOT / irname)
        assert op['status'] == rr['status'] == ir['status'] == 'passed'
        assert op['job_id'] == rr['job_id'] == ir['job_id'] == jid
        assert rr['completed_by_arm'] == dict.fromkeys(('A', 'B'), op['boundary'])
        assert op['allowed_next'] == rr['allowed_next_segment_end'] == ir['allowed_next_segment_end']
        assert op['allowed_next'] == (None if op['boundary'] == 5000 else op['boundary'] + 500)
        include(operation); include(rrname, op['root_sha256']); include(irname)
        for name, checksum in ir['read_evidence_sha256'].items(): include(name, checksum)
        pull = op['pull']; pulls.add(pull); include(pull, op['pull_sha256'])
        for name in ('inventory_before.json', 'inventory_after.json'):
            include((Path(pull).parent / name).as_posix())
        for row in f.read(ROOT / pull)['files']:
            include(row['relative'], row['sha256'])
            if row.get('previous_version'):
                relative = Path(row['previous_version']).relative_to(ROOT).as_posix()
                assert relative == (Path(pull).parent / 'previous_versions' / row['relative']).as_posix()
                include(relative, row['previous_sha256'])
        if op['boundary'] == 5000:
            selected = rrname.removesuffix('.json') + '_selected_server.json'
            include(selected, ir['read_evidence_sha256'][selected])
            assert f.read(ROOT / selected)['status'] == 'passed'
        boundaries.append({k: op[k] for k in ('job_id', 'suite', 'budget', 'seed', 'boundary', 'allowed_next', 'pair_gpu_seconds', 'latest_gpu_seconds')})
        log = '.local/audit_training_segment_' + jid + '.log'
        if (ROOT / log).exists(): include(log)
    guard.validate_boundary_rows(boundaries)
    closed_seconds = sum(row['latest_gpu_seconds'] for row in boundaries)
    assert closed_seconds == 24733
    for name in args.engineering_receipt:
        include(name)
        value = f.read(ROOT / name)
        for key in ('read_evidence_sha256', 'evidence_sha256'):
            for path, checksum in value.get(key, {}).items(): include(path, checksum)
        assert not value.get('current_engineering_pulls'), 'This batch has no new GPU engineering job'
    for name in ('.local/root_pull_review_one_20261017T2250.py', '.local/record_parallel_boundary.py',
            '.local/root_accept_boundary_0018.py', '.local/review_selected_pair_server.py',
            '.local/fill_reviewed_pair_slot.py', '.local/fill_fresh_spatial_pair_0452.py', '.local/reconcile_parallel_progress_20261018.py',
            '.local/read_parallel_live.py', '.local/finalize_parallel_review_20261018T0626.py',
            '.local/stage_verify_parallel_0626_candidate.py', '.local/prepare_guard_0626.py'):
        include(name)
    with f.supervisor_lock():
        state = f.read(f.STATE); jobs = f.refresh(state)
        p = state['formal_main_progress']; par = p['parallel_execution']
        active = [j for j in jobs if j['status'] in f.ACTIVE | {'SUBMITTED'}]
        assert state['pending_submission'] is None
        assert len(active) <= 8 and len({j['stage_key'] for j in active}) == len(active)
        assert all(j['gpus'] == 1 and j['mode'] == 'train' for j in active)
        assert p['trained_arms'] == 20
        updates = sum(2*x.get('last_root_reviewed_common_update', 0) for x in par['pair_tracking'].values())
        assert updates == 138000
        assert p['completed_final_trials'] == 0 and not p['test_opened'] and not p['global_release_created']
        assert p['completed_development_trials'] == 1212
        new_jobs = [j for j in jobs if j['job_id'] not in old_ids]
        assert {j['job_id'] for j in new_jobs} == EXPECTED_NEW
        for job in new_jobs:
            include(job['invocation'], job['invocation_sha256']); include(BASE+'submissions/'+job['job_id']+'.json')
            pair = par['pair_tracking'][f"{job['suite']}_k{job['visual_tokens']}_seed{job['training_seed']}"]
            field = 'fresh_start_precheck' if job['segment_end'] == 500 else 'latest_resume_precheck'
            name = pair[field]
            include(name, pair[field+'_sha256'])
            for path, checksum in f.read(ROOT / name).get('evidence_sha256', {}).items(): include(path, checksum)
        for path in (ROOT / (BASE+'live')).glob('parallel_20261018T*.json'):
            rel = path.relative_to(ROOT).as_posix()
            tracked = subprocess.run(['git','ls-files','--error-unmatch','--',rel],cwd=ROOT,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL).returncode == 0
            if not tracked: include(rel)
        changed = subprocess.check_output(['git','ls-files','--modified','--others','--exclude-standard','-z','--',
            'results/formal_main_v2','results/slurm'],cwd=ROOT).decode().split('\0')
        unknown = [x for x in changed if x and not x.endswith('.md') and x != BASE+'slurm_accounting.json' and x not in refs]
        assert not unknown, ('unbound_current_nonweight_files', unknown)
        folder = ROOT / (BASE+'preflight/parallel_review_batch_20261018T0626_sources')
        assert not folder.exists(); folder.mkdir()
        source_map = {}
        for ix, name in enumerate(sorted(sources), 1):
            dest = folder / (f'{ix:03d}_'+Path(name).name+'.txt')
            dest.write_bytes((ROOT/name).read_bytes())
            archived=dest.relative_to(ROOT).as_posix(); refs.add(archived); source_map[name]=archived
        now=datetime.now(timezone.utc); local=now.astimezone(timezone(timedelta(hours=8)))
        issue=p['downstream_action_range_issue']; assert issue['new_rollouts_held'] and issue['training_continues']
        value=dict(schema='formal_main_v2',kind='root_incremental_parallel_training_review',status='passed',
            reviewed_utc=now.isoformat(),parent_delivered_commit=PARENT,
            completed_training_arms=20,expected_training_arms=72,newly_completed_training_arms=2,
            cumulative_reviewed_arm_updates=updates,newly_reviewed_arm_updates=6000,reviewed_boundaries=boundaries,
            current_batch_closed_GPU_seconds=closed_seconds,closed_cost_is_included_in_snapshot_not_additional=True,
            all_formal_cost_snapshot=state['current_round_cost'],completed_development_trials=1212,new_development_trials=0,
            completed_final_trials=0,final_states_sealed=True,new_submission_job_ids=[j['job_id'] for j in new_jobs],
            active_jobs=[{k:j.get(k) for k in ('job_id','status','mode','suite','visual_tokens','training_seed','segment_end')} for j in active],
            current_terminal_pulls=sorted(pulls),audit_source_archive_map=source_map,new_engineering_evidence=sorted(set(args.engineering_receipt)),
            evidence_sha256={name:f.sha(ROOT/name) for name in sorted(refs)},
            additional_downstream_rollouts_held_for_runtime_diagnosis=True,training_continues=True,
            scientific_settings_changed=False,weights_copied=False,selected_offline_errors_do_not_establish_closed_loop_benefit=True)
        f.write(ROOT/BATCH,value)
        p['latest_incremental_review']=dict(path=BATCH,sha256=f.sha(ROOT/BATCH))
        p['root_and_independent_reviewed_arm_updates']=updates
        state['current_work']=(f'Formal completed20/72 arms; {updates}/360000 arm updates reviewed; {len(active)} registered GPU trainings active. '
            'All1212 historical development records audited and retained. Uniform action execution compatibility remains under engineering acceptance; new rollouts held, training continues. Final0/35200 sealed.')
        state['next_actions']=[issue['next'],
            'Root fill up to8 independent singleGPU jobs after own suite acceptance and terminal/root/independent review. Four Long pairs now run4500-to5000; never extend a completed5000 pair.',
            'Retain all72 checkpoints+16references+four-suite development before global root final release.',
            'Preserve original Object600 including1Dense policy_error; no overwrite, automatic retry, or new independent initial-state claim.']
        state['updated_utc']=now.isoformat(); state['updated_date']=local.strftime('%Y-%m-%d'); f.write(f.STATE,state)
        text=(f'\n{local:%Y-%m-%d %H:%M}进度：完整训练20/72组，新增Spatial128 seed41的A/B完成5000步；11次验证、完整恢复状态及所选权重实际SHA已通过root与独立复核。'
            f'累计{updates}/360000次臂更新已验收，当前{len(active)}个单GPU作业并行。'
            f"正式轮累计分配{state['current_round_cost']['allocated_gpu_hours']:.6f} GPU小时（含运行中、工程验收及失败）。\n\n"
            '本批另外5对已通过分段复核并续跑：Long256/64的seed23/41进入4500→5000，Spatial64 seed17进入500→1000；新空闲槽由Spatial64 seed41从头训练补齐。'
            '原1212条开发记录、35028个已记录动作chunk和276594步动作已逐条核验；Object原policy_error及缺失的第8个chunk仍按未知保留。'
            '动作执行适配器v1的独立CPU审计发现未知构造身份校验缺口，已保留失败证据并作有界修正；实际控制器和新执行身份GPU验收完成前不开放新增闭环。'
            '训练与科学设置不变。最终0/35200，继续封存。'
            f'\n\n[本批真实产物与成本](../{BATCH})；[动作执行CPU审计](../results/formal_main_v2/preflight/action_execution_CPU_root_0626.json)。\n\n')
        for name in ('正式主实验规划.md','完整实验计划.md','实验进度.md'):
            doc=ROOT/'docs'/name; first,sep,body=doc.read_bytes().partition(b'\n'); assert sep
            doc.write_bytes(first+sep+text.encode()+body)
    print(json.dumps(dict(receipt=BATCH,sha256=f.sha(ROOT/BATCH),completed_arms=20,active_jobs=len(active),cost=state['current_round_cost'])))

if __name__ == '__main__': main()
