def require(ok, message):
    if not ok:
        raise ValueError(message)


def digest(data):
    return hashlib.sha256(data).hexdigest()


def file_sha(path):
    h = hashlib.sha256()
    with path.open("rb") as source:
        for chunk in iter(lambda: source.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def no_duplicate_keys(pairs):
    out = {}
    for key, value in pairs:
        require(key not in out, f"duplicate JSON key {key}")
        out[key] = value
    return out


def decode(raw):
    def bad_constant(value):
        raise ValueError(f"nonfinite JSON constant {value}")
    return json.loads(raw, object_pairs_hook=no_duplicate_keys, parse_constant=bad_constant)


def guard_ledger(relative):
    # Check exact whitelist before any filesystem operation, including resolve.
    require(relative in LEDGERS, "ledger must be one of the three pinned development paths")
    require(re.fullmatch(r"results/formal_main_v2/(libero_10|libero_object|libero_spatial)/downstream/development/[a-f0-9]+/attempt[0-9]+/trials\.jsonl", relative) is not None,
            "non-development ledger forbidden")
    path = ROOT / relative
    for part in [path, *path.parents]:
        if part == ROOT.parent:
            break
        require(not part.is_symlink() and not (hasattr(part, "is_junction") and part.is_junction()),
                "symlink/junction ledger path forbidden")
    return path


class Evidence:
    def __init__(self):
        self.seen = {}

    def read(self, rel, expected):
        require(not Path(rel).is_absolute() and ".." not in Path(rel).parts, "invalid evidence path")
        require(rel in PINS or rel in {x["root"] for x in SPECS}
                or re.fullmatch(r"results/formal_main_v2/pulls/[A-Za-z0-9_]+/receipt\.json", rel),
                "unregistered metadata read")
        data = (ROOT / rel).read_bytes()
        require(digest(data) == expected, f"evidence SHA mismatch: {rel}")
        self.seen[rel] = expected
        return data


def numbers(matrix, rows, label):
    require(isinstance(matrix, list) and len(matrix) == rows, f"{label}: row count")
    flat = []
    for row in matrix:
        require(isinstance(row, list) and len(row) == 7, f"{label}: column count")
        for value in row:
            require(isinstance(value, (int, float)) and not isinstance(value, bool)
                    and math.isfinite(value), f"{label}: nonfinite/non-numeric value")
            flat.append(float(value))
    return flat


def f64(values):
    return struct.pack("<" + "d" * len(values), *values)


def inspect_terminal(row, horizon):
    chunks, executed = row["chunks"], row["executed_actions"]
    require(isinstance(chunks, list) and bool(chunks), "missing recorded chunks")
    require(type(row["control_steps"]) is int and 0 < row["control_steps"] <= horizon, "control count/horizon")
    require(len(executed) == row["control_steps"], "executed/control count mismatch")
    require(row["action_chunk"] == 8 and row["configured_wait_steps"] == row["wait_steps"] == 10,
            "chunk/wait settings changed")
    actual = numbers(executed, row["control_steps"], "executed_actions")
    queue, normal_values, metadata_hashes, checkpoint_shas = [], [], [], set()
    calls = Counter()
    method = row["method"]
    expected_method = "uniform" if method.startswith("uniform") else method[0] if method[:1] in ("A", "B") else "dense"
    budget = int(method[7:]) if method.startswith("uniform") else 128 if expected_method in ("A", "B") else 512
    for index, chunk in enumerate(chunks):
        require(chunk["control_step"] == 8 * index, "chunk does not start at fixed queue boundary")
        require(re.fullmatch(r"[a-f0-9]{64}", chunk["observation_sha256"]) is not None, "observation hash missing")
        values = numbers(chunk["environment_actions"], 8, "environment_actions")
        require(all(abs(value) <= THRESHOLD for value in values), "recorded environment action exceeds old threshold")
        queue.extend(values)
        # Normalized head outputs need only be finite. Their range is not the
        # postprocessed environment-action contract.
        normal_values.extend(numbers(chunk["normalized_actions"], 8, "normalized_actions"))
        md = chunk["metadata"]
        require(md["suite"] == row["suite"] and md["method"] == expected_method, "chunk method/suite mismatch")
        require(md["visual_tokens"] == budget and md["per_camera_budgets"] == [budget // 2] * 2,
                "chunk token budget mismatch")
        for call in ("policy_calls", "vision_calls", "packing_calls"):
            require(md[call] == 1, f"{call} must be exactly one")
            calls[call] += md[call]
        require(md["future_head_calls"] == 0 and md["language_cache"] == "disabled", "future/cache deployment change")
        require(md["scorer_calls"] == int(expected_method in ("A", "B")), "scorer call mismatch")
        if expected_method in ("A", "B"):
            require(md["training_seed"] == 17, "selector seed mismatch")
            require(re.fullmatch(r"[a-f0-9]{64}", md["selector_checkpoint_sha256"]) is not None, "checkpoint hash invalid")
            checkpoint_shas.add(md["selector_checkpoint_sha256"])
        metadata_hashes.append(digest(json.dumps(md, sort_keys=True, separators=(",", ":")).encode()))
    require(len(checkpoint_shas) <= 1, "checkpoint changes within trial")
    require(len(actual) <= len(queue) and f64(actual) == f64(queue[:len(actual)]),
            "executed actions are not the exact float64 queue prefix")
    require(all(abs(value) <= THRESHOLD for value in actual), "executed action exceeds old threshold")
    status = row["status"]
    require(status in ("completed", "policy_error"), "unexpected historical terminal status")
    missing = row["policy_calls"] - len(chunks)
    if status == "completed":
        require(missing == 0 and len(chunks) == math.ceil(row["control_steps"] / 8), "normal policy/chunk coverage")
        require(row.get("error") is None and row.get("error_phase") is None, "completed row contains error")
        require((row["success"] is True and row["termination"] == "environment_success")
                or (row["success"] is False and row["termination"] == "task_horizon" and row["control_steps"] == horizon),
                "terminal status/horizon semantics mismatch")
    else:
        require(row["suite"] == "libero_object" and method == "dense"
                and row["trial_id"] == "libero_object:pick_up_the_chocolate_pudding_and_place_it_in_the_basket:state-0049:env-0:dense",
                "unregistered historical policy error")
        require(row["success"] is False and row.get("error_phase") == "policy"
                and row["termination"] == "ValueError"
                and row.get("error") == "environment action outside normalized controller [-1, 1] range",
                "historical policy-error identity mismatch")
        require(missing == 1 and len(chunks) == 7 and row["control_steps"] == 56
                and len(actual) == len(queue), "failed eighth chunk coverage mismatch")
    return {
        "recorded_chunks": len(chunks), "policy_calls": row["policy_calls"],
        "executed_control_actions": row["control_steps"], "unrecorded_policy_chunks": missing,
        "recorded_environment_shape_finite_range_pass": True,
        "normalized_shape_finite_pass": True,
        "normalized_range_is_not_an_execution_gate": True,
        "maximum_abs_normalized": max(map(abs, normal_values)),
        "maximum_abs_environment": max(map(abs, queue)),
        "maximum_abs_environment_first_six_components": max(abs(value) for i, value in enumerate(queue) if i % 7 != 6),
        "executed_exact_float64_queue_prefix": True,
        "all_queued_actions_f64le_sha256": digest(f64(queue)),
        "executed_actions_f64le_sha256": digest(f64(actual)),
        "chunk_metadata_sequence_sha256": digest(json.dumps(metadata_hashes, separators=(",", ":")).encode()),
        "selector_checkpoint_sha256": next(iter(checkpoint_shas), None),
        "recorded_deployment_calls": dict(calls, future_head_calls=0),
        "all_policy_calls_have_recorded_chunks": missing == 0,
        "missing_chunk_values": "unknown_not_reconstructed" if missing else "none",
        "new_adapter_equivalence_proven": False, "reuse_authorized": False,
    }


def line_ref(line, offset, raw):
    return {"line_number": line, "byte_offset": offset, "bytes_including_newline": len(raw),
            "sha256_including_newline": digest(raw)}


def audit_suite(spec, evidence):
    review = decode(evidence.read(spec["root"], spec["root_sha"]))
    require(review["status"] == "passed" and review["stage"] == "development", "root review not passed development")
    require(review["source_sha256"] == SOURCE_SHA and review["adapter_sha256"] == ADAPTER_SHA, "old execution source identity")
    require(review["files"].get(spec["ledger"]) == spec["ledger_sha"], "root does not bind exact ledger")
    require(review["run_root"].endswith("/" + spec["ledger"].rsplit("/", 1)[0]), "run-root mismatch")
    require(review["completed_trials"] == spec["terminals"] and review["expected_trials"] == 600, "root coverage mismatch")
    require(review["reconstructed_summary"]["test_opened"] is False and review["final_state_values_read"] is False,
            "root review claims final-state use")
    jobs = [x for x in review["registered_jobs_at_review"] if x["job_id"] == spec["job"]]
    require(len(jobs) == 1 and jobs[0]["method_ids"] == METHODS and jobs[0]["mode"] == "development", "registered job/method mismatch")
    pull_rel = review["pull_receipt"]
    pull = decode(evidence.read(pull_rel, review["evidence_sha256"][pull_rel]))
    require(pull["status"] == "verified_nonweight_snapshot" and pull["weights_copied"] is False and pull["sealed_initial_states_copied"] is False,
            "pull not verified nonweight")
    pulled = [x for x in pull["files"] if x["relative"] == spec["ledger"]]
    require(len(pulled) == 1 and pulled[0]["sha256"] == spec["ledger_sha"], "pull ledger binding mismatch")
    path = guard_ledger(spec["ledger"])
    require(file_sha(path) == spec["ledger_sha"], "actual ledger SHA differs before audit")
    evidence.seen[spec["ledger"]] = spec["ledger_sha"]
    h = hashlib.sha256(); offset = 0; current_job = None; current_invocation_ref = None
    active = None; ids = set(); results = []; invocations = []; states = {}; events = Counter()
    total_counts = Counter(); statuses = Counter(); terminations = Counter(); per_method = Counter()
    with path.open("rb") as source:
        for number, raw in enumerate(source, 1):
            require(raw.endswith(b"\n"), "truncated final JSONL record")
            ref = line_ref(number, offset, raw); offset += len(raw); h.update(raw)
            row = decode(raw); event = row.get("event"); events[event] += 1
            require(row.get("run_signature") == review["run_signature"], "run signature differs from root")
            if event == "invocation":
                require(active is None, "new invocation interrupts unfinished attempt")
                current_job = row["slurm_job_id"]; current_invocation_ref = ref
                require(current_job not in invocations, "duplicate invocation marker")
                invocations.append(current_job); continue
            require(current_job is not None and event in ("start", "result"), "unexpected event outside invocation")
            require(row.get("split") == "development" and row.get("suite") == spec["suite"], "nondevelopment or wrong-suite row")
            require(row["method"] in METHODS and row["environment_seed"] == 0 and row["runtime_seed"] == 17,
                    "method or environment identity mismatch")
            if event == "start":
                require(active is None and row["trial_id"] not in ids, "overlapping/duplicate trial start")
                active = (row, ref); continue
            require(active is not None, "terminal row without start")
            start, start_ref = active; active = None
            require(all(start[k] == row[k] for k in IDENTITY_KEYS), "start/terminal identity mismatch")
            require(row["method"] == METHODS[len(results) % 6], "fixed six-method order differs")
            require(row["trial_id"] not in ids, "duplicate terminal trial")
            if len(results) % 6:
                require(row["state_trial_id"] == results[-1]["state_trial_id"], "state changed inside six-method block")
            state = states.setdefault(row["state_trial_id"], {"sha": row["initial_state_sha256"], "methods": []})
            require(state["sha"] == row["initial_state_sha256"], "initial-state hash changes")
            state["methods"].append(row["method"])
            checks = inspect_terminal(row, spec["horizon"])
            item = {k: row[k] for k in IDENTITY_KEYS}
            item.update(status=row["status"], success=row["success"], termination=row["termination"],
                        error=row.get("error"), error_phase=row.get("error_phase"),
                        ledger=spec["ledger"], ledger_sha256=spec["ledger_sha"],
                        root_review=spec["root"], root_review_sha256=spec["root_sha"],
                        old_source_sha256=review["source_sha256"], old_adapter_sha256=review["adapter_sha256"],
                        old_execution_manifest_sha256=review["execution_manifest_sha256"],
                        invocation_job_id=current_job, invocation_line=current_invocation_ref,
                        start_line=start_ref, terminal_line=ref, checks=checks)
            results.append(item); ids.add(row["trial_id"]); per_method[row["method"]] += 1
            statuses[row["status"]] += 1; terminations[row["termination"]] += 1
            for key in ("recorded_chunks", "policy_calls", "executed_control_actions", "unrecorded_policy_chunks"):
                total_counts[key] += checks[key]
    require(active is None and len(results) == spec["terminals"], "unfinished or wrong terminal coverage")
    require(h.hexdigest() == spec["ledger_sha"] and offset == pulled[0]["bytes"], "ledger changed during audit")
    require(invocations == review["raw_invocation_job_ids"], "historical invocation sequence mismatch")
    require(len(states) == spec["states"] and all(x["methods"] == METHODS for x in states.values()), "state/method coverage mismatch")
    require(all(per_method[x] == spec["terminals"] // 6 for x in METHODS), "method count mismatch")
    expected_errors = 1 if spec["suite"] == "libero_object" else 0
    require(statuses["policy_error"] == expected_errors, "unexpected historical error count")
    summary = {"suite": spec["suite"], "terminal_records": len(results), "unique_development_states": len(states),
               "statuses": dict(statuses), "terminations": dict(terminations), "per_method_terminal_counts": dict(per_method),
               "calls": dict(total_counts), "invocation_job_ids": invocations, "events": dict(events),
               "ledger": spec["ledger"], "ledger_sha256": spec["ledger_sha"], "ledger_bytes": offset,
               "root_review": spec["root"], "root_review_sha256": spec["root_sha"],
               "pull_receipt": pull_rel, "legacy_recorded_range_scope_pass": True,
               "new_adapter_equivalence_proven": False, "reuse_authorized": False}
    return summary, results


def run(name):
    require(re.fullmatch(r"run_[a-z0-9_]+", name) is not None, "invalid local output name")
    target = HERE / name
    require(not target.exists(), "refuse to overwrite prior candidate output")
    evidence = Evidence()
    for rel, sha in PINS.items():
        evidence.read(rel, sha)
    evidence.seen[str(Path(__file__).relative_to(ROOT)).replace("\\", "/")] = file_sha(Path(__file__))
    summaries, trials = [], []
    for spec in SPECS:
        summary, rows = audit_suite(spec, evidence)
        summaries.append(summary); trials.extend(rows)
        print(json.dumps(summary, ensure_ascii=False, sort_keys=True), flush=True)
    require(len(trials) == 1212 and len({row['trial_id'] for row in trials}) == 1212, "global1212 coverage mismatch")
    failures = [row for row in trials if row["status"] != "completed"]
    require(len(failures) == 1, "expected exactly one retained historical error")
    for rel, sha in evidence.seen.items():
        require(file_sha(ROOT / rel) == sha, f"read evidence changed before publication: {rel}")
    trial_bytes = b"".join((json.dumps(row, ensure_ascii=False, sort_keys=True, separators=(",", ":")) + "\n").encode() for row in trials)
    failures_bytes = (json.dumps(failures, ensure_ascii=False, sort_keys=True, indent=2) + "\n").encode()
    report = {
        "schema": "formal_main_v2", "kind": "historical_development_action_range_scope_audit_candidate",
        "status": "passed_historical_recorded_scope_only_pending_independent_review",
        "created_utc": datetime.now(timezone.utc).isoformat(),
        "terminal_records": 1212, "normal_terminal_records": 1211, "retained_policy_errors": 1,
        "complete_recorded_policy_call_coverage_trials": 1211,
        "missing_policy_chunks_unknown": 1, "environment_action_threshold": THRESHOLD,
        "suites": summaries,
        "per_trial_audit": {"path": "per_trial_audit.jsonl", "sha256": digest(trial_bytes), "bytes": len(trial_bytes), "records": 1212},
        "original_failure_audit": {"path": "retained_policy_error.json", "sha256": digest(failures_bytes), "bytes": len(failures_bytes)},
        "new_adapter_equivalence_proven": False, "reuse_authorized": False,
        "no_original_trial_replaced": True, "no_success_rate_table_recomputed": True,
        "limitations": [
            "Only historical recorded action scope is checked. No new adapter implementation, equivalence, reuse, simulator/controller numerical identity or GPU acceptance is established.",
            "The original Object eighth failing chunk is absent. Its values and new-execution outcome remain unknown; seven recorded chunks and56 executed actions are retained exactly.",
            "Range applies to postprocessed environment actions, not normalized head outputs. No values are clipped or altered.",
            "Root receipts are SHA-bound evidence for the old execution identity. No sealed panel, weights, live state or remote files are opened; no source/history is rewritten.",
            "Original success flags are copied only as per-trial evidence. No replacement success rate, causality claim or scientific release is produced."
        ],
        "read_evidence_sha256": dict(sorted(evidence.seen.items())),
        "new_GPU_seconds": 0, "SSH_used": False, "Git_used": False, "state_modified": False,
        "output_scope": str(HERE.relative_to(ROOT)).replace("\\", "/"),
    }
    target.mkdir()
    for filename, content in [("per_trial_audit.jsonl", trial_bytes), ("retained_policy_error.json", failures_bytes),
                              ("report.json", (json.dumps(report, ensure_ascii=False, sort_keys=True, indent=2) + "\n").encode())]:
        with (target / filename).open("xb") as output:
            output.write(content)
    print(json.dumps({"report": str(target / "report.json"), "sha256": file_sha(target / "report.json"),
                      "status": report["status"], "new_adapter_equivalence_proven": False}, ensure_ascii=False), flush=True)
    return report


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output-name", default="run_v1")
    run(parser.parse_args().output_name)
