def check(ok, message):
    if not ok:
        raise AssertionError(message)


def digest(data):
    return hashlib.sha256(data).hexdigest()


def sha(path):
    h = hashlib.sha256()
    with path.open("rb") as f:
        for block in iter(lambda: f.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()


def safe_path(rel):
    p = Path(rel)
    check(not p.is_absolute() and ".." not in p.parts, "nonlocal metadata")
    low = str(rel).lower().replace("\\", "/")
    check(not any(x in low for x in ("sealed", "/initial_states/", "/downstream/final/", "/weights/", "/checkpoints/")), "forbidden read")
    check(p.suffix.lower() not in (".pt", ".pth", ".safetensors", ".npz"), "forbidden payload")
    return ROOT / p


def read(rel, expected=None):
    data = safe_path(rel).read_bytes()
    actual = digest(data)
    check(expected is None or actual == expected, "SHA mismatch: " + rel)
    check(rel not in seen or seen[rel] == actual, "changed evidence: " + rel)
    seen[rel] = actual
    return data


def decode(data):
    def pairs(items):
        out = {}
        for key, value in items:
            check(key not in out, "duplicate JSON key")
            out[key] = value
        return out
    def bad(value):
        raise AssertionError("nonfinite JSON constant " + value)
    return json.loads(data, object_pairs_hook=pairs, parse_constant=bad)


def js(rel, expected=None):
    return decode(read(rel, expected))


def compact(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":")).encode()


def matrix(value, rows):
    check(isinstance(value, list) and len(value) == rows, "matrix rows")
    check(all(isinstance(row, list) and len(row) == 7 for row in value), "matrix columns")
    check(all(type(v) in (float, int) and math.isfinite(v) for row in value for v in row), "matrix finite numeric")
    return np.asarray(value, dtype="<f8")


def line_ref(number, offset, raw):
    return dict(line_number=number, byte_offset=offset, bytes_including_newline=len(raw), sha256_including_newline=digest(raw))


def inspect_record(row, ident, horizon):
    n = row["control_steps"]
    check(type(n) is int and 0 < n <= horizon, "control range")
    check(row["action_chunk"] == 8 and row["wait_steps"] == row["configured_wait_steps"] == 10, "wait/chunk")
    chunks = row["chunks"]
    check(chunks, "no chunks")
    executed = matrix(row["executed_actions"], n)
    queued, normalized, md_hashes = [], [], []
    cp_sha = None
    for i, c in enumerate(chunks):
        check(c["control_step"] == i * 8, "queue boundary")
        check(re.fullmatch(r"[0-9a-f]{64}", c["observation_sha256"]) is not None, "observation SHA")
        env = matrix(c["environment_actions"], 8)
        norm = matrix(c["normalized_actions"], 8)
        check(np.all(np.abs(env) <= 1.00001), "environment range")
        # The frozen head explicitly emits float32. Its JSON roundtrip must
        # retain every original float32 value, including signed zero.
        check(norm.tobytes() == norm.astype("<f4").astype("<f8").tobytes(), "normalized head float32 roundtrip")
        md = c["metadata"]
        for key in ("suite", "method", "training_seed", "visual_tokens"):
            check(md[key] == ident[key], "chunk/manifest identity: " + key)
        check(md["per_camera_budgets"] == [ident["visual_tokens"] // 2] * 2, "camera budgets")
        check(md["future_head_calls"] == 0 and md["language_cache"] == "disabled", "future/cache")
        check(all(md[k] == 1 for k in ("policy_calls", "vision_calls", "packing_calls")), "deployment count")
        check(md["scorer_calls"] == int(ident["method"] in ("A", "B")), "scorer")
        if ident["method"] in ("A", "B"):
            check(md["selector_checkpoint_sha256"] == ident["checkpoint_sha256"], "checkpoint identity")
            check(md["pair_id"] == ident["pair_id"], "pair identity")
            cp_sha = ident["checkpoint_sha256"]
        indices = md["retained_indices"]
        check(len(indices) == ident["visual_tokens"] and indices == sorted(set(indices)), "packing indices")
        check(all(type(v) is int and 0 <= v < 512 for v in indices), "index range")
        check(sum(v < 256 for v in indices) == ident["visual_tokens"] // 2, "indices camera balance")
        queued.append(env); normalized.append(norm); md_hashes.append(digest(compact(md)))
    queue = np.concatenate(queued, axis=0)
    normal = np.concatenate(normalized, axis=0)
    check(n <= len(queue), "too many executed actions")
    check(executed.tobytes() == queue[:n].tobytes(), "exact environment float64 prefix")
    # No tolerance and no float32 downcast are used to accept the actual data.
    missing = row["policy_calls"] - len(chunks)
    if row["status"] == "completed":
        check(missing == 0 and len(chunks) == (n + 7) // 8, "normal call/chunk coverage")
        check(row.get("error") is None and row.get("error_phase") is None, "error hidden by completed")
        check((row["success"] is True and row["termination"] == "environment_success") or
              (row["success"] is False and row["termination"] == "task_horizon" and n == horizon), "termination semantics")
    else:
        check(row["status"] == "policy_error" and row["suite"] == "libero_object" and row["method"] == "dense", "unexpected error")
        check(row["task_id"] == 8 and row["source_state_index"] == 49 and row["success"] is False, "failed trial identity")
        check(row["error_phase"] == "policy" and row["termination"] == "ValueError" and
              row["error"] == "environment action outside normalized controller [-1, 1] range", "original failure")
        check(missing == 1 and row["policy_calls"] == 8 and len(chunks) == 7 and n == 56 == len(queue), "unrecorded failed chunk")
    return {
        "recorded_chunks": len(chunks), "policy_calls": row["policy_calls"], "executed_control_actions": n,
        "unrecorded_policy_chunks": missing, "recorded_environment_shape_finite_range_pass": True,
        "normalized_shape_finite_pass": True, "normalized_range_is_not_an_execution_gate": True,
        "maximum_abs_normalized": float(np.max(np.abs(normal))),
        "maximum_abs_environment": float(np.max(np.abs(queue))),
        "maximum_abs_environment_first_six_components": float(np.max(np.abs(queue[:, :6]))),
        "executed_exact_float64_queue_prefix": True,
        "all_queued_actions_f64le_sha256": digest(queue.tobytes()),
        "executed_actions_f64le_sha256": digest(executed.tobytes()),
        "chunk_metadata_sequence_sha256": digest(json.dumps(md_hashes, separators=(",", ":")).encode()),
        "selector_checkpoint_sha256": cp_sha,
        "recorded_deployment_calls": {"policy_calls": len(chunks), "vision_calls": len(chunks), "packing_calls": len(chunks), "future_head_calls": 0},
        "all_policy_calls_have_recorded_chunks": missing == 0,
        "missing_chunk_values": "unknown_not_reconstructed" if missing else "none",
        "new_adapter_equivalence_proven": False, "reuse_authorized": False,
    }


def focused_negatives(sample, ident, candidate):
    cases = []
    def reject(name, f):
        try:
            f()
        except (AssertionError, ValueError, KeyError):
            cases.append({"case": name, "result": "rejected_as_required"})
        else:
            raise AssertionError("negative accepted: " + name)
    for label, replacement in [("one_float64_ULP", math.nextafter(sample["executed_actions"][0][0], math.inf)),
                               ("NaN", float("nan")), ("positive_inf", float("inf")), ("boolean", True)]:
        x = copy.deepcopy(sample); x["executed_actions"][0][0] = replacement
        reject(label + "_independent", lambda x=x: inspect_record(x, ident, 520))
        reject(label + "_candidate", lambda x=x: candidate.inspect_terminal(x, 520))
    for label, replacement in [("wrong_checkpoint", "0" * 64), ("wrong_pair", "0" * 64)]:
        x = copy.deepcopy(sample)
        x["chunks"][0]["metadata"]["selector_checkpoint_sha256" if label == "wrong_checkpoint" else "pair_id"] = replacement
        reject(label + "_manifest_crossbinding", lambda x=x: inspect_record(x, ident, 520))
    z = copy.deepcopy(sample)
    z["chunks"][0]["environment_actions"][0][0] = 0.0
    z["executed_actions"][0][0] = -0.0
    reject("signed_zero_independent", lambda: inspect_record(z, ident, 520))
    reject("signed_zero_candidate", lambda: candidate.inspect_terminal(z, 520))
    for label, mutate in [
        ("missing_normal_chunk", lambda x: x.update(policy_calls=x["policy_calls"] + 1)),
        ("future_head", lambda x: x["chunks"][0]["metadata"].update(future_head_calls=1)),
        ("wrong_shape", lambda x: x["chunks"][0]["environment_actions"][0].pop()),
    ]:
        x = copy.deepcopy(sample); mutate(x)
        reject(label + "_independent", lambda x=x: inspect_record(x, ident, 520))
        reject(label + "_candidate", lambda x=x: candidate.inspect_terminal(x, 520))
    for value in (b'{"x":1,"x":2}', b'{"x":NaN}'):
        reject("strict_JSON_independent", lambda value=value: decode(value))
        reject("strict_JSON_candidate", lambda value=value: candidate.decode(value))
    for bad in ("results/formal_main_v2/libero_10/downstream/final/abc/trials.jsonl", "../trials.jsonl", "F:/weights.pt"):
        with patch.object(Path, "open", side_effect=RuntimeError("I/O forbidden")), patch.object(Path, "resolve", side_effect=RuntimeError("resolve forbidden")):
            reject("candidate_path_before_IO:" + bad, lambda bad=bad: candidate.guard_ledger(bad))
    return cases
