"""Integrate the immutable CPU rejection and historical trace review only."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import sys

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / 'scripts'))
import formal_supervisor as f

RECEIPTS = {
    'results/formal_main_v2/preflight/action_execution_compat_CPU_independent_0626.json':
        'fb60c2d881ba26805693e91d9952209ed9592df2b475956f7681fc1cab863a48',
    'results/formal_main_v2/preflight/action_execution_legacy_independent_0626.json':
        '4bdc98c8c40921792172207090ebfdb85aef40689e54cbd2b3b962a7dbb269f7',
}
OUT = 'results/formal_main_v2/preflight/action_execution_CPU_root_0626.json'
FOLDER = 'results/formal_main_v2/preflight/action_execution_CPU_evidence_0626'
assert not (ROOT / OUT).exists() and not (ROOT / FOLDER).exists()
evidence = {}

def include(name, digest=None):
    assert not Path(name).is_absolute() and ':' not in name and '..' not in Path(name).parts
    assert not any('sealed' in part.lower() or part.lower() in ('weights', 'checkpoints', 'test') for part in Path(name).parts)
    path = ROOT / name
    assert path.is_file() and not path.is_symlink()
    actual = f.sha(path)
    assert digest is None or actual == digest, name
    assert name not in evidence or evidence[name] == actual
    evidence[name] = actual

for name, digest in RECEIPTS.items():
    include(name, digest)
    for path, checksum in f.read(ROOT / name)['read_evidence_sha256'].items():
        include(path, checksum)
cpu, legacy = [f.read(ROOT / path) for path in RECEIPTS]
assert cpu['status'] == 'blocked_unknown_PolicyChunk_constructor_identity'
assert len(cpu['blocking_findings']) == 1
assert cpu['candidate_sha256'] == 'd7132596293958f35bc029707136fddf4450d76efad49b8ab96b65d6c8a9f0d9'
assert len(cpu['blocking_findings'][0]['reproductions']) == 2
assert legacy['candidate_all_1212_per_trial_fields_exact_equal'] is True
assert legacy['recorded_chunks'] == 35028 and legacy['executed_control_actions'] == 276594
assert legacy['new_adapter_equivalence_proven'] is False and legacy['reuse_authorized'] is False
assert legacy['new_GPU_seconds'] == 0

bundle = '.local/action_execution_compat_legacy_audit_0452/candidate_bundle.json'
include(bundle, legacy['candidate_bundle_sha256'])
b = f.read(ROOT / bundle)
assert b['coverage']['terminal_records'] == 1212
assert b['coverage']['retained_policy_error_records'] == 1
for name, row in b['files'].items():
    include((Path(bundle).parent / name).as_posix(), row['sha256'])
include('.local/root_integrate_execution_cpu_0626.py')
folder = ROOT / FOLDER
folder.mkdir()
archives = {}
for i, name in enumerate(sorted(p for p in evidence if p.startswith('.local/')), 1):
    dest = folder / (f'{i:03d}_' + Path(name).name + '.txt')
    dest.write_bytes((ROOT / name).read_bytes())
    assert f.sha(dest) == evidence[name]
    archives[name] = dest.relative_to(ROOT).as_posix()
    evidence[archives[name]] = evidence[name]

value = dict(schema='formal_main_v2', kind='root_CPU_scope_and_history_integration',
    status='history_scope_passed_adapter_v1_rejected_pending_bounded_revision',
    reviewed_utc=datetime.now(timezone.utc).isoformat(),
    independent_receipts=RECEIPTS, evidence_sha256=evidence,
    source_archive_map=archives, all_input_bytes_rehashed=True,
    historical_terminal_records=1212, recorded_chunks=35028, executed_actions=276594,
    historical_scope_only=True, new_adapter_equivalence_proven=False, reuse_authorized=False,
    original_Object_error_preserved=True, missing_eighth_chunk_remains_unknown=True,
    CPU_v1_blocking_finding='Unknown PolicyChunk construction could bypass pinned post-init validation in two synthetic tests; no evidence that real trials used such construction.',
    finite_action_transforms_added=False, frozen_source_modified=False,
    new_GPU_seconds=0, new_scientific_trials=0, final_states_opened=False,
    next='Retain v1 and failed tests; bounded v2 construction-identity repair and independent CPU review, then separately preregister actual-controller GPU acceptance. No downstream release yet.')
f.write(ROOT / OUT, value)
with f.supervisor_lock():
    state = f.read(f.STATE)
    issue = state['formal_main_progress']['downstream_action_range_issue']
    assert issue['new_rollouts_held'] and issue['training_continues']
    issue['CPU_history_review_0626'] = dict(path=OUT, sha256=f.sha(ROOT / OUT))
    issue['compatibility_status'] = value['status']
    issue['next'] = value['next']
    state['next_actions'][0] = issue['next']
    state['updated_utc'] = datetime.now(timezone.utc).isoformat()
    f.write(f.STATE, state)
print(json.dumps(dict(path=OUT, sha256=f.sha(ROOT / OUT), source_archives=len(archives), evidence_files=len(evidence), GPU_seconds=0)))
