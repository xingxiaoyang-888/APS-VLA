"""Make a batch-specific copy of the already accepted delivery helper."""
from pathlib import Path
import hashlib
import re

root = Path(__file__).resolve().parents[1]
source = root / '.local/stage_verify_parallel_0452_candidate.py'
assert hashlib.sha256(source.read_bytes()).hexdigest() == '5e255e610abc9b806ff7c24852b8caceaec65ad238096d27026add2d037bb7e7'
text = source.read_text(encoding='utf-8')
text = text.replace('0452', '0626')
text = text.replace('fad47a89ecaf1ccb697a10454f79aaf36bcf19cd', '1bbaff1e2cb29d0e7d23d2857c11be326e0c1193')
block = '''EXPECTED_BOUNDARIES = {
    "3023228": ("libero_spatial", 64, 17, 500),
    "3023516": ("libero_10", 64, 41, 4500),
    "3023299": ("libero_10", 64, 23, 4500),
    "3023274": ("libero_10", 256, 23, 4500),
    "3023285": ("libero_10", 256, 41, 4500),
    "3023249": ("libero_spatial", 128, 41, 5000),
}'''
text, count = re.subn(r'EXPECTED_BOUNDARIES = \{.*?\n\}', block, text, count=1, flags=re.S)
assert count == 1
text = text.replace('for row in rows}) == 8', 'for row in rows}) == 6')
text = text.replace('for row in rows) == 4', 'for row in rows) == 1')
text = text.replace('len(training_pulls) == 8', 'len(training_pulls) == 6')
text = text.replace('exactly these eight boundaries', 'exactly these six boundaries')
text = text.replace('Four are final5000', 'One is final5000')
# The prior helper source and history remain immutable and byte-identical.
destination = root / '.local/stage_verify_parallel_0626_candidate.py'
assert not destination.exists()
destination.write_text(text, encoding='utf-8', newline='\n')
compile(text, str(destination), 'exec')
print(hashlib.sha256(destination.read_bytes()).hexdigest())
