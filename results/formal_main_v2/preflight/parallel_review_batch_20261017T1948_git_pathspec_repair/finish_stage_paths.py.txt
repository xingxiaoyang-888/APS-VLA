"""Finish the already-registered delivery using a NUL-delimited exact path list."""
from pathlib import Path, PurePosixPath
from datetime import datetime, timezone
import hashlib, json, subprocess

root = Path(__file__).resolve().parents[1]
base = 'results/formal_main_v2/'
required_path = base + 'preflight/parallel_review_batch_20261017T1948_git_required.json'
snapshot_path = base + 'nonweight_archives/v1/snapshots/long_dev600_20261018.json'
required_bytes = (root / required_path).read_bytes()
required = json.loads(required_bytes)
assert required['schema'] == 'cti_nonweight_archive_stage_v1' and required['kind'] == 'required_files'
raw = {r['path'] for r in json.loads((root / snapshot_path).read_bytes())['files']}
assert len(raw) == 2
paths = []
for row in required['files']:
    name = row['path']
    path = PurePosixPath(name)
    assert not path.is_absolute() and str(path) == name and '..' not in path.parts and '\\' not in name
    data = (root / name).read_bytes()
    assert len(data) == row['bytes'] and hashlib.sha256(data).hexdigest() == row['sha256']
    if name not in raw:
        paths.append(name)
assert len(paths) == len(set(paths))
pathspec_bytes = b''.join((':(literal)' + name).encode('utf-8') + b'\0' for name in sorted(paths))
pathspec = root / '.local' / 'parallel_review_1948_required_paths.nul'
assert not pathspec.exists()
pathspec.write_bytes(pathspec_bytes)
command = ['git', 'add', '-f', '--pathspec-from-file=' + str(pathspec), '--pathspec-file-nul']
r = subprocess.run(command, cwd=root, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
assert r.returncode == 0, r.stderr.decode('utf-8')

folder = root / (base + 'preflight/parallel_review_batch_20261017T1948_git_pathspec_repair')
assert not folder.exists()
folder.mkdir()
helper = folder / 'finish_stage_paths.py.txt'
helper.write_bytes(Path(__file__).read_bytes())
original = root / (base + 'preflight/root_operational_source/stage_parallel_review_20261017T1948.py.txt')
failure = folder / 'original_failure.txt'
failure.write_text('The first ordinary git add completed. The subsequent force-add of individually verified required files was not launched by Windows.\n'
    'Traceback source: .local/stage_parallel_review_20261017T1948.py line56\n'
    "subprocess.run(['git', 'add', '-f', '--', *sorted(set(required) - raw_names)], cwd=root, check=True)\n"
    'FileNotFoundError: [WinError 206] 文件名或扩展名太长。\n'
    'No experiment source, job, weight, raw original or original required manifest was changed for this correction.\n', encoding='utf-8', newline='\n')
value = dict(schema='formal_main_v2', kind='root_git_delivery_pathspec_repair', status='passed',
    recorded_utc=datetime.now(timezone.utc).isoformat(), original_error='Windows CreateProcess WinError206 before force-add process launch',
    original_stage_source_archive=original.relative_to(root).as_posix(), original_stage_source_sha256=hashlib.sha256(original.read_bytes()).hexdigest(),
    required_manifest=required_path, required_manifest_sha256=hashlib.sha256(required_bytes).hexdigest(),
    pathspec_sha256=hashlib.sha256(pathspec_bytes).hexdigest(), pathspec_bytes=len(pathspec_bytes), exact_path_count=len(paths),
    omitted_archived_raw_paths=sorted(raw), original_required_hashes_all_rechecked=True,
    retry_command=command, retry_exit_code=r.returncode, retry_stdout=r.stdout.decode('utf-8'), retry_stderr=r.stderr.decode('utf-8'),
    source_archive_sha256=hashlib.sha256(helper.read_bytes()).hexdigest(), failure_record_sha256=hashlib.sha256(failure.read_bytes()).hexdigest(),
    scientific_settings_changed=False, frozen_runtime_source_changed=False, raw_originals_changed=False,
    weights_added=False, GPU_seconds=0, index_only_full_reconstruction_still_required=True)
receipt = folder / 'root_review.json'
receipt.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n', encoding='utf-8', newline='\n')
subprocess.run(['git', 'add', '--', *[p.relative_to(root).as_posix() for p in (helper, failure, receipt)]], cwd=root, check=True)
print(json.dumps(dict(status='exact_paths_staged', paths=len(paths), required_sha256=value['required_manifest_sha256'],
    repair_receipt=receipt.relative_to(root).as_posix(), repair_sha256=hashlib.sha256(receipt.read_bytes()).hexdigest())))
