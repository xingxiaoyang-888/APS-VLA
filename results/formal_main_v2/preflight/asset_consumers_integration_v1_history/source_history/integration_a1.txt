def save(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, sort_keys=True, indent=2) + "\n", encoding="utf-8", newline="\n")


def read(path):
    return json.loads(path.read_text(encoding="utf-8"))


def translate(value, old, new):
    if isinstance(value, dict):
        return {translate(k, old, new): translate(v, old, new) for k, v in value.items()}
    if isinstance(value, list):
        return [translate(v, old, new) for v in value]
    if isinstance(value, str):
        normalized = value.replace("\\", "/")
        if normalized == old or normalized.startswith(old + "/"):
            return new + normalized[len(old):]
    return value

def synthetic_source(p, suite, rows, input_sha):
    """Explicit GPU-source stand-in; actual helper source tests are separate."""
    a = p.actual.adapter
    result = dict(schema=a.SCHEMA, kind="asset_downstream_source_provenance", status="verified", suite=suite,
        metadata_implementation=a.metadata_contract(), registered_source_jobs=[],
        shared_projection=dict(input_acceptance_sha256=input_sha),
        method_bindings={r["identity"]["method_id"]: r["identity"] for r in rows},
        synthetic_upstream_provenance_standin=True)
    result["shared_projection_sha256"] = p.d.signature(result["shared_projection"])
    if suite == "libero_goal":
        result["component_implementation"] = a.component_contract()
    return result

def execute_plan(c, monkeypatch, *, job_id=None):
    p, t, case = c.p, c.t, c.case
    if job_id:
        case.job = job_id
    original = deepcopy(c.plan["invocation"])
    inv = translate(original, str(t.P), p.actual.project.as_posix())
    # Official model locations are not local project inputs and are never
    # fabricated as aliases. Only tiny Goal test tensor I/O is substituted.
    checkpoint_index = inv["argv"].index("--checkpoint") + 1
    inv["argv"][checkpoint_index] = original["argv"][checkpoint_index]
    inv["source_provenance"] = deepcopy(original["source_provenance"])
    inv["source_inventory_sha256"] = t.digest(p.actual.project / t.SOURCE_INVENTORY)
    # A resumed fixture has two representations of retained evidence: original
    # Windows writer bytes and the explicit server-path transport. Keep each
    # internally bound; never rewrite the original producer's old archives.
    historical = {}
    previous_ledger = b""
    previous_prefix = None
    if original["resume"]:
        prior = read(p.root / original["prior_review"])
        historical = {name: (p.root / name).read_bytes() for name in prior["files"]}
        previous_ledger = (p.root / c.plan["output_relative"] / "trials.jsonl").read_bytes()
        from cti_vla.evaluation import formal_trials
        old = scheduler_fixtures.module(t.ADAPTER)
        old_config = read(p.root / c.plan["output_relative"] / "config.json")
        previous_prefix = old.checked_ledger(p.root / c.plan["output_relative"] / "trials.jsonl",
            t.signature(old_config), panel=read(p.root / c.plan["output_relative"] / "panel.json"),
            identifiers=original["method_ids"], validators=formal_trials)
    for path in inv["watched_files"]:
        target = Path(path)
        if not target.exists():
            rel = target.relative_to(p.actual.project).as_posix()
            target.parent.mkdir(parents=True, exist_ok=True)
            shutil.copyfile(p.root / rel, target)
        inv["watched_files"][path] = t.digest(target)
    case.value = inv
    case.path = p.actual.project / c.plan["invocation_relative"]
    case.output = p.actual.project / c.plan["output_relative"]
    case.args["output"] = case.output
    save(case.path, inv)
    monkeypatch.setattr(p.writer, "CPU_PROGRAM", scientific_program(p.writer.CPU_PROGRAM))
    p.writer.run_case(p.actual, case)
    actual = runtime_fixtures.transport_downstream(p, case)
    # Undo no original producer bytes: restore only the previous transported
    # snapshot that the generic first-attempt fixture copier does not preserve.
    # Rebind the current attempt to the exact actual E04 plan below.
    for name, raw in historical.items():
        (p.root / name).write_bytes(raw)
    actual.invocation = deepcopy(original)
    invrel = c.plan["invocation_relative"]
    save(p.root / invrel, actual.invocation)
    invsha = t.digest(p.root / invrel)
    actual.job.update(invocation=invrel, invocation_sha256=invsha)
    actual.identity["invocation_sha256"] = actual.complete["invocation_sha256"] = invsha
    actual.identity["watched_files"] = {str(t.P / invrel): invsha, **original["watched_files"]}
    for row in actual.events:
        row["invocation_sha256"] = invsha
    # Transport helper retains original raw bytes. Add the real CPU trial
    # outputs; server-path rewriting requires recomputing their run signature.
    old_config = read(case.output / "config.json")
    config = translate(old_config, p.actual.project.as_posix(), str(t.P))
    signature = t.signature(config)
    target = p.root / actual.relative
    save(target / "config.json", config)
    panel = read(case.output / "panel.json")
    save(target / "panel.json", panel)
    events = [json.loads(line) for line in (case.output / "trials.jsonl").read_text(encoding="utf-8").splitlines()]
    for row in events:
        row["run_signature"] = signature
    old_count = len(previous_ledger.splitlines())
    suffix = "".join(json.dumps(row, sort_keys=True) + "\n" for row in events[old_count:]).encode()
    (target / "trials.jsonl").write_bytes(previous_ledger + suffix)
    from cti_vla.evaluation import formal_trials
    from cti_vla.evaluation.libero_runtime import TrialLedger
    ledger = TrialLedger(target / "trials.jsonl", run_signature=signature)
    report = formal_trials.summarize_formal_trials(panel, actual.invocation["method_ids"], ledger,
        provenance={**config, "run_signature": signature, "checkpoint": config["synthetic_checkpoint_metadata"],
            "runtime": {**config["runtime"], "slurm_job_id": case.job}}, checkpoint_unchanged=True)
    save(target / "closed_loop_report.json", report)
    old = scheduler_fixtures.module(t.ADAPTER)
    actual.oldreceipt["after_ledger"] = old.checked_ledger(target / "trials.jsonl", signature,
        panel=panel, identifiers=actual.invocation["method_ids"], validators=formal_trials)
    if previous_prefix is not None:
        actual.oldreceipt["before_ledger_raw"] = {k: previous_prefix[k] for k in ("bytes", "sha256")}
        actual.oldreceipt["before_ledger"] = deepcopy(previous_prefix)
        for row in actual.oldevents:
            if row["event"] == "resume_preflight_passed":
                row["evidence"] = deepcopy(previous_prefix)
    actual.source_proof = deepcopy(c.proof)
    actual.job.update(name="cti-formal-downstream", method_ids=actual.invocation["method_ids"],
        source_provenance_sha256=t.signature(c.proof), submitted_utc="2026-01-01")
    runtime_fixtures.seal_downstream(actual)
    actual.wrapper_receipt["wall_seconds"] = 500.
    save(p.root / f"results/{t.ROUND}/job_receipts/{case.job}.json", actual.wrapper_receipt)
    for ext in ("out", "err"):
        save(p.root / f"results/slurm/cti-formal-downstream_{case.job}.{ext}", dict(synthetic=True))
    c.state["registered_jobs"].append(actual.job)
    # Use the original complete-pull fixture writer, including all run bytes,
    # both journals, Slurm logs, terminal accounting and parent job receipt.
    pull_context = SimpleNamespace(root=p.root, scheduler=t, args=c.args,
        save=lambda name, data: save(p.root / name, data))
    audit_context = SimpleNamespace(output=actual.relative, job=actual.job,
        pull_relative=actual.pull.relative_to(p.root).as_posix())
    science_fixtures.pull(pull_context, audit_context)
    reviewer = scheduler_fixtures.module("scripts/review_formal_downstream.py")
    result = reviewer.review(case.job, audit_context.pull_relative, project=p.root,
        state=c.state, scheduler=t, helper=c.helper)
    ref = f"results/{t.ROUND}/downstream_reviews/{case.job}.json"
    save(p.root / ref, result)
    for name, raw in historical.items():
        if name not in (actual.relative + "/trials.jsonl", actual.relative + "/closed_loop_report.json"):
            assert (p.root / name).read_bytes() == raw
    assert (target / "trials.jsonl").read_bytes().startswith(previous_ledger)
    c.actual, c.report, c.review_ref = actual, result, ref
    return result

@pytest.mark.parametrize("suite,count", [("libero_object", 10), ("libero_goal", 14)])
def test_production_plan_actual_writer_and_original_science_root_review(actual_project, monkeypatch, suite, count):
    c = prepare(actual_project, monkeypatch, suite)
    inv = c.plan["invocation"]
    assert len(inv["support_files"]) == count and inv["entrypoint"] == c.t.ASSET_ADAPTER
    assert inv["support_files"].items() <= c.plan["publish_files"].items()
    result = execute_plan(c, monkeypatch)
    assert result["status"] == "passed" and not result["scientific_complete"]
    assert result["completed_trials"] == 1 and result["source_provenance"] == c.proof
    assert result["asset_runtime_review"]["metadata_loads"] == 1
    assert result["asset_runtime_review"]["component_loads"] == (2 if suite == "libero_goal" else 0)
    assert c.t.STATE not in result["evidence_sha256"]
    c.t.require_root_review(c.helper.Evidence(c.p.root), c.review_ref,
        c.t.formal_jobs(c.state), expected_stage="development")
    reviewer = scheduler_fixtures.module("scripts/review_formal_downstream.py")
    for field in ("source_provenance", "asset_runtime_review", "registered_jobs_at_review", "cost"):
        changed = deepcopy(result)
        changed.pop(field)
        with pytest.raises(ValueError):
            c.t.require_root_review(reviewer.CandidateEvidence(c.helper.Evidence(c.p.root), changed),
                "in_memory_candidate", c.t.formal_jobs(c.state), expected_stage="development")
    # Execute the actual E04 continuation on the same persistent raw ledger.
    c.args.prior_review, c.args.resume = c.review_ref, True
    c.case.job = str(int(c.case.job) + 1000)
    resumed = c.t.plan(c.args, c.state, project=c.p.root, helper=c.helper)
    prefix = c.t.resume_prefix_inputs(resumed["invocation"])
    assert set(prefix) == {inv["run_root"] + "/trials.jsonl", inv["run_root"] + "/closed_loop_report.json"}
    for path, checksum in prefix.items():
        relative = PurePosixPath(path).relative_to(c.t.P).as_posix()
        assert checksum == result["files"][relative] == resumed["invocation"]["root_evidence_sha256"][relative]
        assert path not in resumed["invocation"]["watched_files"]
    assert all(str(c.t.P / name) in resumed["invocation"]["watched_files"]
        for name in result["files"] if str(c.t.P / name) not in prefix)
    previous_report, previous_id = deepcopy(result), c.actual.job["job_id"]
    c.plan = resumed
    continued = execute_plan(c, monkeypatch)
    assert continued["completed_trials"] == 2 and not continued["scientific_complete"]
    assert continued["raw_invocation_job_ids"] == [previous_id, c.case.job]
    assert continued["cost"]["all_attempt_gpu_seconds"] == 1200
    assert len(continued["cost"]["allocations"]) == 2
    assert continued["asset_runtime_review"]["legacy_adapter_attempt_id"].startswith(c.case.job + "_")
    assert previous_report["reconstructed_summary"]["completed_trials"] == 1
    c.t.require_root_review(c.helper.Evidence(c.p.root), c.review_ref,
        c.t.formal_jobs(c.state), expected_stage="development")

def test_legacy_runtime_still_requires_source_provenance(tmp_path, monkeypatch, suite):
    p = scheduler_fixtures.project.__wrapped__(tmp_path, monkeypatch)
    rows = p.helper.suite_rows(p.helper.Evidence(p.root), p.registration, suite, p.sources, ["dense"])
    p.save(p.args.execution_manifest, p.helper.make_execution(p.registration, suite, rows))
    called = []
    original = p.scheduler.source_provenance
    def counted(*args, **kwargs):
        called.append(kwargs["suite"])
        return original(*args, **kwargs)
    monkeypatch.setattr(p.scheduler, "source_provenance", counted)
    plan = scheduler_fixtures.make_plan(p)
    assert called == [suite] and plan["invocation"]["entrypoint"] == p.scheduler.ADAPTER
    assert "downstream_asset_compatibility" not in plan["invocation"]
    assert "source_provenance" in plan["invocation"]
    def rejected(*args, **kwargs):
        raise ValueError("Synthetic incomplete upstream training provenance")
    monkeypatch.setattr(p.scheduler, "source_provenance", rejected)
    with pytest.raises(ValueError, match="upstream training provenance"):
        scheduler_fixtures.make_plan(p)
