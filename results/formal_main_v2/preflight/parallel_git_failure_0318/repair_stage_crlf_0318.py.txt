"""Root-only delivery repair for one observed historical CRLF input version.

Never rewrite the original diagnostic or batch. Preserve the rejected attempt,
deliver both exact historical bytes and the existing canonical Git bytes, and
rerun the unchanged full index/archive guard with explicit version resolution.
"""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import importlib.util
import json
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / 'scripts'))
import formal_supervisor as f

PARENT = 'fee1befa20681fd87228e17d3be04577d0e11430'
BASE = 'results/formal_main_v2/preflight/'
OLD_REQUIRED = BASE + 'parallel_review_batch_20261018T0318_git_required.json'
BATCH = BASE + 'parallel_review_batch_20261018T0318.json'
DIAGNOSIS = BASE + 'object_action_range_diagnosis_0318.json'
OLD_HELPER = '.local/stage_verify_parallel_0318.py'
SOURCE = 'configs/assets/runtime_sources.json'
OLD_SHA = '2dd246313eb56cb799d01fd097ff9608a56b8fa26255fe2a876d0fb0eac5ad07'
CANON_SHA = '4e699692dfbee08c2b41a0997ce8225ac07df429eb30c03e4baf12bf23c27c5f'
FOLDER = BASE + 'parallel_git_failure_0318/'
ARCHIVE = FOLDER + 'runtime_sources_original_crlf.json'
ALIAS = BASE + 'parallel_git_source_version_resolution_0318.json'
REQUIRED = BASE + 'parallel_review_batch_20261018T0318_git_required_v2.json'
PROOF = BASE + 'parallel_review_batch_20261018T0318_git_delivery_v2.json'

def sha(b):
    return hashlib.sha256(b).hexdigest()

def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)

def row(n):
    p = ROOT / n
    assert p.is_file() and not p.is_symlink(), n
    b = p.read_bytes()
    return dict(path=n, sha256=sha(b), bytes=len(b))

def write_new(n, b):
    (ROOT / n).parent.mkdir(parents=True, exist_ok=True)
    with (ROOT / n).open('xb') as stream:
        stream.write(b)

def json_new(n, v):
    write_new(n, (json.dumps(v, ensure_ascii=False, sort_keys=True, indent=2) + '\n').encode())

def main():
    assert git('rev-parse', 'HEAD').decode().strip() == PARENT
    pins = {
        OLD_REQUIRED: '5ab3fe1781718ffd3e364f2d2cebdfd4e9cc6ccda2e28d33b6a44c1bbaaa5402',
        BATCH: 'e34308d71409ed8eb1f1b6b995a5b758458b631409365e123444af59287ee430',
        DIAGNOSIS: '799c9e4380a31de8b614c12592871373af6b8e7948259010eaf8c938cfdf0e48',
        OLD_HELPER: '2ea0ef588fec78561d3fd4f08df2530b1171cc812402fb61143a5ba9bfcc5f61',
    }
    for n, h in pins.items():
        assert row(n)['sha256'] == h, n
    spec = importlib.util.spec_from_file_location('old_stage_contract', ROOT / OLD_HELPER)
    old = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(old)
    for n, h in old.PINS.items():
        assert row(n)['sha256'] == h
    inventory = json.loads((ROOT / 'results/formal_main_v2/source/formal_main_v2_run_564be0995e.json').read_bytes())
    assert SOURCE not in inventory['files']
    original = (ROOT / SOURCE).read_bytes()
    canonical = git('show', 'HEAD:' + SOURCE)
    assert len(original) == 682 and sha(original) == OLD_SHA
    assert len(canonical) == 663 and sha(canonical) == CANON_SHA
    assert canonical == git('show', ':' + SOURCE)
    assert original.count(b'\r\n') == 19 and original.replace(b'\r\n', b'\n') == canonical
    assert json.loads(original) == json.loads(canonical)
    assert not git('status', '--porcelain', '--', SOURCE).strip()
    original_required = json.loads((ROOT / OLD_REQUIRED).read_bytes())
    required = {r['path']: r for r in original_required['files']}
    assert len(required) == len(original_required['files']) == 1783
    assert required[SOURCE] == dict(path=SOURCE, sha256=OLD_SHA, bytes=682)
    for n, r in required.items():
        assert row(n) == r, n
    assert json.loads((ROOT / BATCH).read_bytes())['evidence_sha256'][SOURCE] == OLD_SHA
    assert json.loads((ROOT / DIAGNOSIS).read_bytes())['read_evidence_sha256'][SOURCE] == OLD_SHA
    assert not (ROOT / FOLDER).exists()
    for n in (ALIAS, REQUIRED, PROOF):
        assert not (ROOT / n).exists(), n
    before_repair = git('ls-files', '--stage', '-z')
    old_panels = {}
    for n, h in old.PANEL_PINS.items():
        assert row(n)['sha256'] == h == sha(git('show', 'HEAD:' + n)) == sha(git('show', ':' + n))
        assert not git('status', '--porcelain', '--', n).strip()
        old_panels[n] = row(n)

    # All old receipts remain immutable, including the originally rejected
    # requirements; this folder records the observed tool output, not a rerun.
    write_new(ARCHIVE, original)
    for n, name in [(OLD_REQUIRED, 'original_required.json'), (OLD_HELPER, 'original_stage_helper.py.txt')]:
        write_new(FOLDER + name, (ROOT / n).read_bytes())
    self_archive = FOLDER + 'repair_stage_crlf_0318.py.txt'
    write_new(self_archive, Path(__file__).read_bytes())
    json_new(FOLDER + 'observed_guard_failure.json', dict(
        schema='formal_main_v2', kind='retained_CPU_Git_delivery_failure',
        observed_from='exec session13673 terminal output; original complete subprocess streams were not saved by that helper',
        exit_code=1, guard_status='rejected', error_type='StageError',
        error='Working tree differs from index: configs/assets/runtime_sources.json',
        original_required=row(OLD_REQUIRED), original_stage_helper=row(OLD_HELPER),
        index_snapshot_sha256=sha(before_repair), GPU_seconds=0,
        scientific_runtime_changed=False, experimental_reruns=0))
    resolution = dict(schema='formal_main_v2', kind='single_historical_input_version_resolution',
        status='exact_version_archived', original_path=SOURCE, original_sha256=OLD_SHA, original_bytes=682,
        exact_original_archive=row(ARCHIVE), existing_HEAD_and_index_sha256=CANON_SHA, canonical_bytes=663,
        parent_commit=PARENT, batch=row(BATCH), diagnosis=row(DIAGNOSIS), rejected_required=row(OLD_REQUIRED),
        conversion=dict(operation='replace CRLF by LF', replacements=19, byte_equal_to_existing_HEAD=True,
            JSON_value_equal=True, not_a_frozen_runtime_file=True),
        interpretation='Only the historical (path,SHA,bytes) tuple resolves to this exact archive. Old batch and diagnosis are unchanged and do not directly rehash against the now-canonical current path.',
        arbitrary_SHA_mismatches_allowed=False, guard_changed=False, GPU_seconds=0)
    json_new(ALIAS, resolution)
    (ROOT / SOURCE).write_bytes(canonical)
    assert git('show', ':' + SOURCE) == git('show', 'HEAD:' + SOURCE) == canonical
    assert not git('status', '--porcelain', '--', SOURCE).strip()
    assert row(ARCHIVE) == dict(path=ARCHIVE, sha256=OLD_SHA, bytes=682)
    assert git('ls-files', '--stage', '-z') == before_repair

    with f.supervisor_lock():
        state_name = 'results/experiment_supervisor/state.json'
        assert row(state_name) == required[state_name]
        original_state_bytes = f.STATE.read_bytes()
        original_state = json.loads(original_state_bytes)
        state = json.loads(original_state_bytes)
        assert 'nonweight_delivery_source_version_resolution' not in state['formal_main_progress']
        write_new(FOLDER + 'original_state_before_alias.json', original_state_bytes)
        state['formal_main_progress']['nonweight_delivery_source_version_resolution'] = dict(
            path=ALIAS, sha256=row(ALIAS)['sha256'], failed_original_guard_preserved=True,
            historical_evidence_is_resolved_via_exact_archive=True, scientific_settings_changed=False,
            current_config_equals_existing_HEAD=True, GPU_seconds=0)
        f.write(f.STATE, state)
        verified_state = f.read(f.STATE)
        assert verified_state['formal_main_progress'].pop('nonweight_delivery_source_version_resolution') == state['formal_main_progress']['nonweight_delivery_source_version_resolution']
        assert verified_state == original_state
    for n, r in list(required.items()):
        if n in (SOURCE, 'results/experiment_supervisor/state.json'):
            required[n] = row(n)
        else:
            assert row(n) == r, n
    added = [ALIAS, OLD_REQUIRED]
    added.extend(p.relative_to(ROOT).as_posix() for p in (ROOT / FOLDER).iterdir())
    for n in added:
        old.safe_name(n)
        required[n] = row(n)
    json_new(REQUIRED, dict(schema='cti_nonweight_archive_stage_v1', kind='required_files',
        files=[required[n] for n in sorted(required)]))
    raw_names = {r['path'] for r in json.loads((ROOT / old.SNAPSHOT_NAME).read_bytes())['files']}
    # Precisely these new forensic records and the state pointer are staged.
    names = sorted(set(added + [REQUIRED, 'results/experiment_supervisor/state.json']))
    specpath = ROOT / '.local/stage_0318_repair_v2.nul'
    with specpath.open('xb') as stream:
        stream.write(b''.join((':(literal)' + n).encode() + b'\0' for n in names))
    subprocess.run(['git', 'add', '-f', '--pathspec-from-file=' + str(specpath), '--pathspec-file-nul'], cwd=ROOT, check=True)
    assert git('show', ':' + ARCHIVE) == original
    assert git('show', ':' + SOURCE) == canonical
    before, changes = git('ls-files', '--stage', '-z'), git('diff', '--cached', '--name-status', '-z')
    command = [sys.executable, '-B', 'scripts/ops/check_nonweight_archive_stage.py', '--project-root', str(ROOT),
        '--snapshot', old.SNAPSHOT_NAME, '--snapshot-sha256', old.SNAPSHOT_SHA, '--required-files', REQUIRED]
    for n in sorted(raw_names):
        command.extend(['--raw-path', n])
    print(json.dumps(dict(status='guard_retry_running', required_count=len(required), historical_alias=ALIAS)), flush=True)
    process = subprocess.run(command, cwd=ROOT, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    if process.returncode:
        json_new(FOLDER + 'guard_retry_failure.json', dict(returncode=process.returncode,
            stdout=process.stdout.decode('utf-8', errors='replace'), stderr=process.stderr.decode('utf-8', errors='replace'), GPU_seconds=0))
        raise RuntimeError(process.stderr.decode('utf-8', errors='replace'))
    report = json.loads(process.stdout)
    assert report['status'] == 'verified_index_archive_delivery' and report['head'] == PARENT
    assert report['index_sha256'] == sha(before) and report['staged_changes_sha256'] == sha(changes)
    assert report['required_count'] == len(required) and report['frozen_files_verified_in_worktree_and_index'] == 173
    assert len(report['raw_paths_restored_exactly']) == 2 and report['gpu_seconds'] == 0
    assert before == git('ls-files', '--stage', '-z') and changes == git('diff', '--cached', '--name-status', '-z')
    for n, r in old_panels.items():
        assert row(n) == r and sha(git('show', ':' + n)) == sha(git('show', 'HEAD:' + n)) == r['sha256']
        assert not git('status', '--porcelain', '--', n).strip()
    for n, h in pins.items():
        assert row(n)['sha256'] == h, n
    assert row(ARCHIVE)['sha256'] == sha(git('show', ':' + ARCHIVE)) == OLD_SHA
    assert row(SOURCE)['sha256'] == sha(git('show', ':' + SOURCE)) == sha(git('show', 'HEAD:' + SOURCE)) == CANON_SHA
    json_new(PROOF, dict(schema='formal_main_v2', kind='root_git_index_nonweight_delivery', status='passed',
        reviewed_utc=datetime.now(timezone.utc).isoformat(), guard_result=report, guard_command=command,
        historical_source_resolution=row(ALIAS), exact_original_archive=row(ARCHIVE), canonical_current_source=row(SOURCE),
        failed_original_requirements=row(OLD_REQUIRED), source_canonicalization_only=True,
        old_batch_and_diagnosis_unchanged=True, other_original_required_files_verified=True,
        unrelated_evidence_mismatch_exemptions=0, unchanged_index_until_guard_return=True,
        separate_development_panels_verified_before_and_after=list(old_panels),
        no_weights_delivered=True, final_states_opened=False, scientific_settings_changed=False, GPU_seconds=0))
    subprocess.run(['git', 'add', '--', PROOF], cwd=ROOT, check=True)
    oldrows, newrows = set(before.split(b'\0')) - {b''}, set(git('ls-files', '--stage', '-z').split(b'\0')) - {b''}
    assert not oldrows - newrows and len(newrows - oldrows) == 1
    assert next(iter(newrows - oldrows)).split(b'\t', 1)[1].decode() == PROOF
    assert git('show', ':' + PROOF) == (ROOT / PROOF).read_bytes()
    subprocess.run(['git', 'diff', '--cached', '--check'], cwd=ROOT, check=True)
    assert not git('diff', '--name-only').strip()
    print(json.dumps(dict(status='passed', proof=row(PROOF), required_count=len(required),
        frozen_files=173, raw_archives_restored=2, commit_not_performed=True)), flush=True)

if __name__ == '__main__':
    main()
