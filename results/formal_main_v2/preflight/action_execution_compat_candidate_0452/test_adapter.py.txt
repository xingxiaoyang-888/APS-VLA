class Clock:
    def __init__(self):self.value = 0.
    def perf_counter(self):
        self.value += .001
        return self.value
    def reset(self):self.value = 0.

def scope(modules, rows=None, **overrides):
    runtime, trials, _ = modules
    options = dict(project_root=ROOT, source_root=ROOT, runtime_module=runtime, formal_trials_module=trials,
        expected_adapter_sha256=ADAPTER_SHA, emit=lambda event, **fields: rows.append(dict(event=event, **fields)) if rows is not None else None)
    options.update(overrides)
    return m.install_action_execution_compat(**options)


def observation(step):
    return dict(agentview_image=np.full((2, 2, 3), step % 255, dtype=np.uint8),
        robot0_eye_in_hand_image=np.zeros((2, 2, 3), dtype=np.uint8), robot0_eef_pos=np.array([step, 0., 0.]),
        robot0_eef_quat=np.array([0., 0., 0., 1.]), robot0_gripper_qpos=np.array([0., 0.]))


class Environment:
    def __init__(self, *, success_step=None, fail_step=None):
        self.actions = []; self.success_step = success_step; self.fail_step = fail_step
        self.resets = 0; self.state = None
    def reset(self):self.resets += 1
    def set_init_state(self, state):self.state = state.copy(); return observation(0)
    def step(self, action):
        self.actions.append(copy.deepcopy(action))
        if len(self.actions) == self.fail_step:raise RuntimeError("synthetic environment exception")
        return observation(len(self.actions)), 0., len(self.actions) == self.success_step, {}


class Policy:
    def __init__(self, runtime, *, values=None, method="dense", failure=None, clock=None, draw_rng=False):
        self.runtime = runtime; self.values = np.zeros((8, 7), dtype=np.float64) if values is None else values
        self.method = method; self.failure = failure; self.clock = clock; self.draw_rng = draw_rng
        self.calls = 0; self.resets = 0
    def reset(self):self.resets += 1
    def __call__(self, obs, instruction):
        self.calls += 1
        if self.draw_rng:random.random(); np.random.random()
        if self.failure == "exception":raise ValueError("synthetic policy exception")
        if self.failure == "wrong_type":return object()
        if self.failure == "watchdog":self.clock.value += 61.
        return self.runtime.PolicyChunk(np.zeros((8, 7)), np.asarray(self.values), {"method": self.method})


def episode(function, runtime, clock, *, values=None, method="dense", failure=None, success_step=None,
        fail_step=None, max_steps=19, draw_rng=False):
    env = Environment(success_step=success_step, fail_step=fail_step)
    policy = Policy(runtime, values=values, method=method, failure=failure, clock=clock, draw_rng=draw_rng)
    clock.reset()
    result = function(env, policy, "synthetic task", [1., 2.], max_steps=max_steps)
    return result, env, policy


@pytest.mark.parametrize("method", ["dense", "uniform", "A", "B"])
@pytest.mark.parametrize("success_step", [None, 27])
def test_inrange_every_action_and_result_exact(modules, method, success_step):
    runtime, trials, clock = modules
    original = runtime.run_episode
    values = np.linspace(-1., 1., 56).reshape(8, 7)
    before, env_before, policy_before = episode(original, runtime, clock, values=values, method=method, success_step=success_step)
    rows = []
    with scope(modules, rows) as identity:
        assert runtime.run_episode is trials.run_episode and runtime.run_episode is not original
        after, env_after, policy_after = episode(trials.run_episode, runtime, clock, values=values, method=method, success_step=success_step)
        assert identity["candidate_only"] and not identity["deployment_authorized"]
        assert identity["ast_proof"]["removed_if_count"] == 1 and identity["ast_proof"]["other_AST_nodes_unchanged"]
    assert before == after
    assert env_before.actions == env_after.actions and np.array_equal(env_before.state, env_after.state)
    assert (policy_before.calls, policy_before.resets) == (policy_after.calls, policy_after.resets)
    assert runtime.run_episode is original and trials.run_episode is original
    assert rows[-1]["original_bindings_restored"] and rows[-1]["source_inputs_unchanged"]


@pytest.mark.parametrize("value", [1.010742197890625, -1.010742197890625, 2.])
def test_finite_overrange_original_value_reaches_environment(modules, value):
    runtime, trials, clock = modules
    values = np.zeros((8, 7)); values[0, 2] = value
    before, old_env, _ = episode(runtime.run_episode, runtime, clock, values=values, max_steps=8)
    assert before["status"] == "policy_error" and before["control_steps"] == 0 and len(old_env.actions) == 10
    with scope(modules):
        after, env, _ = episode(trials.run_episode, runtime, clock, values=values, max_steps=8)
    assert after["status"] == "completed" and after["control_steps"] == 8
    assert env.actions[10:] == values.tolist() and env.actions[10][2] == value
    assert after["executed_actions"] == values.tolist() and after["chunks"][0]["environment_actions"] == values.tolist()
    assert not after["success"]  # Synthetic horizon, not a real task success claim.


@pytest.mark.parametrize("bad", ["nan", "inf", "negative_inf", "shape", "wrong_type", "exception", "watchdog"])
def test_original_invalid_actions_watchdog_errors_unchanged(modules, bad):
    runtime, trials, clock = modules
    values = np.zeros((8, 7)); failure = None
    if bad in ("nan", "inf", "negative_inf"):values[0, 0] = {"nan": np.nan, "inf": np.inf, "negative_inf": -np.inf}[bad]
    elif bad == "shape":values = np.zeros((7, 7))
    else:failure = bad
    original, env_a, _ = episode(runtime.run_episode, runtime, clock, values=values, failure=failure)
    with scope(modules):adapted, env_b, _ = episode(trials.run_episode, runtime, clock, values=values, failure=failure)
    assert original == adapted and original["status"] == "policy_error"
    assert original["control_steps"] == 0 and env_a.actions == env_b.actions

def test_scope_does_not_draw_rng_and_policy_rng_is_identical(modules):
    runtime, trials, clock = modules
    random.seed(17); np.random.seed(17)
    before = (random.getstate(), copy.deepcopy(np.random.get_state()))
    with scope(modules):pass
    assert random.getstate() == before[0]
    assert all(np.array_equal(a, b) for a, b in zip(np.random.get_state(), before[1]))
    original, _, _ = episode(runtime.run_episode, runtime, clock, draw_rng=True)
    original_end = (random.getstate(), copy.deepcopy(np.random.get_state()))
    random.setstate(before[0]); np.random.set_state(before[1])
    with scope(modules):adapted, _, _ = episode(trials.run_episode, runtime, clock, draw_rng=True)
    assert original == adapted and random.getstate() == original_end[0]
    assert all(np.array_equal(a, b) for a, b in zip(np.random.get_state(), original_end[1]))

@pytest.mark.parametrize("target", ["runtime_binding", "formal_alias", "replacement_code", "original_code", "original_defaults"])
def test_in_scope_tampering_is_rejected_and_original_restored(modules, target):
    runtime, trials, _ = modules; original = runtime.run_episode; code = original.__code__; defaults = copy.deepcopy(original.__kwdefaults__)
    with pytest.raises(RuntimeError, match="tampering"):
        with scope(modules):
            if target == "runtime_binding":runtime.run_episode = lambda: None
            elif target == "formal_alias":trials.run_episode = lambda: None
            elif target == "replacement_code":runtime.run_episode.__code__ = (lambda: None).__code__
            elif target == "original_code":original.__code__ = (lambda: None).__code__
            else:original.__kwdefaults__["max_steps"] = 1
    assert runtime.run_episode is original and trials.run_episode is original
    assert original.__code__ is code and original.__kwdefaults__ == defaults

@pytest.mark.parametrize("part", ["source", "registration"])
def test_file_tampering_rejected_before_binding(tmp_path, modules, part):
    runtime, trials, _ = modules; original = runtime.run_episode
    project = tmp_path / "project"
    for relative in (m.REGISTRATION, m.INVENTORY):
        path = project / relative; path.parent.mkdir(parents=True, exist_ok=True); shutil.copyfile(ROOT / relative, path)
    source = ROOT
    if part == "registration":(project / m.REGISTRATION).write_bytes(b"{}\n")
    else:
        source = project / "frozen_source"
        inventory = json.loads((ROOT / m.INVENTORY).read_bytes())
        for relative in inventory["files"]:
            path = source / relative; path.parent.mkdir(parents=True, exist_ok=True); shutil.copyfile(ROOT / relative, path)
        (source / m.RUNTIME).write_bytes((source / m.RUNTIME).read_bytes() + b"\n# synthetic tampering\n")
    with pytest.raises(ValueError, match="SHA mismatch"):
        with scope(modules, project_root=project, source_root=source):pass
    assert runtime.run_episode is original and trials.run_episode is original
