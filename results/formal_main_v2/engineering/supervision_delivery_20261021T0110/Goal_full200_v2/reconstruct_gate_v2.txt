def reconstruct_gate(project,rows,*,specification,job_id):
 """Rebuild all raw observation phases and the complete original trial chain.
 v2 closes the independent empty-array counterexample without changing runtime.
 """
 import numpy as np
 root=Path(project)/OUT;read_refs={};events=jsonl(root/'observer/events.jsonl');gate=jsonl(root/'observer/gate.jsonl')
 keys=('agentview_image','robot0_eye_in_hand_image','robot0_eef_pos','robot0_eef_quat','robot0_gripper_qpos')
 identity_keys=('trial_id','state_trial_id','task_id','task_name','source_state_index','initial_state_sha256','method')
 obs_phases={'reset_after','set_init_before','set_init_after','wait_before','wait_after','t0_query_before','t0_query_after','query_before','query_after'}
 plain_phases={'trial_begin_before_reseed','reseed_after','factory_before','factory_after','reset_before','original_runner_returned'}
 loader_phases=['existing_loader_event:'+x for x in ('asset_metadata_compat_load_start','asset_metadata_compat_load_complete','deployment_base_load_start','goal_component_load_start','goal_component_load_complete','goal_component_load_start','goal_component_load_complete','deployment_base_load_complete')]
 captures=[r for r in events if 'phase' in r]
 require(len(captures)<=21032 and not any(r.get('event')=='observer_abort' for r in events),'observer abort/bound')
 require(len(specification['rows'])==200 and len({r['trial_id'] for r in specification['rows']})==200,'fixed200 unique specification')
 expected={r['trial_id']:{k:r[k] for k in identity_keys} for r in specification['rows']}
 completed=[r for r in rows if r.get('event')=='result']
 require(len(completed)==200 and all(r['status']=='completed' for r in completed),'whole200 completed records required; policy_error never admitted')
 require([{k:r[k] for k in identity_keys} for r in completed]==list(expected.values()),'full200 original identity/order exact')
 for r in completed:
  require(type(r['control_steps']) is int and 1<=r['control_steps']<=300 and type(r['policy_calls']) is int and r['policy_calls']==(r['control_steps']+7)//8 and type(r['wait_steps']) is int and r['wait_steps']==10,'pertrial original300/chunk8/warmup10 bounds')
  require(len(r['chunks'])==r['policy_calls'] and len(r['executed_actions'])==r['control_steps'],'pertrial raw chunk and executed counts')
 require(sum(r['control_steps'] for r in completed)<=60000 and sum(r['policy_calls'] for r in completed)<=7600 and sum(r['wait_steps'] for r in completed)==2000,'control/query/warmup counts distinct')
 capidx={};raw_signatures={}
 for i,e in enumerate(captures,1):
  require(type(e['event_index']) is int and e['event_index']==i and len(canonical(e))<=16384,'capture sequence/size')
  require(e.get('invariants')=={'object_identity_and_bytes_unchanged':True,'passive_sim_and_environment_rng_unchanged':True,'global_rng_unchanged':True},'passive capture invariant missing')
  phase=e['phase'];require(phase in obs_phases|plain_phases|set(loader_phases),'unknown capture phase')
  arrays=e['observation_arrays'];require(type(arrays) is dict,'observation arrays must be a mapping')
  if phase in obs_phases:
   require(set(arrays)==set(keys) and 'observation_missing' not in e,'required observation phase has all five raw arrays')
   expected_schema={'agentview_image':('uint8',[256,256,3],196608),'robot0_eye_in_hand_image':('uint8',[256,256,3],196608),'robot0_eef_pos':('float64',[3],24),'robot0_eef_quat':('float64',[4],32),'robot0_gripper_qpos':('float64',[2],16)}
   require(all((arrays[k]['dtype'],arrays[k]['shape'],arrays[k]['bytes'])==v for k,v in expected_schema.items()),'exact actual five-component storage schema')
   aggregate=hashlib.sha256();components={}
   for key in keys:
    a=arrays[key];raw=read_blob(a['blob'],observation=True,output_root=root,read_refs=read_refs)
    require(type(a['bytes']) is int and a['bytes']==len(raw) and a['sha256']==digest(raw) and len(raw)==int(np.prod(a['shape']))*np.dtype(a['dtype']).itemsize,'array dtype/shape/byte metadata')
    aggregate.update(key.encode());aggregate.update(a['dtype'].encode());aggregate.update(str(tuple(a['shape'])).encode());aggregate.update(raw)
    components[key]={'dtype':str(np.dtype(a['dtype'])),'shape':a['shape'],'bytes':len(raw),'sha256':digest(raw)}
   require(aggregate.hexdigest()==e['observation_sha256'],'five-component aggregate reconstruction')
   raw_signatures[i]=components
  else:
   require(arrays=={} and e.get('observation_missing')=='no returned observation exists at this marker' and 'observation_sha256' not in e,'non-observation marker cannot claim unbacked observation bytes/hash')
  require(sum(v['bytes'] for v in e.get('sim_array_blobs',{}).values())<=8192,'sim byte bound')
  for v in e.get('sim_array_blobs',{}).values():read_blob(v,observation=False,output_root=root,read_refs=read_refs)
  rb=read_blob(e['global_rng'],observation=False,output_root=root,read_refs=read_refs);require(len(rb)<=65536,'RNG bound')
  capidx[i]=e
 # Exact real loader prefix is also independently tied to the real outer proof
 # by review_terminal. CPU tests explicitly derive this one fixture assertion.
 prefix=[];pos=0
 while pos<len(events) and events[pos].get('phase','').startswith('existing_loader_event:'):
  prefix.append(events[pos]['phase']);e=events[pos]
  require(e['trial_identity'] is None and e['ordinal'] is None and e['control_step'] is None and e['wait_steps'] is None,'loader capture outside all trials')
  pos+=1
 require(prefix==loader_phases,'complete original loader capture sequence')
 trialcaps={};trial_end_count=0
 for ordinal,row in enumerate(completed,1):
  ident=expected[row['trial_id']]
  shape=[(x,None,None) for x in ('trial_begin_before_reseed','reseed_after','factory_before','factory_after','reset_before','reset_after','set_init_before','set_init_after')]
  for wait in range(10):shape.extend([('wait_before',None,wait),('wait_after',None,wait+1)])
  for qi in range(row['policy_calls']):
   shape.extend([('t0_query_before' if qi==0 else 'query_before',8*qi,None),('t0_query_after' if qi==0 else 'query_after',8*qi,None)])
  observed=[]
  for phase,step,wait in shape:
   require(pos<len(events),'missing capture chain tail');e=events[pos];pos+=1
   require(e.get('phase')==phase and e.get('trial_identity')==ident and type(e.get('ordinal')) is int and e['ordinal']==ordinal and e.get('control_step')==step and e.get('wait_steps')==wait,'complete original capture phase/order/identity/counters')
   require((step is None or type(e['control_step']) is int) and (wait is None or type(e['wait_steps']) is int),'capture counter strict integer')
   observed.append(e)
  trialcaps[row['trial_id']]=observed
  require(pos<len(events),'missing original trial-end witness');end=events[pos];pos+=1
  require(end.get('event')=='trial_end' and 'phase' not in end and end.get('trial_identity')==ident and type(end.get('ordinal')) is int and end['ordinal']==ordinal and end.get('status')==row['status'] and end.get('success_not_used_for_admission') is True and end.get('scientific_trials_added')==0,'one original trial-end witness in order')
  require(all(type(end.get(k)) is int and end[k]==row[k] for k in ('control_steps','policy_calls','wait_steps')),'trial-end counts bound to original ledger')
  before,after=observed[28:30]
  require(end['t0_hashes']=={'t0_query_before':before['observation_sha256'],'t0_query_after':after['observation_sha256']} and after['observation_sha256']==row['chunks'][0]['observation_sha256'],'trial-end t0 exact original bytes')
  trial_end_count+=1
 require(pos+1==len(events),'unknown/extra event or missing final return')
 last=events[pos];require(last.get('phase')=='original_runner_returned' and last['trial_identity'] is None and last['ordinal'] is None and last['control_step'] is None and last['wait_steps'] is None,'single original return after all200')
 require(gate[0]['event']=='paired_input_gate_open' and gate[-1]['event']=='paired_input_gate_finished' and not any(x['event']=='paired_input_gate_abort' for x in gate),'complete single gate attempt')
 require(gate[0]['CPU_fixture_enabled'] is False and gate[0]['runtime_context']=='real_environment_engineering','CPU fixture forbidden')
 require(len(gate)<=68032 and all(len(canonical(x))<=16384 for x in gate),'gate storage bound')
 queries=[e for e in gate if e['event']=='query_integrity_passed'];steps=[e for e in gate if e['event']=='policy_step_executed'];pairs=[e for e in gate if e['event']=='t0_pair_gate']
 require(len(pairs)==200 and len(queries)==sum(r['policy_calls'] for r in completed)<=7600 and len(steps)==sum(r['control_steps'] for r in completed)<=60000,'event totals equal original raw ledger')
 # Event order is the original per-trial pair, query, and executed action order.
 expected_gate=[]
 for ordinal,row in enumerate(completed,1):
  expected_gate.append(('t0_pair_gate',ordinal,None,row['trial_id']))
  for step in range(row['control_steps']):
   if step%8==0:expected_gate.append(('query_integrity_passed',ordinal,step,row['trial_id']))
   expected_gate.append(('policy_step_executed',ordinal,step,row['trial_id']))
 require([(e['event'],e.get('ordinal'),e.get('control_step'),e.get('trial_identity',{}).get('trial_id')) for e in gate[1:-1]]==expected_gate,'complete ordered gate chain; no duplicate, foreign or missing witness')
 for e in gate[1:-1]:
  require(e['trial_identity']==expected[e['trial_identity']['trial_id']] and type(e['ordinal']) is int and e.get('scientific_trials_added')==0,'gate full immutable trial identity')
 bytrial={r['trial_id']:r for r in completed};baselines={};matched=set();pair_by_trial={}
 for e in pairs:
  t=e['trial_identity'];cap=capidx[e['capture_event_index']]
  require(cap is trialcaps[t['trial_id']][28] and cap['phase']=='t0_query_before' and cap['trial_identity']==t,'pair is exact before first query')
  instruction=bytes.fromhex(e['instruction_utf8_hex']);instruction.decode('utf-8')
  sig={'components':raw_signatures[cap['event_index']],'instruction_utf8_sha256':digest(instruction)}
  require(sig==e['content'],'pair content and instruction rebuilt from raw bytes')
  pair_by_trial[t['trial_id']]=e
  state=t['state_trial_id']
  if t['method']=='A256_seed17':require(state not in baselines and e['pairing']=='baseline_recorded_before_first_policy','unique A baseline');baselines[state]=sig
  else:require(t['method']=='B256_seed17' and state in baselines and state not in matched and e['pairing']=='exact_before_policy_pair_passed' and sig==baselines[state],'exact preB five-array and instruction pair');matched.add(state)
 require(len(baselines)==len(matched)==100,'all100 pairs')
 qseen={tid:0 for tid in bytrial};sseen={tid:0 for tid in bytrial}
 for e in queries:
  tid=e['trial_identity']['trial_id'];row=bytrial[tid];qi=qseen[tid]
  require(type(e['control_step']) is int and e['control_step']==8*qi and qi<len(row['chunks']),'query original cadence')
  after=capidx[e['capture_event_index']];before=capidx[e['capture_event_index']-1]
  require(before is trialcaps[tid][28+2*qi] and after is trialcaps[tid][29+2*qi] and after['trial_identity']==before['trial_identity']==e['trial_identity'],'query exact original before/after witnesses')
  require(set(before['observation_arrays'])==set(after['observation_arrays'])==set(keys),'every query pre/post raw five components required')
  for key in keys:require(before['observation_arrays'][key]==after['observation_arrays'][key],'query input byte/identity metadata unchanged')
  instruction=bytes.fromhex(e['instruction_utf8_hex']);instruction.decode('utf-8')
  sig_before={'components':raw_signatures[before['event_index']],'instruction_utf8_sha256':digest(instruction)}
  sig_after={'components':raw_signatures[after['event_index']],'instruction_utf8_sha256':digest(instruction)}
  require(sig_before==sig_after==e['content'],'every query content rebuilt from complete raw before and after')
  pair=pair_by_trial[tid];require(e['instruction_utf8_hex']==pair['instruction_utf8_hex'],'original instruction unchanged throughout trial')
  if qi==0:require(e['content']==pair['content'],'t0 query exactly follows paired prequery')
  ident=e['input_identity_before'];require(type(ident) is dict and set(ident)=={'mapping_id','components'} and type(ident['mapping_id']) is int and ident['mapping_id']>0 and set(ident['components'])==set(keys),'query mapping and allfive object identities')
  require(all(type(v) is dict and set(v)=={'id','pointer'} and all(type(n) is int and n>0 for n in v.values()) for v in ident['components'].values()) and ident==e['input_identity_after'],'prepost exact mapping/array pointers')
  c=row['chunks'][qi];require(len(canonical(c['metadata']))<=32768,'original chunk metadata storage bound');require(c['observation_sha256']==after['observation_sha256'],'ledger postpolicy hash')
  for name in ('normalized_actions','environment_actions'):
   meta=e['returned_actions'][name];raw=bytes.fromhex(meta['bytes_hex']);require(digest(raw)==meta['sha256'] and len(raw)==meta['bytes'],'returned actions raw')
   arr=np.frombuffer(raw,dtype=np.dtype(meta['dtype'])).reshape(meta['shape']);require(arr.shape==(8,7) and np.isfinite(arr).all() and np.array_equal(arr,np.asarray(c[name])),'unchanged returned chunk values')
  qseen[tid]+=1
 for e in steps:
  tid=e['trial_identity']['trial_id'];i=sseen[tid];require(type(e['control_step']) is int and e['control_step']==i,'step order')
  a=e['action'];raw=bytes.fromhex(a['bytes_hex']);require(digest(raw)==a['sha256'] and len(raw)==a['bytes'],'step raw bytes');arr=np.frombuffer(raw,dtype=np.dtype(a['dtype'])).reshape(a['shape'])
  require(arr.shape==(7,) and np.isfinite(arr).all() and np.array_equal(arr,np.asarray(bytrial[tid]['executed_actions'][i])),'original action unchanged')
  sseen[tid]+=1
 require(all(qseen[k]==v['policy_calls'] and sseen[k]==v['control_steps'] for k,v in bytrial.items()),'pertrial exact queries/actions')
 end=gate[-1];require(end['exact_paired_t0_checks']==100 and end['all_query_checks']==len(queries) and end['executed_policy_steps_recorded']==len(steps) and end['original_prefix_trials']==200 and end['observer_events']==len(captures),'finish totals exact')
 require(end['event_sha256']==digest(b''.join(canonical(e)+b'\n' for e in events)),'gate footer binds complete raw event bytes')
 return {'trials':200,'paired_t0_checks':100,'queries':len(queries),'control_steps':len(steps),'warmup_steps':2000,'complete_trial_end_witnesses':trial_end_count,'complete_observation_phase_raw_rebuilt':True,'all_query_content_rebuilt_from_before_and_after_raw':True,'raw_blob_refs':read_refs,'old200_HOLD':True,'final_release':False}
