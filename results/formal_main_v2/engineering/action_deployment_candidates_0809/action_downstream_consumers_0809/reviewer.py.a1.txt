def require(condition, message):
    if not condition:
        raise ValueError(message)


def sha(value):
    require(isinstance(value, str) and re.fullmatch("[0-9a-f]{64}", value) is not None, "Invalid SHA256")
    return value


def signature(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, ensure_ascii=False,
        separators=(",", ":"), allow_nan=False).encode()).hexdigest()


def digest(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(8 << 20), b""):
            h.update(block)
    return h.hexdigest()


def relative(value):
    require(isinstance(value, str), "Evidence path must be a string")
    p = PurePosixPath(value)
    require(value and not p.is_absolute() and str(p) == value and ".." not in p.parts
            and ":" not in value and "\\" not in value and "\0" not in value, "Noncanonical evidence path")
    low = value.lower()
    require(not any("sealed" in part.lower() or part.lower() in
                    ("weights", "checkpoints", "test", "final") for part in p.parts),
            "Final/sealed/weight/test payload is forbidden before I/O")
    require(not (set(p.suffixes) & WEIGHT_SUFFIXES), "Weight payload forbidden")
    require(p.suffix == ".py" or not any(stem in low for stem in STATE_STEMS)
            or value in DEV_PANELS, "Only exact registered development state paths allowed")
    require(value != STATE, "Mutable supervisor state is not immutable evidence")
    return value


def remote_relative(value):
    p = PurePosixPath(value)
    require(P in p.parents, "Remote path outside project")
    return relative(p.relative_to(P).as_posix())


def json_bytes(data):
    def duplicate(pairs):
        out = {}
        for key, value in pairs:
            require(key not in out, "Duplicate JSON key")
            out[key] = value
        return out
    return json.loads(data, object_pairs_hook=duplicate,
        parse_constant=lambda x: (_ for _ in ()).throw(ValueError("Nonfinite JSON")))


def ref(value):
    require(type(value) is dict and set(value) == {"path", "sha256"}, "Exact path/SHA reference required")
    relative(value["path"]); sha(value["sha256"])
    return value


class Evidence:
    """All bytes consumed are hashed, with lexical guards before stat/open."""
    def __init__(self, project):
        self.project = Path(project).resolve()
        self.files = {}
        self.forbidden_paths = set()

    def relative(self, value):
        if isinstance(value, Path) and value.is_absolute():
            try:
                value = value.relative_to(self.project).as_posix()
            except ValueError:
                raise ValueError("Evidence outside control project") from None
        return relative(str(value))

    def path(self, value):
        name = self.relative(value)
        p = self.project / name
        require(p not in self.forbidden_paths, "Explicit forbidden evidence path")
        # Check each component, not just the final file; never follow symlinks.
        current = self.project
        for part in PurePosixPath(name).parts:
            current = current / part
            require(not current.is_symlink(), "Symlink evidence forbidden")
        require(p.resolve() == p, "Noncanonical evidence path")
        return p

    def watch(self, value):
        name = self.relative(value)
        p = self.path(name)
        require(p.is_file(), "Missing evidence: " + name)
        actual = digest(p)
        require(name not in self.files or self.files[name] == actual, "Evidence changed while reading: " + name)
        self.files[name] = actual
        return actual

    def read(self, value):
        self.watch(value)
        return json_bytes(self.path(value).read_bytes())

    def read_ref(self, value):
        ref(value)
        require(self.watch(value["path"]) == value["sha256"], "Evidence reference SHA mismatch: " + value["path"])
        return self.read(value["path"])

    def unchanged(self):
        for name, checksum in tuple(self.files.items()):
            require(digest(self.path(name)) == checksum, "Evidence changed during reconstruction: " + name)


def lines(evidence, value):
    evidence.watch(value)
    data = evidence.path(value).read_bytes()
    require(not data or data.endswith(b"\n"), "Truncated JSONL tail")
    out = []
    for line in data.splitlines():
        require(bool(line.strip()), "Blank raw JSONL line")
        row = json_bytes(line)
        require(type(row) is dict, "Raw JSONL object required")
        out.append(row)
    return out


def load(evidence, path, checksum=None):
    name = relative(path)
    actual = evidence.watch(name)
    if checksum is not None:
        require(actual == checksum, "Implementation SHA differs: " + name)
    module_name = "_action_g5_" + signature([str(evidence.project), name, actual])[:24]
    module = sys.modules.get(module_name)
    if module is None:
        module = types.ModuleType(module_name)
        module.__file__ = str(evidence.path(name))
        module.__package__ = ""
        sys.modules[module_name] = module
        try:
            exec(compile(evidence.path(name).read_bytes(), module.__file__, "exec", dont_inherit=True), module.__dict__)
        except BaseException:
            sys.modules.pop(module_name, None)
            raise
    require(Path(module.__file__) == evidence.path(name), "Imported module path differs")
    return module


def clone(function, **changes):
    """Use the original code object; only explicit call dependencies vary."""
    namespace = {**function.__globals__, **changes}
    result = types.FunctionType(function.__code__, namespace, function.__name__,
        function.__defaults__, function.__closure__)
    result.__kwdefaults__ = copy.copy(function.__kwdefaults__)
    result.__annotations__ = dict(function.__annotations__)
    return result


def trajectory_validator(old, analyzer, runtime):
    """Two exact offline amplitude edits, no action conversion or global patch."""
    _, chunk, chunk_proof = runtime.derive_validator(analyzer, kind="analyzer_chunk")
    source = Path(old.__file__).read_bytes()
    require(hashlib.sha256(source).hexdigest() == PINNED[OLD_REVIEW], "Original trajectory source changed")
    tree = ast.parse(source)
    original = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == "trajectory")
    changed = copy.deepcopy(original)
    target = ast.parse("np.all(np.abs(actions) <= 1.00001)", mode="eval").body
    matches = [(node, i) for node in ast.walk(changed)
        if isinstance(node, ast.BoolOp) and isinstance(node.op, ast.And)
        for i, value in enumerate(node.values)
        if ast.dump(value, include_attributes=False) == ast.dump(target, include_attributes=False)]
    require(len(matches) == 1, "Trajectory amplitude clause no longer exact")
    node, index = matches[0]
    removed = node.values.pop(index)
    node.values.insert(index, removed)
    require(ast.dump(changed, include_attributes=False) == ast.dump(original, include_attributes=False),
            "Unexpected trajectory derivation")
    node.values.pop(index)
    require(len(node.values) >= 2, "Unexpected trajectory conjunction")
    original_loader = old.load
    def dependencies(path):
        return types.SimpleNamespace(validate_chunk=chunk) if path == ANALYZER else original_loader(path)
    namespace = {**old.__dict__, "load": dependencies}
    exec(compile(ast.Module(body=[changed], type_ignores=[]), str(old.__file__),
        "exec", dont_inherit=True), namespace)
    result = namespace["trajectory"]
    return result, dict(chunk=chunk_proof, trajectory={
        "source_sha256": PINNED[OLD_REVIEW], "original_ast_sha256": signature(ast.dump(original, include_attributes=False)),
        "derived_ast_sha256": signature(ast.dump(changed, include_attributes=False)),
        "removed_ast_sha256": signature(ast.dump(removed, include_attributes=False)),
        "changes": "one executed-action amplitude conjunct", "action_values_modified": False,
        "shape_finite_equality_cadence_preserved": True})


def semantic_identity(contract):
    """Stage is the only stage-dependent field; target/model/implementation stay bound."""
    scope = contract["runtime_contract"]
    return signature({"execution_version": VERSION, "files": contract["files"],
        "runtime_contract_without_stage": {k: v for k, v in scope.items() if k != "stage"},
        "GPU_acceptance": contract["GPU_acceptance"], "source_inventory": contract["source_inventory"]})


def _same_fields(actual, expected, message):
    require(all(actual.get(k) == v for k, v in expected.items()), message)


def validate_envelope(evidence, job, invocation, expected_contract_ref):
    """Validate the real outer identity after execution, without pre-run rehashes."""
    require(invocation.get("stage") in ALLOWED_STAGES and job.get("mode") == invocation["stage"],
            "Final or unknown stage forbidden")
    ref(expected_contract_ref)
    require(invocation.get("version") == VERSION and job.get("execution_version") == VERSION,
            "Exact action execution version required")
    require(invocation.get("action_execution_contract") == expected_contract_ref, "Action contract mismatch")
    contract = evidence.read_ref(expected_contract_ref)
    files = contract.get("files", {})
    require(CONTRACT in files and RUNTIME in files and REVIEWER in files and CONSUMER in files,
            "Complete versioned implementation closure required")
    module = load(evidence, CONTRACT, files[CONTRACT])
    original = evidence.read_ref(invocation["scientific_plan"])
    readiness = evidence.read_ref(invocation["runtime_readiness"])
    expected = dict(schema=SCHEMA, kind="formal_action_runtime_readiness", status="passed", reviewed_by="root",
        execution_version=VERSION, scientific_settings_changed=False, final_evaluation_enabled=False,
        raw_consumer_ready=True)
    _same_fields(readiness, expected, "Missing executed-version root readiness")
    effective = evidence.read_ref(readiness["effective_plan"])
    module.validate_effective(original, effective, scientific_plan_ref=invocation["scientific_plan"],
        execution_contract_ref=expected_contract_ref, execution_contract=contract)
    base = effective["invocation"]
    require(set(invocation) == set(base) | {"registered_utc", "plan_sha256", "runtime_readiness"},
            "Prepared invocation field set changed")
    require(all(invocation[k] == v for k, v in base.items()) and
        invocation["plan_sha256"] == effective["plan_sha256"] == readiness.get("plan_sha256"),
        "Actual prepared invocation differs from effective plan")
    require(isinstance(invocation["registered_utc"], str) and invocation["registered_utc"], "Missing registration time")
    require(readiness.get("runtime_sources") == files, "Readiness covers another runtime")
    roles = []
    for role in ("root_review", "independent_review"):
        item = readiness.get(role)
        report = evidence.read_ref(item)
        _same_fields(report, dict(status="passed", execution_version=VERSION,
            runtime_sources=files, production_chain_complete=True), "Incomplete runtime chain readiness")
        roles.append(item)
    require(roles[0] != roles[1], "Root and independent readiness must be distinct")
    for name, checksum in module.execution_dependencies(contract).items():
        require(evidence.watch(name) == checksum, "Action implementation changed")
    for name, checksum in contract["GPU_acceptance"].items():
        require(evidence.watch(name) == checksum, "Prior GPU gate identity changed")
    require(evidence.watch(contract["source_inventory"]["path"]) == contract["source_inventory"]["sha256"],
            "Frozen inventory changed")
    require(evidence.watch(REVIEWER) == files[REVIEWER] and Path(__file__).resolve() == evidence.path(REVIEWER),
            "Consumer must run the exact contract reviewer at its published path")
    for field in ("wrapper", "wrapper_sha256", "entrypoint", "action_execution_contract", "scientific_plan",
                  "runtime_readiness", "plan_sha256"):
        require(job.get(field) == invocation[field], "Registered outer field differs: " + field)
    # Only these two completed-output bytes may grow during a registered resume.
    mutable = {remote_relative(invocation["run_root"]) + "/" + n for n in
        ("trials.jsonl", "closed_loop_report.json")} if invocation["resume"] and invocation["stage"] == "development" else set()
    for path, checksum in invocation["watched_files"].items():
        name = remote_relative(path)
        require(name not in mutable, "Mutable resume outputs cannot be post-run immutable watches")
        require(evidence.watch(name) == checksum, "Post-run immutable watch differs: " + name)
    for path, checksum in invocation["root_evidence_sha256"].items():
        relative(path)
        if path not in mutable:
            require(evidence.watch(path) == checksum, "Original root evidence changed: " + path)
    runtime = load(evidence, RUNTIME, files[RUNTIME])
    # validate_contract verifies only immutable metadata/source bytes; it loads no policy.
    runtime.validate_contract(contract["runtime_contract"], project_root=evidence.project,
        source_root=Path(invocation["source_root"]))
    # Record all metadata read by runtime's validator, not just the runtime file.
    scope = contract["runtime_contract"]
    for key in ("source_inventory", "registration"):
        if key in scope:
            evidence.read_ref(scope[key])
    for path, checksum in scope.get("dependencies", {}).items():
        require(evidence.watch(path) == checksum, "Runtime closure changed")
    for item in scope.get("engineering_acceptance", {}).values():
        evidence.read_ref(item)
    return original, effective, contract, runtime


def verify_pull(evidence, job, pull_ref):
    """Rebuild transport evidence directly; no synthetic old passed report."""
    require(job.get("mode") in ALLOWED_STAGES, "Forbidden stage before pull I/O")
    pull = evidence.read_ref(pull_ref)
    require(pull_ref["path"].startswith("results/formal_main_v2/pulls/"), "Formal verified pull required")
    _same_fields(pull, dict(schema=SCHEMA, kind="nonweight_pull", status="verified_nonweight_snapshot",
        final_evidence=True, weights_copied=False, sealed_initial_states_copied=False), "Unverified nonweight pull")
    selected = [r for r in pull.get("selected_jobs", []) if r.get("job_id") == job["job_id"]]
    require(len(selected) == 1 and selected[0].get("include_run_root") is True
        and selected[0].get("run_root") == job["run_root"]
        and selected[0].get("source_sha256") == job["source_sha256"], "Pull selected-job identity differs")
    folder = str(PurePosixPath(pull_ref["path"]).parent)
    before = evidence.read(folder + "/inventory_before.json")
    after = evidence.read(folder + "/inventory_after.json")
    require(before == after and evidence.watch(folder + "/inventory_before.json") == pull.get("inventory_sha256"),
            "Pull changed during transfer")
    for when in ("accounting_before", "accounting_after"):
        actual = pull.get(when, {}).get("jobs", {}).get(job["job_id"], {})
        require(all(actual.get(k) == job.get(k) for k in ("status", "exit_code", "elapsed_seconds")),
                "Pull accounting differs")
    require(type(job.get("elapsed_seconds")) is int and job["elapsed_seconds"] >= 0 and job.get("gpus") == 1,
            "One complete GPU allocation required")
    for row in pull.get("files", []) + before.get("files", []):
        relative(row["relative"])
    rows = {r["relative"]: r for r in pull.get("files", [])}
    originals = {r["relative"]: r for r in before.get("files", [])}
    require(len(rows) == len(pull.get("files", [])) and len(originals) == len(before.get("files", [])),
            "Duplicate transport row")
    output = remote_relative(job["run_root"])
    required = {n for n, r in originals.items() if n.startswith(output + "/") and job["job_id"] in r.get("job_ids", [])}
    required |= {f"results/formal_main_v2/job_receipts/{job['job_id']}.json",
        *[f"results/slurm/{job['name']}_{job['job_id']}.{ext}" for ext in ("out", "err")]}
    require(any(n.startswith(output + "/adapter_invocations/") for n in required), "Missing original adapter raw evidence")
    require(any(n.startswith(output + "/action_execution/" + job["job_id"] + "/") for n in required),
            "Missing new actual execution evidence")
    for row in before.get("missing", []):
        require(str(row.get("job_id")) != job["job_id"], "Missing downstream evidence")
    for row in before.get("excluded", []):
        if str(row.get("job_id")) == job["job_id"]:
            require(row.get("path") == str(P / output / ".evaluation.lock") and
                row.get("reason") == "unregistered_nonweight_extension" and job["mode"] == "development",
                "Unexpected excluded downstream evidence")
    checked = {}
    for name in sorted(required):
        relative(name)
        row, old = rows.get(name, {}), originals.get(name, {})
        actual = evidence.watch(name)
        size = evidence.path(name).stat().st_size
        require(row.get("sha256") == old.get("sha256") == actual
            and row.get("path") == old.get("path") == str(P / name)
            and row.get("bytes") == old.get("bytes") == size
            and job["job_id"] in row.get("job_ids", []) and job["job_id"] in old.get("job_ids", [])
            and row.get("action") in ("same_sha_skipped", "downloaded_verified"), "Transport/file identity differs: " + name)
        checked[name] = actual
    parent = evidence.read(f"results/formal_main_v2/job_receipts/{job['job_id']}.json")
    original_artifacts = parent.get("adapter_artifacts", [])
    require(len(original_artifacts) == 2, "Original adapter receipt/journal pair required")
    seen = set()
    for artifact in original_artifacts:
        name = remote_relative(artifact.get("path", "/invalid"))
        filename = PurePosixPath(name).name
        require(re.fullmatch(re.escape(job["job_id"]) + r"_[0-9a-f]{32}\.(json|jsonl)", filename) is not None
            and name == output + "/adapter_invocations/" + filename
            and artifact.get("source") == str(P / "results/formal_main_v2/adapter_invocations" / filename)
            and name in checked and artifact.get("sha256") == checked[name]
            and artifact.get("bytes") == evidence.path(name).stat().st_size, "Foreign or corrupted original adapter archive")
        require(name not in seen, "Duplicate original adapter archive")
        seen.add(name)
    return checked

def validate_transition_after(evidence, invocation, contract, adapter):
    if not invocation["resume"]:
        require(contract["transition"] is None, "Fresh run cannot claim an old ledger")
        return None
    require(invocation["stage"] == "development", "Only explicit development continuation supported")
    transition = evidence.read_ref(contract["transition"])
    prior_path = invocation["prior_review"]
    prior = evidence.read_ref({"path": prior_path, "sha256": invocation["root_evidence_sha256"][prior_path]})
    previous_version = prior.get("execution_version", "formal-downstream-root-v1")
    _same_fields(transition, dict(schema=SCHEMA, kind="formal_action_runtime_transition", status="root_reviewed",
        from_version=previous_version, to_version=VERSION, stage_key=invocation["stage_key"],
        run_root=invocation["run_root"], execution_manifest_sha256=invocation["execution_manifest_sha256"],
        prior_job_ids=invocation["prior_job_ids"], prior_review=prior_path, terminal_trials_repeated=False,
        prior_outcomes_replaced=False, new_trial_ids_only=True), "Transition does not describe actual prior execution")
    require(previous_version in ("formal-downstream-root-v1", VERSION), "Unknown prior execution")
    if previous_version == VERSION:
        require(prior.get("action_semantics_sha256") == semantic_identity(contract), "Same-version resume changed action semantics")
    files = transition.get("files", {})
    output = remote_relative(invocation["run_root"])
    mutable = {output + "/trials.jsonl", output + "/closed_loop_report.json"}
    require(mutable | {output + "/config.json", output + "/panel.json"} <= set(files),
            "Transition omits prior ledger/config/panel")
    previous_pull = evidence.read(prior["pull_receipt"])
    rows = {r["relative"]: r for r in previous_pull["files"]}
    for name, checksum in files.items():
        relative(name); sha(checksum)
        require(prior["files"].get(name) == rows.get(name, {}).get("sha256") == checksum,
                "Transition is not bound to historical pull/root bytes")
        if name not in mutable:
            require(evidence.watch(name) == checksum, "Immutable transition input changed")
    ledger = output + "/trials.jsonl"
    prefix = adapter.get("before_ledger_raw", {})
    require(prefix.get("bytes") == rows[ledger]["bytes"] and prefix.get("sha256") == files[ledger],
            "Actual before-ledger differs from root transition prefix")
    n = prefix["bytes"]
    require(type(n) is int and n > 0, "Resume prefix must be positive")
    h = hashlib.sha256()
    with evidence.path(ledger).open("rb") as stream:
        while n:
            block = stream.read(min(n, 8 << 20))
            require(block, "Truncated original ledger prefix")
            h.update(block); n -= len(block)
    require(h.hexdigest() == prefix["sha256"], "Historical raw prefix overwritten")
    independent = evidence.read_ref(transition["independent_review"])
    _same_fields(independent, dict(kind="formal_action_runtime_transition_review", status="passed",
        review_role="independent", stage_key=invocation["stage_key"], files=files,
        prior_review=prior_path, from_version=previous_version, to_version=VERSION),
        "Transition lacks exact independent prefix review")
    return dict(reference=contract["transition"], from_version=previous_version,
        preserved_prefix_bytes=prefix["bytes"], preserved_prefix_sha256=prefix["sha256"],
        historical_terminal_outcomes_replaced=False)


def checked_asset_identities(evidence, runtime, contract, invocation):
    """Original metadata-only validators read P/S; bind every byte to its R copy.

    No identity path rewriting: the expected identity remains the real P/S
    identity. The explicit mirror map is audit evidence, not an I/O alias.
    """
    if invocation["suite"] not in ("libero_object", "libero_goal"):
        require(invocation.get("downstream_asset_compatibility") is None, "Unexpected asset contract")
        return None, []
    expected = runtime.expected_asset_identities(project_root=Path(P), source_root=Path(invocation["source_root"]),
        contract=contract["runtime_contract"], asset_contract=invocation["downstream_asset_compatibility"])
    require(type(expected) is dict and "metadata" in expected, "Missing independently derived asset identities")
    source = PurePosixPath(invocation["source_root"])
    mirrors = {}
    for identity in expected.values():
        for path, checksum in identity["watched_files"].items():
            p = PurePosixPath(path)
            if source in p.parents:
                name = relative(p.relative_to(source).as_posix())
            else:
                name = remote_relative(path)
            require(evidence.watch(name) == checksum, "P/S metadata differs from retained R evidence")
            key = (path, name)
            require(key not in mirrors or mirrors[key] == checksum, "Conflicting metadata mirror")
            mirrors[key] = checksum
    return expected, [dict(server_path=a, control_relative=b, sha256=v) for (a, b), v in sorted(mirrors.items())]

    _same_fields(parent, dict(schema=SCHEMA, kind="formal_downstream_job_receipt", execution_version=VERSION,
        job_id=job["job_id"], invocation_sha256=job["invocation_sha256"], run_root=job["run_root"],
        stage=job["mode"], stage_key=job["stage_key"], source_sha256=invocation["source_sha256"],
        adapter_sha256=invocation["adapter_sha256"], wrapper_sha256=invocation["wrapper_sha256"],
        action_execution_contract=invocation["action_execution_contract"], scientific_plan=invocation["scientific_plan"],
        runtime_readiness=invocation["runtime_readiness"], stage_completed_claimed=False,
        original_scientific_wrapper_executed=False, status="returned", exit_code=0),
        "Actual parent identity/outcome differs")
    require(job["status"] == "COMPLETED" and job["exit_code"] == "0:0",
            "Failed/partial action execution needs a separate failure audit; never passed here")
    artifacts = parent.get("action_execution_artifacts", [])
    expected_paths = {output + "/action_execution/" + job["job_id"] + "/" + n
        for n in ("identity.json", "events.jsonl", "completion.json")}
    require(len(artifacts) == 3 and {remote_relative(a["path"]) for a in artifacts} == expected_paths,
            "Exact three action archive files required")
    for item in artifacts:
        name = remote_relative(item["path"])
        expected_source = str(P / "results/formal_main_v2/action_execution_invocations" / job["job_id"] / PurePosixPath(name).name)
        require(item.get("source") == expected_source and files.get(name) == item.get("sha256") == evidence.watch(name)
            and item.get("bytes") == evidence.path(name).stat().st_size, "Action archive copy identity differs")
    prefix = output + "/action_execution/" + job["job_id"] + "/"
    identity = evidence.read(prefix + "identity.json")
    complete = evidence.read(prefix + "completion.json")
    events = lines(evidence, prefix + "events.jsonl")
    shared = dict(schema=SCHEMA, version=VERSION, job_id=job["job_id"], invocation_sha256=job["invocation_sha256"],
        **{k: invocation[k] for k in ("stage", "stage_key", "suite", "source_root", "source_sha256",
                                      "execution_manifest_sha256", "action_execution_contract", "scientific_plan", "runtime_readiness")})
    for value, kind in ((identity, "formal_action_execution_identity"), (complete, "formal_action_execution_completion")):
        _same_fields(value, {**shared, "kind": kind}, "Mixed actual execution identity")
    require(events and events[0].get("event") == "action_execution_start"
            and events[-1].get("event") == "action_execution_end", "Missing outer execution endpoints")
    times = []
    for index, event in enumerate(events):
        _same_fields(event, {**shared, "kind": "formal_action_execution_event", "seq": index}, "Mixed/noncontiguous action journal")
        require(isinstance(event.get("event"), str) and isinstance(event.get("utc"), str), "Malformed event")
        parsed = datetime.fromisoformat(event["utc"])
        require(parsed.tzinfo is not None, "Journal timestamp must have timezone")
        times.append(parsed)
    require(times == sorted(times), "Nonmonotone actual runtime events")
    require(complete.get("events_sha256") == evidence.watch(prefix + "events.jsonl")
            and complete.get("events_path") == str(P / "results/formal_main_v2/action_execution_invocations" / job["job_id"] / "events.jsonl")
            and complete.get("identity_sha256") == evidence.watch(prefix + "identity.json"), "Actual completion hash closure differs")
    ending = dict(status="returned", exit_code=0, runtime_restored=True, immutable_inputs_unchanged=True,
        errors=[], scientific_completion_not_inferred=True, runner_status=adapter.get("runner_status"))
    _same_fields(complete, ending, "Incomplete or failed actual runtime")
    _same_fields(events[-1], ending, "Actual journal end differs")
    _same_fields(identity, dict(actual_entrypoint=invocation["entrypoint"], actual_wrapper=invocation["wrapper"],
        runtime_sources=contract["files"], runtime_contract=contract["runtime_contract"], expected_base_load_calls=1,
        original_evaluation_adapter=str(P / "scripts/ops/formal_evaluation_adapter.py"),
        original_evaluation_adapter_sha256=PINNED["scripts/ops/formal_evaluation_adapter.py"],
        original_scientific_plan_sha256=invocation["scientific_plan_sha256"], original_scientific_wrapper_executed=False,
        source_inventory_sha256=invocation["source_inventory_sha256"], scientific_settings_changed=False,
        scientific_completion_not_inferred=True), "Actual child identity differs")
    starts = [i for i, x in enumerate(events) if x["event"] == "legacy_adapter_start"]
    ends = [i for i, x in enumerate(events) if x["event"] == "legacy_adapter_return"]
    require(len(starts) == len(ends) == 1 and starts[0] < ends[0]
        and not any(x["event"] == "legacy_adapter_exception" for x in events), "One returned original scientific adapter required")
    start, end = events[starts[0]], events[ends[0]]
    require(start.get("identity_sha256") == evidence.watch(prefix + "identity.json")
        and start.get("original_run_code_sha256") == identity.get("original_run_code_sha256")
        and end.get("runner_status") == adapter.get("runner_status"), "Outer/inner execution binding differs")
    sha(identity.get("original_run_code_sha256"))
    inner_events = lines(evidence, output + "/adapter_invocations/" + adapter["attempt_id"] + ".jsonl")
    inner_times = [datetime.fromisoformat(x["utc"]) for x in inner_events]
    require(inner_times and inner_times == sorted(inner_times)
        and times[starts[0]] <= inner_times[0] <= inner_times[-1] <= times[ends[0]], "Inner adapter is not in the same actual call")
    # An exact outer whitelist plus the runtime verifier rejects silently added scopes.
    outer_names = {"action_execution_start", "action_execution_end", "legacy_adapter_start", "legacy_adapter_return"}
    prefixes = ("deployment_", "action_execution_candidate_", "asset_metadata_compat_", "goal_component_")
    require(all(x["event"] in outer_names or x["event"].startswith(prefixes) for x in events), "Unknown execution event")
    require([x["event"] for x in events if x["event"] in outer_names] ==
        ["action_execution_start", "legacy_adapter_start", "legacy_adapter_return", "action_execution_end"],
        "Duplicated/misordered outer scope")
    assets, mirrors = checked_asset_identities(evidence, runtime, contract, invocation)
    structural = runtime.verify_runtime_events(events, contract["runtime_contract"], expected_asset_identities=assets)
    require(structural.get("all_source_and_binding_contexts_restored") is True and
            structural.get("original_base_loads") == 1 and structural.get("synthetic_input") is False,
            "Runtime scope did not restore or contains synthetic input")
    # Base loading must happen inside the actual legacy call, not an unrelated preflight.
    loading = [i for i, x in enumerate(events) if x["event"] in ("deployment_base_load_start", "deployment_base_load_complete")]
    require(len(loading) == 2 and starts[0] < loading[0] < loading[1] < ends[0], "Base belongs to a different call")
    require(type(complete.get("wall_seconds")) in (float, int) and 0 <= complete["wall_seconds"] <= parent["wall_seconds"] + 1
        <= job["elapsed_seconds"] + 2, "Nested runtime wall exceeds allocated cost")
    return dict(execution_version=VERSION, action_semantics_sha256=semantic_identity(contract),
        identity_ref={"path": prefix + "identity.json", "sha256": files[prefix + "identity.json"]},
        events_ref={"path": prefix + "events.jsonl", "sha256": files[prefix + "events.jsonl"]},
        completion_ref={"path": prefix + "completion.json", "sha256": files[prefix + "completion.json"]},
        structural=structural, original_adapter_attempt=adapter["attempt_id"], metadata_mirror_checks=mirrors,
        actual_source_unchanged=True, scientific_completion_not_inferred=True)


def audit_freeze(evidence, helper, directory, registration, consume):
    # Original 88 identities/72 selectors+16 references and four development
    # receipts remain mandatory. This only consumes a freeze; it never releases.
    result = helper.release_global(evidence, registration, directory, release=False)
    matrix = evidence.read(directory + "/global_freeze.json")
    for suite in matrix["suites"]:
        roots = evidence.read(directory + "/development_root_reviews/" + suite + ".json")
        for item in roots["reviews"]:
            report = consume(evidence, item["reference"], expected_stage="development", suite=suite)
            require(report.get("execution_version") == VERSION,
                "Old action semantics require an explicitly registered equivalence map; not yet supported")
    return result


def science_context(evidence, old, scheduler, helper, job, invocation, registered_jobs, original, contract, consume):
    """Unchanged scientific validation code, with only explicit new outer gates."""
    # The supplied original is an immutable genuine scientific plan. It is not
    # passed off as proof that this new invocation ran the old wrapper.
    original_inv = original["invocation"]
    def shape(ev, actual_job, actual_inv, proof):
        require(actual_inv is invocation and actual_job == job, "Unexpected shape validation target")
        require(actual_inv.get("source_provenance") == proof == original_inv.get("source_provenance"),
                "Training source proof changed")
        if actual_inv["suite"] in scheduler.ASSET_SUITES:
            expected = scheduler.asset_contract(ev, actual_inv["suite"], proof)
            require(actual_inv.get("downstream_asset_compatibility") == expected, "Suite asset contract changed")
        else:
            require("downstream_asset_compatibility" not in actual_inv, "Foreign asset compatibility")
        argv = actual_inv.get("argv", [])
        require(argv == original_inv["argv"] and argv and all(isinstance(x, str) and "\0" not in x for x in argv),
                "Scientific argv changed")
        flags = [x for x in argv[1:] if x.startswith("--")]
        require(len(flags) == len(set(flags)) and "--prepare-only" not in flags
            and "--sealed-final" not in flags and "--release" not in flags, "Unregistered/final argv")
    def root_output(ev, path, jobs, *, expected_stage):
        reference = {"path": path, "sha256": ev.watch(path)}
        saved = ev.read(path)
        rebuilt = consume(ev, reference, expected_stage=expected_stage,
            suite=invocation["suite"], run_root=saved["run_root"])
        require(rebuilt.get("execution_version") == VERSION and
            rebuilt.get("action_semantics_sha256") == semantic_identity(contract),
            "O2A collection and inference action semantics differ")
        return rebuilt
    helper_proxy = types.SimpleNamespace(**helper.__dict__)
    helper_proxy.Evidence = Evidence
    def global_gate(ev, registration, directory, *, release=False, **kwargs):
        require(release is False and not kwargs, "Final release is forbidden")
        return audit_freeze(ev, helper, directory, registration, consume)
    helper_proxy.release_global = global_gate
    scheduler_proxy = types.SimpleNamespace(**scheduler.__dict__)
    scheduler_proxy.validate_invocation_shape = shape
    scheduler_proxy.validate_inputs = clone(scheduler.validate_inputs, require_root_review=root_output)
    state = {"registered_jobs": registered_jobs}
    value = old.validate_invocation(evidence, scheduler_proxy, helper_proxy, state, job, invocation)
    require(value["source_provenance"] == invocation["source_provenance"], "Actual trained-source provenance differs")
    return value, scheduler_proxy


def _merge(evidence, mapping):
    require(isinstance(mapping, dict), "Evidence map required")
    for name, checksum in mapping.items():
        require(evidence.watch(name) == checksum, "Evidence dependency changed")


def review(project, *, job, invocation_ref, pull_ref, registered_jobs, expected_contract_ref):
    """Rebuild a successful terminal action-v1 development/collect/timing run.

    Root may publish the returned result only after its own review. A successful
    engineering prefix remains scientific_complete=False. Failures raise and
    retain their original allocation/bytes; this function never retries.
    """
    require(job.get("mode") in ALLOWED_STAGES, "Final evaluation is hard forbidden before I/O")
    evidence = Evidence(project)
    ref(invocation_ref); ref(pull_ref); ref(expected_contract_ref)
    require(job.get("invocation") == invocation_ref["path"] and job.get("invocation_sha256") == invocation_ref["sha256"],
            "Actual invocation reference differs")
    require(type(registered_jobs) is list and registered_jobs, "Immutable registered-job snapshot required")
    ids = [str(j.get("job_id")) for j in registered_jobs]
    require(len(ids) == len(set(ids)) and ids.count(str(job["job_id"])) == 1
        and registered_jobs[ids.index(str(job["job_id"]))] == job, "Duplicate or inconsistent snapshot job")
    for path, checksum in PINNED.items():
        require(evidence.watch(path) == checksum, "Original audit chain changed")
    root_src = str(evidence.project / "src")
    if root_src not in sys.path:
        sys.path.insert(0, root_src)
    old = load(evidence, OLD_REVIEW, PINNED[OLD_REVIEW])
    scheduler = load(evidence, OLD_SCHEDULER, PINNED[OLD_SCHEDULER])
    helper = load(evidence, OLD_HELPER, PINNED[OLD_HELPER])
    require(all(j.get("experiment") == SCHEMA and j.get("status") in scheduler.TERMINAL for j in registered_jobs),
            "Only immutable terminal formal jobs may enter result proof")
    require(job.get("status") == "COMPLETED" and job.get("exit_code") == "0:0",
            "Failed or nonterminal execution cannot receive a successful new raw review")
    invocation = evidence.read_ref(invocation_ref)
    original, effective, contract, runtime = validate_envelope(evidence, job, invocation, expected_contract_ref)
    consumer = load(evidence, CONSUMER, contract["files"][CONSUMER])
    files = verify_pull(evidence, job, pull_ref)
    output = remote_relative(job["run_root"])
    value, scheduler_proxy = science_context(evidence, old, scheduler, helper, job, invocation,
        registered_jobs, original, contract, consumer.review_downstream_output)
    config = evidence.read(output + "/config.json")
    value["checkpoint"] = old.runtime_inputs(evidence, scheduler_proxy, value, config)
    # Genuine unchanged inner-adapter journal/receipt from this actual new call.
    adapter = old.adapters(evidence, scheduler_proxy, value, job, config, files)
    require(adapter.get("status") == "returned", "Inner scientific adapter failed")
    runtime_proof = actual_runtime(evidence, job, invocation, files, contract, runtime, adapter)
    transition = validate_transition_after(evidence, invocation, contract, adapter)
    analyzer = load(evidence, ANALYZER, PINNED[ANALYZER])
    trajectory, derivation = trajectory_validator(old, analyzer, runtime)
    # Every JSONL is parsed strictly before unchanged scientific iterators use it.
    for name in files:
        if name.endswith(".jsonl"):
            lines(evidence, name)
    if job["mode"] == "development":
        result = clone(old.rollout, trajectory=trajectory)(evidence, scheduler_proxy, value, job, config, adapter)
    else:
        old.profile_config(evidence, scheduler_proxy, value, job, config)
        primitive = clone(old.collect, trajectory=trajectory) if job["mode"] == "collect" else old.profile
        result = primitive(evidence, scheduler_proxy, value, job, config, adapter)
        duration = evidence.read(output + "/summary.json").get("execution_wall_seconds_including_model_loading")
        old.number(duration, "profile/collection execution wall")
        require(duration <= job["elapsed_seconds"] + 1, "Profile wall exceeds allocation")
    cost = old.allocation_cost(evidence, scheduler_proxy, registered_jobs, job, invocation, result["raw_invocation_job_ids"])
    candidate = dict(schema=SCHEMA, kind="root_downstream_review", status="passed", reviewed_by="root",
        execution_version=VERSION, version=REVIEW_VERSION, job_id=job["job_id"], stage=job["mode"], suite=job["suite"],
        stage_key=job["stage_key"], source_sha256=job["source_sha256"], adapter_sha256=job["adapter_sha256"],
        invocation=invocation_ref["path"], invocation_sha256=invocation_ref["sha256"], invocation_ref=invocation_ref,
        run_root=job["run_root"], execution_manifest_sha256=job["execution_manifest_sha256"],
        terminal_status=job["status"], exit_code=job["exit_code"], pull_receipt=pull_ref["path"], pull_ref=pull_ref,
        files=files, registered_jobs_at_review=copy.deepcopy(registered_jobs), source_provenance=value["source_provenance"],
        action_execution_contract=expected_contract_ref, action_semantics_sha256=semantic_identity(contract),
        actual_runtime_review=runtime_proof, historical_transition=transition, offline_validator_derivation=derivation,
        cost=cost, **result, reviewed_utc=datetime.now(timezone.utc).isoformat(),
        final_state_values_read=False, final_release=False, weights_read=False,
        original_scientific_wrapper_executed=False, old_report_or_invocation_fabricated=False,
        reviewer_source_sha256=evidence.watch(REVIEWER))
    for name, checksum in evidence.files.items():
        if name.startswith(output + "/"):
            require(files.get(name) == checksum, "Scientific artifact is absent from the verified pull: " + name)
    candidate["audit_source_sha256"] = {**PINNED, **contract["files"]}
    evidence.unchanged()
    candidate["evidence_sha256"] = dict(sorted(evidence.files.items()))
    return candidate
