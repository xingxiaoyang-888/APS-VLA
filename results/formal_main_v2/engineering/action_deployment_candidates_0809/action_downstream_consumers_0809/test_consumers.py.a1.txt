def module(path, name):
    spec = importlib.util.spec_from_file_location(name, path)
    result = importlib.util.module_from_spec(spec);sys.modules[name] = result
    spec.loader.exec_module(result);return result

def save(root, path, value):
    p = root / path;p.parent.mkdir(parents=True, exist_ok=True)
    data = value if isinstance(value, bytes) else json.dumps(value, sort_keys=True, allow_nan=False).encode()
    p.write_bytes(data)
    return {"path": path, "sha256": hashlib.sha256(data).hexdigest()}

    originals = (old.trajectory, analyzer.validate_chunk, old.trajectory.__code__, analyzer.validate_chunk.__code__)
    current,proof = g.trajectory_validator(old,analyzer,runtime)

def test_raw_prefix_transition_rechecks_old_bytes_without_reopening_old_report(tmp_path):
    output="results/formal_main_v2/libero_10/downstream/development/x/attempt01"
    olddata=b'{"event":"old"}\n';newdata=olddata+b'{"event":"new"}\n'
    checks={output+"/trials.jsonl":hashlib.sha256(olddata).hexdigest(),output+"/closed_loop_report.json":"d"*64}
    for name in ("config.json","panel.json"):
        row=save(tmp_path,output+"/"+name,{"synthetic":True});checks[row["path"]]=row["sha256"]
    save(tmp_path,output+"/trials.jsonl",newdata);save(tmp_path,output+"/closed_loop_report.json",{"new":True})
    pull=save(tmp_path,"results/formal_main_v2/pulls/old/receipt.json",
        {"files":[dict(relative=p,sha256=s,bytes=len(olddata) if p.endswith("jsonl") else 1) for p,s in checks.items()]})
    prior=save(tmp_path,"results/formal_main_v2/downstream_reviews/1.json",dict(files=checks,pull_receipt=pull["path"]))
    inv=dict(resume=True,stage="development",run_root=str(g.P/output),stage_key="key",
        execution_manifest_sha256="a"*64,prior_job_ids=["1"],prior_review=prior["path"],root_evidence_sha256={prior["path"]:prior["sha256"]})
    identity=dict(schema=g.SCHEMA,kind="formal_action_runtime_transition",status="root_reviewed",
        from_version="formal-downstream-root-v1",to_version=g.VERSION,stage_key="key",run_root=inv["run_root"],
        execution_manifest_sha256="a"*64,prior_job_ids=["1"],prior_review=prior["path"],
        terminal_trials_repeated=False,prior_outcomes_replaced=False,new_trial_ids_only=True,files=checks)
    independent=save(tmp_path,"results/formal_main_v2/preflight/ind.json",dict(kind="formal_action_runtime_transition_review",
        status="passed",review_role="independent",stage_key="key",files=checks,prior_review=prior["path"],
        from_version="formal-downstream-root-v1",to_version=g.VERSION))
    identity["independent_review"]=independent
