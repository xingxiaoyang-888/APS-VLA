class AdditionalControlTests(ControllerTests):
    # Inherited tests are intentionally not duplicated by pytest collection below.
    def control(self,**data):
        state=c.read_json(self.state);state["autonomous_training_v1"]=data;c.atomic_json(self.state,state)
    def ref(self):
        p=self.root/"root_hold.json";c.atomic_json(p,{"kind":"synthetic_root_request"})
        return dict(path="root_hold.json",sha256=c.file_sha(p))
    def test_hold_allows_terminal_review_but_never_submits(self):
        self.write([job(1,PAIRS[0],status="COMPLETED")]);self.b.complete=True
        self.control(root_submission_hold=dict(enabled=True,reason="registered engineering slot",root_record=self.ref(),requested_utc="2026-10-19T00:00:00Z"))
        r=self.tick();self.assertEqual(r["status"],"root_submission_hold")
        self.assertEqual(c.read_json(self.state)["formal_main_progress"]["trained_arms"],2);self.assertFalse(self.s.calls)
    def test_reservation_preserves_one_slot(self):
        self.write([job(i,PAIRS[i+1]) for i in range(7)])
        self.control(reserved_GPU_slots=1,reservation_root_record=self.ref())
        self.assertEqual(self.tick()["status"],"capacity_full");self.assertFalse(self.s.calls)
    def test_hold_requires_exact_provenance(self):
        self.control(root_submission_hold=dict(enabled=True))
        with self.assertRaises(ValueError):self.tick()
        self.assertFalse(self.s.calls)
    def test_clear_stale_active_tracking(self):
        self.write([job(1,PAIRS[0],status="COMPLETED")]);self.b.complete=True
        s=c.read_json(self.state);s["formal_main_progress"]={"parallel_execution":{"pair_tracking":{c.pair_key(PAIRS[0]):{"active_job_id":"101"}}}};c.atomic_json(self.state,s)
        self.tick()
        row=c.read_json(self.state)["formal_main_progress"]["parallel_execution"]["pair_tracking"][c.pair_key(PAIRS[0])]
        self.assertNotIn("active_job_id",row)
    def test_blocked_history_not_reaudited_until_root_clears(self):
        self.b.bad=True;self.tick();n=len(self.b.calls);self.tick();self.assertEqual(n,len(self.b.calls))

class BridgeTests(unittest.TestCase):
    def setUp(self):
        (D/"synthetic").mkdir(exist_ok=True)
        self.root=Path(tempfile.mkdtemp(prefix="bridge_",dir=D/"synthetic"))
    def test_journal_rotation_and_restart_retains_every_event(self):
        j=c.EventJournal(self.root/"journal",limit=700)
        for i in range(12):j.append(dict(event="sample",payload="x"*100,i=i))
        index=c.read_json(self.root/"journal/index.json");self.assertGreater(len(index["sealed"]),1)
        again=c.EventJournal(self.root/"journal",limit=700);self.assertEqual(again.seq,12)
        again.append(dict(event="after_restart"));self.assertEqual(c.EventJournal(self.root/"journal",700).seq,13)
        self.assertTrue(all((self.root/"journal"/x["path"]).stat().st_size<=700 for x in index["sealed"]))
    def test_journal_truncated_tail_fails_without_repair(self):
        j=c.EventJournal(self.root/"journal",700);j.append(dict(event="ok"))
        with j.path.open("ab") as f:f.write(b'{"partial":')
        before=j.path.read_bytes()
        with self.assertRaises(ValueError):c.EventJournal(self.root/"journal",700)
        self.assertEqual(before,j.path.read_bytes())
    def test_journal_sealed_tamper_fails(self):
        j=c.EventJournal(self.root/"journal",700)
        for i in range(5):j.append(dict(event="sample",payload="x"*100))
        p=self.root/"journal"/c.read_json(j.index_path)["sealed"][0]["path"];p.write_bytes(p.read_bytes().replace(b"sample",b"tamper",1))
        with self.assertRaises(ValueError):c.EventJournal(self.root/"journal",700)
    def test_current_pull_selected_without_reexpanding_historical_payload(self):
        prefix="results/formal_main_v2/pulls/"
        refs={}
        for name,ids in (("old",["1"]),("current",["2"])):
            rel=prefix+name+"/receipt.json";c.atomic_json(self.root/rel,dict(selected_jobs=[dict(job_id=i) for i in ids]));refs[rel]=c.file_sha(self.root/rel)
        got=w.pull_reference(self.root,dict(job_id="2",evidence_sha256=refs))
        self.assertEqual(got["path"],prefix+"current/receipt.json")
    def test_current_pull_ambiguous_rejected(self):
        refs={}
        for name in ("one","two"):
            rel=f"results/formal_main_v2/pulls/{name}/receipt.json";c.atomic_json(self.root/rel,dict(selected_jobs=[dict(job_id="2")]))
            refs[rel]=c.file_sha(self.root/rel)
        with self.assertRaises(ValueError):w.pull_reference(self.root,dict(job_id="2",evidence_sha256=refs))
    def test_cache_rehashes_same_size_mtime_preserved_change_using_ctime(self):
        p=self.root/"value.json";p.write_bytes(b"aaaa");before=p.stat()
        x=b.Backend.__new__(b.Backend);x.root=self.root;x.cache={}
        ref=dict(path="value.json",sha256=c.file_sha(p));x.watch(ref)
        p.write_bytes(b"bbbb");os.utime(p,ns=(before.st_atime_ns,before.st_mtime_ns))
        with self.assertRaises(ValueError):x.watch(ref)
    def test_evidence_rejects_mutable_state_and_sealed_before_read(self):
        x=b.Backend.__new__(b.Backend);x.root=self.root;x.cache={}
        for name in ("results/experiment_supervisor/state.json","results/final/initial_states.json","weights/w.pt"):
            with self.subTest(name=name),self.assertRaises(ValueError):x.verify_evidence([dict(path=name,sha256="a"*64)])
    def test_resume_probe_hashes_only_exact_registered_two_files(self):
        h=self.root/"H";pair=dict(checkpoint_root_server=str(h/"checkpoints/formal_main_v2/suite/k128/seed17"))
        specs=[]
        for arm in ("A","B"):
            p=Path(pair["checkpoint_root_server"])/arm/"resume.pt";p.parent.mkdir(parents=True);p.write_bytes(arm.encode())
            specs.append(dict(arm=arm,path=str(p),must_exist=True,sha256=c.file_sha(p)))
        with patch.object(c,"H",h):
            result=w.checkpoint_checks(pair,specs,"resume.pt");self.assertEqual(len(result),2)
            specs[0]["path"]=str(h/"unregistered.pt")
            with self.assertRaises(ValueError):w.checkpoint_checks(pair,specs,"resume.pt")
    def test_wrong_checkpoint_sha_rejects_without_deserializing(self):
        h=self.root/"H";pair=dict(checkpoint_root_server=str(h/"checkpoints/formal_main_v2/suite/k128/seed17"))
        specs=[]
        for arm in ("A","B"):
            p=Path(pair["checkpoint_root_server"])/arm/"resume.pt";p.parent.mkdir(parents=True);p.write_bytes(b"not torch")
            specs.append(dict(arm=arm,path=str(p),must_exist=True,sha256="0"*64))
        with patch.object(c,"H",h),self.assertRaises(ValueError):w.checkpoint_checks(pair,specs,"resume.pt")
    def test_existing_invocation_cannot_publish_or_overwrite(self):
        x=b.Backend.__new__(b.Backend);x.root=self.root;x.cache={};x.verify_evidence=lambda refs:None
        p=copy.deepcopy(plan(PAIRS[0],[]));wrapper=self.root/"wrapper";wrapper.write_text("wrapper")
        p.update(wrapper=str(wrapper),wrapper_sha256=c.file_sha(wrapper),invocation_contents={"watched_files":{}},preflight={"require_no_training_summary":str(self.root/"summary"),"require_absent_if_fresh":[]},server_resume_checks=[])
        p["invocation_sha256"]=c.sha_bytes(c.canonical(p["invocation_contents"])+b"\n");x.registration={"pairs":[PAIRS[0]]}
        target=self.root/p["invocation"];target.parent.mkdir(parents=True);target.write_bytes(b"prior")
        with self.assertRaisesRegex(ValueError,"invocation exists"):x.verify_prepared_plan(p)
        self.assertEqual(target.read_bytes(),b"prior")
