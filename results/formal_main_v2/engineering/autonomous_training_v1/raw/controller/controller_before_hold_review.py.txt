def require(c,m):
    if not c: raise ValueError(m)
def canonical(v):
    return json.dumps(v,sort_keys=True,ensure_ascii=False,allow_nan=False,separators=(",",":")).encode()
def sha_bytes(v): return hashlib.sha256(v).hexdigest()
def no_links(path):
    p=Path(path).absolute()
    for x in (p,*p.parents): require(not x.is_symlink(),"symlink forbidden: "+str(x))
    return p
def file_sha(path):
    with no_links(path).open("rb") as f:
        require(stat.S_ISREG(os.fstat(f.fileno()).st_mode),"regular file required")
        h=hashlib.sha256()
        for b in iter(lambda:f.read(8<<20),b""): h.update(b)
    return h.hexdigest()
def read_json(path):
    return json.loads(no_links(path).read_bytes(),parse_constant=lambda v:(_ for _ in ()).throw(ValueError("nonfinite JSON")))
def atomic_json(path,value,expected_sha=None):
    p=no_links(path)
    if expected_sha is not None: require(p.exists() and file_sha(p)==expected_sha,"state changed outside writer transaction")
    p.parent.mkdir(parents=True,exist_ok=True)
    tmp=p.with_name(p.name+"."+uuid.uuid4().hex+".partial")
    with tmp.open("xb") as f:
        f.write(canonical(value)+b"\n");f.flush();os.fsync(f.fileno())
    if expected_sha is not None: require(file_sha(p)==expected_sha,"state changed before atomic publication")
    os.replace(tmp,p)
    fd=os.open(p.parent,os.O_RDONLY|os.O_DIRECTORY)
    try: os.fsync(fd)
    finally: os.close(fd)
@contextmanager
def exclusive_lock(path):
    p=no_links(path);p.parent.mkdir(parents=True,exist_ok=True)
    with p.open("a+b") as f:
        fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
        try: yield f
        finally: fcntl.flock(f,fcntl.LOCK_UN)
def utc(): return datetime.now(timezone.utc).isoformat()

def state_name(v):
    require(isinstance(v,str) and v,"scheduler state missing")
    return "CANCELLED" if re.fullmatch(r"CANCELLED by [0-9]+",v) else v
def parse_sacct(raw,ids):
    out={}
    for line in raw.splitlines():
        if not line.strip(): continue
        c=line.split("|")
        if c and c[-1]=="": c.pop()
        require(len(c)==8,"unexpected sacct columns")
        jid,status,exitcode,elapsed,alloc,node,name,comment=c
        require(jid in ids and jid not in out,"foreign or duplicate sacct row")
        require(elapsed.isdigit() and re.fullmatch(r"[0-9]+:[0-9]+",exitcode),"invalid accounting scalar")
        resources=dict(i.split("=",1) for i in alloc.split(",") if "=" in i)
        gpus=resources.get("gres/gpu")
        if gpus is None: gpus=sum(int(v) for k,v in resources.items() if k.startswith("gres/gpu:"))
        require(str(gpus).isdigit(),"invalid GPU allocation")
        out[jid]=dict(job_id=jid,status=state_name(status),exit_code=exitcode,elapsed_seconds=int(elapsed),allocation=alloc,node=node,gpus=int(gpus),scheduler_name=name,scheduler_comment=comment)
    return out
class Slurm:
    """Queries registered IDs; only submits one verified plan."""
    def __init__(self,runner=subprocess.run,timeout=60): self.runner,self.timeout=runner,timeout
    def run(self,argv): return self.runner(argv,capture_output=True,text=True,timeout=self.timeout)
    def observe(self,jobs):
        ids=[str(j["job_id"]) for j in jobs]
        require(len(ids)==len(set(ids)) and all(re.fullmatch("[0-9]+",x) for x in ids),"invalid/duplicate registered IDs")
        if not ids: return dict(rows={},queue={},raw_sacct="",raw_squeue="")
        a=self.run(["sacct","-n","-X","-P","-j",",".join(ids),"--format=JobIDRaw,State,ExitCode,ElapsedRaw,AllocTRES,NodeList,JobName%128,Comment%128"])
        require(a.returncode==0,"sacct failed: "+a.stderr)
        rows=parse_sacct(a.stdout,set(ids))
        idsq=[j["job_id"] for j in jobs if j["job_id"] not in rows or rows[j["job_id"]]["status"] not in TERMINAL or state_name(j.get("status","UNKNOWN")) not in TERMINAL]
        queue={};rawq=""
        if idsq:
            q=self.run(["squeue","-h","-j",",".join(idsq),"-o","%i|%T|%k"])
            require(q.returncode==0,"squeue failed: "+q.stderr);rawq=q.stdout
            for line in rawq.splitlines():
                if not line.strip():continue
                c=line.split("|");require(len(c)==3 and c[0] in idsq and c[0] not in queue,"foreign/duplicate/malformed queue row")
                queue[c[0]]=dict(status=state_name(c[1]),comment=c[2])
        return dict(rows=rows,queue=queue,raw_sacct=a.stdout,raw_squeue=rawq)
    def submit(self,argv): return self.run(argv)
def capacity(jobs,snapshot,coordinator_job_id=None):
    rows,queue=snapshot["rows"],snapshot["queue"]
    occupied=[];unknown=[];pairs={};findings=[]
    for j in jobs:
        jid=str(j["job_id"]);r=rows.get(jid);old=state_name(j.get("status","UNKNOWN"))
        now=r["status"] if r else "MISSING";q=queue.get(jid,{}).get("status")
        if old in TERMINAL and r and (now!=old or r["exit_code"]!=j.get("exit_code",r["exit_code"])): findings.append("terminal_identity_drift:"+jid)
        if r and r["gpus"] not in (0,1):findings.append("unsupported_GPU_allocation:"+jid)
        exempt=(j.get("gpus")==0 and r is not None and r["gpus"]==0)  # Only confirmed CPU allocations; unknown still blocks all submissions.
        if now not in TERMINAL|KNOWN_LIVE or (q and q not in KNOWN_LIVE):unknown.append(jid)
        if r and now in TERMINAL and q is not None:findings.append("accounting_queue_disagreement:"+jid)
        if (now not in TERMINAL or q is not None) and not exempt:
            occupied.append(jid);key=job_pair(j)
            if key:pairs.setdefault(key,[]).append(jid)
    findings.extend("duplicate_pair:"+k for k,v in pairs.items() if len(v)>1)
    if len(occupied)>8:findings.append("over_capacity")
    return dict(occupied=occupied,unknown=unknown,active_pairs=pairs,findings=findings)
def validate_matrix(pairs):
    require(len(pairs)==36 and {(p["suite"],p["visual_tokens"],p["seed"]) for p in pairs}==PAIR_KEYS,"exact36-pair matrix required")
    for p in pairs: require(re.fullmatch("[0-9a-f]{64}",p["sha256"]),"pair config SHA missing")
def validate_plan(p,pair,history):
    require(p.get("pair_key")==pair_key(pair),"plan pair differs")
    require(p.get("source_root")==str(S) and p.get("source_sha256")==SOURCE_SHA,"plan source differs")
    require(p.get("config_sha256")==pair["sha256"],"plan config differs")
    require(p.get("target_updates")==5000 and p.get("soft_wall_seconds")==169200 and p.get("wall_time")=="48:00:00" and p.get("gpus")==1,"execution/resource policy differs")
    require(p.get("resume") is bool(history),"fresh/resume plan differs")
    failed=[j["job_id"] for j in history if j["status"]!="COMPLETED" or j.get("exit_code")!="0:0"]
    if failed:
        repair=p.get("failed_history_repair",{})
        require(repair.get("status")=="verified_root_repair" and sorted(repair.get("failed_job_ids",[]))==sorted(failed) and repair.get("pair_key")==pair_key(pair),"failed history requires exact root repair")
        require(repair.get("root_repair_ref") in p.get("evidence_refs",[]),"repair SHA absent from verified evidence")
    require(p.get("run_root")==pair["run_root_server"]+"/training","run root differs")
    wrapper={"libero_10":S/"scripts/server/formal_main.slurm","libero_spatial":S/"scripts/server/formal_main.slurm","libero_object":P/"scripts/server/formal_asset_main.slurm","libero_goal":P/"scripts/server/formal_goal_asset_main.slurm"}[pair["suite"]]
    require(p.get("wrapper")==str(wrapper),"unregistered wrapper")
    inv=p.get("invocation","")
    require(re.fullmatch(r"results/formal_main_v2/invocations/[a-zA-Z0-9_-]+\.json",inv),"noncanonical invocation")
    require(re.fullmatch("[0-9a-f]{64}",p.get("invocation_sha256","")),"invocation SHA missing")
    e=p.get("exports",{})
    require(set(e) in ({"SOURCE_ROOT","INVOCATION"},{"SOURCE_ROOT","INVOCATION","INVOCATION_SHA256"}),"unexpected export")
    require(e.get("SOURCE_ROOT")==str(S) and e.get("INVOCATION")==str(P/inv),"wrong invocation export")
    if "INVOCATION_SHA256" in e:require(e["INVOCATION_SHA256"]==p["invocation_sha256"],"export SHA differs")
    require(p.get("scientific_verification",{}).get("status")=="passed_full_verification","scientific gate absent")
    require(isinstance(p.get("evidence_refs"),list) and p["evidence_refs"],"evidence missing")
    require(isinstance(p.get("job_record"),dict) and isinstance(p.get("autonomous_execution"),dict),"job/execution contract missing")
    require(p.get("stage_key")==f"{pair['suite']}_train_k{pair['visual_tokens']}_seed{pair['seed']}","stage differs")
    return p
def sbatch_argv(p,token):
    require(re.fullmatch("[0-9a-f]{64}",token),"bad submission token")
    e="ALL,"+",".join(k+"="+v for k,v in sorted(p["exports"].items()))
    require(not any(c in e for c in ("\n","\r",";","\0")),"unsafe exports")
    return ["sbatch","--parsable","--no-requeue","--time=48:00:00","--gres=gpu:l40:1","--exclude=gpu4020","--job-name=cti-formal","--comment=cti-auto-v1:"+token,"--chdir="+str(P),"--export="+e,p["wrapper"]]

    def leadership(self):
        with exclusive_lock(self.lease_path):
            self._lease=True
            old=read_json(self.owner_path) if self.owner_path.exists() else None
            if old:require(old["activation_sha256"]==self.activation_sha,"owner activation differs")

            self.event("leader_acquired",prior_owner=old)
            try:yield
            finally:self.event("leader_released");self._lease=False

    def tick(self):
        require(self._lease,"leadership lock not held")
        with exclusive_lock(self.writer_path):
            before_sha=file_sha(self.state_path);state=read_json(self.state_path)
            jobs=state["registered_jobs"]
            require(isinstance(jobs,list),"registered jobs must be list")
            before_jobs={str(j["job_id"]):copy.deepcopy(j) for j in jobs}
            require(len(before_jobs)==len(jobs),"duplicate registered ID")

            require(control.get("activation_sha256",self.activation_sha)==self.activation_sha,"different controller activation owns state")

            except Exception as exc:
                control["last_result"]=dict(status="scheduler_unavailable",error=repr(exc))
                self._save(state,before_sha);self.event("scheduler_unavailable",error=repr(exc))
                return control["last_result"]
            self.event("scheduler_observation",raw_sacct=snap["raw_sacct"],raw_squeue=snap["raw_squeue"],capacity=caps,registered_job_ids=sorted(before_jobs))
            for job in jobs:
                row=snap["rows"].get(job["job_id"])
                if row and job.get("experiment")==ROUND and not (state_name(job.get("status","UNKNOWN")) in TERMINAL and (row["status"]!=state_name(job["status"]) or row["exit_code"]!=job.get("exit_code",row["exit_code"]))):
                    job.update({k:row[k] for k in ("status","exit_code","elapsed_seconds","allocation","node")})
                    # No new field is added to historical job snapshots; accounting stays in the observation journal.
            control["allocated_gpu_seconds_observed"]=sum(row["gpus"]*row["elapsed_seconds"] for jid,row in snap["rows"].items() if before_jobs[jid].get("experiment")==ROUND)

            if state.get("pending_submission") is not None:

    def _submission_hold(self,control):
        value=control.get("root_submission_hold")
        if value is None:return False
        require(isinstance(value,dict) and set(value)=={"enabled","reason","root_record","requested_utc"} and type(value["enabled"]) is bool,"invalid root hold")
        reference(self.root,value["root_record"])
        require(isinstance(value["reason"],str) and value["reason"] and isinstance(value["requested_utc"],str),"root hold provenance missing")
        return value["enabled"]
    def _reserved_slots(self,control):
        value=control.get("reserved_GPU_slots",0)
        require(type(value) is int and 0<=value<=8,"invalid root reserved slots")
        if value:
            reference(self.root,control.get("reservation_root_record"))
        return value
    def _advance(self,state,jobs,caps,before_sha):
        control=state["autonomous_training_v1"]
        for pair in self.pairs:
            key=pair_key(pair)
            if key in caps["active_pairs"]:continue
            if key in control["completed_pairs"]:
                self.backend.verify_evidence(control["completed_pairs"][key]["evidence_refs"])
                continue
            history=[j for j in jobs if job_pair(j)==key]
            if any(state_name(j["status"]) not in TERMINAL for j in history):continue
            try:
                o=self.backend.inspect_and_prepare(pair,history,copy.deepcopy(state))
                require(o.get("decision") in {"complete","ready","blocked"},"unknown backend decision")
                if o["decision"]=="blocked":
                    control["blocked_pairs"][key]=o;continue
                require(o.get("evidence_refs"),"backend evidence missing")
                self.backend.verify_evidence(o["evidence_refs"])
                completed=o.get("completed_by_arm")
                require(isinstance(completed,dict) and set(completed)=={"A","B"} and all(type(x) is int and 0<=x<=5000 for x in completed.values()),"verified progress missing")
                control.setdefault("verified_pairs",{})[key]=dict(completed_by_arm=completed,evidence_refs=o["evidence_refs"],verification_kind="machine_automatic_not_independent")
                if o["decision"]=="complete":
                    require(o.get("completed_by_arm")=={"A":5000,"B":5000},"partial pair cannot be complete")
                    require(o.get("validation_updates")==list(range(0,5001,500)),"eleven-node schedule incomplete")
                    require(o.get("selected_checkpoints_verified") is True,"actual selected checkpoint verification missing")
                    control["completed_pairs"][key]=o;control["blocked_pairs"].pop(key,None)
                    self.event("pair_complete",pair_key=key,evidence_refs=o["evidence_refs"])
                    continue
                plan=validate_plan(o["plan"],pair,history)
                self.backend.verify_prepared_plan(plan)
            except Exception as exc:
                control["blocked_pairs"][key]=dict(decision="blocked",reason=repr(exc),utc=utc())
                self.event("pair_blocked",pair_key=key,error=repr(exc));continue
            control["blocked_pairs"].pop(key,None)
            token=sha_bytes(canonical(dict(plan=plan,activation=self.activation_sha,nonce=uuid.uuid4().hex)))
            argv=sbatch_argv(plan,token)
            pending=dict(version=VERSION,experiment=ROUND,pair_key=key,stage_key=plan["stage_key"],invocation=plan["invocation"],invocation_sha256=plan["invocation_sha256"],request_token=token,actual_sbatch_argv=argv,plan=plan,created_utc=utc(),state="submission_started_no_acknowledgment",retry_permitted=False,requested_gpus=1)

            self._save(state,before_sha);pending_sha=file_sha(self.state_path)
            self.event("submission_intent_durable",pending=pending)
            try:answer=self.slurm.submit(argv)
            except Exception as exc:
                self.event("submission_outcome_unresolved",pair_key=key,error=repr(exc))
                return dict(status="unresolved_submission",pair_key=key,error=repr(exc))

            if answer.returncode!=0 or not match:

            if any(j["job_id"]==jid for j in jobs):
                self.event("submission_outcome_unresolved",reason="duplicate_returned_job_id",job_id=jid)
                return dict(status="unresolved_submission",pair_key=key,reason="duplicate_job_id")
            job=copy.deepcopy(plan["job_record"])

def reference(root,ref):
    require(isinstance(ref,dict) and set(ref)=={"path","sha256"},"invalid SHA reference")
    rel=ref["path"]
    require(isinstance(rel,str) and rel and not rel.startswith("/") and "\\" not in rel and ":" not in rel and ".." not in Path(rel).parts,"noncanonical evidence path")
    low=rel.lower()
    require(not any(x in Path(low).parts for x in ("sealed","test","weights","checkpoints")) and not any(x in low for x in ("initial_states","init_states","final_registry")),"sealed/weight input forbidden")
    p=no_links(root/rel);require(file_sha(p)==ref["sha256"],"evidence SHA differs: "+rel)
    return p
