def run_episode(env, predict_chunk: Callable, instruction: str, initial_state, *, wait_steps=10, max_steps=220, policy_watchdog_seconds=60.0) -> dict[str, Any]:
    """One independent reset. The watchdog is a hang guard, not real-time control.

    A synchronously returned request exceeding the guard is rejected. A wedged
    CUDA process still requires the enclosing Slurm time limit; no unsafe thread
    cancellation is attempted.
    """
    if wait_steps < 0 or max_steps <= 0 or policy_watchdog_seconds <= 0:
        raise ValueError('invalid episode limits')
    started = time.perf_counter()
    result = {'status': 'completed', 'success': False, 'termination': 'task_horizon', 'control_steps': 0, 'wait_steps': 0, 'policy_calls': 0, 'chunks': [], 'executed_actions': []}
    phase = 'environment'
    try:
        _cti_prefix_observer.capture('reset_before', env)
        _cti_prefix_reset_return = env.reset()
        _cti_prefix_observer.capture('reset_after', env, _cti_prefix_reset_return)
        _cti_prefix_observer.capture('set_init_before', env, _cti_prefix_reset_return)
        del _cti_prefix_reset_return
        observation = env.set_init_state(np.asarray(initial_state, dtype=np.float64))
        _cti_prefix_observer.capture('set_init_after', env, observation)
        if hasattr(predict_chunk, 'reset'):
            predict_chunk.reset()
        for _ in range(wait_steps):
            _cti_prefix_observer.capture('wait_before', env, observation, wait_steps=result['wait_steps'])
            observation, _, _, _ = env.step(list(DUMMY_ACTION))
            result['wait_steps'] += 1
            _cti_prefix_observer.capture('wait_after', env, observation, wait_steps=result['wait_steps'])
        queue = deque()
        for control_step in range(max_steps):
            if not queue:
                phase = 'policy'
                _cti_prefix_observer.capture('t0_query_before' if control_step == 0 else 'query_before', env, observation, control_step=control_step, instruction=instruction)
                query_started = time.perf_counter()
                result['policy_calls'] += 1
                chunk = predict_chunk(observation, instruction)
                elapsed = time.perf_counter() - query_started
                _cti_prefix_observer.capture('t0_query_after' if control_step == 0 else 'query_after', env, observation, control_step=control_step, chunk=chunk, instruction=instruction)
                if not isinstance(chunk, PolicyChunk):
                    raise TypeError('policy must return a validated PolicyChunk')
                if elapsed > policy_watchdog_seconds:
                    raise TimeoutError(f'policy watchdog exceeded: {elapsed:.3f}s')
                result['chunks'].append({'control_step': control_step, 'observation_sha256': _observation_hash(observation), 'normalized_actions': chunk.normalized_actions.tolist(), 'environment_actions': chunk.environment_actions.tolist(), 'engineering_seconds': elapsed, 'metadata': dict(chunk.metadata)})
                queue.extend(np.asarray(chunk.environment_actions).copy())
            action = queue.popleft()
            result['executed_actions'].append(action.tolist())
            phase = 'environment'
            observation, _, done, _ = env.step(action.tolist())
            result['control_steps'] += 1
            _cti_prefix_observer.step_after(action, control_step, result['control_steps'], result['policy_calls'])
            if bool(done):
                result.update(success=True, termination='environment_success')
                break
    except Exception as exc:
        result.update(status='policy_error' if phase == 'policy' else 'infrastructure_error', termination=type(exc).__name__, error=str(exc), error_phase=phase)
    result['wall_seconds'] = time.perf_counter() - started
    return result
